1//! Bounded nearest-name suggestions for unresolved identifiers, fields and types.
2//! Suggestions only decorate diagnostics; they never change resolution.
3
4/// Names longer than this are not compared, keeping edit-distance work small and predictable.
5const MAX_NAME_CHARS: usize = 64;
6/// Candidate sets are truncated deterministically before comparison.
7const MAX_CANDIDATES: usize = 4_096;
8
9/// Return the closest candidate to `name`, or `None` when nothing is plausibly a typo.
10///
11/// A candidate qualifies when its edit distance is at most one third of the longer
12/// spelling (at least one edit), or when it differs only by letter case. Ties select
13/// the lexicographically smallest candidate so messages stay deterministic.
14pub fn nearest<'a>(name: &str, candidates: impl IntoIterator<Item = &'a str>) -> Option<String> {
15    let target: Vec<char> = name.chars().take(MAX_NAME_CHARS + 1).collect();
16    if target.is_empty() || target.len() > MAX_NAME_CHARS {
17        return None;
18    }
19    let mut best: Option<(usize, &str)> = None;
20    for candidate in candidates.into_iter().take(MAX_CANDIDATES) {
21        if candidate == name || candidate.is_empty() || candidate.starts_with('$') {
22            continue;
23        }
24        let other: Vec<char> = candidate.chars().take(MAX_NAME_CHARS + 1).collect();
25        if other.len() > MAX_NAME_CHARS {
26            continue;
27        }
28        let distance = if candidate.eq_ignore_ascii_case(name) {
29            0
30        } else {
31            let allowed = (target.len().max(other.len()) / 3).max(1);
32            if target.len().abs_diff(other.len()) > allowed {
33                continue;
34            }
35            let distance = levenshtein(&target, &other);
36            if distance > allowed {
37                continue;
38            }
39            distance
40        };
41        let better = match best {
42            None => true,
43            Some((best_distance, best_name)) => {
44                distance < best_distance || (distance == best_distance && candidate < best_name)
45            }
46        };
47        if better {
48            best = Some((distance, candidate));
49        }
50    }
51    best.map(|(_, candidate)| candidate.to_owned())
52}

54/// Format the suffix appended to a diagnostic, or an empty string without a suggestion.
55pub fn hint<'a>(name: &str, candidates: impl IntoIterator<Item = &'a str>) -> String {
56    nearest(name, candidates).map_or_else(String::new, |found| format!("; did you mean '{found}'?"))
57}

111/// Suggest a builtin module member: nearest spelling first, then a well-known synonym.
112/// `members` holds fully qualified spellings sharing `prefix`; the result is qualified too.
113pub fn nearest_member<'a>(
114    prefix: &str,
115    member: &str,
116    members: impl IntoIterator<Item = &'a str>,
117) -> Option<String> {
118    let members: Vec<&str> = members.into_iter().take(MAX_CANDIDATES).collect();
119    let qualified = format!("{prefix}.{member}");
120    if let Some(found) = nearest(&qualified, members.iter().copied()) {
121        return Some(found);
122    }
123    let lowered = member.to_ascii_lowercase();
124    SYNONYMS
125        .iter()
126        .find(|(alias, _)| *alias == lowered)
127        .and_then(|(_, targets)| {
128            targets.iter().find_map(|target| {
129                let candidate = format!("{prefix}.{target}");
130                members
131                    .iter()
132                    .any(|existing| *existing == candidate)
133                    .then_some(candidate)
134            })
135        })
136}

138/// Return the unqualified tail of a module-qualified declaration name.
139pub fn tail(name: &str) -> &str {
140    name.rsplit('.').next().unwrap_or(name)
141}

143/// Optimal string alignment distance: insertions, deletions, substitutions and adjacent
144/// transpositions each cost one edit, so `cuont` is one edit from `count`.
145fn levenshtein(a: &[char], b: &[char]) -> usize {
146    let width = b.len() + 1;
147    let mut rows: Vec<Vec<usize>> = vec![(0..width).collect()];
148    for (i, &left) in a.iter().enumerate() {
149        let mut current = vec![0; width];
150        current[0] = i + 1;
151        for (j, &right) in b.iter().enumerate() {
152            let previous = &rows[i];
153            let substitution = previous[j] + usize::from(left != right);
154            let mut best = substitution.min(previous[j + 1] + 1).min(current[j] + 1);
155            if i > 0 && j > 0 && left == b[j - 1] && a[i - 1] == right {
156                best = best.min(rows[i - 1][j - 1] + 1);
157            }
158            current[j + 1] = best;
159        }
160        rows.push(current);
161    }
162    rows[a.len()][b.len()]
163}

165#[cfg(test)]
166mod tests {
167    use super::*;
168
169    #[test]
170    fn finds_single_edit_typos_and_case_differences() {
171        assert_eq!(
172            nearest("cuont", ["count", "total"]),
173            Some("count".to_owned())
174        );
175        assert_eq!(
176            nearest("Lenght", ["len", "length"]),
177            Some("length".to_owned())
178        );
179        assert_eq!(nearest("string", ["String"]), Some("String".to_owned()));
180        assert_eq!(
181            nearest("helpr", ["helper", "main"]),
182            Some("helper".to_owned())
183        );
184        assert_eq!(nearest("Gren", ["Green", "Red"]), Some("Green".to_owned()));
185    }
186
187    #[test]
188    fn transpositions_cost_one_edit() {
189        assert_eq!(levenshtein(&['a', 'b'], &['b', 'a']), 1);
190        assert_eq!(levenshtein(&['c', 'a'], &['a', 'b', 'c']), 3);
191        assert_eq!(tail("main.helper"), "helper");
192        assert_eq!(tail("helper"), "helper");
193    }
194
195    #[test]
196    fn rejects_distant_or_identical_names() {
197        assert_eq!(nearest("zzzzzzzzzz", ["count", "total"]), None);
198        assert_eq!(nearest("count", ["count"]), None);
199        assert_eq!(nearest("", ["count"]), None);
200        assert_eq!(nearest("ab", ["xyz"]), None);
201    }
202
203    #[test]
204    fn ties_resolve_lexicographically() {
205        assert_eq!(nearest("cat", ["cut", "bat"]), Some("bat".to_owned()));
206    }
207
208    #[test]
209    fn bounds_long_names_and_candidate_counts() {
210        let long = "a".repeat(MAX_NAME_CHARS + 1);
211        assert_eq!(nearest(&long, [long.as_str()]), None);
212        let many = std::iter::repeat_n("zzzzzzzzzz", MAX_CANDIDATES).chain(["count"]);
213        assert_eq!(nearest("cuont", many), None);
214        let few = std::iter::repeat_n("zzzzzzzzzz", MAX_CANDIDATES - 1).chain(["count"]);
215        assert_eq!(nearest("cuont", few), Some("count".to_owned()));
216    }
217
218    #[test]
219    fn members_fall_back_to_synonyms_only_when_the_target_exists() {
220        let members = ["Map.put", "Map.get", "Map.len"];
221        assert_eq!(
222            nearest_member("Map", "insert", members),
223            Some("Map.put".to_owned())
224        );
225        assert_eq!(
226            nearest_member("Map", "Size", members),
227            Some("Map.len".to_owned())
228        );
229        assert_eq!(
230            nearest_member("Map", "gett", members),
231            Some("Map.get".to_owned())
232        );
233        assert_eq!(nearest_member("Map", "upper", members), None);
234        assert_eq!(nearest_member("Map", "zzzzzzzz", members), None);
235    }
236
237    #[test]
238    fn hint_formats_the_suffix() {
239        assert_eq!(hint("helpr", ["helper"]), "; did you mean 'helper'?");
240        assert_eq!(hint("zzz", ["helper"]), "");
241    }
