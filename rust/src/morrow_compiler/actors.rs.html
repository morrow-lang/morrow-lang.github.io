8/// Peel only the explicit actor-effect wrapper, retaining the ordinary source signature.
9pub(crate) fn function(ty: &Type) -> Option<(Option<&Type>, &[Type], &Type)> {
10    match ty {
11        Type::Function(args, result) => Some((None, args, result)),
12        Type::ActorFunction(mailbox, function) => {
13            let Type::Function(args, result) = function.as_ref() else {
14                return None;
15            };
16            Some((Some(mailbox), args, result))
17        }
18        _ => None,
19    }
20}

22/// Enumerate direct syntax children exactly once; callers own depth and aggregate limits.
23pub(crate) fn source_children(expr: &ast::Expr) -> Vec<&ast::Expr> {
24    use ast::ExprKind::*;
25    match &expr.kind {
26        Receive { .. } | With { .. } => scoped_children(expr),
27        Match { value, arms } => {
28            let mut out = vec![value.as_ref()];
29            out.extend(arm_children(arms));
30            out
31        }
32        If {
33            condition,
34            then_branch,
35            else_branch,
36        } => {
37            let mut out = vec![condition.as_ref(), then_branch.as_ref()];
38            out.extend(else_branch.as_deref());
39            out
40        }
41        Lambda { body, .. }
42        | Try(body)
43        | Return(body)
44        | Defer(body)
45        | Unary { value: body, .. }
46        | Field { value: body, .. } => vec![body],
47        Binary { left, right, .. }
48        | Range {
49            start: left,
50            end: right,
51            ..
52        }
53        | PostfixIf {
54            value: left,
55            condition: right,
56        } => vec![left, right],
57        Tuple(values) | List(values) => values.iter().collect(),
58        Map(entries) => entries.iter().flat_map(|(k, v)| [k, v]).collect(),
59        RecordUpdate { value, fields } => std::iter::once(value.as_ref())
60            .chain(fields.iter().map(|f| &f.value))
61            .collect(),
62        Call { args, .. } | GlobalCall { args, .. } => args.iter().map(|a| &a.value).collect(),
63        Apply { callee, args } => std::iter::once(callee.as_ref())
64            .chain(args.iter().map(|a| &a.value))
65            .collect(),
66        Pipe { value, args, .. } | GlobalPipe { value, args, .. } => {
67            std::iter::once(value.as_ref())
68                .chain(args.iter().map(|a| &a.value))
69                .collect()
70        }
71        Block(stmts) => stmts.iter().flat_map(statement_children).collect(),
72        Interpolate(parts) | MultilineString(parts) => parts
73            .iter()
74            .filter_map(|part| match part {
75                ast::StringPart::Value(v) => Some(v),
76                _ => None,
77            })
78            .collect(),
79        For { iterable, body, .. } => vec![iterable, body],
80        ConditionMatch(arms) => arms
81            .iter()
82            .flat_map(|a| a.condition.iter().chain([&a.body]))
83            .collect(),
84        _ => Vec::new(),
85    }
86}

88/// Preserve initializer-before-failure order for statement-owned children.
89fn statement_children(stmt: &ast::Stmt) -> Vec<&ast::Expr> {
90    match stmt {
91        ast::Stmt::Let { value, .. }
92        | ast::Stmt::LetPattern { value, .. }
93        | ast::Stmt::Expr(value) => vec![value],
94        ast::Stmt::LetElse {
95            value, else_branch, ..
96        } => vec![value, else_branch],
97    }
98}

100/// Guards belong to the same receive/match arm and precede its body.
101fn arm_children(arms: &[ast::MatchArm]) -> Vec<&ast::Expr> {
102    arms.iter()
103        .flat_map(|a| a.guard.iter().chain([&a.body]))
104        .collect()
105}

107/// Actor children remain visible to publication checks, capture discovery and resource accounting.
108pub(crate) fn children(actor: &ir::ActorExpr) -> Vec<&ir::Expr> {
109    match actor {
110        ir::ActorExpr::Process(value) => value.children(),
111        ir::ActorExpr::Lowered(value) => value.children(),
112        ir::ActorExpr::Spawn {
113            entry,
114            max_restarts,
115            ..
116        } => std::iter::once(entry.as_ref())
117            .chain(max_restarts.as_deref())
118            .collect(),
119        ir::ActorExpr::SupervisedCurrent { pid } => vec![pid],
120        ir::ActorExpr::Send { pid, message } => vec![pid, message],
121        ir::ActorExpr::Call { args, .. } => args.iter().collect(),
122        ir::ActorExpr::Receive { arms, timeout, .. } => {
123            let mut out: Vec<_> = arms
124                .iter()
125                .flat_map(|a| a.guard.iter().chain([&a.body]))
126                .collect();
127            if let Some((duration, body)) = timeout {
128                out.extend([duration.as_ref(), body.as_ref()]);
129            }
130            out
131        }
132    }
133}

135/// Mutable actor children allow ordinary substitution and closure lifting without replay.
136pub(crate) fn children_mut(actor: &mut ir::ActorExpr) -> Vec<&mut ir::Expr> {
137    match actor {
138        ir::ActorExpr::Process(value) => value.children_mut(),
139        ir::ActorExpr::Lowered(value) => value.children_mut(),
140        ir::ActorExpr::Spawn {
141            entry,
142            max_restarts,
143            ..
144        } => std::iter::once(entry.as_mut())
145            .chain(max_restarts.as_deref_mut())
146            .collect(),
147        ir::ActorExpr::SupervisedCurrent { pid } => vec![pid],
148        ir::ActorExpr::Send { pid, message } => vec![pid, message],
149        ir::ActorExpr::Call { args, .. } => args.iter_mut().collect(),
150        ir::ActorExpr::Receive { arms, timeout, .. } => {
151            let mut out: Vec<_> = arms
152                .iter_mut()
153                .flat_map(|a| a.guard.iter_mut().chain([&mut a.body]))
154                .collect();
155            if let Some((duration, body)) = timeout {
156                out.extend([duration.as_mut(), body.as_mut()]);
157            }
158            out
159        }
160    }
161}

163/// Opaque compiler continuation operation; source/public IR cannot mint executable steps.
164///
165/// ```compile_fail
166/// let forged = morrow_compiler::actors::Lowered {};
167/// ```
168#[derive(Clone, Debug)]
169pub struct Lowered {
170    pub(crate) operation: Operation,
171}
172#[derive(Clone, Debug)]
173pub(crate) enum Operation {
174    ListBuilder {
175        capacity: Box<ir::Expr>,
176        item: Type,
177    },
178    ListAppend {
179        list: Box<ir::Expr>,
180        value: Box<ir::Expr>,
181    },
182    ClosureIdentity {
183        value: Box<ir::Expr>,
184        function: ir::FunctionId,
185    },
186    ClosureCapture {
187        value: Box<ir::Expr>,
188        index: usize,
189        ty: Type,
190    },
191    ScopeEnter,
192    ScopeDefer(Box<ir::Expr>),
193    ScopeLeave,
194    CleanupInvoke {
195        function: ir::FunctionId,
196        closure: Box<ir::Expr>,
197    },
198    /// Audited iteration fields; inaccessible to authored/public IR.
199    IterateField {
200        value: Box<ir::Expr>,
201        field: usize,
202    },
203    IterateItem {
204        value: Box<ir::Expr>,
205        index: Box<ir::Expr>,
206        item: Type,
207    },
208    Continue(Box<ir::Expr>),
209    /// Private nonescaping Unit helper self-tail frame; never authored IR.
210    ContinueReusable(Box<ir::Expr>),
211    /// Terminal process operation: its status returns directly to the scheduler.
212    ProcessExit(Box<ir::Expr>),
213    Pointer(Box<ir::Expr>),
214    Register {
215        view: crate::processes::ReceiveView,
216        mailbox: Type,
217        selector: Box<ir::Expr>,
218        timeout: Option<Box<ir::Expr>>,
219        duration: Box<ir::Expr>,
220    },
221    Select {
222        value: Box<ir::Expr>,
223        arms: Vec<ir::MatchArm>,
224    },
225}
226impl Lowered {
227    /// Expose every owned child to bounded compiler publication walkers.
228    fn children(&self) -> Vec<&ir::Expr> {
229        match &self.operation {
230            Operation::ScopeEnter | Operation::ScopeLeave => vec![],
231            Operation::ListBuilder { capacity, .. } => vec![capacity],
232            Operation::ListAppend { list, value } => vec![list, value],
233            Operation::ScopeDefer(value)
234            | Operation::CleanupInvoke { closure: value, .. }
235            | Operation::Continue(value)
236            | Operation::ContinueReusable(value)
237            | Operation::ProcessExit(value)
238            | Operation::Pointer(value)
239            | Operation::IterateField { value, .. }
240            | Operation::ClosureIdentity { value, .. }
241            | Operation::ClosureCapture { value, .. } => vec![value],
242            Operation::IterateItem { value, index, .. } => vec![value, index],
243            Operation::Register {
244                selector,
245                timeout,
246                duration,
247                ..
248            } => std::iter::once(selector.as_ref())
249                .chain(timeout.as_deref())
250                .chain([duration.as_ref()])
251                .collect(),
252            Operation::Select { value, arms } => std::iter::once(value.as_ref())
253                .chain(arms.iter().flat_map(|a| a.guard.iter().chain([&a.body])))
254                .collect(),
255        }
256    }
257    /// Preserve shared substitution visibility without exposing constructors to callers.
258    fn children_mut(&mut self) -> Vec<&mut ir::Expr> {
259        match &mut self.operation {
260            Operation::ScopeEnter | Operation::ScopeLeave => vec![],
261            Operation::ListBuilder { capacity, .. } => vec![capacity],
262            Operation::ListAppend { list, value } => vec![list, value],
263            Operation::ScopeDefer(value)
264            | Operation::CleanupInvoke { closure: value, .. }
265            | Operation::Continue(value)
266            | Operation::ContinueReusable(value)
267            | Operation::ProcessExit(value)
268            | Operation::Pointer(value)
269            | Operation::IterateField { value, .. }
270            | Operation::ClosureIdentity { value, .. }
271            | Operation::ClosureCapture { value, .. } => vec![value],
272            Operation::IterateItem { value, index, .. } => vec![value, index],
273            Operation::Register {
274                selector,
275                timeout,
276                duration,
277                ..
278            } => std::iter::once(selector.as_mut())
279                .chain(timeout.as_deref_mut())
280                .chain([duration.as_mut()])
281                .collect(),
282            Operation::Select { value, arms } => std::iter::once(value.as_mut())
283                .chain(
284                    arms.iter_mut()
285                        .flat_map(|a| a.guard.iter_mut().chain([&mut a.body])),
286                )
287                .collect(),
288        }
289    }
290}

292/// Preserve arm and binding evaluation order for syntax owning multiple independent scopes.
293fn scoped_children(expr: &ast::Expr) -> Vec<&ast::Expr> {
294    use ast::ExprKind::*;
295    match &expr.kind {
296        Receive { arms, timeout, .. } => {
297            let mut out = arm_children(arms);
298            if let Some((duration, body)) = timeout {
299                out.extend([duration.as_ref(), body.as_ref()]);
300            }
301            out
302        }
303        With {
304            bindings,
305            body,
306            arms,
307        } => {
308            let mut out: Vec<_> = bindings.iter().map(|b| &b.value).collect();
309            out.push(body);
310            if let Some(arms) = arms {
311                out.extend(arm_children(arms));
312            }
313            out
314        }
315        _ => Vec::new(),
316    }
