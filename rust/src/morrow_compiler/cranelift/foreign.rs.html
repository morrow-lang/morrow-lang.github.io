1//! Explicit physical C signatures; logical Morrow registers never determine foreign widths.
2use super::*;
3use crate::ffi::{AbiType, Declaration};
4
5pub(super) fn abi_param(ty: &AbiType) -> Result<AbiParam, String> {
6    use AbiType::*;
7    let physical = match ty {
8        I8 | U8 | Bool => types::I8,
9        I16 | U16 => types::I16,
10        I32 | U32 => types::I32,
11        I64 | U64 | Pointer(_) => types::I64,
12        F32 => types::F32,
13        F64 => types::F64,
14        Void => return Err("void has no physical value".into()),
15    };
16    let param = AbiParam::new(physical);
17    Ok(match ty {
18        I8 | I16 | I32 => param.sext(),
19        U8 | U16 | U32 | Bool => param.uext(),
20        _ => param,
21    })
22}

24impl Backend {
25    pub(super) fn declare_foreign(&mut self, declaration: &Declaration) -> Result<(), String> {
26        declaration
27            .validate(crate::Span::default())
28            .map_err(|error| error.message)?;
29        if let Some((_, existing)) = self.foreign.get(&declaration.symbol) {
30            return if existing == declaration {
31                Ok(())
32            } else {
33                Err(format!(
34                    "conflicting foreign declarations for {}",
35                    declaration.symbol
36                ))
37            };
38        }
39        if self.functions.contains_key(&declaration.symbol)
40            || self.data.contains_key(&declaration.symbol)
41            || runtime_abi::signature(&declaration.symbol).is_some()
42        {
43            return Err(format!(
44                "foreign symbol conflicts with compiler/runtime symbol: {}",
45                declaration.symbol
46            ));
47        }
48        let mut signature = self.module.make_signature();
49        signature.params = declaration
50            .params
51            .iter()
52            .map(abi_param)
53            .collect::<Result<_, _>>()?;
54        if declaration.result != AbiType::Void {
55            signature.returns.push(abi_param(&declaration.result)?);
56        }
57        let id = self
58            .module
59            .declare_function(&declaration.symbol, Linkage::Import, &signature)
60            .map_err(|error| error.to_string())?;
61        self.foreign
62            .insert(declaration.symbol.clone(), (id, declaration.clone()));
63        Ok(())
64    }
65}
