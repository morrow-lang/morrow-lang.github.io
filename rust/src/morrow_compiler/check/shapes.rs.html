1//! Delayed structural evidence stays in non-executable probes until source is rechecked.
2use super::*;

4pub(super) struct Obligation {
5    token: ir::ProbeToken,
6    kind: Shape,
7    span: Span,
8}
9enum Shape {
10    Field {
11        subject: Type,
12        name: String,
13        result: Type,
14    },
15    Update {
16        subject: Type,
17        fields: Vec<(String, Type)>,
18    },
19    Iterable {
20        subject: Type,
21        item: Type,
22    },
23    TupleRest {
24        subject: Type,
25        prefix: Vec<Type>,
26        tail: Type,
27    },
28    Binding {
29        subject: Type,
30        pattern: ir::Pattern,
31    },
32}
33
34impl Checker<'_> {
35    /// Delay only existential roots in the whole-signature pass, never rigid universals.
36    pub(super) fn unknown_shape(&self, ty: &Type, span: Span) -> Checked<bool> {
37        Ok(self.inference.whole_signature
38            && matches!(self.inference.resolve(ty, span)?, Type::Infer(_)))
39    }
40
41    /// Retain the actual receiver while its layout determines the field index later.
42    pub(super) fn defer_field(
43        &mut self,
44        value: ir::Expr,
45        name: &str,
46        span: Span,
47    ) -> Checked<TypedKind> {
48        let result = self.inference.fresh();
49        let token = self.obligation(
50            Shape::Field {
51                subject: value.ty.clone(),
52                name: name.into(),
53                result: result.clone(),
54            },
55            span,
56        )?;
57        Ok((
58            ir::ExprKind::Probe {
59                token,
60                children: vec![value],
61                bindings: vec![],
62            },
63            result,
64        ))
65    }
66
67    /// Capture ordered update operands before a later use identifies the nominal record.
68    pub(super) fn defer_update(
69        &mut self,
70        value: ir::Expr,
71        fields: &[ast::RecordField],
72        span: Span,
73        depth: usize,
74    ) -> Checked<TypedKind> {
75        let subject = value.ty.clone();
76        let mut children = vec![value];
77        let mut typed = Vec::new();
78        let mut names = HashSet::new();
79        for field in fields {
80            if !names.insert(&field.name) {
81                return Err(Diagnostic::new(field.span, "duplicate record field"));
82            }
83            let value = self.expression(&field.value, depth)?;
84            typed.push((field.name.clone(), value.ty.clone()));
85            children.push(value);
86        }
87        let token = self.obligation(
88            Shape::Update {
89                subject: subject.clone(),
90                fields: typed,
91            },
92            span,
93        )?;
94        Ok((
95            ir::ExprKind::Probe {
96                token,
97                children,
98                bindings: vec![],
99            },
100            subject,
101        ))
102    }
103
104    /// Iteration retains an independent item type until List, Map, or Range is established.
105    pub(super) fn defer_item(&mut self, subject: &Type, span: Span) -> Checked<Type> {
106        let item = self.inference.fresh();
107        self.obligation(
108            Shape::Iterable {
109                subject: subject.clone(),
110                item: item.clone(),
111            },
112            span,
113        )?;
114        Ok(item)
115    }
116
117    /// Tuple suffix evidence may flow in either direction without inventing the tuple arity.
118    pub(super) fn defer_tuple(
119        &mut self,
120        subject: &Type,
121        count: usize,
122        span: Span,
123    ) -> Checked<(Vec<Type>, Type)> {
124        let prefix: Vec<_> = (0..count).map(|_| self.inference.fresh()).collect();
125        let tail = self.inference.fresh();
126        self.obligation(
127            Shape::TupleRest {
128                subject: subject.clone(),
129                prefix: prefix.clone(),
130                tail: tail.clone(),
131            },
132            span,
133        )?;
134        Ok((prefix, tail))
135    }
136
137    /// Keep pattern-created IDs visible without fabricating executable projections.
138    pub(super) fn defer_binding(
139        &mut self,
140        value: ir::Expr,
141        pattern: ir::Pattern,
142        first_id: usize,
143        span: Span,
144    ) -> Checked<ir::Expr> {
145        let mut bindings = Vec::new();
146        for (id, ty) in self.scopes.iter().flat_map(|s| s.values()) {
147            returns::charge(&self.inference, span)?;
148            if id.0 >= first_id {
149                returns::charge_output(&self.inference, ty, span)?;
150                bindings.push(ir::Param {
151                    id: *id,
152                    ty: ty.clone(),
153                });
154            }
155        }
156        let token = self.obligation(
157            Shape::Binding {
158                subject: value.ty.clone(),
159                pattern,
160            },
161            span,
162        )?;
163        Ok(ir::Expr {
164            kind: ir::ExprKind::Probe {
165                token,
166                children: vec![value],
167                bindings,
168            },
169            ty: Type::Unit,
170            span,
171        })
172    }
173
174    /// Charge every retained obligation and keep its construction inaccessible outside the crate.
175    fn obligation(&mut self, kind: Shape, span: Span) -> Checked<ir::ProbeToken> {
176        returns::charge(&self.inference, span)?;
177        let token = ir::ProbeToken::new(self.inference.next_shape);
178        self.inference.next_shape += 1;
179        self.inference.shapes.push(Obligation {
180            token: token.clone(),
181            kind,
182            span,
183        });
184        Ok(token)
185    }
186}

188/// No provisional expression can be finalized into published typed IR.
189pub(super) fn reject(expr: &ir::Expr) -> Checked<()> {
190    if matches!(expr.kind, ir::ExprKind::Probe { .. }) {
191        Err(Diagnostic::new(
192            expr.span,
193            "inference probe cannot enter executable IR",
194        ))
195    } else {
196        Ok(())
197    }
198}

200/// Settle each component before quantification; stalled shapes require source annotations.
201pub(super) fn finish(inference: &mut Inference, registry: &nominal::Registry) -> Checked<()> {
202    loop {
203        let revision = inference.revision;
204        let mut pending = Vec::new();
205        for obligation in std::mem::take(&mut inference.shapes) {
206            returns::charge(inference, obligation.span)?;
207            if !solve(inference, registry, &obligation)? {
208                pending.push(obligation);
209            }
210        }
211        inference.shapes = pending;
212        returns::settle_calls(inference)?;
213        if inference.shapes.is_empty() {
214            return Ok(());
215        }
216        if inference.revision == revision {
217            return Err(Diagnostic::new(
218                inference.shapes[0].span,
219                "cannot infer private signature shape or tuple arity; add a parameter or local type annotation",
220            ));
221        }
222    }
223}

225/// Resolve a known structural owner, preserving unknown existential subjects for another pass.
226fn solve(
227    inference: &mut Inference,
228    registry: &nominal::Registry,
229    obligation: &Obligation,
230) -> Checked<bool> {
231    let span = obligation.span;
232    debug_assert!(obligation.token.id() < inference.next_shape);
233    if let Shape::TupleRest {
234        subject,
235        prefix,
236        tail,
237    } = &obligation.kind
238    {
239        return solve_tuple(inference, subject, prefix, tail, span);
240    }
241    let subject = match &obligation.kind {
242        Shape::Field { subject, .. }
243        | Shape::Update { subject, .. }
244        | Shape::Iterable { subject, .. }
245        | Shape::Binding { subject, .. } => subject,
246        Shape::TupleRest { .. } => unreachable!("tuple obligations dispatched above"),
247    };
248    let subject = inference.resolve(subject, span)?;
249    if matches!(subject, Type::Infer(_)) {
250        return Ok(false);
251    }
252    match &obligation.kind {
253        Shape::Field { name, result, .. } => {
254            let field = field_type(registry, inference, &subject, name, span)?;
255            inference.unify(result, &field, span, "record/tuple field")?;
256        }
257        Shape::Update { fields, .. } => {
258            registry.record_shape(&subject, span)?;
259            for (name, actual) in fields {
260                let field = field_type(registry, inference, &subject, name, span)?;
261                inference.unify(actual, &field, span, "record update field")?;
262            }
263        }
264        Shape::Iterable { item, .. } => inference.unify(
265            item,
266            &iteration::item_type(&subject, span)?,
267            span,
268            "iteration item",
269        )?,
270        Shape::Binding { pattern, .. } => {
271            iteration::irrefutable(pattern, &subject, span, registry)?
272        }
273        Shape::TupleRest { .. } => unreachable!("tuple obligations dispatched above"),
274    }
275    Ok(true)
276}

278/// A numeric projection requires a known tuple arity; names require a known nominal owner.
279fn field_type(
280    registry: &nominal::Registry,
281    inference: &Inference,
282    subject: &Type,
283    name: &str,
284    span: Span,
285) -> Checked<Type> {
286    if let Type::List(item) = subject
287        && name == "enumerate"
288    {
289        return Ok(Type::Function(
290            vec![],
291            Box::new(Type::List(Box::new(Type::Tuple(vec![
292                Type::Int,
293                *item.clone(),
294            ])))),
295        ));
296    }
297    if let Type::Tuple(fields) = subject {
298        let index = name
299            .parse::<usize>()
300            .map_err(|_| Diagnostic::new(span, "tuple field must be a numeric index"))?;
301        return fields
302            .get(index)
303            .cloned()
304            .ok_or_else(|| Diagnostic::new(span, "tuple field index is out of range"));
305    }
306    Ok(registry.project_field(subject, name, inference, span)?.1)
307}

309/// Relate an exact tuple to its prefix and exact suffix, using occurs-checked unification.
310fn solve_tuple(
311    inference: &mut Inference,
312    subject: &Type,
313    prefix: &[Type],
314    tail: &Type,
315    span: Span,
316) -> Checked<bool> {
317    let resolved = inference.resolve(subject, span)?;
318    if matches!(resolved, Type::Infer(_)) {
319        let suffix = inference.resolve(tail, span)?;
320        if matches!(suffix, Type::Infer(_)) {
321            return Ok(false);
322        }
323        let mut fields = prefix.to_vec();
324        fields.extend_from_slice(sequences::tuple_fields(&suffix, span)?);
325        inference.unify(
326            subject,
327            &sequences::tuple_type(fields),
328            span,
329            "tuple-rest shape",
330        )?;
331        return Ok(true);
332    }
333    let fields = sequences::tuple_fields(&resolved, span)?;
334    if prefix.len() > fields.len() {
335        return Err(Diagnostic::new(
336            span,
337            "tuple pattern prefix exceeds tuple arity",
338        ));
339    }
340    for (actual, expected) in prefix.iter().zip(fields) {
341        inference.unify(actual, expected, span, "tuple prefix")?;
342    }
343    inference.unify(
344        tail,
345        &sequences::tuple_type(fields[prefix.len()..].to_vec()),
346        span,
347        "tuple suffix",
348    )?;
349    Ok(true)
350}
