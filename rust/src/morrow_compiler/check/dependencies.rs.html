5#[derive(Debug)]
6pub(super) struct Group {
7    pub name: String,
8    pub group_start: usize,
9    pub span: Span,
10    pub clauses: Range<usize>,
11    pub callees: Vec<usize>,
12}
13#[derive(Debug)]
14pub(super) struct Graph {
15    pub groups: Vec<Group>,
16    pub components: Vec<Vec<usize>>,
17    pub work: usize,
18}
19struct Budget(usize);
20impl Budget {
21    /// Charge traversal, lookup, copied names and graph algorithms to one aggregate limit.
22    fn charge(&mut self, work: usize, span: Span) -> Checked<()> {
23        self.0 = self
24            .0
25            .checked_sub(work)
26            .ok_or_else(|| Diagnostic::new(span, "dependency work limit exceeded"))?;
27        Ok(())
28    }
29}

31/// Analyze validated source groups; edges refer to source-order group indices.
32#[cfg(test)]
33pub(super) fn analyze(program: &ast::Program) -> Checked<Graph> {
34    analyze_with_work(program, 0)
35}

37/// Alias expansion and dependency analysis spend one aggregate inference budget.
38pub(super) fn analyze_with_work(program: &ast::Program, work: usize) -> Checked<Graph> {
39    analyze_mode(program, work, false)
40}
41/// Direct lexical calls owned by each function exclude separately executed closure bodies.
42pub(super) fn owned_calls(program: &ast::Program) -> Checked<Graph> {
43    analyze_mode(program, 0, true)
44}
45fn analyze_mode(program: &ast::Program, work: usize, owned_calls: bool) -> Checked<Graph> {
46    preflight::check(program)?;
47    let mut budget = Budget(MAX_EXPR_COUNT * 4);
48    budget.charge(work, Span::default())?;
49    let (mut groups, names) = groups(program, &mut budget)?;
50    for group in &mut groups {
51        let mut walker = Walker {
52            owned_calls,
53            tuples: HashSet::new(),
54            names: &names,
55            locals: HashMap::new(),
56            scopes: Vec::new(),
57            pending: Vec::new(),
58            edges: HashSet::new(),
59            budget: &mut budget,
60            span: group.span,
61        };
62        for clause in &program.functions[group.clauses.clone()] {
63            walker.clause(clause)?;
64        }
65        group.callees = walker.edges.into_iter().collect();
66        sort_indices(&mut group.callees, &mut budget, group.span)?;
67    }
68    let components = components(&groups, &mut budget)?;
69    Ok(Graph {
70        groups,
71        components,
72        work: MAX_EXPR_COUNT * 4 - budget.0,
73    })
74}

76/// Reject separated or conflicting identities rather than guessing which declaration is meant.
77fn groups<'a>(
78    program: &'a ast::Program,
79    budget: &mut Budget,
80) -> Checked<(Vec<Group>, HashMap<&'a str, usize>)> {
81    let mut groups = Vec::new();
82    let mut names = HashMap::new();
83    let mut index = 0;
84    while index < program.functions.len() {
85        let first = &program.functions[index];
86        budget.charge(first.name.len() + 1, first.span)?;
87        if names.insert(first.name.as_str(), groups.len()).is_some() {
88            return Err(Diagnostic::new(
89                first.span,
90                "dependency function clauses must be adjacent",
91            ));
92        }
93        let start = index;
94        index += 1;
95        while index < program.functions.len() && program.functions[index].name == first.name {
96            budget.charge(1, program.functions[index].span)?;
97            if program.functions[index].group_start != first.group_start {
98                return Err(Diagnostic::new(
99                    program.functions[index].span,
100                    "dependency clause identity mismatch",
101                ));
102            }
103            index += 1;
104        }
105        groups.push(Group {
106            name: first.name.clone(),
107            group_start: first.group_start,
108            span: first.span,
109            clauses: start..index,
110            callees: Vec::new(),
111        });
112    }
113    Ok((groups, names))
114}

116/// Account for deterministic sorting without depending on hash iteration order.
117fn sort_indices(values: &mut [usize], budget: &mut Budget, span: Span) -> Checked<()> {
118    let levels = usize::BITS as usize - values.len().leading_zeros() as usize;
119    budget.charge(values.len().saturating_mul(levels), span)?;
120    values.sort_unstable();
121    Ok(())
122}

124enum Task<'a> {
125    Expression(&'a ast::Expr),
126    Statement(&'a ast::Stmt),
127    Pattern(&'a ast::Pattern),
128    Name(&'a str, Span),
129    Arm(&'a ast::MatchArm),
130    Enter,
131    Leave,
132}
133struct Walker<'a, 'b> {
134    owned_calls: bool,
135    tuples: HashSet<usize>,
136    names: &'b HashMap<&'a str, usize>,
137    locals: HashMap<&'a str, usize>,
138    scopes: Vec<Vec<&'a str>>,
139    pending: Vec<Task<'a>>,
140    edges: HashSet<usize>,
141    budget: &'b mut Budget,
142    span: Span,
143}

145/// Reuse lexical dependency resolution to activate an implicit prelude only for free names.
146pub(super) fn uses_trait_prelude(program: &ast::Program) -> Checked<bool> {
147    let names: HashMap<_, _> = [
148        "show",
149        "eq",
150        "neq",
151        "compare",
152        "clone",
153        "to_json",
154        "from_json",
155        "Less",
156        "Equal",
157        "Greater",
158    ]
159    .into_iter()
160    .filter(|name| {
161        !program.functions.iter().any(|f| f.name == *name)
162            && !program
163                .types
164                .iter()
165                .flat_map(|d| &d.variants)
166                .any(|v| v.name == *name)
167    })
168    .enumerate()
169    .map(|(index, name)| (name, index))
170    .collect();
171    let mut budget = Budget(MAX_EXPR_COUNT * 4);
172    let mut walker = Walker {
173        owned_calls: false,
174        tuples: HashSet::new(),
175        names: &names,
176        locals: HashMap::new(),
177        scopes: Vec::new(),
178        pending: Vec::new(),
179        edges: HashSet::new(),
180        budget: &mut budget,
181        span: Span::default(),
182    };
183    for function in &program.functions {
184        walker.span = function.span;
185        walker.clause(function)?;
186        if !walker.edges.is_empty() {
187            return Ok(true);
188        }
189    }
190    Ok(false)
191}

193/// Generate tuple instances for pairs and every arity present in source, retaining the language's bound.
194pub(super) fn trait_tuple_arities(
195    program: &ast::Program,
196) -> Checked<std::collections::BTreeSet<usize>> {
197    let names = HashMap::new();
198    let mut budget = Budget(MAX_EXPR_COUNT * 4);
199    let mut walker = Walker {
200        owned_calls: false,
201        tuples: HashSet::new(),
202        names: &names,
203        locals: HashMap::new(),
204        scopes: Vec::new(),
205        pending: Vec::new(),
206        edges: HashSet::new(),
207        budget: &mut budget,
208        span: Span::default(),
209    };
210    for function in &program.functions {
211        walker.span = function.span;
212        walker.clause(function)?;
213    }
214    let mut tuples = walker.tuples;
215    // Pairs also arise without tuple syntax, from List.zip, List.enumerate and runtime results.
216    tuples.insert(2);
217    let mut pending: Vec<_> = program
218        .functions
219        .iter()
220        .flat_map(|f| {
221            f.params
222                .iter()
223                .filter_map(|p| p.annotation.as_ref())
224                .chain(f.return_type.as_ref())
225        })
226        .chain(
227            program
228                .types
229                .iter()
230                .flat_map(|d| &d.variants)
231                .flat_map(|v| &v.fields)
232                .map(|f| &f.ty),
233        )
234        .chain(program.newtypes.iter().map(|d| &d.inner))
235        .chain(program.aliases.iter().map(|a| &a.target))
236        .collect();
237    while let Some(ty) = pending.pop() {
238        budget.charge(1, Span::default())?;
239        match ty {
240            Type::Tuple(items) => {
241                tuples.insert(items.len());
242                pending.extend(items);
243            }
244            Type::List(a) | Type::Option(a) | Type::Pid(a) => pending.push(a),
245            Type::Map(a, b) | Type::Result(a, b) | Type::ActorFunction(a, b) => {
246                pending.extend([a.as_ref(), b.as_ref()])
247            }
248            Type::Named(_, args) | Type::Union(args) => pending.extend(args),
249            Type::Function(args, result) => {
250                pending.extend(args);
251                pending.push(result);
252            }
253            _ => {}
254        }
255    }
256    Ok(tuples.into_iter().filter(|n| *n > 0).collect())
257}
258impl<'a> Walker<'a, '_> {
259    /// Clause binders are independent even though dependencies share one function identity.
260    fn clause(&mut self, function: &'a ast::Function) -> Checked<()> {
261        self.pending.push(Task::Leave);
262        self.pending.push(Task::Expression(&function.body));
263        if let Some(guard) = &function.guard {
264            self.pending.push(Task::Expression(guard));
265        }
266        for param in function.params.iter().rev() {
267            self.pending.push(Task::Pattern(&param.pattern));
268        }
269        self.pending.push(Task::Enter);
270        while let Some(task) = self.pending.pop() {
271            self.budget.charge(1, self.span)?;
272            match task {
273                Task::Expression(expr) => self.expression(expr)?,
274                Task::Statement(stmt) => self.statement(stmt),
275                Task::Pattern(pattern) => self.pattern(pattern),
276                Task::Name(name, span) => self.bind(name, span)?,
277                Task::Arm(arm) => self.arm(arm),
278                Task::Enter => self.scopes.push(Vec::new()),
279                Task::Leave => self.leave(),
280            }
281        }
282        Ok(())
283    }
284
285    /// A local dotted root takes precedence over a qualified source function name.
286    fn reference(&mut self, name: &str, span: Span) -> Checked<()> {
287        self.budget.charge(name.len() + 1, span)?;
288        let root = name.split('.').next().unwrap_or(name);
289        if !self.locals.contains_key(root) {
290            self.reference_path(name);
291        }
292        Ok(())
293    }
294
295    /// A resolved declaration edge is unaffected by canonical-prefix local bindings.
296    fn global_reference(&mut self, name: &str, span: Span) -> Checked<()> {
297        self.budget.charge(name.len() + 1, span)?;
298        self.reference_path(name);
299        Ok(())
300    }
301
302    /// A global constant may be followed by record fields in one dotted token.
303    fn reference_path(&mut self, mut name: &str) {
304        loop {
305            if let Some(index) = self.names.get(name) {
306                self.edges.insert(*index);
307                return;
308            }
309            let Some((prefix, _)) = name.rsplit_once('.') else {
310                return;
311            };
312            name = prefix;
313        }
314    }
315
316    /// Store borrowed names with shadow counts; wildcard bindings never become accessible.
317    fn bind(&mut self, name: &'a str, span: Span) -> Checked<()> {
318        self.budget.charge(name.len() + 1, span)?;
319        if name != "_" {
320            *self.locals.entry(name).or_default() += 1;
321            self.scopes
322                .last_mut()
323                .expect("binding has lexical scope")
324                .push(name);
325        }
326        Ok(())
327    }
328
329    /// Restore precisely the bindings introduced by this lexical scope.
330    fn leave(&mut self) {
331        for name in self.scopes.pop().expect("balanced lexical scope") {
332            let count = self.locals.get_mut(name).expect("registered local binding");
333            *count -= 1;
334            if *count == 0 {
335                self.locals.remove(name);
336            }
337        }
338    }
339
340    /// Queue a child whose declarations cannot escape into a sibling expression.
341    fn scoped(&mut self, expr: &'a ast::Expr) {
342        self.pending.push(Task::Leave);
343        self.pending.push(Task::Expression(expr));
344        self.pending.push(Task::Enter);
345    }
346
347    /// Initializers and failure branches run before the successful binding exists.
348    fn statement(&mut self, stmt: &'a ast::Stmt) {
349        match stmt {
350            ast::Stmt::Let {
351                name, value, span, ..
352            } => {
353                self.pending.push(Task::Name(name, *span));
354                self.pending.push(Task::Expression(value));
355            }
356            ast::Stmt::LetElse {
357                pattern,
358                value,
359                else_branch,
360                ..
361            } => {
362                self.pending.push(Task::Pattern(pattern));
363                self.scoped(else_branch);
364                self.pending.push(Task::Expression(value));
365            }
366            ast::Stmt::LetPattern { pattern, value, .. } => {
367                self.pending.push(Task::Pattern(pattern));
368                self.pending.push(Task::Expression(value));
369            }
370            ast::Stmt::Expr(value) => self.pending.push(Task::Expression(value)),
371        }
372    }
373
374    /// Bind every nested payload/rest name without interpreting constructors as value calls.
375    fn pattern(&mut self, pattern: &'a ast::Pattern) {
376        match &pattern.kind {
377            ast::PatternKind::Typed { pattern, .. } => self.pending.push(Task::Pattern(pattern)),
378            ast::PatternKind::Bind(name) => self.pending.push(Task::Name(name, pattern.span)),
379            ast::PatternKind::Constructor {
380                binding: Some(name),
381                ..
382            } => {
383                self.pending.push(Task::Name(name, pattern.span));
384            }
385            ast::PatternKind::Tuple(fields) | ast::PatternKind::NamedConstructor { fields, .. } => {
386                self.pending.extend(fields.iter().rev().map(Task::Pattern));
387            }
388            ast::PatternKind::List { prefix, rest } => {
389                if let Some(rest) = rest {
390                    self.pending.push(Task::Pattern(rest));
391                }
392                self.pending.extend(prefix.iter().rev().map(Task::Pattern));
393            }
394            ast::PatternKind::TupleRest { prefix, rest } => {
395                self.pending.push(Task::Pattern(rest));
396                self.pending.extend(prefix.iter().rev().map(Task::Pattern));
397            }
398            ast::PatternKind::Wildcard
399            | ast::PatternKind::Int(_)
400            | ast::PatternKind::Bool(_)
401            | ast::PatternKind::String(_)
402            | ast::PatternKind::Constructor { binding: None, .. } => {}
403        }
404    }
405
406    /// A pattern binds before its guard and body, and ends before the next arm.
407    fn arm(&mut self, arm: &'a ast::MatchArm) {
408        self.pending.push(Task::Leave);
409        self.pending.push(Task::Expression(&arm.body));
410        if let Some(guard) = &arm.guard {
411            self.pending.push(Task::Expression(guard));
412        }
413        self.pending.push(Task::Pattern(&arm.pattern));
414        self.pending.push(Task::Enter);
415    }
416
417    /// Handle lexical boundaries explicitly; the remaining expressions only carry references.
418    fn expression(&mut self, expr: &'a ast::Expr) -> Checked<()> {
419        if let ast::ExprKind::Tuple(items) = &expr.kind {
420            self.tuples.insert(items.len());
421        }
422        match &expr.kind {
423            ast::ExprKind::Block(stmts) => {
424                self.pending.push(Task::Leave);
425                self.pending.extend(stmts.iter().rev().map(Task::Statement));
426                self.pending.push(Task::Enter);
427            }
428            ast::ExprKind::Lambda { .. } if self.owned_calls => {}
429            ast::ExprKind::Lambda { params, body } => {
430                self.pending.push(Task::Leave);
431                self.pending.push(Task::Expression(body));
432                self.pending
433                    .extend(params.iter().rev().map(|p| Task::Name(&p.name, p.span)));
434                self.pending.push(Task::Enter);
435            }
436            ast::ExprKind::For {
437                pattern,
438                iterable,
439                body,
440            } => {
441                self.pending.push(Task::Leave);
442                self.pending.push(Task::Expression(body));
443                self.pending.push(Task::Pattern(pattern));
444                self.pending.push(Task::Enter);
445                self.pending.push(Task::Expression(iterable));
446            }
447            ast::ExprKind::Receive { arms, timeout, .. } => {
448                if let Some((duration, body)) = timeout {
449                    self.scoped(body);
450                    self.pending.push(Task::Expression(duration));
451                }
452                self.pending.extend(arms.iter().rev().map(Task::Arm));
453            }
454            ast::ExprKind::Match { value, arms } => {
455                self.pending.extend(arms.iter().rev().map(Task::Arm));
456                self.pending.push(Task::Expression(value));
457            }
458            ast::ExprKind::With {
459                bindings,
460                body,
461                arms,
462            } => self.with(bindings, body, arms.as_deref()),
463            ast::ExprKind::If {
464                condition,
465                then_branch,
466                else_branch,
467            } => {
468                if let Some(branch) = else_branch {
469                    self.scoped(branch);
470                }
471                self.scoped(then_branch);
472                self.pending.push(Task::Expression(condition));
473            }
474            ast::ExprKind::ConditionMatch(arms) => {
475                for arm in arms.iter().rev() {
476                    self.scoped(&arm.body);
477                    if let Some(condition) = &arm.condition {
478                        self.pending.push(Task::Expression(condition));
479                    }
480                }
481            }
482            ast::ExprKind::Defer(value) => self.scoped(value),
483            _ => self.plain(expr)?,
484        }
485        Ok(())
486    }
487
488    /// Successful with-bindings are sequential; each error arm sees the original outer scope.
489    fn with(
490        &mut self,
491        bindings: &'a [ast::WithBinding],
492        body: &'a ast::Expr,
493        arms: Option<&'a [ast::MatchArm]>,
494    ) {
495        if let Some(arms) = arms {
496            self.pending.extend(arms.iter().rev().map(Task::Arm));
497        }
498        self.pending.push(Task::Leave);
499        self.pending.push(Task::Expression(body));
500        for binding in bindings.iter().rev() {
501            self.pending.push(Task::Pattern(&binding.pattern));
502            self.pending.push(Task::Expression(&binding.value));
503        }
504        self.pending.push(Task::Enter);
505    }
506
507    /// Queue map key/value expressions in source order without changing dependency scopes.
508    fn map_entries(&mut self, entries: &'a [(ast::Expr, ast::Expr)]) {
509        for (key, value) in entries.iter().rev() {
510            self.pending.push(Task::Expression(value));
511            self.pending.push(Task::Expression(key));
512        }
513    }
514
515    /// Queue source-written argument values; labels introduce no dependency edges.
516    fn arguments(&mut self, args: &'a [ast::Argument]) {
517        self.pending
518            .extend(args.iter().rev().map(|arg| Task::Expression(&arg.value)));
519    }
520
521    /// Queue explicit global edges independently of canonical-prefix local bindings.
522    fn global(&mut self, expr: &'a ast::Expr) -> Checked<bool> {
523        use ast::ExprKind::*;
524        match &expr.kind {
525            GlobalName { resolved, .. } => {
526                if !self.owned_calls {
527                    self.global_reference(resolved, expr.span)?;
528                }
529            }
530            GlobalCall { resolved, args, .. } => {
531                self.global_reference(resolved, expr.span)?;
532                self.arguments(args);
533            }
534            GlobalPipe {
535                value,
536                resolved,
537                args,
538                ..
539            } => {
540                self.global_reference(resolved, expr.span)?;
541                self.arguments(args);
542                self.pending.push(Task::Expression(value));
543            }
544            _ => return Ok(false),
545        }
546        Ok(true)
547    }
548
549    /// Queue value children exhaustively so new syntax cannot silently lose dependencies.
550    fn string_parts(&mut self, parts: &'a [ast::StringPart]) {
551        for part in parts.iter().rev() {
552            if let ast::StringPart::Value(value) = part {
553                self.pending.push(Task::Expression(value));
554            }
555        }
556    }
557
558    /// Queue ordinary expression children in source evaluation order.
559    fn plain(&mut self, expr: &'a ast::Expr) -> Checked<()> {
560        use ast::ExprKind::*;
561        if self.global(expr)? {
562            return Ok(());
563        }
564        match &expr.kind {
565            TypeTarget(_) => {}
566            Name(name) => {
567                if !self.owned_calls {
568                    self.reference(name, expr.span)?;
569                }
570            }
571            Call { name, args } => {
572                self.reference(name, expr.span)?;
573                self.arguments(args);
574            }
575            Pipe {
576                value, name, args, ..
577            } => {
578                self.reference(name, expr.span)?;
579                self.arguments(args);
580                self.pending.push(Task::Expression(value));
581            }
582            Apply { callee, args } => {
583                self.arguments(args);
584                if self.owned_calls {
585                    match &callee.kind {
586                        Name(name) => self.reference(name, callee.span)?,
587                        GlobalName { resolved, .. } => {
588                            self.global_reference(resolved, callee.span)?
589                        }
590                        _ => self.pending.push(Task::Expression(callee)),
591                    }
592                } else {
593                    self.pending.push(Task::Expression(callee));
594                }
595            }
596            ForeignCall { args: values, .. } | Tuple(values) | List(values) => self
597                .pending
598                .extend(values.iter().rev().map(Task::Expression)),
599            Map(entries) => self.map_entries(entries),
600            RecordUpdate { value, fields } => {
601                self.pending
602                    .extend(fields.iter().rev().map(|f| Task::Expression(&f.value)));
603                self.pending.push(Task::Expression(value));
604            }
605            Interpolate(parts) | MultilineString(parts) => self.string_parts(parts),
606            Field { value, .. } | Unary { value, .. } | Try(value) | Return(value) => {
607                self.pending.push(Task::Expression(value));
608            }
609            Binary { left, right, .. }
610            | PostfixIf {
611                condition: left,
612                value: right,
613            }
614            | Range {
615                start: left,
616                end: right,
617                ..
618            } => {
619                self.pending.push(Task::Expression(right));
620                self.pending.push(Task::Expression(left));
621            }
622            GlobalName { .. } | GlobalCall { .. } | GlobalPipe { .. } => {
623                unreachable!("global handled above")
624            }
625            Int(_) | Float(_) | Bool(_) | String(_) | Unit | Break | Continue => {}
626            Block(_)
627            | Lambda { .. }
628            | For { .. }
629            | Receive { .. }
630            | Match { .. }
631            | With { .. }
632            | If { .. }
633            | ConditionMatch(_)
634            | Defer(_) => unreachable!("scoped expression handled first"),
635        }
636        Ok(())
637    }
638}

640/// Kosaraju traversal uses explicit DFS stacks, including for 4096-node recursive components.
641fn components(groups: &[Group], budget: &mut Budget) -> Checked<Vec<Vec<usize>>> {
642    let finish = finish_order(groups, budget)?;
643    let mut incoming = vec![Vec::new(); groups.len()];
644    for (caller, group) in groups.iter().enumerate() {
645        for &callee in &group.callees {
646            budget.charge(1, group.span)?;
647            incoming[callee].push(caller);
648        }
649    }
650    let mut seen = vec![false; groups.len()];
651    let mut components = Vec::new();
652    for start in finish.into_iter().rev() {
653        if seen[start] {
654            continue;
655        }
656        let mut component = Vec::new();
657        let mut pending = vec![start];
658        seen[start] = true;
659        while let Some(node) = pending.pop() {
660            budget.charge(1, groups[node].span)?;
661            component.push(node);
662            for &caller in &incoming[node] {
663                budget.charge(1, groups[node].span)?;
664                if !seen[caller] {
665                    seen[caller] = true;
666                    pending.push(caller);
667                }
668            }
669        }
670        sort_indices(&mut component, budget, groups[start].span)?;
671        components.push(component);
672    }
673    components.reverse();
674    Ok(components)
675}

677/// Obtain forward DFS finish order without recursion on the host stack.
678fn finish_order(groups: &[Group], budget: &mut Budget) -> Checked<Vec<usize>> {
679    let mut seen = vec![false; groups.len()];
680    let mut finish = Vec::new();
681    for start in 0..groups.len() {
682        if seen[start] {
683            continue;
684        }
685        seen[start] = true;
686        let mut pending = vec![(start, 0)];
687        while let Some((node, cursor)) = pending.last_mut() {
688            budget.charge(1, groups[*node].span)?;
689            if let Some(&callee) = groups[*node].callees.get(*cursor) {
690                *cursor += 1;
691                if !seen[callee] {
692                    seen[callee] = true;
693                    pending.push((callee, 0));
694                }
695            } else {
696                finish.push(*node);
697                pending.pop();
698            }
699        }
700    }
701    Ok(finish)
702}
