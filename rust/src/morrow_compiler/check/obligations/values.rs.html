1//! Shared value regions and conditional coverage keep aliases correlated across source paths.

4#[derive(Clone, Debug, PartialEq, Eq)]
5pub(super) enum Key {
6    Int(i64),
7    Bool(bool),
8    String(String),
9}
10#[derive(Clone, Debug)]
11pub(super) struct Value {
12    pub node: Rc<Node>,
13    pub complete: bool,
14}
15#[derive(Debug)]
16pub(super) struct Node {
17    pub id: usize,
18    pub kind: Region,
19}
20#[derive(Debug)]
21pub(super) enum Region {
22    Empty,
23    EditorBorrow,
24    Symbolic,
25    UnknownCallable,
26    Callable {
27        function: usize,
28        captures: Vec<Value>,
29    },
30    Scalar(Key),
31    Boolean(Predicate),
32    Product(Vec<Value>),
33    Sum {
34        origin: Option<usize>,
35        tag: Option<usize>,
36        guards: Vec<Predicate>,
37        variants: Vec<Vec<Value>>,
38    },
39    RecursiveCut {
40        layout: usize,
41        origin: Option<usize>,
42        retained: Vec<Value>,
43    },
44    Nominal {
45        layout: usize,
46        expanded: RefCell<Option<Value>>,
47    },
48    Union {
49        members: Vec<Type>,
50        value: Value,
51    },
52    List {
53        nonempty: Predicate,
54        items: Vec<Value>,
55        exact: bool,
56    },
57    Map {
58        nonempty: Predicate,
59        entries: Vec<(Option<Key>, Value)>,
60        exact: bool,
61    },
62    Choice(Vec<(Predicate, Value)>),
63}
64impl Value {
65    /// A partial projection never upgrades the universal coverage of its underlying family.
66    pub(super) fn partial(&self) -> Self {
67        Self {
68            node: self.node.clone(),
69            complete: false,
70        }
71    }

73impl Engine<'_> {
74    /// Expand concrete shapes with independent duties only on active payload paths.
75    pub(super) fn fresh(
76        &mut self,
77        ty: &Type,
78        input: Option<usize>,
79        span: Span,
80        depth: usize,
81    ) -> Checked<Value> {
82        self.charge(1, span)?;
83        if depth >= DEPTH_LIMIT {
84            return Err(Diagnostic::new(
85                span,
86                "Result obligation type depth limit exceeded",
87            ));
88        }
89        let kind = match ty {
90            Type::Function(..) => {
91                self.input_callable |= input.is_some();
92                Region::UnknownCallable
93            }
94            Type::Bool => Region::Boolean(self.predicates.variable(&mut self.work, span)?),
95            Type::Result(ok, err) => {
96                return self.fresh_sum(vec![vec![ok], vec![err]], input, true, span, depth);
97            }
98            Type::Option(inner) => {
99                return self.fresh_sum(vec![vec![inner], vec![]], input, false, span, depth);
100            }
101            Type::Tuple(types) => {
102                Region::Product(self.fresh_values(types.iter(), input, span, depth + 1)?)
103            }
104            Type::Union(types) => {
105                self.charge(types.len(), span)?;
106                let value = self.fresh_sum(
107                    types.iter().map(|ty| vec![ty]).collect(),
108                    input,
109                    false,
110                    span,
111                    depth,
112                )?;
113                return self.union_value(types, value, span);
114            }
115            Type::List(inner) => return self.fresh_list(inner, input, span, depth + 1),
116            Type::Map(_, inner) => return self.fresh_map(inner, input, span, depth + 1),
117            Type::Named(..) => return self.fresh_nominal(ty, input, span, depth),
118            Type::Generic(_) if self.mode != Mode::Concrete => Region::Symbolic,
119            Type::Infer(_) | Type::Generic(_) | Type::Never => return self.unsupported(span),
120            _ => Region::Empty,
121        };
122        self.node(kind, span)
123    }
124    /// Product width is charged before allocating child regions.
125    fn fresh_values<'b>(
126        &mut self,
127        types: impl ExactSizeIterator<Item = &'b Type>,
128        input: Option<usize>,
129        span: Span,
130        depth: usize,
131    ) -> Checked<Vec<Value>> {
132        self.charge(types.len(), span)?;
133        types.map(|ty| self.fresh(ty, input, span, depth)).collect()
134    }
135    /// Sum alternatives are exclusive and exhaustive; nested origins inherit their exact guard.
136    pub(super) fn fresh_sum(
137        &mut self,
138        types: Vec<Vec<&Type>>,
139        input: Option<usize>,
140        result: bool,
141        span: Span,
142        depth: usize,
143    ) -> Checked<Value> {
144        self.charge(types.len(), span)?;
145        let origin = if result {
146            Some(self.origin(input, span)?)
147        } else {
148            None
149        };
150        let guards = self.alternatives(types.len(), span)?;
151        let parent = self.path;
152        let mut variants = Vec::new();
153        for (guard, fields) in guards.iter().zip(types) {
154            self.path = self.predicates.and(parent, *guard, &mut self.work, span)?;
155            variants.push(self.fresh_values(fields.into_iter(), input, span, depth + 1)?);
156        }
157        self.path = parent;
158        self.node(
159            Region::Sum {
160                origin,
161                tag: None,
162                guards,
163                variants,
164            },
165            span,
166        )
167    }
168    /// Concrete nominal payloads keep storage identity separate from semantic error layers.
169    fn fresh_nominal(
170        &mut self,
171        ty: &Type,
172        input: Option<usize>,
173        span: Span,
174        depth: usize,
175    ) -> Checked<Value> {
176        if let Some(layout) = self.lazy_layout(ty, span)? {
177            return self.node(
178                Region::Nominal {
179                    layout,
180                    expanded: RefCell::new(None),
181                },
182                span,
183            );
184        }
185        let index = gate::layout_index(self.program, ty, &mut self.work, span)?;
186        self.charge(self.active_nominals.len(), span)?;
187        if let Some((_, anchor)) = self.active_nominals.iter().find(|(id, _)| *id == index) {
188            return self.recursive_cut(index, *anchor, input, span);
189        }
190        let anchor = self.node(Region::Empty, span)?.node.id;
191        self.active_nominals.push((index, anchor));
192        let value = self.fresh_nominal_fields(index, input, span, depth);
193        self.active_nominals.pop();
194        if let Ok(value) = &value {
195            self.nominal_roots.insert(value.node.id, anchor);
196            self.certify_ancestors(value.node.id, None, span)?;
197        }
198        value
199    }
200    /// Expand stored fields once; repeated layout edges become finite, separately accountable cuts.
201    fn fresh_nominal_fields(
202        &mut self,
203        index: usize,
204        input: Option<usize>,
205        span: Span,
206        depth: usize,
207    ) -> Checked<Value> {
208        let layout = &self.program.types[index];
209        if layout.storage == ir::LayoutStorage::Unboxed {
210            let inner = layout
211                .variants
212                .first()
213                .and_then(|v| v.first())
214                .ok_or_else(|| Diagnostic::new(span, "invalid newtype obligation layout"))?;
215            return self.fresh(inner, input, span, depth + 1);
216        }
217        let count = layout
218            .variants
219            .iter()
220            .try_fold(layout.variants.len(), |n, v| n.checked_add(v.len()))
221            .ok_or_else(|| Diagnostic::new(span, "Result obligation layout size limit exceeded"))?;
222        self.charge(count, span)?;
223        let types = layout
224            .variants
225            .iter()
226            .map(|fields| fields.iter().collect())
227            .collect();
228        self.fresh_sum(types, input, false, span, depth)
229    }
230    /// Allocate bounded mutually exclusive guards without exponential path enumeration.
231    pub(super) fn alternatives(&mut self, count: usize, span: Span) -> Checked<Vec<Predicate>> {
232        self.charge(count, span)?;
233        let mut remaining = Predicate::TRUE;
234        let mut guards = Vec::new();
235        for i in 0..count {
236            if i + 1 == count {
237                guards.push(remaining);
238                break;
239            }
240            let variable = self.predicates.variable(&mut self.work, span)?;
241            guards.push(
242                self.predicates
243                    .and(remaining, variable, &mut self.work, span)?,
244            );
245            let inverse = self.predicates.not(variable, &mut self.work, span)?;
246            remaining = self
247                .predicates
248                .and(remaining, inverse, &mut self.work, span)?;
249        }
250        Ok(guards)
251    }
252    /// Coverage maps each original duty to the paths on which this value definitely carries it.
253    pub(super) fn origins_of(
254        &mut self,
255        value: &Value,
256        outer: bool,
257        span: Span,
258    ) -> Checked<BTreeMap<usize, Predicate>> {
259        self.coverage(value, outer, span, 0, &mut HashMap::new())
260    }
261    /// Cache shared regions while preserving predicates; a MAY-alias union is never a guarantee.
262    fn coverage(
263        &mut self,
264        value: &Value,
265        outer: bool,
266        span: Span,
267        depth: usize,
268        cache: &mut HashMap<(usize, bool), BTreeMap<usize, Predicate>>,
269    ) -> Checked<BTreeMap<usize, Predicate>> {
270        self.charge(1, span)?;
271        if depth >= DEPTH_LIMIT {
272            return Err(Diagnostic::new(
273                span,
274                "Result obligation region depth limit exceeded",
275            ));
276        }
277        if !value.complete {
278            return Ok(BTreeMap::new());
279        }
280        let key = (value.node.id, outer);
281        if let Some(found) = cache.get(&key) {
282            self.charge(found.len(), span)?;
283            return Ok(found.clone());
284        }
285        let mut result = BTreeMap::new();
286        if let Region::Sum {
287            origin: Some(id), ..
288        } = value.node.kind
289        {
290            result.insert(id, Predicate::TRUE);
291        }
292        match &value.node.kind {
293            Region::RecursiveCut {
294                origin, retained, ..
295            } if !outer => {
296                if let Some(id) = origin {
297                    result.insert(*id, Predicate::TRUE);
298                }
299                for child in retained {
300                    let found = self.coverage(child, false, span, depth + 1, cache)?;
301                    self.merge_coverage(&mut result, found, Predicate::TRUE, span)?;
302                }
303            }
304            Region::Nominal { expanded, .. } => {
305                if let Some(value) = expanded.borrow().as_ref() {
306                    let found = self.coverage(value, outer, span, depth + 1, cache)?;
307                    self.merge_coverage(&mut result, found, Predicate::TRUE, span)?;
308                }
309            }
310            Region::Choice(choices) => {
311                for (guard, child) in choices {
312                    let found = self.coverage(child, outer, span, depth + 1, cache)?;
313                    self.merge_coverage(&mut result, found, *guard, span)?;
314                }
315            }
316            Region::Sum {
317                guards, variants, ..
318            } if !outer => {
319                for (guard, fields) in guards.iter().zip(variants) {
320                    for child in fields {
321                        let found = self.coverage(child, false, span, depth + 1, cache)?;
322                        self.merge_coverage(&mut result, found, *guard, span)?;
323                    }
324                }
325            }
326            _ if !outer => {
327                for child in children(&value.node.kind) {
328                    let found = self.coverage(child, false, span, depth + 1, cache)?;
329                    self.merge_coverage(&mut result, found, Predicate::TRUE, span)?;
330                }
331            }
332            _ => {}
333        }
334        self.charge(result.len(), span)?;
335        cache.insert(key, result.clone());
336        Ok(result)
337    }
338    /// Join guarded provenance for one origin while charging every set and predicate operation.
339    fn merge_coverage(
340        &mut self,
341        target: &mut BTreeMap<usize, Predicate>,
342        source: BTreeMap<usize, Predicate>,
343        guard: Predicate,
344        span: Span,
345    ) -> Checked<()> {
346        self.charge(source.len(), span)?;
347        for (id, path) in source {
348            let path = self.predicates.and(path, guard, &mut self.work, span)?;
349            let prior = target.get(&id).copied().unwrap_or(Predicate::FALSE);
350            target.insert(id, self.predicates.or(prior, path, &mut self.work, span)?);
351        }
352        Ok(())
353    }
354}
355/// Borrow unconditionally included product/container children without allocating a work queue.
356pub(super) fn children(region: &Region) -> Box<dyn Iterator<Item = &Value> + '_> {
357    match region {
358        Region::Product(values)
359        | Region::List { items: values, .. }
360        | Region::RecursiveCut {
361            retained: values, ..
362        } => Box::new(values.iter()),
363        Region::Map { entries, .. } => Box::new(entries.iter().map(|(_, value)| value)),
364        Region::Union { value, .. } => Box::new(std::iter::once(value)),
365        _ => Box::new(std::iter::empty()),
366    }
