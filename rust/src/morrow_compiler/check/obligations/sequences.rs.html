1//! Sequence patterns partition retained identities; a prefix never acknowledges its suffix.
2use super::*;
3impl Engine<'_> {
4    /// Known layouts split exactly; unknown-length prefixes require a separate family partition proof.
5    pub(super) fn sequence_pattern(
6        &mut self,
7        pattern: &ir::Pattern,
8        value: &Value,
9        span: Span,
10        depth: usize,
11    ) -> Checked<Predicate> {
12        self.charge(1, span)?;
13        if depth >= DEPTH_LIMIT {
14            return self.unsupported(span);
15        }
16        if let Region::Choice(choices) = &value.node.kind {
17            return self.sequence_choices(pattern, choices, value.complete, span, depth + 1);
18        }
19        match (pattern, &value.node.kind) {
20            (
21                ir::Pattern::List { prefix, rest },
22                Region::List {
23                    items, exact: true, ..
24                },
25            ) => {
26                if items.len() < prefix.len() || (rest.is_none() && items.len() != prefix.len()) {
27                    return Ok(Predicate::FALSE);
28                }
29                self.sequence_fields(
30                    (prefix, rest.as_deref()),
31                    items,
32                    (value.complete, true),
33                    span,
34                    depth,
35                )
36            }
37            (ir::Pattern::List { prefix, rest }, Region::List { nonempty, .. })
38                if prefix.is_empty() =>
39            {
40                if let Some(rest) = rest {
41                    self.pattern(rest, value, span, depth)
42                } else {
43                    self.predicates.not(*nonempty, &mut self.work, span)
44                }
45            }
46            (ir::Pattern::List { prefix, rest }, Region::List { .. }) => {
47                self.dynamic_sequence(prefix, rest.as_deref(), value, span, depth + 1)
48            }
49            (ir::Pattern::TupleRest { prefix, rest }, Region::Product(items)) => {
50                if items.len() < prefix.len() {
51                    return self.unsupported(span);
52                }
53                self.sequence_fields(
54                    (prefix, Some(rest)),
55                    items,
56                    (value.complete, false),
57                    span,
58                    depth,
59                )
60            }
61            _ => self.unsupported(span),
62        }
63    }
64    /// Partition fixed storage while propagating partial-family status into every derived view.
65    fn sequence_fields(
66        &mut self,
67        pattern: (&[ir::Pattern], Option<&ir::Pattern>),
68        items: &[Value],
69        mode: (bool, bool),
70        span: Span,
71        depth: usize,
72    ) -> Checked<Predicate> {
73        let (prefix, rest) = pattern;
74        let (complete, list) = mode;
75        self.charge(items.len(), span)?;
76        let mut items: Vec<_> = items
77            .iter()
78            .map(|v| if complete { v.clone() } else { v.partial() })
79            .collect();
80        let suffix = items.split_off(prefix.len());
81        let mut selected = self.pattern_fields(prefix, &items, Predicate::TRUE, span, depth)?;
82        if let Some(rest) = rest {
83            let kind = if list {
84                let nonempty = if suffix.is_empty() {
85                    Predicate::FALSE
86                } else {
87                    Predicate::TRUE
88                };
89                Region::List {
90                    items: suffix,
91                    exact: true,
92                    nonempty,
93                }
94            } else if suffix.is_empty() {
95                Region::Empty
96            } else {
97                Region::Product(suffix)
98            };
99            let value = self.node(kind, span)?;
100            let parent = self.path;
101            self.path = self
102                .predicates
103                .and(parent, selected, &mut self.work, span)?;
104            let matched = self.pattern(rest, &value, span, depth)?;
105            self.path = parent;
106            selected = self
107                .predicates
108                .and(selected, matched, &mut self.work, span)?;
109        }
110        Ok(selected)
111    }
112    /// A branch-selected sequence binds each name to its guarded value, never a union of aliases.
113    fn sequence_choices(
114        &mut self,
115        pattern: &ir::Pattern,
116        choices: &[(Predicate, Value)],
117        complete: bool,
118        span: Span,
119        depth: usize,
120    ) -> Checked<Predicate> {
121        let ids = self.sequence_bindings(pattern, span)?;
122        let parent = self.path;
123        let mut selected = Predicate::FALSE;
124        let mut bindings: BTreeMap<usize, Vec<(Predicate, Value)>> = BTreeMap::new();
125        self.charge(choices.len(), span)?;
126        for (guard, value) in choices {
127            self.path = self.predicates.and(parent, *guard, &mut self.work, span)?;
128            if self.path == Predicate::FALSE {
129                continue;
130            }
131            let value = if complete {
132                value.clone()
133            } else {
134                value.partial()
135            };
136            let matched = self.sequence_pattern(pattern, &value, span, depth)?;
137            let matched = self.predicates.and(*guard, matched, &mut self.work, span)?;
138            selected = self
139                .predicates
140                .or(selected, matched, &mut self.work, span)?;
141            if matched == Predicate::FALSE {
142                continue;
143            }
144            for id in &ids {
145                self.charge(1, span)?;
146                let value = self
147                    .locals
148                    .get(id)
149                    .cloned()
150                    .ok_or_else(|| Diagnostic::new(span, "missing Result sequence binding"))?;
151                bindings.entry(*id).or_default().push((matched, value));
152            }
153        }
154        self.path = parent;
155        for (id, values) in bindings {
156            let value = self.node(Region::Choice(values), span)?;
157            self.locals.insert(id, value);
158        }
159        Ok(selected)
160    }
161    /// Collect only pattern-owned local identities using the already bounded typed pattern graph.
162    fn sequence_bindings(&mut self, pattern: &ir::Pattern, span: Span) -> Checked<Vec<usize>> {
163        let mut pending = vec![pattern];
164        let mut ids = Vec::new();
165        while let Some(pattern) = pending.pop() {
166            self.charge(1, span)?;
167            match pattern {
168                ir::Pattern::Bind(id)
169                | ir::Pattern::Constructor {
170                    binding: Some(id), ..
171                } => ids.push(id.0),
172                ir::Pattern::UnionSelect {
173                    binding: Some(param),
174                    ..
175                } => ids.push(param.id.0),
176                ir::Pattern::Newtype(inner) => pending.push(inner),
177                ir::Pattern::Tuple(fields) | ir::Pattern::Variant { fields, .. } => {
178                    self.charge(fields.len(), span)?;
179                    pending.extend(fields);
180                }
181                ir::Pattern::List { prefix, rest } => {
182                    self.charge(prefix.len().saturating_add(1), span)?;
183                    pending.extend(prefix);
184                    if let Some(rest) = rest {
185                        pending.push(rest);
186                    }
187                }
188                ir::Pattern::TupleRest { prefix, rest } => {
189                    self.charge(prefix.len().saturating_add(1), span)?;
190                    pending.extend(prefix);
191                    pending.push(rest);
192                }
193                _ => {}
194            }
195        }
196        Ok(ids)
197    }
