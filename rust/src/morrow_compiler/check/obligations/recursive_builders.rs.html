1//! Finite recursive builders preserve complete input duties without promising to handle them.
2use super::*;
3/// Nominal storage is a finite graph; arbitrary fresh data contains no hidden callable contracts.
4pub(super) fn closed_data(
5    program: &ir::Program,
6    ty: &Type,
7    results: bool,
8    work: &mut usize,
9    span: Span,
10) -> Checked<bool> {
11    closed_data_mode(program, ty, results, false, work, span)
12}
13/// Template payloads stay symbolic; concrete specializations must prove their actual duties again.
14pub(super) fn template_output(
15    program: &ir::Program,
16    ty: &Type,
17    work: &mut usize,
18    span: Span,
19) -> Checked<bool> {
20    closed_data_mode(program, ty, true, true, work, span)
21}
22fn closed_data_mode(
23    program: &ir::Program,
24    ty: &Type,
25    results: bool,
26    generics: bool,
27    work: &mut usize,
28    span: Span,
29) -> Checked<bool> {
30    let mut pending = vec![ty];
31    let mut layouts = HashSet::new();
32    while let Some(ty) = pending.pop() {
33        gate::type_cost(ty, work, span)?;
34        match ty {
35            Type::Named(..) => {
36                let index = gate::layout_index(program, ty, work, span)?;
37                if !layouts.insert(index) {
38                    continue;
39                }
40                for fields in &program.types[index].variants {
41                    charge(work, fields.len(), span)?;
42                    pending.extend(fields);
43                }
44            }
45            Type::List(inner) | Type::Option(inner) | Type::Pid(inner) => pending.push(inner),
46            Type::Result(ok, error) if results => {
47                charge(work, 2, span)?;
48                pending.extend([ok.as_ref(), error.as_ref()]);
49            }
50            Type::Map(key, value) => {
51                charge(work, 2, span)?;
52                pending.extend([key.as_ref(), value.as_ref()]);
53            }
54            Type::Tuple(fields) | Type::Union(fields) => {
55                charge(work, fields.len(), span)?;
56                pending.extend(fields);
57            }
58            Type::Generic(_) if generics => {}
59            Type::Unit
60            | Type::Int
61            | Type::Float
62            | Type::Bool
63            | Type::String
64            | Type::Range
65            | Type::Native(_) => {}
66            _ => return Ok(false),
67        }
68    }
69    Ok(true)
70}
71/// A backedge preserves the full input family and may add independently accountable fresh values.
72pub(super) fn apply(
73    engine: &mut Engine<'_>,
74    function: &ir::Function,
75    indices: &[usize],
76    args: &[Value],
77    span: Span,
78) -> Checked<Value> {
79    engine.charge(indices.len(), span)?;
80    let retained = indices
81        .iter()
82        .map(|index| {
83            args.get(*index)
84                .cloned()
85                .ok_or_else(|| Diagnostic::new(span, "missing recursive builder input"))
86        })
87        .collect::<Checked<Vec<_>>>()?;
88    if matches!(function.return_type, Type::Named(..)) {
89        let layout = gate::layout_index(
90            engine.program,
91            &function.return_type,
92            &mut engine.work,
93            span,
94        )?;
95        let origin = engine.origin(None, span)?;
96        engine.origins[origin].aggregate = true;
97        return engine.node(
98            Region::RecursiveCut {
99                layout,
100                origin: Some(origin),
101                retained,
102            },
103            span,
104        );
105    }
106    list_output(engine, function, indices, &retained, span)
107}
108/// Preserve each actual element shape and only those nonempty facts justified by returned duties.
109fn list_output(
110    engine: &mut Engine<'_>,
111    function: &ir::Function,
112    indices: &[usize],
113    retained: &[Value],
114    span: Span,
115) -> Checked<Value> {
116    let fresh = engine.fresh(&function.return_type, None, span, 0)?;
117    let Region::List {
118        items, nonempty, ..
119    } = &fresh.node.kind
120    else {
121        return engine.unsupported(span);
122    };
123    engine.charge(items.len().saturating_add(retained.len()), span)?;
124    let mut output = items.clone();
125    let mut nonempty = *nonempty;
126    for (index, input) in indices.iter().zip(retained) {
127        engine.charge(index.saturating_add(1), span)?;
128        let ty = &function
129            .params
130            .iter()
131            .chain(&function.captures)
132            .nth(*index)
133            .ok_or_else(|| Diagnostic::new(span, "missing recursive builder parameter"))?
134            .ty;
135        let (item, present) = if ty == &function.return_type {
136            (
137                substitute::Substitution::family(engine, input, false, span, 0)?,
138                duty_present(engine, input, span)?,
139            )
140        } else {
141            (input.clone(), duty_present(engine, input, span)?)
142        };
143        output.push(item);
144        nonempty = engine
145            .predicates
146            .or(nonempty, present, &mut engine.work, span)?;
147    }
148    engine.node(
149        Region::List {
150            items: output,
151            exact: false,
152            nonempty,
153        },
154        span,
155    )
156}
157/// Only present retained duties imply a nonempty output; an absent Option may validly produce [].
158fn duty_present(engine: &mut Engine<'_>, input: &Value, span: Span) -> Checked<Predicate> {
159    let mut present = Predicate::FALSE;
160    for (origin, selected) in engine.origins_of(input, false, span)? {
161        engine.charge(1, span)?;
162        // A recursive cut requires complete treatment even when its concrete subtree is Empty.
163        // Its aggregate origin therefore says nothing about actual Result or element existence.
164        if engine.origins[origin].aggregate {
165            continue;
166        }
167        let exists = engine.predicates.and(
168            engine.origins[origin].exists,
169            selected,
170            &mut engine.work,
171            span,
172        )?;
173        present = engine
174            .predicates
175            .or(present, exists, &mut engine.work, span)?;
176    }
177    Ok(present)
178}
179/// Every ordinary return must carry the original accumulator, not just a selected element or subset.
180pub(super) fn verify(indices: &[usize], summary: &mut Summary, span: Span) -> Checked<()> {
181    for origin in &summary.origins {
182        charge(&mut summary.work, 1, span)?;
183        charge(&mut summary.work, indices.len(), span)?;
184        if !origin.input.is_some_and(|index| indices.contains(&index)) {
185            continue;
186        }
187        let required =
188            summary
189                .predicates
190                .and(origin.exists, summary.exits, &mut summary.work, span)?;
191        if !summary
192            .predicates
193            .implies(required, origin.returned, &mut summary.work, span)?
194        {
195            return Err(Diagnostic::new(
196                span,
197                "Result obligation recursive builder does not retain its complete input",
198            ));
199        }
200    }
201    Ok(())
202}

204/// Builders may embed nominal payloads, List elements, or complete List accumulators.
205pub(super) fn inputs(
206    program: &ir::Program,
207    function: &ir::Function,
208    work: &mut usize,
209) -> Checked<Option<Vec<usize>>> {
210    let span = function.body.span;
211    let supported = match &function.return_type {
212        Type::List(_) => true,
213        ty @ Type::Named(..) => recursive_trees::storage(program, ty, work, span)?,
214        _ => false,
215    };
216    if !supported || !closed_data(program, &function.return_type, true, work, span)? {
217        return Ok(None);
218    }
219    let mut indices = Vec::new();
220    for (index, param) in function.params.iter().chain(&function.captures).enumerate() {
221        if !closed_data(program, &param.ty, true, work, span)? {
222            return Ok(None);
223        }
224        if gate::contains(program, &param.ty, work, span)? {
225            if let Type::List(element) = &function.return_type
226                && param.ty != function.return_type
227                && &param.ty != element.as_ref()
228            {
229                return Ok(None);
230            }
231            charge(work, 1, span)?;
232            indices.push(index);
233        }
234    }
235    Ok((!indices.is_empty()).then_some(indices))
236}

258    #[test]
259    fn aggregate_obligations_do_not_establish_actual_result_presence() {
260        let program = program();
261        let mut engine = Engine::new(&program);
262        let span = Span::default();
263        let cut = engine.recursive_cut(0, 0, Some(0), span).unwrap();
264        assert_eq!(
265            duty_present(&mut engine, &cut, span).unwrap(),
266            Predicate::FALSE
267        );
268        let list = engine
269            .node(
270                Region::List {
271                    items: vec![cut],
272                    exact: true,
273                    nonempty: Predicate::TRUE,
274                },
275                span,
276            )
277            .unwrap();
278        assert_eq!(
279            duty_present(&mut engine, &list, span).unwrap(),
280            Predicate::FALSE
281        );
282        let result = engine

284                &Type::Result(Box::new(Type::Int), Box::new(Type::String)),
285                Some(1),
286                span,
287                0,
288            )
289            .unwrap();
290        assert_eq!(
291            duty_present(&mut engine, &result, span).unwrap(),
292            Predicate::TRUE
