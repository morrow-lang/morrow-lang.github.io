1//! Eager map/filter traversal proves every callback while preserving produced and retained families.
2use super::*;
3impl Engine<'_> {
4    /// Evaluate one typed callback summary universally; empty input never executes that callback.
5    pub(super) fn collection_transform(
6        &mut self,
7        builtin: ir::Builtin,
8        values: &[Value],
9        source: &[ir::Expr],
10        result: &Type,
11        span: Span,
12    ) -> Checked<Value> {
13        let [list, callback] = values else {
14            return self.unsupported(span);
15        };
16        let [list_expr, callback_expr] = source else {
17            return self.unsupported(span);
18        };
19        let Type::List(item) = &list_expr.ty else {
20            return self.unsupported(span);
21        };
22        let Type::List(output) = result else {
23            return self.unsupported(span);
24        };
25        let callback_result = if builtin == ir::Builtin::ListFilter {
26            &Type::Bool
27        } else {
28            output.as_ref()
29        };
30        let family = substitute::Substitution::family(self, list, false, span, 0)?;
31        let plan = tree_iteration::prepare(self, &family, item, span)?;
32        let summary = self.collection_callback(
33            callback,
34            &callback_expr.ty,
35            (item, callback_result),
36            plan.as_ref(),
37            span,
38        )?;
39        self.work = summary.work;
40        let nonempty = self.list_nonempty(list, span, 0)?;
41        if nonempty == Predicate::FALSE {
42            return self.node(
43                Region::List {
44                    items: vec![],
45                    exact: true,
46                    nonempty,
47                },
48                span,
49            );
50        }
51        let parent = self.path;
52        self.path = self
53            .predicates
54            .and(parent, nonempty, &mut self.work, span)?;
55        let mut substitution = substitute::Substitution::traversal(&summary);
56        let output = substitution.apply(self, &[family.clone(), callback.clone()], span)?;
57        let output = if builtin == ir::Builtin::ListFilter {
58            self.filtered_family(family, output, nonempty, span)?
59        } else {
60            self.node(
61                Region::List {
62                    items: vec![output],
63                    exact: false,
64                    nonempty,
65                },
66                span,
67            )?
68        };
69        self.path = parent;
70        Ok(output)
71    }
72    /// The callable environment is a formal input so captures preserve actual Boolean/alias facts.
73    pub(super) fn collection_callback(
74        &mut self,
75        callback: &Value,
76        callable_type: &Type,
77        signature: (&Type, &Type),
78        plan: Option<&tree_iteration::Plan>,
79        span: Span,
80    ) -> Checked<Summary> {
81        let mut engine = Engine::new(self.program);
82        engine.work = self.work;
83        engine.mode = self.mode;
84        engine.summaries = self.summaries;
85        engine.relevance = self.relevance;
86        engine.effect_cache = self.effect_cache.clone();
87        let input = tree_iteration::input(&mut engine, signature.0, plan, span)?;
88        let shape = engine.effect_shape(callback, span, 0)?;
89        let callable = engine.fresh_shaped(callable_type, Some(1), Some(&shape), span, 0)?;
90        engine.inputs = vec![input.clone(), callable.clone()];
91        engine.used_inputs.extend([0, 1]);
92        let output = engine.invoke(&callable, &[input], signature.1, span, 0)?;
93        engine.exit(&output, span, 0)?;
94        let output = engine.output(span)?;
95        engine.finish(output)
96    }
97    /// A filter's retained output covers every input only when its predicate is unconditionally true.
98    fn filtered_family(
99        &mut self,
100        family: Value,
101        selected: Value,
102        nonempty: Predicate,
103        span: Span,
104    ) -> Checked<Value> {
105        let selected = self.condition(&selected, span)?;
106        if selected == Predicate::FALSE {
107            return self.node(
108                Region::List {
109                    items: vec![],
110                    exact: true,
111                    nonempty: Predicate::FALSE,
112                },
113                span,
114            );
115        }
116        let (family, nonempty) = if selected == Predicate::TRUE {
117            (family, nonempty)
118        } else {
119            let retained = self.predicates.variable(&mut self.work, span)?;
120            let retained = self
121                .predicates
122                .and(nonempty, retained, &mut self.work, span)?;
123            (family.partial(), retained)
124        };
125        self.node(
126            Region::List {
127                items: vec![family],
128                exact: false,
129                nonempty,
130            },
131            span,
132        )
133    }
134}
