1//! Callable effect specialization keys retain code identity, never runtime scalar values.
2use super::*;
3
4#[derive(Clone, Debug, PartialEq, Eq, Hash)]
5pub(super) enum Shape {
6    Plain,
7    Callable(usize, Vec<Shape>),
8}
9#[derive(Default)]
10pub(super) struct Cache {
11    ready: HashMap<(usize, Vec<Shape>), Rc<Summary>>,
12    active: HashSet<(usize, Vec<Shape>)>,
13}
14impl Engine<'_> {
15    /// Resolve a bounded target signature without deriving ownership from function representation.
16    fn effect_function(&mut self, id: usize, span: Span) -> Checked<&ir::Function> {
17        self.charge(self.program.functions.len(), span)?;
18        self.program
19            .functions
20            .iter()
21            .find(|f| f.id.0 == id)
22            .ok_or_else(|| Diagnostic::new(span, "missing Result obligation callable definition"))
23    }
24    /// Abstract known environments recursively; scalar choices stay parameters of the cached proof.
25    pub(super) fn effect_shape(
26        &mut self,
27        value: &Value,
28        span: Span,
29        depth: usize,
30    ) -> Checked<Shape> {
31        self.charge(1, span)?;
32        if depth >= DEPTH_LIMIT {
33            return self.unsupported(span);
34        }
35        let Region::Callable { function, captures } = &value.node.kind else {
36            return Ok(Shape::Plain);
37        };
38        self.charge(captures.len(), span)?;
39        let captures = captures
40            .iter()
41            .map(|v| self.effect_shape(v, span, depth + 1))
42            .collect::<Checked<_>>()?;
43        Ok(Shape::Callable(*function, captures))
44    }
45    /// Rebuild formal callable environments using checked capture types and symbolic scalar values.
46    pub(super) fn fresh_shaped(
47        &mut self,
48        ty: &Type,
49        input: Option<usize>,
50        shape: Option<&Shape>,
51        span: Span,
52        depth: usize,
53    ) -> Checked<Value> {
54        self.charge(1, span)?;
55        if depth >= DEPTH_LIMIT {
56            return self.unsupported(span);
57        }
58        let Some(Shape::Callable(function, shapes)) = shape else {
59            return self.fresh(ty, input, span, depth);
60        };
61        if !matches!(ty, Type::Function(..)) {
62            return self.unsupported(span);
63        }
64        if self.abstract_trait(*function) && shapes.is_empty() {
65            return self.node(
66                Region::Callable {
67                    function: *function,
68                    captures: Vec::new(),
69                },
70                span,
71            );
72        }
73        let target = self.effect_function(*function, span)?;
74        if target.captures.len() != shapes.len() {
75            return self.unsupported(span);
76        }
77        // The program is immutable and independently bounded by the checker entry boundary.
78        let program = self.program;
79        let target = program
80            .functions
81            .iter()
82            .find(|f| f.id.0 == *function)
83            .ok_or_else(|| Diagnostic::new(span, "missing callable capture types"))?;
84        self.charge(shapes.len(), span)?;
85        let captures = target
86            .captures
87            .iter()
88            .zip(shapes)
89            .map(|(p, s)| self.fresh_shaped(&p.ty, input, Some(s), span, depth + 1))
90            .collect::<Checked<_>>()?;
91        self.node(
92            Region::Callable {
93                function: *function,
94                captures,
95            },
96            span,
97        )
98    }
99    /// Cache complete summaries per target/environment shape with no recursive self-discharge.
100    pub(super) fn effect_summary(
101        &mut self,
102        id: usize,
103        args: &[Value],
104        span: Span,
105    ) -> Checked<Option<Rc<Summary>>> {
106        self.charge(args.len(), span)?;
107        let shapes = args
108            .iter()
109            .map(|v| self.effect_shape(v, span, 0))
110            .collect::<Checked<Vec<_>>>()?;
111        if shapes.iter().all(|s| *s == Shape::Plain) {
112            return Ok(None);
113        }
114        let key = (id, shapes);
115        if let Some(summary) = self.effect_cache.borrow().ready.get(&key) {
116            return Ok(Some(summary.clone()));
117        }
118        if self.effect_cache.borrow().active.len() >= DEPTH_LIMIT {
119            return self.unsupported(span);
120        }
121        if !self.effect_cache.borrow_mut().active.insert(key.clone()) {
122            return Err(Diagnostic::new(
123                span,
124                "Result obligation recursive callable summary requires a fixed point",
125            ));
126        }
127        let result = self.build_effect_summary(id, &key.1, span);
128        self.effect_cache.borrow_mut().active.remove(&key);
129        let summary = Rc::new(result?);
130        self.work = summary.work;
131        self.effect_cache
132            .borrow_mut()
133            .ready
134            .insert(key, summary.clone());
135        Ok(Some(summary))
136    }
137    /// A shape proof shares the entire caller budget and substitutes fresh formal inputs once.
138    fn build_effect_summary(
139        &mut self,
140        id: usize,
141        shapes: &[Shape],
142        span: Span,
143    ) -> Checked<Summary> {
144        self.charge(self.program.functions.len(), span)?;
145        let function = self
146            .program
147            .functions
148            .iter()
149            .find(|f| f.id.0 == id)
150            .ok_or_else(|| Diagnostic::new(span, "missing Result obligation effect target"))?;
151        let mut engine = Engine::new(self.program);
152        engine.work = self.work;
153        engine.mode = self.mode;
154        engine.summaries = self.summaries;
155        engine.relevance = self.relevance;
156        engine.effect_cache = self.effect_cache.clone();
157        analyze_shaped_body(engine, function, shapes)
158    }
159}
