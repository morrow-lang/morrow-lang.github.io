109struct Summary {

112    work: usize,
113    exits: Predicate,
114    terminated: Predicate,
115    breaks: Predicate,
116    iteration_returns: Vec<(Predicate, Value)>,

127    /// Count universally acknowledged parameter duties; partial views never count.
128    fn handled_inputs(&self) -> usize {

136    fn decision_nodes(&self) -> usize {

179/// Analyze supported typed bodies while refusing forms without a complete handling proof.
180#[cfg(test)]
181fn analyze_function(program: &ir::Program, function: &ir::Function) -> Checked<Summary> {
182    calls::analyze(program, function)
183}
184/// Record the original input graph alongside the checked effects and all returned output paths.
185fn analyze_body(engine: Engine<'_>, function: &ir::Function) -> Checked<Summary> {
186    analyze_shaped_body(engine, function, &[])
187}
188/// Callable shapes refine effects while scalar captures remain independent formal input regions.
189fn analyze_shaped_body(
190    mut engine: Engine<'_>,
191    function: &ir::Function,
192    shapes: &[effect_shapes::Shape],
193) -> Checked<Summary> {
194    engine.source_function = Some(function.id.0);
195    engine.charge(
196        function
197            .params
198            .len()
199            .saturating_add(function.captures.len()),
200        function.body.span,
201    )?;
202    for (index, param) in function.params.iter().chain(&function.captures).enumerate() {
203        let value = engine.fresh_shaped(
204            &param.ty,
205            Some(index),
206            shapes.get(index),
207            function.body.span,
208            0,
209        )?;
210        engine.inputs.push(value.clone());
211        engine.locals.insert(param.id.0, value);
212        engine.parameters.insert(param.id.0, index);
213    }
214    let value = engine.expression(&function.body, 0)?;
215    engine.exit(&value, function.body.span, 0)?;
216    let output = engine.output(function.body.span)?;
217    engine.finish(output)
218}
219impl<'a> Engine<'a> {
220    /// Begin one bounded function proof with an unconditional reachable entry.
221    fn new(program: &'a ir::Program) -> Self {
222        Self {
223            program,
224            source_function: None,
225            locals: HashMap::new(),
226            parameters: HashMap::new(),
227            used_inputs: HashSet::new(),
228            origins: Vec::new(),
229            work: 0,
230            nodes: 0,
231            predicates: Predicates::default(),
232            path: Predicate::TRUE,
233            exits: Predicate::FALSE,
234            terminated: Predicate::FALSE,
235            cleanups: Vec::new(),
236            returned_values: Vec::new(),
237            inputs: Vec::new(),
238            summaries: None,
239            relevance: None,
240            mode: Mode::Concrete,
241            input_callable: false,
242            iteration_breaks: None,
243            iteration_returns: Vec::new(),
244            pending_callbacks: BTreeMap::new(),
245            effect_cache: Rc::new(RefCell::new(effect_shapes::Cache::default())),
246            partitions: HashMap::new(),
247            sequence_offsets: HashMap::new(),
248            active_nominals: Vec::new(),
249            nominal_roots: HashMap::new(),
250            nominal_descendants: HashMap::new(),
251            tree_context: None,
252        }
253    }
254    /// Bound every traversal, allocation and copied edge before doing its work.
255    fn charge(&mut self, amount: usize, span: Span) -> Checked<()> {
256        charge(&mut self.work, amount, span)?;
257        Ok(())
258    }
259    /// Region creation is shared by aliases and carries a stable analysis-only identity.
260    fn node(&mut self, kind: Region, span: Span) -> Checked<Value> {
261        self.charge(1, span)?;
262        let id = self.nodes;
263        self.nodes += 1;
264        Ok(Value {
265            node: Rc::new(values::Node { id, kind }),
266            complete: true,
267        })
268    }
269    /// An outer Result construction is separate from all Results in its payloads.
270    fn origin(&mut self, input: Option<usize>, span: Span) -> Checked<usize> {
271        self.charge(1, span)?;
272        let id = self.origins.len();
273        self.origins.push(Origin {
274            aggregate: false,
275            input,
276            span,
277            exists: self.path,
278            handled: Predicate::FALSE,
279            returned: Predicate::FALSE,
280        });
281        Ok(id)
282    }
283    /// Preserve the existing unused-input rule independently of borrowing summaries.
284    fn finish(mut self, output: Value) -> Checked<Summary> {
285        #[cfg(test)]
286        let mut handled_input_count = 0;
287        #[cfg(test)]
288        let mut borrowed_input_count = 0;
289        for (id, origin) in self.origins.iter().enumerate() {
290            if origin.input.is_some_and(|i| !self.used_inputs.contains(&i)) {
291                return Err(Diagnostic::new(
292                    origin.span,
293                    "Result binding parameter is never used",
294                ));
295            }
296            let required =
297                self.predicates
298                    .and(origin.exists, self.exits, &mut self.work, origin.span)?;
299            let disposed =
300                self.predicates
301                    .or(origin.handled, origin.returned, &mut self.work, origin.span)?;
302            let complete =
303                self.predicates
304                    .implies(required, disposed, &mut self.work, origin.span)?;
305            let possible = self
306                .pending_callbacks
307                .get(&id)
308                .copied()
309                .unwrap_or(Predicate::FALSE);
310            let possible = self
311                .predicates
312                .or(disposed, possible, &mut self.work, origin.span)?;
313            let conditional = self.input_callable
314                && self
315                    .predicates
316                    .implies(required, possible, &mut self.work, origin.span)?;
317            if origin.input.is_none() && !complete && !conditional {
318                return Err(Diagnostic::new(
319                    origin.span,
320                    "Result obligation remains unhandled on this exit",
321                ));
322            }
323            #[cfg(test)]
324            if origin.input.is_some() && required != Predicate::FALSE {
325                handled_input_count += usize::from(self.predicates.implies(
326                    required,
327                    origin.handled,
328                    &mut self.work,
329                    origin.span,
330                )?);
331                borrowed_input_count += usize::from(!complete);
332            }
333        }
334        Ok(Summary {
335            origins: self.origins,
336            inputs: self.inputs,
337            work: self.work,
338            exits: self.exits,
339            terminated: self.terminated,
340            breaks: self.iteration_breaks.unwrap_or(Predicate::FALSE),
341            iteration_returns: self.iteration_returns,
342            output,
343            predicates: self.predicates,
344            pending_callbacks: self.pending_callbacks,
345            #[cfg(test)]
346            handled_input_count,
347            #[cfg(test)]
348            borrowed_input_count,
349        })
350    }
351    /// Unsupported proof forms fail closed until their separate semantic stage is implemented.
352    fn unsupported<T>(&self, span: Span) -> Checked<T> {
353        Err(Diagnostic::new(
354            span,
355            "Result obligation analysis does not yet prove this operation",
356        ))
357    }
358}

360/// One shared budget covers region work and all path-predicate operations.
361fn charge(work: &mut usize, amount: usize, span: Span) -> Checked<()> {
362    *work = work
363        .checked_add(amount)
364        .filter(|n| *n <= WORK_LIMIT)
365        .ok_or_else(|| Diagnostic::new(span, "Result obligation analysis work limit exceeded"))?;
366    Ok(())
