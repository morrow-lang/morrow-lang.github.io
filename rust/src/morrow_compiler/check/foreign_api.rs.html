1//! Checked conversions preserve narrow scalar ranges and retain borrowed pointer ownership.
2use super::*;
3use ir::{Expr as E, ExprKind as K};
4fn expression(kind: K, ty: Type, span: Span) -> E {
5    E { kind, ty, span }
6}
7fn result(ty: Type) -> Type {
8    Type::Result(Box::new(ty), Box::new(Type::String))
9}
10fn pointer(item: Type) -> Type {
11    Type::Named("Ptr".into(), vec![item])
12}
13fn byte_pointer() -> Type {
14    pointer(Type::Named("CUInt8".into(), vec![]))
15}
16fn field(value: E, index: usize, ty: Type) -> E {
17    let span = value.span;
18    expression(
19        K::Field {
20            value: Box::new(value),
21            index,
22        },
23        ty,
24        span,
25    )
26}
27impl Checker<'_> {
28    fn foreign_api_signature(&mut self, name: &str) -> (Vec<Type>, Type) {
29        let pointer = pointer(self.inference.fresh());
30        match name {
31            "Ptr.null" => (vec![], pointer),
32            "Ptr.is_null" => (vec![pointer], Type::Bool),
33            "Ptr.equal" => (vec![pointer.clone(), pointer], Type::Bool),
34            "Ptr.to_string" => (vec![byte_pointer(), Type::Int], result(Type::String)),
35            "String.as_ptr" => (vec![Type::String], byte_pointer()),
36            _ => {
37                let (owner, method) = name.split_once('.').expect("validated foreign API");
38                let nominal = Type::Named(owner.into(), vec![]);
39                let primitive = if owner == "CFloat32" {
40                    Type::Float
41                } else {
42                    Type::Int
43                };
44                if method.starts_with("from_") {
45                    (vec![primitive], result(nominal))
46                } else {
47                    (
48                        vec![nominal],
49                        if owner == "CUInt64" {
50                            result(primitive)
51                        } else {
52                            primitive
53                        },
54                    )
55                }
56            }
57        }
58    }
59    pub(super) fn foreign_api(
60        &mut self,
61        name: &str,
62        args: &[ast::Argument],
63        expected: Option<&Type>,
64        span: Span,
65        depth: usize,
66    ) -> Checked<TypedKind> {
67        labels::positional(args)?;
68        let (params, result) = self.foreign_api_signature(name);
69        self.constrain_result(&result, expected, span)?;
70        let args = self.call_arguments(args, &params, span, depth)?;
71        let value = self.foreign_api_body(name, args, result, span)?;
72        Ok((value.kind, value.ty))
73    }
74    pub(super) fn foreign_api_value(&mut self, name: &str, span: Span) -> Checked<TypedKind> {
75        let (types, result) = self.foreign_api_signature(name);
76        let params: Vec<_> = types
77            .iter()
78            .map(|ty| {
79                let id = ir::LocalId(self.local_count);
80                self.local_count += 1;
81                ir::Param { id, ty: ty.clone() }
82            })
83            .collect();
84        let args = params
85            .iter()
86            .map(|p| expression(K::Local(p.id), p.ty.clone(), span))
87            .collect();
88        let body = self.foreign_api_body(name, args, result.clone(), span)?;
89        Ok((
90            K::Lambda {
91                params,
92                captures: vec![],
93                body: Box::new(body),
94                local_count: self.local_count,
95            },
96            Type::Function(types, Box::new(result)),
97        ))
98    }
99    pub(super) fn foreign_string_borrow(
100        &mut self,
101        value: E,
102        args: &[ast::Argument],
103        span: Span,
104    ) -> Checked<TypedKind> {
105        if !args.is_empty() {
106            return Err(Diagnostic::new(
107                span,
108                "String.as_ptr method takes no arguments",
109            ));
110        }
111        self.inference
112            .unify(&value.ty, &Type::String, span, "String.as_ptr receiver")?;
113        let value = self.foreign_api_body("String.as_ptr", vec![value], byte_pointer(), span)?;
114        Ok((value.kind, value.ty))
115    }
116    fn foreign_api_body(
117        &mut self,
118        name: &str,
119        args: Vec<E>,
120        result: Type,
121        span: Span,
122    ) -> Checked<E> {
123        if let Some(index) = args.iter().position(|a| a.ty == Type::Never) {
124            return Ok(expression(
125                K::Block(
126                    args.into_iter()
127                        .take(index + 1)
128                        .map(ir::Stmt::Expr)
129                        .collect(),
130                ),
131                Type::Never,
132                span,
133            ));
134        }
135        let mut statements = Vec::new();
136        let args: Vec<_> = args
137            .into_iter()
138            .map(|value| {
139                let id = ir::LocalId(self.local_count);
140                self.local_count += 1;
141                let local = expression(K::Local(id), value.ty.clone(), span);
142                statements.push(ir::Stmt::Let { id, value });
143                local
144            })
145            .collect();
146        let value = match name {
147            "Ptr.null" => expression(
148                K::CustomConstruct {
149                    tag: 0,
150                    fields: vec![
151                        expression(K::Int(0), Type::Int, span),
152                        expression(K::List(vec![]), Type::List(Box::new(Type::String)), span),
153                    ],
154                },
155                result,
156                span,
157            ),
158            "Ptr.is_null" => compare(
159                ast::BinaryOp::Eq,
160                field(args[0].clone(), 0, Type::Int),
161                expression(K::Int(0), Type::Int, span),
162                span,
163            ),
164            "Ptr.equal" => compare(
165                ast::BinaryOp::Eq,
166                field(args[0].clone(), 0, Type::Int),
167                field(args[1].clone(), 0, Type::Int),
168                span,
169            ),
170            "String.as_ptr" => internal("$ffi.borrow_string", args, result, span)?,
171            "Ptr.to_string" => internal(
172                "$ffi.read_string",
173                vec![field(args[0].clone(), 0, Type::Int), args[1].clone()],
174                result,
175                span,
176            )?,
177            _ => scalar_conversion(name, args[0].clone(), result, span)?,
178        };
179        let ty = value.ty.clone();
180        statements.push(ir::Stmt::Expr(value));
181        Ok(expression(K::Block(statements), ty, span))
182    }
183}
184fn internal(name: &str, args: Vec<E>, ty: Type, span: Span) -> Checked<E> {
185    let id = runtime::resolve(name)
186        .ok_or_else(|| Diagnostic::new(span, "missing internal foreign runtime boundary"))?;
187    Ok(expression(
188        K::Call {
189            target: ir::CallTarget::Runtime(id),
190            args,
191        },
192        ty,
193        span,
194    ))
195}
196fn compare(op: ast::BinaryOp, left: E, right: E, span: Span) -> E {
197    expression(
198        K::Binary {
199            op,
200            left: Box::new(left),
201            right: Box::new(right),
202        },
203        Type::Bool,
204        span,
205    )
206}
207fn checked_value(condition: E, value: E, result: Type, span: Span) -> E {
208    let ok = expression(
209        K::Construct {
210            constructor: Constructor::Ok,
211            value: Some(Box::new(value)),
212        },
213        result.clone(),
214        span,
215    );
216    let error = expression(
217        K::String("foreign scalar conversion is outside its representable range".into()),
218        Type::String,
219        span,
220    );
221    let error = expression(
222        K::Construct {
223            constructor: Constructor::Err,
224            value: Some(Box::new(error)),
225        },
226        result.clone(),
227        span,
228    );
229    expression(
230        K::If {
231            condition: Box::new(condition),
232            then_branch: Box::new(ok),
233            else_branch: Some(Box::new(error)),
234        },
235        result,
236        span,
237    )
238}
239fn scalar_conversion(name: &str, value: E, result: Type, span: Span) -> Checked<E> {
240    let (owner, method) = name.split_once('.').expect("validated scalar API");
241    if method.starts_with("to_") {
242        let inner = if owner == "CFloat32" {
243            Type::Float
244        } else {
245            Type::Int
246        };
247        let value = expression(K::Unwrap(Box::new(value)), inner, span);
248        return Ok(if owner == "CUInt64" {
249            let condition = compare(
250                ast::BinaryOp::Ge,
251                value.clone(),
252                expression(K::Int(0), Type::Int, span),
253                span,
254            );
255            checked_value(condition, value, result, span)
256        } else {
257            value
258        });
259    }
260    let nominal = Type::Named(owner.into(), vec![]);
261    let wrapped = expression(K::Wrap(Box::new(value.clone())), nominal, span);
262    if owner == "CFloat32" {
263        // Rounding occurs at the explicit CFloat32 conversion boundary, not later native calls.
264        return internal("$ffi.float32", vec![value], result, span);
265    }
266    let (min, max) = match owner {
267        "CInt8" => (i8::MIN as i64, i8::MAX as i64),
268        "CInt16" => (i16::MIN as i64, i16::MAX as i64),
269        "CInt32" => (i32::MIN as i64, i32::MAX as i64),
270        "CUInt8" => (0, u8::MAX as i64),
271        "CUInt16" => (0, u16::MAX as i64),
272        "CUInt32" => (0, u32::MAX as i64),
273        "CUInt64" => (0, i64::MAX),
274        _ => return Err(Diagnostic::new(span, "unknown foreign scalar conversion")),
275    };
276    let low = compare(
277        ast::BinaryOp::Ge,
278        value.clone(),
279        expression(K::Int(min), Type::Int, span),
280        span,
281    );
282    let high = compare(
283        ast::BinaryOp::Le,
284        value,
285        expression(K::Int(max), Type::Int, span),
286        span,
287    );
288    let condition = compare(ast::BinaryOp::And, low, high, span);
289    Ok(checked_value(condition, wrapped, result, span))
290}
