1//! Bounded source type descriptions; inference ownership is never rewritten in error strings.
2use crate::{Type, presentation};

4/// Describe a checked type mismatch, retaining source quantifiers and distinct identities.
5pub(super) fn mismatch(expected: &Type, actual: &Type) -> String {
6    let mut budget = Budget {
7        nodes: 4096,
8        bytes: 65_536,
9    };
10    let pair = budget.ty(expected, 0).zip(budget.ty(actual, 0));
11    let Some((left, right)) = pair else {
12        return "types differ (details exceed the diagnostic limit)".into();
13    };
14    let left_text = description(&left);
15    let right_text = description(&right);
16    let distinct = if left == right && expected != actual {
17        " (these are distinct type parameters)"
18    } else {
19        ""
20    };
21    format!("expected {left_text}, found {right_text}{distinct}")
22}

24/// Keep the known outer shape when unresolved children prevent a complete source signature.
25fn description(ty: &Type) -> String {
26    presentation::render_type(ty, presentation::Limits::default()).unwrap_or_else(|error| {
27        if error.message.contains("limit") {
28            return "a type whose details exceed the display limit".into();
29        }
30        match ty {
31            Type::Function(_, _) => "Function with unresolved type variables".into(),
32            Type::List(_) => "List with an unresolved element type".into(),
33            Type::Option(_) => "Option with an unresolved payload type".into(),
34            Type::Map(_, _) => "Map with unresolved key/value types".into(),
35            Type::Result(_, _) => "Result with unresolved payload types".into(),
36            Type::Tuple(_) => "Tuple with unresolved field types".into(),
37            Type::Named(name, _) => format!("{name} with unresolved type arguments"),
38            Type::Generic(_) => "a generic type".into(),
39            _ => "an unresolved type".into(),
40        }
41    })
42}

44struct Budget {
45    nodes: usize,
46    bytes: usize,
47}
48impl Budget {
49    /// Copy only bounded semantic type nodes, decoding owned names solely in Generic variants.
50    fn ty(&mut self, ty: &Type, depth: usize) -> Option<Type> {
51        if depth >= 128 {
52            return None;
53        }
54        self.nodes = self.nodes.checked_sub(1)?;
55        Some(match ty {
56            Type::Generic(name) => Type::Generic(self.name(source_generic(name))?),
57            Type::Named(name, args) => Type::Named(self.name(name)?, self.types(args, depth)?),
58            Type::Union(args) => Type::Union(self.types(args, depth)?),
59            Type::Tuple(args) => Type::Tuple(self.types(args, depth)?),
60            Type::Function(args, result) => Type::Function(
61                self.types(args, depth)?,
62                Box::new(self.ty(result, depth + 1)?),
63            ),
64            Type::List(a) => Type::List(Box::new(self.ty(a, depth + 1)?)),
65            Type::Option(a) => Type::Option(Box::new(self.ty(a, depth + 1)?)),
66            Type::Map(a, b) => Type::Map(
67                Box::new(self.ty(a, depth + 1)?),
68                Box::new(self.ty(b, depth + 1)?),
69            ),
70            Type::Result(a, b) => Type::Result(
71                Box::new(self.ty(a, depth + 1)?),
72                Box::new(self.ty(b, depth + 1)?),
73            ),
74            _ => ty.clone(),
75        })
76    }
77    /// Charge name bytes before allocation, including nominal names that must remain unchanged.
78    fn name(&mut self, name: &str) -> Option<String> {
79        self.bytes = self.bytes.checked_sub(name.len())?;
80        Some(name.into())
81    }
82    /// Copy compound children under one aggregate diagnostic storage budget.
83    fn types(&mut self, types: &[Type], depth: usize) -> Option<Vec<Type>> {
84        types.iter().map(|ty| self.ty(ty, depth + 1)).collect()
85    }
86}

88/// Decode only the exact compiler-owned rigid-variable form; all other names retain identity.
89fn source_generic(name: &str) -> &str {
90    let Some((owner, source)) = name.strip_prefix("$rigid").and_then(|s| s.split_once(':')) else {
91        return name;
92    };
93    let mut chars = source.chars();
94    let valid = chars.next().is_some_and(char::is_lowercase)
95        && chars.all(|c| {
96            c == '_' || c.is_ascii_alphanumeric() || (!c.is_ascii() && !c.is_whitespace())
97        });
98    if !owner.is_empty() && owner.bytes().all(|b| b.is_ascii_digit()) && valid {
99        source
100    } else {
101        name
102    }
103}

105#[cfg(test)]
106mod tests {
107    use super::*;
108    #[test]
109    fn name_decoding_is_structural_and_does_not_conflate_owned_parameters() {
110        let name = "$rigid12:a";
111        let mut budget = Budget {
112            nodes: 20,
113            bytes: 100,
114        };
115        assert_eq!(
116            budget.ty(&Type::Named(name.into(), vec![]), 0),
117            Some(Type::Named(name.into(), vec![]))
118        );
119        assert_eq!(source_generic("$rigidx:a"), "$rigidx:a");
120        assert_eq!(source_generic("$rigid1:a.bad"), "$rigid1:a.bad");
121        let error = mismatch(
122            &Type::Generic("$rigid0:a".into()),
123            &Type::Generic("$rigid1:a".into()),
124        );
125        assert!(error.contains("distinct type parameters"));
126        assert!(!error.contains("$rigid"));
127    }
128    #[test]
129    fn diagnostic_name_storage_is_bounded_before_copying() {
130        let error = mismatch(&Type::Generic("a".repeat(65_537)), &Type::Int);
131        assert!(error.contains("diagnostic limit"));
132        let error = mismatch(&Type::Tuple(vec![Type::Int; 3000]), &Type::Int);
133        assert!(error.contains("display limit"), "{error}");
134    }
