1//! Resolve private parameter slots from complete, caller-independent clause evidence.
2use super::*;
3use std::borrow::Cow;
4
5type Constraint<'a> = (&'a ast::Pattern, Type);
6
7/// Fill only private omissions, keeping the original source and fully annotated fast path.
8pub(super) fn resolve<'a>(
9    program: &'a ast::Program,
10    registry: &nominal::Registry,
11) -> Checked<Cow<'a, ast::Program>> {
12    if !program
13        .functions
14        .iter()
15        .any(|f| f.params.iter().any(|p| p.annotation.is_none()))
16    {
17        return Ok(Cow::Borrowed(program));
18    }
19    let mut prepared = program.clone();
20    let signatures = HashMap::new();
21    let mut checker = Checker {
22        mailbox: None,
23        editor: None,
24        recovery: None,
25        signatures: &signatures,
26        registry,
27        scopes: Vec::new(),
28        local_count: 0,
29        expr_count: 0,
30        inference: Inference {
31            newtypes: registry.newtype_definitions(),
32            newtype_work: registry.newtype_budget(),
33            ..Inference::default()
34        },
35        function_return: Type::Unit,
36        deferred: false,
37        loop_depth: 0,
38    };
39    checker.inference.probing = true;
40    let result = prepare_groups(program, &mut prepared, &mut checker);
41    result.map_err(|error| {
42        if error.message.contains("return inference work limit") {
43            Diagnostic::new(
44                error.span,
45                "parameter inference work limit exceeded; add parameter annotations",
46            )
47        } else {
48            error
49        }
50    })?;
51    Ok(Cow::Owned(prepared))
52}

54/// Keep inference variables independent across groups while retaining one aggregate budget.
55fn prepare_groups(
56    source: &ast::Program,
57    prepared: &mut ast::Program,
58    checker: &mut Checker<'_>,
59) -> Checked<()> {
60    let mut start = 0;
61    while start < source.functions.len() {
62        let name = &source.functions[start].name;
63        let mut end = start + 1;
64        while end < source.functions.len() && source.functions[end].name == *name {
65            end += 1;
66        }
67        let group = &source.functions[start..end];
68        if group
69            .iter()
70            .any(|f| f.params.iter().any(|p| p.annotation.is_none()))
71        {
72            let types = infer_group(group, checker)?;
73            for function in &mut prepared.functions[start..end] {
74                for (param, ty) in function.params.iter_mut().zip(&types) {
75                    returns::charge_output(&checker.inference, ty, param.span)?;
76                    param.annotation = Some(ty.clone());
77                }
78            }
79        }
80        start = end;
81    }
82    Ok(())
83}

85/// Constrain all annotations before patterns so rigid generics never depend on clause order.
86pub(super) fn infer_group(
87    group: &[ast::Function],
88    checker: &mut Checker<'_>,
89) -> Checked<Vec<Type>> {
90    let first = &group[0];
91    let slots: Vec<_> = first
92        .params
93        .iter()
94        .map(|_| checker.inference.fresh())
95        .collect();
96    let mut pending = Vec::new();
97    for function in group {
98        if function.params.len() != slots.len() || function.group_start != first.group_start {
99            return Err(Diagnostic::new(
100                function.span,
101                "function clauses must be adjacent and have the same arity",
102            ));
103        }
104        for (param, slot) in function.params.iter().zip(&slots) {
105            returns::charge(&checker.inference, param.span)?;
106            if let Some(annotation) = &param.annotation {
107                checker.inference.unify(
108                    slot,
109                    annotation,
110                    param.span,
111                    "parameter clause annotation",
112                )?;
113            } else if function.public {
114                return Err(Diagnostic::new(
115                    param.span,
116                    "public function parameters require type annotations",
117                ));
118            }
119            pending.push((&param.pattern, slot.clone()));
120        }
121    }
122    constrain(checker, pending)?;
123    slots
124        .iter()
125        .enumerate()
126        .map(|(index, slot)| {
127            let ty = checker.inference.resolve(slot, first.params[index].span)?;
128            if returns::has_infer(&ty) && !checker.inference.whole_signature {
129                return Err(Diagnostic::new(
130                    first.params[index].span,
131                    format!(
132                        "cannot infer parameter {} of '{}'; add a type annotation",
133                        index + 1,
134                        first.name
135                    ),
136                ));
137            }
138            if !checker.inference.whole_signature {
139                validate_type(&ty, first.params[index].span)?;
140            }
141            Ok(ty)
142        })
143        .collect()
144}

146/// Revisit only tuple-rest obligations whose fixed arity another clause can establish.
147pub(super) fn constrain(
148    checker: &mut Checker<'_>,
149    mut pending: Vec<Constraint<'_>>,
150) -> Checked<()> {
151    loop {
152        let mut deferred = Vec::new();
153        let mut progressed = false;
154        while let Some((pattern, ty)) = pending.pop() {
155            returns::charge(&checker.inference, pattern.span)?;
156            if let ast::PatternKind::TupleRest { prefix, .. } = &pattern.kind {
157                let shape = checker.inference.resolve(&ty, pattern.span)?;
158                if matches!(shape, Type::Infer(_)) {
159                    deferred.push((pattern, ty));
160                    continue;
161                }
162                let fields = sequences::tuple_fields(&shape, pattern.span)?;
163                if prefix.len() > fields.len() {
164                    return Err(Diagnostic::new(
165                        pattern.span,
166                        "tuple parameter pattern prefix exceeds tuple arity",
167                    ));
168                }
169                pending.extend(prefix.iter().zip(fields.iter().cloned()));
170            } else {
171                constrain_pattern(checker, pattern, &ty, &mut pending)?;
172            }
173            progressed = true;
174        }
175        if deferred.is_empty() {
176            return Ok(());
177        }
178        if !progressed && checker.inference.whole_signature {
179            return Ok(());
180        }
181        if !progressed {
182            return Err(Diagnostic::new(
183                deferred[0].0.span,
184                "cannot infer tuple-rest parameter arity; add a tuple type annotation",
185            ));
186        }
187        pending = deferred;
188    }
189}

191/// Build scalar/container constraints without binding source names or consuming Results.
192fn constrain_pattern<'a>(
193    checker: &mut Checker<'_>,
194    pattern: &'a ast::Pattern,
195    ty: &Type,
196    pending: &mut Vec<Constraint<'a>>,
197) -> Checked<()> {
198    use ast::PatternKind::*;
199    let expected = match &pattern.kind {
200        Typed { .. } => {
201            return Err(Diagnostic::new(
202                pattern.span,
203                "typed parameter subpatterns require an explicit enclosing annotation",
204            ));
205        }
206        Bind(_) | Wildcard => return Ok(()),
207        Int(_) => Type::Int,
208        Bool(_) => Type::Bool,
209        String(_) => Type::String,
210        Tuple(fields) => {
211            let types: Vec<_> = fields.iter().map(|_| checker.inference.fresh()).collect();
212            pending.extend(fields.iter().zip(types.iter().cloned()));
213            sequences::tuple_type(types)
214        }
215        List { prefix, .. } => {
216            let element = checker.inference.fresh();
217            pending.extend(prefix.iter().map(|p| (p, element.clone())));
218            Type::List(Box::new(element))
219        }
220        NamedConstructor { name, fields } => {
221            let (expected, _, types) = checker.pattern_signature(name, pattern.span)?;
222            if fields.len() != types.len() {
223                return Err(Diagnostic::new(
224                    pattern.span,
225                    "constructor parameter pattern field arity mismatch",
226                ));
227            }
228            for field in &types {
229                returns::charge_output(&checker.inference, field, pattern.span)?;
230            }
231            pending.extend(fields.iter().zip(types));
232            expected
233        }
234        Constructor { constructor, .. } => {
235            let name = match constructor {
236                crate::Constructor::Some => "Some",
237                crate::Constructor::None => "None",
238                crate::Constructor::Ok => "Ok",
239                crate::Constructor::Err => "Err",
240            };
241            checker.pattern_signature(name, pattern.span)?.0
242        }
243        TupleRest { .. } => {
244            unreachable!("tuple-rest constraints are handled before structural patterns")
245        }
246    };
247    checker
248        .inference
249        .unify(ty, &expected, pattern.span, "parameter pattern")
250}
