1//! Bound caller-created syntax before recursive cloning for monomorphization.
2use super::{
3    Checked, MAX_EXPR_COUNT, MAX_EXPR_DEPTH, MAX_FUNCTIONS, MAX_PARAMETERS, validate_type_structure,
4};
5use crate::{Diagnostic, Span, Type, ast};

7struct Budget {
8    nodes: usize,
9    bytes: usize,
10}
11
12fn trait_bounds(bounds: &[ast::TraitBound], budget: &mut Budget) -> Checked<()> {
13    if bounds.len() > 32 {
14        return Err(Diagnostic::new(
15            Span::default(),
16            "trait constraint limit exceeded",
17        ));
18    }
19    for bound in bounds {
20        budget.charge(bound.name.len(), bound.span)?;
21        budget.ty(&bound.ty, bound.span)?;
22    }
23    Ok(())
24}
25impl Budget {
26    /// Charge a source node and its owned text before copying it.
27    fn charge(&mut self, bytes: usize, span: Span) -> Checked<()> {
28        self.nodes += 1;
29        self.bytes = self.bytes.saturating_add(bytes);
30        if self.nodes > MAX_EXPR_COUNT || self.bytes > 1024 * 1024 {
31            return Err(Diagnostic::new(
32                span,
33                "prototype syntax size limit exceeded",
34            ));
35        }
36        Ok(())
37    }
38    /// Check recursive annotations iteratively and account for their names.
39    fn ty(&mut self, ty: &Type, span: Span) -> Checked<()> {
40        validate_type_structure(ty, span, true)?;
41        let mut pending = vec![ty];
42        while let Some(ty) = pending.pop() {
43            let bytes = match ty {
44                Type::Named(name, _) | Type::Generic(name) => name.len(),
45                _ => 0,
46            };
47            self.charge(bytes, span)?;
48            match ty {
49                Type::Named(_, args) => pending.extend(args),
50                Type::Function(args, result) => {
51                    pending.extend(args);
52                    pending.push(result);
53                }
54                Type::Union(args) | Type::Tuple(args) => pending.extend(args),
55                Type::List(a) | Type::Option(a) => pending.push(a),
56                Type::ActorFunction(a, b) | Type::Result(a, b) | Type::Map(a, b) => {
57                    pending.push(a);
58                    pending.push(b);
59                }
60                _ => {}
61            }
62        }
63        Ok(())
64    }
65    /// Validate one bounded pattern tree, including otherwise unused generic bodies.
66    fn pattern(&mut self, pattern: &ast::Pattern) -> Checked<()> {
67        let mut pending = vec![(pattern, 0)];
68        while let Some((pattern, depth)) = pending.pop() {
69            if depth >= MAX_EXPR_DEPTH {
70                return Err(Diagnostic::new(
71                    pattern.span,
72                    "pattern nesting limit exceeded",
73                ));
74            }
75            let bytes = match &pattern.kind {
76                ast::PatternKind::Typed {
77                    pattern: inner,
78                    annotation,
79                } => {
80                    self.ty(annotation, pattern.span)?;
81                    pending.push((inner, depth + 1));
82                    0
83                }
84                ast::PatternKind::List { prefix, rest } => {
85                    self.sequence(prefix, rest.as_deref(), &mut pending, depth, pattern.span)?;
86                    0
87                }
88                ast::PatternKind::TupleRest { prefix, rest } => {
89                    self.sequence(prefix, Some(rest), &mut pending, depth, pattern.span)?;
90                    0
91                }
92                ast::PatternKind::Tuple(fields) => {
93                    if fields.len() > 128 {
94                        return Err(Diagnostic::new(
95                            pattern.span,
96                            "sequence pattern prefix limit exceeded",
97                        ));
98                    }
99                    pending.extend(fields.iter().map(|p| (p, depth + 1)));
100                    0
101                }
102                ast::PatternKind::Bind(n) | ast::PatternKind::String(n) => n.len(),
103                ast::PatternKind::Constructor { binding, .. } => {
104                    binding.as_ref().map_or(0, String::len)
105                }
106                ast::PatternKind::NamedConstructor { name, fields } => {
107                    pending.extend(fields.iter().map(|p| (p, depth + 1)));
108                    name.len()
109                }
110                _ => 0,
111            };
112            self.charge(bytes, pattern.span)?;
113        }
114        Ok(())
115    }
116    /// Bound flat sequence fields and reject recursively structured rest payloads before copies.
117    fn sequence<'a>(
118        &mut self,
119        prefix: &'a [ast::Pattern],
120        rest: Option<&'a ast::Pattern>,
121        pending: &mut Vec<(&'a ast::Pattern, usize)>,
122        depth: usize,
123        span: Span,
124    ) -> Checked<()> {
125        if prefix.len() > 128 {
126            return Err(Diagnostic::new(
127                span,
128                "sequence pattern prefix limit exceeded",
129            ));
130        }
131        pending.extend(prefix.iter().map(|p| (p, depth + 1)));
132        if let Some(rest) = rest {
133            if !matches!(
134                rest.kind,
135                ast::PatternKind::Bind(_) | ast::PatternKind::Wildcard
136            ) {
137                return Err(Diagnostic::new(
138                    rest.span,
139                    "sequence rest must be a binding or wildcard",
140                ));
141            }
142            pending.push((rest, depth + 1));
143        }
144        Ok(())
145    }
146    /// Check block annotations and enqueue initializers without recursive traversal.
147    fn block<'a>(
148        &mut self,
149        stmts: &'a [ast::Stmt],
150        pending: &mut Vec<(&'a ast::Expr, usize)>,
151        depth: usize,
152    ) -> Checked<()> {
153        for stmt in stmts {
154            if let ast::Stmt::LetElse { else_branch, .. } = stmt {
155                pending.push((else_branch, depth + 1));
156            }
157            match stmt {
158                ast::Stmt::Let {
159                    name,
160                    annotation,
161                    value,
162                    span,
163                } => {
164                    self.charge(name.len(), *span)?;
165                    if let Some(ty) = annotation {
166                        self.ty(ty, *span)?;
167                    }
168                    pending.push((value, depth + 1));
169                }
170                ast::Stmt::LetElse {
171                    pattern,
172                    annotation,
173                    value,
174                    span,
175                    ..
176                }
177                | ast::Stmt::LetPattern {
178                    pattern,
179                    annotation,
180                    value,
181                    span,
182                } => {
183                    self.pattern(pattern)?;
184                    if let Some(ty) = annotation {
185                        self.ty(ty, *span)?;
186                    }
187                    pending.push((value, depth + 1));
188                }
189                ast::Stmt::Expr(value) => pending.push((value, depth + 1)),
190            }
191        }
192        Ok(())
193    }

195    /// Charge interpolation text and enqueue expressions at the enclosing traversal depth.
196    fn interpolation<'a>(
197        &mut self,
198        parts: &'a [ast::StringPart],
199        pending: &mut Vec<(&'a ast::Expr, usize)>,
200        depth: usize,
201        span: Span,
202    ) -> Checked<()> {
203        for part in parts {
204            match part {
205                ast::StringPart::Text(text) => self.charge(text.len(), span)?,
206                ast::StringPart::Value(value) => pending.push((value, depth + 1)),
207            }
208        }
209        Ok(())
210    }

212    /// Check lambda annotations and duplicate names even in uninstantiated generic templates.
213    fn lambda_params(&mut self, params: &[ast::LambdaParam], span: Span) -> Checked<()> {
214        if params.len() > MAX_PARAMETERS {
215            return Err(Diagnostic::new(span, "lambda parameter limit exceeded"));
216        }
217        let mut names = std::collections::HashSet::new();
218        for param in params {
219            if !names.insert(&param.name) {
220                return Err(Diagnostic::new(param.span, "duplicate lambda parameter"));
221            }
222            self.charge(param.name.len(), param.span)?;
223            if let Some(ty) = &param.annotation {
224                self.ty(ty, param.span)?;
225            }
226        }
227        Ok(())
228    }

230    /// Check patterns before enqueueing each arm body and guard at its inherited depth.
231    fn match_arms<'a>(
232        &mut self,
233        arms: &'a [ast::MatchArm],
234        pending: &mut Vec<(&'a ast::Expr, usize)>,
235        depth: usize,
236    ) -> Checked<()> {
237        for arm in arms {
238            self.pattern(&arm.pattern)?;
239            pending.push((&arm.body, depth + 1));
240            if let Some(guard) = &arm.guard {
241                pending.push((guard, depth + 1));
242            }
243        }
244        Ok(())
245    }

247    /// Charge source/resolved names and all named-reference children before checker cloning.
248    fn reference_expression<'a>(
249        &mut self,
250        expr: &'a ast::Expr,
251        pending: &mut Vec<(&'a ast::Expr, usize)>,
252        depth: usize,
253    ) -> Checked<bool> {
254        match &expr.kind {
255            ast::ExprKind::TypeTarget(_) => {
256                return Err(Diagnostic::new(
257                    expr.span,
258                    "compile-time type target cannot be used as a value",
259                ));
260            }
261
262            ast::ExprKind::GlobalName { name, resolved } => {
263                self.charge(name.len(), expr.span)?;
264                self.charge(resolved.len(), expr.span)?;
265            }
266            ast::ExprKind::GlobalCall {
267                name,
268                resolved,
269                args,
270            } => {
271                self.charge(name.len(), expr.span)?;
272                self.charge(resolved.len(), expr.span)?;
273                self.reference_arguments(expr, args, pending, depth)?;
274            }
275            ast::ExprKind::GlobalPipe {
276                name,
277                resolved,
278                value,
279                args,
280                ..
281            } => {
282                self.charge(name.len(), expr.span)?;
283                self.charge(resolved.len(), expr.span)?;
284                pending.push((value, depth + 1));
285                self.reference_arguments(expr, args, pending, depth)?;
286            }
287            ast::ExprKind::Pipe {
288                value, name, args, ..
289            } => {
290                self.charge(name.len(), expr.span)?;
291                pending.push((value, depth + 1));
292                self.reference_arguments(expr, args, pending, depth)?;
293            }
294            ast::ExprKind::Call { name, args } => {
295                self.charge(name.len(), expr.span)?;
296                self.reference_arguments(expr, args, pending, depth)?;
297            }
298            _ => return Ok(false),
299        }
300        Ok(true)
301    }

303    /// A static type leaf is permitted only as the canonical decoder's complete target argument.
304    fn reference_arguments<'a>(
305        &mut self,
306        expr: &'a ast::Expr,
307        args: &'a [ast::Argument],
308        pending: &mut Vec<(&'a ast::Expr, usize)>,
309        depth: usize,
310    ) -> Checked<()> {
311        let slot = crate::codec_syntax::static_slot(expr);
312        for (index, arg) in args.iter().enumerate() {
313            if let ast::ExprKind::TypeTarget(ty) = &arg.value.kind {
314                if slot != Some(index) {
315                    return Err(Diagnostic::new(
316                        arg.span,
317                        "compile-time type target cannot be used as a value",
318                    ));
319                }
320                self.ty(ty, arg.span)?;
321            } else {
322                pending.push((&arg.value, depth + 1));
323            }
324        }
325        Ok(())
326    }

328    /// Charge expression nodes only after checking their current traversal depth.
329    fn expression_node(&mut self, span: Span, depth: usize) -> Checked<()> {
330        if depth >= MAX_EXPR_DEPTH {
331            return Err(Diagnostic::new(
332                span,
333                "prototype expression nesting limit exceeded (128)",
334            ));
335        }
336        self.charge(0, span)
337    }

339    /// Charge record field names and queue initializer expressions before cloning.
340    fn update_fields<'a>(
341        &mut self,
342        fields: &'a [ast::RecordField],
343        pending: &mut Vec<(&'a ast::Expr, usize)>,
344        depth: usize,
345    ) -> Checked<()> {
346        for field in fields {
347            self.charge(field.name.len(), field.span)?;
348            pending.push((&field.value, depth + 1));
349        }
350        Ok(())

353    /// Queue flat control flow and charge each newly introduced binding pattern.
354    fn iteration<'a>(
355        &mut self,
356        expr: &'a ast::Expr,
357        pending: &mut Vec<(&'a ast::Expr, usize)>,
358        depth: usize,
359    ) -> Checked<bool> {
360        match &expr.kind {
361            ast::ExprKind::Range { start, end, .. } => {
362                pending.push((start, depth + 1));
363                pending.push((end, depth + 1));
364            }
365            ast::ExprKind::For {
366                pattern,
367                iterable,
368                body,
369            } => {
370                self.pattern(pattern)?;
371                pending.push((iterable, depth + 1));
372                pending.push((body, depth + 1));
373            }
374            ast::ExprKind::With {
375                bindings,
376                body,
377                arms,
378            } => {
379                for binding in bindings {
380                    self.pattern(&binding.pattern)?;
381                    pending.push((&binding.value, depth + 1));
382                }
383                pending.push((body, depth + 1));
384                if let Some(arms) = arms {
385                    self.match_arms(arms, pending, depth)?;
386                }
387            }
388            _ => return Ok(false),
389        }
390        Ok(true)
391    }

393    /// Bound receive metadata before pushing its independently scoped expression children.
394    fn receive<'a>(
395        &mut self,
396        arms: &'a [ast::MatchArm],
397        timeout: &'a Option<(Box<ast::Expr>, Box<ast::Expr>)>,
398        span: Span,
399        pending: &mut Vec<(&'a ast::Expr, usize)>,
400        depth: usize,
401    ) -> Checked<()> {
402        if arms.is_empty() || arms.len() > 128 {
403            return Err(Diagnostic::new(span, "receive arm limit exceeded"));
404        }
405        self.match_arms(arms, pending, depth)?;
406        if let Some((duration, body)) = timeout {
407            pending.extend([(duration.as_ref(), depth + 1), (body.as_ref(), depth + 1)]);
408        }
409        Ok(())
410    }

412    /// Validate expression depth and annotations before source-instance cloning.
413    fn expression(&mut self, expr: &ast::Expr) -> Checked<()> {
414        let mut pending = vec![(expr, 0)];
415        while let Some((expr, depth)) = pending.pop() {
416            self.expression_node(expr.span, depth)?;
417            self.argument_metadata(expr)?;
418            if self.iteration(expr, &mut pending, depth)? {
419                continue;
420            }
421            if self.reference_expression(expr, &mut pending, depth)? {
422                continue;
423            }
424            match &expr.kind {
425                ast::ExprKind::ForeignCall { declaration, args } => {
426                    declaration.validate(expr.span)?;
427                    self.charge(
428                        declaration.symbol.len()
429                            + declaration.library.as_ref().map_or(0, |s| s.len()),
430                        expr.span,
431                    )?;
432                    if args.len() != declaration.params.len() {
433                        return Err(Diagnostic::new(expr.span, "foreign AST arity mismatch"));
434                    }
435                    for abi in declaration.params.iter().chain([&declaration.result]) {
436                        self.ty(&abi.source_type(), expr.span)?;
437                    }
438                    pending.extend(args.iter().map(|arg| (arg, depth + 1)));
439                }
440                ast::ExprKind::Receive { arms, timeout, .. } => {
441                    self.receive(arms, timeout, expr.span, &mut pending, depth)?
442                }
443                ast::ExprKind::ConditionMatch(arms) => queue_conditions(arms, &mut pending, depth),
444                ast::ExprKind::Interpolate(parts) | ast::ExprKind::MultilineString(parts) => {
445                    self.interpolation(parts, &mut pending, depth, expr.span)?
446                }
447                ast::ExprKind::Map(entries) => queue_map(entries, &mut pending, depth),
448                ast::ExprKind::RecordUpdate { value, fields } => {
449                    pending.push((value, depth + 1));
450                    self.update_fields(fields, &mut pending, depth)?;
451                }
452                ast::ExprKind::Lambda { params, body } => {
453                    self.lambda_params(params, expr.span)?;
454                    pending.push((body, depth + 1));
455                }
456                ast::ExprKind::Apply { callee, args } => {
457                    pending.push((callee, depth + 1));
458                    pending.extend(args.iter().map(|a| (&a.value, depth + 1)));
459                }
460                ast::ExprKind::Name(n) | ast::ExprKind::String(n) => {
461                    self.charge(n.len(), expr.span)?
462                }
463                ast::ExprKind::Return(value)
464                | ast::ExprKind::Defer(value)
465                | ast::ExprKind::Unary { value, .. }
466                | ast::ExprKind::Try(value) => pending.push((value, depth + 1)),
467                ast::ExprKind::Field { value, name } => {
468                    self.charge(name.len(), expr.span)?;
469                    pending.push((value, depth + 1));
470                }
471                ast::ExprKind::PostfixIf {
472                    value: left,
473                    condition: right,
474                }
475                | ast::ExprKind::Binary { left, right, .. } => {
476                    pending.push((left, depth + 1));
477                    pending.push((right, depth + 1));
478                }
479                ast::ExprKind::Tuple(args) | ast::ExprKind::List(args) => {
480                    pending.extend(args.iter().map(|e| (e, depth + 1)))
481                }
482                ast::ExprKind::If { .. } => queue_if(expr, &mut pending, depth),
483                ast::ExprKind::Match { value, arms } => {
484                    pending.push((value, depth + 1));
485                    self.match_arms(arms, &mut pending, depth)?;
486                }
487                ast::ExprKind::Block(stmts) => self.block(stmts, &mut pending, depth)?,
488                _ => {}
489            }
490        }
491        Ok(())
492    }
493}

495/// Bound the entire source AST before registry copies and specialization work begin.
496pub(super) fn check(program: &ast::Program) -> Checked<()> {
497    if program.functions.len() > MAX_FUNCTIONS
498        || program
499            .types
500            .len()
501            .saturating_add(program.aliases.len())
502            .saturating_add(program.newtypes.len())
503            > MAX_FUNCTIONS
504    {
505        return Err(Diagnostic::new(
506            Span::default(),
507            "prototype declaration count limit exceeded",
508        ));
509    }
510    let mut budget = Budget { nodes: 0, bytes: 0 };
511    if program
512        .traits
513        .len()
514        .saturating_add(program.implementations.len())
515        > MAX_FUNCTIONS
516    {
517        return Err(Diagnostic::new(
518            Span::default(),
519            "trait declaration count limit exceeded",
520        ));
521    }
522    for declaration in &program.traits {
523        budget.charge(
524            declaration
525                .name
526                .len()
527                .saturating_add(declaration.parameter.len()),
528            declaration.span,
529        )?;
530        trait_bounds(&declaration.parents, &mut budget)?;
531        if declaration.methods.len() > MAX_PARAMETERS {
532            return Err(Diagnostic::new(
533                declaration.span,
534                "trait method limit exceeded",
535            ));
536        }
537        for method in &declaration.methods {
538            budget.charge(
539                method
540                    .name
541                    .len()
542                    .saturating_add(method.function.len())
543                    .saturating_add(method.default.as_ref().map_or(0, String::len)),
544                declaration.span,
545            )?;
546        }
547    }
548    for implementation in &program.implementations {
549        trait_bounds(std::slice::from_ref(&implementation.bound), &mut budget)?;
550        trait_bounds(&implementation.constraints, &mut budget)?;
551        if implementation.methods.len() > MAX_PARAMETERS {
552            return Err(Diagnostic::new(
553                implementation.span,
554                "implementation method limit exceeded",
555            ));
556        }
557        for (name, function) in &implementation.methods {
558            budget.charge(
559                name.len().saturating_add(function.len()),
560                implementation.span,
561            )?;
562        }
563    }
564    for doc in program.docs.iter().chain(&program.module_doc) {
565        budget.charge(doc.target.len().saturating_add(doc.text.len()), doc.span)?;
566    }
567    for function in &program.functions {
568        if function.syntax == ast::FunctionSyntax::Constant
569            && (!function.params.is_empty()
570                || function.guard.is_some()
571                || !function.constraints.is_empty()
572                || function.name == "main")
573        {
574            return Err(Diagnostic::new(
575                function.span,
576                "constant declaration cannot have parameters, guards, trait constraints or entry-point syntax",
577            ));
578        }
579        trait_bounds(&function.constraints, &mut budget)?;
580        budget.charge(function.name.len(), function.span)?;
581        for param in &function.params {
582            budget.label(&param.label)?;
583            budget.pattern(&param.pattern)?;
584            if let Some(ty) = &param.annotation {
585                budget.ty(ty, param.span)?;
586            }
587        }
588        if let Some(ty) = &function.return_type {
589            budget.ty(ty, function.span)?;
590        }
591        if let Some(guard) = &function.guard {
592            budget.expression(guard)?;
593        }
594        budget.expression(&function.body)?;
595    }
596    aliases(program, &mut budget)?;
597    newtypes(program, &mut budget)?;
598    for decl in &program.types {
599        budget.charge(decl.name.len(), decl.span)?;
600        derivations(&decl.derives, decl.span, &mut budget)?;
601        if decl.parameters.len() > MAX_PARAMETERS || decl.variants.len() > MAX_PARAMETERS {
602            return Err(Diagnostic::new(
603                decl.span,
604                "type declaration arity limit exceeded",
605            ));
606        }
607        for name in &decl.parameters {
608            budget.charge(name.len(), decl.span)?;
609        }
610        for variant in &decl.variants {
611            budget.charge(variant.name.len(), variant.span)?;
612            if variant.fields.len() > MAX_PARAMETERS {
613                return Err(Diagnostic::new(
614                    variant.span,
615                    "constructor field count limit exceeded",
616                ));
617            }
618            for field in &variant.fields {
619                budget.charge(field.name.as_ref().map_or(0, String::len), field.span)?;
620                budget.ty(&field.ty, field.span)?;
621            }
622        }
623    }
624    Ok(())
625}

627/// Account for alias targets and formal names before any source cloning or expansion.
628fn aliases(program: &ast::Program, budget: &mut Budget) -> Checked<()> {
629    for alias in &program.aliases {
630        budget.charge(alias.name.len(), alias.span)?;
631        if alias.parameters.len() > MAX_PARAMETERS {
632            return Err(Diagnostic::new(
633                alias.span,
634                "type alias arity limit exceeded",
635            ));
636        }
637        for name in &alias.parameters {
638            budget.charge(name.len(), alias.span)?;
639        }
640        budget.ty(&alias.target, alias.span)?;
641    }
642    Ok(())
643}

645/// Queue map keys and values at the same bounded expression depth.
646fn queue_map<'a>(
647    entries: &'a [(ast::Expr, ast::Expr)],
648    pending: &mut Vec<(&'a ast::Expr, usize)>,
649    depth: usize,
650) {
651    pending.extend(
652        entries
653            .iter()
654            .flat_map(|(key, value)| [(key, depth + 1), (value, depth + 1)]),
655    );
656}

658/// Queue lazy condition branches without increasing source nesting for sibling arms.
659fn queue_conditions<'a>(
660    arms: &'a [ast::ConditionArm],
661    pending: &mut Vec<(&'a ast::Expr, usize)>,
662    depth: usize,
663) {
664    for arm in arms {
665        pending.extend(arm.condition.iter().map(|c| (c, depth + 1)));
666        pending.push((&arm.body, depth + 1));
667    }
668}
669/// Queue each branch at its bounded lexical child depth.
670fn queue_if<'a>(expr: &'a ast::Expr, pending: &mut Vec<(&'a ast::Expr, usize)>, depth: usize) {
671    if let ast::ExprKind::If {
672        condition,
673        then_branch,
674        else_branch,
675    } = &expr.kind
676    {
677        pending.push((condition, depth + 1));
678        pending.push((then_branch, depth + 1));
679        pending.extend(else_branch.as_deref().map(|n| (n, depth + 1)));
680    }
681}

683/// Bound newtype declarations before copying or expanding any storage metadata.
684fn newtypes(program: &ast::Program, budget: &mut Budget) -> Checked<()> {
685    for decl in &program.newtypes {
686        budget.charge(decl.name.len() + decl.constructor.len(), decl.span)?;
687        if decl.parameters.len() > MAX_PARAMETERS {
688            return Err(Diagnostic::new(
689                decl.span,
690                "newtype parameter limit exceeded",
691            ));
692        }
693        for name in &decl.parameters {
694            budget.charge(name.len(), decl.span)?;
695        }
696        budget.ty(&decl.inner, decl.inner_span)?;
697        derivations(&decl.derives, decl.span, budget)?;
698    }
699    Ok(())
700}
701
702impl Budget {
703    /// Charge caller-owned label text before cloning or validating a signature interface.
704    fn label(&mut self, label: &Option<ast::ArgumentLabel>) -> Checked<()> {
705        if let Some(label) = label {
706            self.charge(label.name.len(), label.span)?;
707            if !crate::parse::valid_label(&label.name) || label.span.start > label.span.end {
708                return Err(Diagnostic::new(label.span, "invalid argument label"));
709            }
710        }
711        Ok(())
712    }
713
714    /// Bound written arguments and pipe metadata even in inactive caller-created source ASTs.
715    fn argument_metadata(&mut self, expr: &ast::Expr) -> Checked<()> {
716        let args = match &expr.kind {
717            ast::ExprKind::Pipe { args, label, .. }
718            | ast::ExprKind::GlobalPipe { args, label, .. } => {
719                self.label(label)?;
720                args
721            }
722            ast::ExprKind::Call { args, .. }
723            | ast::ExprKind::GlobalCall { args, .. }
724            | ast::ExprKind::Apply { args, .. } => args,
725            _ => return Ok(()),
726        };
727        if args.len() > MAX_PARAMETERS {
728            return Err(Diagnostic::new(expr.span, "call argument limit exceeded"));
729        }
730        for arg in args {
731            self.label(&arg.label)?;
732        }
733        Ok(())
734    }
735}
736
737/// Bound source trait metadata before any nominal registry clones it.
738fn derivations(derives: &[ast::Derivation], span: Span, budget: &mut Budget) -> Checked<()> {
739    if derives.len() > 32 {
740        return Err(Diagnostic::new(
741            span,
742            "derive trait count limit exceeded (32)",
743        ));
744    }
745    for derive in derives {
746        budget.charge(derive.name.len(), derive.span)?;
747    }
748    Ok(())
749}
