1//! A single member hole is checked locally after its enclosing source signature is independently fixed.
2//! Hole-dependent local unknowns never enter SCC inference, reusable schemes, or executable IR.

6pub(super) struct State {
7    pub(super) site: HoleSite,
8    result: Option<Type>,
9    receiver: Option<Type>,
10}

12/// Validate every unaffected function, then prove one local incomplete operation without code generation.
13pub(crate) fn analyze(source: &ast::Program, site: &HoleSite) -> Checked<editor::Facts> {
14    preflight::check(source)?;
15    let expanded = aliases::expand(source)?;
16    let source = expanded.program.as_ref();
17    let registry = nominal::Registry::new(source)?;
18    aliases::validate(source, &registry)?;
19    let selected = source
20        .functions
21        .iter()
22        .find(|f| f.span.start <= site.field().start && site.field().end <= f.span.end)
23        .ok_or_else(|| failure(site, "member hole is outside a function"))?;
24    independent_group(source, &selected.name, site)?;
25    let graph = dependencies::analyze_with_work(source, expanded.work)?;
26    independent_component(source, &graph, &selected.name, site)?;
27    let (prepared, mut signatures, work) = whole::resolve(source, &registry, &graph)?;
28    labels::finalize(&prepared, &mut signatures)?;
29    let mut checked = schemes::validate(&prepared, &registry, &mut signatures, work)?;
30    for function in &prepared.functions {
31        if function.name != selected.name && signatures[&function.name].generics.is_empty() {
32            checked.push(checker(&registry, &signatures, None).function(function)?);
33        }
34    }
35    let function = prepared
36        .functions
37        .iter()
38        .find(|f| f.name == selected.name)
39        .ok_or_else(|| failure(site, "missing source group"))?;
40    let mut checker = checker(&registry, &signatures, Some(site.clone()));
41    checked.push(checker.function(function)?);
42    prove_obligations(checked, &registry, registry.codec_template_work.get())?;
43    let receiver = checker
44        .recovery
45        .and_then(|state| state.receiver)
46        .ok_or_else(|| failure(site, "member receiver has no independent type"))?;
47    let facts = editor::receiver_facts(source, &registry, receiver, site.field())?;
48    if facts
49        .members
50        .iter()
51        .any(|member| member.name == site.original())
52    {
53        return Err(failure(
54            site,
55            "complete known member cannot suppress an unrelated error",
56        ));
57    }
58    Ok(facts)
59}

61/// Keep the private hole explicit while lifting all checked cleanup and closure bodies for proof.
62fn prove_obligations(
63    mut functions: Vec<ir::Function>,
64    registry: &nominal::Registry,
65    work: usize,
66) -> Checked<()> {
67    lift::run(&mut functions)?;
68    let types = registry.layouts(&functions)?;
69    obligations::check_recovery(&ir::Program { functions, types }, work)
70}

72/// Require a complete concrete source signature for every clause of the selected group.
73fn independent_group(source: &ast::Program, name: &str, site: &HoleSite) -> Checked<()> {
74    for function in source.functions.iter().filter(|f| f.name == name) {
75        if !complete(function) {
76            return Err(failure(
77                site,
78                "member recovery requires an independently annotated function signature",
79            ));
80        }
81        let result = function_result(function)?;
82        let types = function
83            .params
84            .iter()
85            .filter_map(|p| p.annotation.clone())
86            .chain([result]);
87        if !nominal::generics(types).is_empty() {
88            return Err(failure(
89                site,
90                "generic outer signatures require ordinary complete source",
91            ));
92        }
93    }
94    Ok(())
95}

97/// Refuse an incomplete recursive peer that would cause ordinary whole inference to inspect this hole.
98fn independent_component(
99    source: &ast::Program,
100    graph: &dependencies::Graph,
101    name: &str,
102    site: &HoleSite,
103) -> Checked<()> {
104    let selected = graph
105        .groups
106        .iter()
107        .position(|g| g.name == name)
108        .ok_or_else(|| failure(site, "unknown source group"))?;
109    for component in &graph.components {
110        if !component.contains(&selected) {
111            continue;
112        }
113        for &index in component {
114            for function in &source.functions[graph.groups[index].clauses.clone()] {
115                if !complete(function) {
116                    return Err(failure(
117                        site,
118                        "recursive recovery requires fixed source signatures",
119                    ));
120                }
121            }
122        }
123    }
124    Ok(())
125}

127/// Source main retains its established Unit default; all other selected slots must be explicit.
128fn complete(function: &ast::Function) -> bool {
129    function.params.iter().all(|p| p.annotation.is_some())
130        && (function.name == "main" || function.return_type.is_some())
131}

133/// All ordinary function checks use the same defaults; recovery is explicit and local to one body.
134fn checker<'a>(
135    registry: &'a nominal::Registry,
136    signatures: &'a HashMap<String, Signature>,
137    site: Option<HoleSite>,
138) -> Checker<'a> {
139    Checker {
140        mailbox: None,
141        editor: None,
142        recovery: site.map(|site| State {
143            site,
144            result: None,
145            receiver: None,
146        }),
147        signatures,
148        registry,
149        scopes: vec![HashMap::new()],
150        local_count: 0,
151        expr_count: 0,
152        inference: Inference::default(),
153        function_return: Type::Unit,
154        deferred: false,
155        loop_depth: 0,
156    }
157}

159/// Keep recovery refusal located at the original operation, never at an inserted token.
160fn failure(site: &HoleSite, message: &str) -> Diagnostic {
161    Diagnostic::new(site.field(), message)
162}

164impl Checker<'_> {
165    /// Ordinary checker finalization never accepts an editor operation, even when unreachable.
166    pub(super) fn finalize_member_hole(&self, receiver: &mut ir::Expr, span: Span) -> Checked<()> {
167        if self.recovery.is_none() {
168            return Err(Diagnostic::new(
169                span,
170                "editor hole cannot enter ordinary finalization",
171            ));
172        }
173        self.finalize(receiver)
174    }
175
176    /// Resolve the receiver completely before creating any member-result inference variable.
177    pub(super) fn editor_member_hole(
178        &mut self,
179        expr: &ast::Expr,
180        depth: usize,
181    ) -> Checked<TypedKind> {
182        let ast::ExprKind::Field { value, .. } = &expr.kind else {
183            return Err(Diagnostic::new(expr.span, "invalid member hole"));
184        };
185        let receiver = self.expression(value, depth)?;
186        let concrete = self.inference.concrete(&receiver.ty, receiver.span)?;
187        if !nominal::generics([concrete.clone()]).is_empty() || concrete == Type::Never {
188            return Err(Diagnostic::new(
189                expr.span,
190                "member receiver is not independently concrete",
191            ));
192        }
193        let result = self.inference.fresh();
194        let state = self
195            .recovery
196            .as_mut()
197            .ok_or_else(|| Diagnostic::new(expr.span, "missing editor proof"))?;
198        if state.receiver.replace(concrete).is_some() {
199            return Err(failure(&state.site, "multiple editor member operations"));
200        }
201        state.result = Some(result.clone());
202        Ok((
203            ir::ExprKind::EditorHole {
204                token: ir::EditorHoleToken::new(),
205                receiver: Box::new(receiver),
206            },
207            result,
208        ))
209    }
210
211    /// Rigidify only unresolved classes reachable from this local result; no scheme is generalized.
212    pub(super) fn seal_member_hole(&mut self) -> Checked<()> {
213        let Some(state) = &self.recovery else {
214            return Ok(());
215        };
216        let result = state
217            .result
218            .clone()
219            .ok_or_else(|| failure(&state.site, "member hole was not checked"))?;
220        let span = state.site.field();
221        let result = self.inference.resolve(&result, span)?;
222        let variables = unresolved(&result, span)?;
223        if !variables.is_empty() {
224            self.inference.template = true;
225            for (index, variable) in variables.into_iter().enumerate() {
226                let name = format!("$editor_member_hole_{index}");
227                self.inference.template_names.insert(name.clone());
228                self.inference.unify(
229                    &Type::Infer(variable),
230                    &Type::Generic(name),
231                    span,
232                    "editor hole result",
233                )?;
234            }
235        }
236        Ok(())
237    }
238}

240/// Collect only unresolved result classes under the ordinary structural type budget.
241fn unresolved(ty: &Type, span: Span) -> Checked<Vec<u32>> {
242    let mut pending = vec![ty];
243    let mut variables = std::collections::BTreeSet::new();
244    let mut nodes = 0;
245    while let Some(ty) = pending.pop() {
246        nodes += 1;
247        if nodes > MAX_TYPE_NODES {
248            return Err(Diagnostic::new(span, "editor hole type limit exceeded"));
249        }
250        match ty {
251            Type::Infer(id) => {
252                variables.insert(*id);
253            }
254            Type::Function(args, result) => {
255                pending.extend(args);
256                pending.push(result);
257            }
258            Type::Union(args) | Type::Tuple(args) | Type::Named(_, args) => pending.extend(args),
259            Type::List(x) | Type::Option(x) => pending.push(x),
260            Type::ActorFunction(a, b) | Type::Map(a, b) | Type::Result(a, b) => {
261                pending.push(a);
262                pending.push(b);
263            }
264            _ => {}
265        }
266    }
267    Ok(variables.into_iter().collect())
268}
