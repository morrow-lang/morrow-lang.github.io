28    /// Validate all forward declarations before checking individual fields.
29    pub(super) fn new(program: &ast::Program) -> Checked<Self> {
30        let mut result = Self {
31            traits: super::traits::Registry::new(program)?,
32            codec_predicate_work: std::cell::Cell::new(0),
33            codec_template_work: std::cell::Cell::new(0),
34            codec_work: std::cell::Cell::new(0),
35            codec_plans: Default::default(),
36            codec_aliases: program
37                .aliases
38                .iter()
39                .map(|a| (a.name.clone(), (a.parameters.clone(), a.target.clone())))
40                .collect(),
41            newtype_definitions: std::rc::Rc::new(
42                program
43                    .newtypes
44                    .iter()
45                    .map(|d| (d.name.clone(), (d.parameters.clone(), d.inner.clone())))
46                    .collect(),
47            ),
48            newtype_work: std::rc::Rc::new(std::cell::Cell::new(0)),
49            newtypes: program
50                .newtypes
51                .iter()
52                .map(|decl| decl.name.clone())
53                .collect(),
54            aliases: program
55                .aliases
56                .iter()
57                .map(|alias| alias.name.clone())
58                .collect(),
59            declarations: HashMap::new(),
60            constructors: HashMap::new(),
61        };
62        let mut declarations = program.types.clone();
63        declarations.extend(
64            program
65                .newtypes
66                .iter()
67                .map(newtypes::declaration)
68                .collect::<Checked<Vec<_>>>()?,
69        );
70        for decl in &declarations {
71            if super::reserved(&decl.name)
72                || result
73                    .declarations
74                    .insert(decl.name.clone(), decl.clone())
75                    .is_some()
76            {
77                return Err(Diagnostic::new(
78                    decl.span,
79                    "duplicate or reserved type name",
80                ));
81            }
82        }
83        result.register_processes();
84        result.register_set();
85        result.register_foreign();
86        for decl in &declarations {
87            result.declaration(decl)?;
88        }
89        result.validate_newtypes(program)?;
90        Ok(result)
91    }

98    /// Check a declaration's shape, parameters, fields and constructor namespace.
99    fn declaration(&mut self, decl: &ast::TypeDecl) -> Checked<()> {
100        let allowed: HashSet<_> = decl.parameters.iter().cloned().collect();
101        if allowed.len() != decl.parameters.len() || decl.variants.is_empty() {
102            return Err(Diagnostic::new(
103                decl.span,
104                "duplicate type parameter or empty type declaration",
105            ));
106        }
107        if decl.record && decl.variants.len() != 1 {
108            return Err(Diagnostic::new(
109                decl.span,
110                "record must have exactly one constructor",
111            ));
112        }
113        let mut variants = HashSet::new();
114        for (tag, variant) in decl.variants.iter().enumerate() {
115            if !variants.insert(&variant.name) {
116                return Err(Diagnostic::new(variant.span, "duplicate constructor"));
117            }
118            let mut fields = HashSet::new();
119            for field in &variant.fields {
120                self.validate(&field.ty, &allowed, field.span)?;
121                if let Some(name) = &field.name {
122                    if !fields.insert(name) {
123                        return Err(Diagnostic::new(field.span, "duplicate record field"));
124                    }
125                } else if decl.record {
126                    return Err(Diagnostic::new(field.span, "record fields require names"));
127                }
128            }
129            self.add_constructor(&variant.name, &decl.name, tag, variant.span)?;
130            let leaf = variant.name.rsplit('.').next().unwrap_or(&variant.name);
131            let qualified = format!("{}.{leaf}", decl.name);
132            if qualified != variant.name {
133                self.add_constructor(&qualified, &decl.name, tag, variant.span)?;
134            }
135            if decl.record && variant.name != decl.name {
136                self.add_constructor(&decl.name, &decl.name, tag, variant.span)?;
137            }
138        }
139        Ok(())
140    }

142    /// Insert one constructor alias while rejecting ambiguous or builtin names.
143    fn add_constructor(&mut self, name: &str, owner: &str, tag: usize, span: Span) -> Checked<()> {
144        if super::reserved(name) {
145            return Err(Diagnostic::new(span, "constructor name is reserved"));
146        }
147        if let Some(existing) = self.constructors.get(name) {
148            if existing != &(owner.to_owned(), tag) {
149                return Err(Diagnostic::new(span, "ambiguous duplicate constructor"));
150            }
151        } else {
152            self.constructors.insert(name.into(), (owner.into(), tag));
153        }
154        Ok(())
155    }

157    /// Validate nominal arities and generic names in an already bounded annotation.
158    pub(super) fn validate(&self, ty: &Type, allowed: &HashSet<String>, span: Span) -> Checked<()> {
159        validate_type(ty, span)?;
160        let mut pending = vec![ty];
161        while let Some(ty) = pending.pop() {
162            match ty {
163                Type::Named(name, args) => {
164                    let decl = self.declarations.get(name).ok_or_else(|| {
165                        Diagnostic::new(
166                            span,
167                            format!("unknown type '{name}'{}", self.type_hint(name)),
168                        )
169                    })?;
170                    if args.len() != decl.parameters.len() {
171                        return Err(Diagnostic::new(
172                            span,
173                            format!(
174                                "type '{name}' expects {} type arguments",
175                                decl.parameters.len()
176                            ),
177                        ));
178                    }
179                    if name == "Set" {
180                        super::maps::validate_key(
181                            &self.representation(&args[0], span)?,
182                            span,
183                            true,
184                        )?;
185                    }
186                    pending.extend(args);
187                }
188                Type::Generic(name) if !allowed.contains(name) => {
189                    return Err(Diagnostic::new(
190                        span,
191                        format!("undeclared generic type '{name}'"),
192                    ));
193                }
194                Type::Function(args, result) => {
195                    pending.extend(args);
196                    pending.push(result);
197                }
198                Type::Union(args) => {
199                    crate::unions::charge(&self.newtype_work, ty, span)?;
200                    pending.extend(args);
201                }
202                Type::Tuple(args) => pending.extend(args),
203                Type::Pid(t) | Type::List(t) | Type::Option(t) => pending.push(t),
204                Type::Map(a, b) => {
205                    super::maps::validate_key(&self.representation(a, span)?, span, true)?;
206                    pending.extend([a.as_ref(), b.as_ref()]);
207                }
208                Type::Result(a, b) => {
209                    pending.push(a);
210                    pending.push(b);
211                }
212                _ => {}
213            }
214        }
215        Ok(())
216    }

223    /// Nominal type spellings, including builtin types, for unresolved-annotation suggestions.
224    pub(super) fn type_hint(&self, name: &str) -> String {
225        const BUILTIN_TYPES: &[&str] = &[
226            "Int", "Float", "Bool", "String", "List", "Map", "Set", "Option", "Result", "Pid",
227        ];
228        crate::suggest::hint(
229            crate::suggest::tail(name),
230            self.declarations
231                .keys()
232                .chain(self.aliases.iter())
233                .chain(self.newtypes.iter())
234                .map(|key| crate::suggest::tail(key))
235                .chain(BUILTIN_TYPES.iter().copied()),
236        )
237    }

258    /// Instantiate one nominal layout without recursively expanding referenced types.
259    pub(super) fn layout(&self, ty: &Type, span: Span) -> Checked<ir::TypeLayout> {
260        validate_layout_type(ty, span)?;
261        let Type::Named(name, args) = ty else {
262            return Err(Diagnostic::new(span, "field access requires a record type"));
263        };
264        let decl = self.declarations.get(name).ok_or_else(|| {
265            Diagnostic::new(
266                span,
267                format!("unknown type '{name}'{}", self.type_hint(name)),
268            )
269        })?;
270        if args.len() != decl.parameters.len() {
271            return Err(Diagnostic::new(span, "wrong nominal type argument count"));
272        }
273        let substitutions = decl
274            .parameters
275            .iter()
276            .cloned()
277            .zip(args.iter().cloned())
278            .collect();
279        let variants = decl
280            .variants
281            .iter()
282            .map(|v| {
283                v.fields
284                    .iter()
285                    .map(|f| substitute(&f.ty, &substitutions))
286                    .collect::<Checked<Vec<_>>>()
287            })
288            .collect::<Checked<Vec<_>>>()?;
289        let fields = if decl.record {
290            decl.variants[0]
291                .fields
292                .iter()
293                .map(|f| f.name.clone().unwrap_or_default())
294                .collect()
295        } else {
296            Vec::new()
297        };
298        Ok(ir::TypeLayout {
299            storage: if self.is_newtype(ty) {
300                ir::LayoutStorage::Unboxed
301            } else {
302                ir::LayoutStorage::Tagged
303            },
304            ty: ty.clone(),
305            variants,
306            variant_names: if decl.record || self.is_newtype(ty) {
307                Vec::new()
308            } else {
309                decl.variants
310                    .iter()
311                    .map(|v| v.name.rsplit('.').next().unwrap().to_owned())
312                    .collect()
313            },
314            fields,
315        })
316    }

318    /// Instantiate requirement fields incrementally, charging input and expanded substitutions.
319    pub(super) fn requirement_fields(
320        &self,
321        ty: &Type,
322        inference: &super::Inference,
323        span: Span,
324    ) -> Checked<Vec<Type>> {
325        validate_layout_type(ty, span)?;
326        let Type::Named(name, args) = ty else {
327            return Err(Diagnostic::new(
328                span,
329                "nominal requirements require a named type",
330            ));
331        };
332        let decl = self
333            .declarations
334            .get(name)
335            .ok_or_else(|| Diagnostic::new(span, "unknown nominal requirement type"))?;
336        if args.len() != decl.parameters.len() {
337            return Err(Diagnostic::new(span, "wrong nominal type argument count"));
338        }
339        let values = decl
340            .parameters
341            .iter()
342            .cloned()
343            .zip(args.iter().cloned())
344            .collect();
345        let mut fields = Vec::new();
346        for field in decl.variants.iter().flat_map(|variant| &variant.fields) {
347            super::returns::charge_output(inference, &field.ty, span)?;
348            let ty = substitute(&field.ty, &values)?;
349            super::returns::charge_output(inference, &ty, span)?;
350            fields.push(ty);
351        }
352        Ok(fields)
353    }

355    /// Require an identified nominal record before any field-level update constraints.
356    pub(super) fn record_shape(&self, ty: &Type, span: Span) -> Checked<()> {
357        if matches!(ty, Type::Named(name, _) if name == "Ptr") {
358            return Err(Diagnostic::new(
359                span,
360                "Ptr storage is private; use the Ptr APIs",
361            ));
362        }
363        let Type::Named(name, args) = ty else {
364            return Err(Diagnostic::new(span, "field access requires a record type"));
365        };
366        let decl = self
367            .declarations
368            .get(name)
369            .ok_or_else(|| Diagnostic::new(span, "unknown record type"))?;
370        if !decl.record || decl.parameters.len() != args.len() {
371            return Err(Diagnostic::new(span, "field access requires a record type"));
372        }
373        Ok(())
374    }

376    /// Resolve only the requested field while charging names and generic substitutions.
377    pub(super) fn project_field(
378        &self,
379        ty: &Type,
380        name: &str,
381        inference: &Inference,
382        span: Span,
383    ) -> Checked<(usize, Type)> {
384        if matches!(ty, Type::Named(owner, _) if matches!(owner.as_str(), "Set" | "Ptr")) {
385            return Err(Diagnostic::new(
386                span,
387                "collection or pointer storage is private; use its public APIs",
388            ));
389        }
390        if self.is_newtype(ty) {
391            if name != "0" {
392                return Err(Diagnostic::new(span, "newtype field must be .0"));
393            }
394            let inner = self.newtype_inner(ty, span)?;
395            returns::charge_output(inference, &inner, span)?;
396            return Ok((0, inner));
397        }
398        self.record_shape(ty, span)?;
399        let Type::Named(owner, args) = ty else {
400            unreachable!("validated record shape")
401        };
402        let decl = &self.declarations[owner];
403        for arg in args {
404            returns::charge_output(inference, arg, span)?;
405        }
406        let substitutions = decl
407            .parameters
408            .iter()
409            .cloned()
410            .zip(args.iter().cloned())
411            .collect();
412        for (index, field) in decl.variants[0].fields.iter().enumerate() {
413            for _ in 0..field.name.as_ref().map_or(0, String::len) + 1 {
414                returns::charge(inference, span)?;
415            }
416            if field.name.as_deref() != Some(name) {
417                continue;
418            }
419            returns::charge_output(inference, &field.ty, span)?;
420            let result = substitute(&field.ty, &substitutions)?;
421            returns::charge_output(inference, &result, span)?;
422            return Ok((index, result));
423        }
424        let declared = decl.variants[0]
425            .fields
426            .iter()
427            .filter_map(|field| field.name.as_deref());
428        Err(Diagnostic::new(
429            span,
430            format!(
431                "unknown record field '{name}'{}",
432                crate::suggest::hint(name, declared)
433            ),
434        ))
435    }

437    /// Resolve constructor payload types for builtin and nominal sums.
438    pub(super) fn variants(&self, ty: &Type, span: Span) -> Checked<Vec<Vec<Type>>> {
439        match ty {
440            Type::Union(members) => Ok(members.iter().map(|ty| vec![ty.clone()]).collect()),
441            Type::Tuple(fields) => Ok(vec![fields.clone()]),
442            Type::Option(a) => Ok(vec![vec![(**a).clone()], vec![]]),
443            Type::Result(a, b) => Ok(vec![vec![(**a).clone()], vec![(**b).clone()]]),
444            Type::Named(..) => Ok(self.layout(ty, span)?.variants),
445            _ => Err(Diagnostic::new(
446                span,
447                "constructor pattern requires a sum or record type",
448            )),
449        }
450    }

452    /// Find hidden Result values while terminating on ordinary recursive nominal types.
453    pub(super) fn contains_result(&self, ty: &Type) -> Checked<bool> {
454        let mut pending = vec![ty.clone()];
455        let mut seen = HashSet::new();
456        while let Some(ty) = pending.pop() {
457            if seen.len() >= MAX_TYPE_NODES {
458                return Err(Diagnostic::new(
459                    Span::default(),
460                    "type expansion limit exceeded",
461                ));
462            }
463            if !seen.insert(ty.clone()) {
464                continue;
465            }
466            match ty {
467                Type::Result(..) => return Ok(true),
468                Type::Map(key, value) => {
469                    pending.push(*key);
470                    pending.push(*value);
471                }
472                Type::Union(fields) | Type::Tuple(fields) => pending.extend(fields),
473                Type::List(a) | Type::Option(a) => pending.push(*a),
474                Type::Named(..) => pending.extend(
475                    self.layout(&ty, Span::default())?
476                        .variants
477                        .into_iter()
478                        .flatten(),
479                ),
480                _ => {}
481            }
482        }
483        Ok(false)
484    }

486    /// Collect every concrete layout reachable from emitted function types and values.
487    pub(super) fn layouts(&self, functions: &[ir::Function]) -> Checked<Vec<ir::TypeLayout>> {
488        let mut pending = Vec::new();
489        for function in functions {
490            pending.push(function.return_type.clone());
491            pending.extend(function.mailbox.iter().cloned());
492            pending.extend(function.captures.iter().map(|p| p.ty.clone()));
493            pending.extend(function.params.iter().map(|p| p.ty.clone()));
494            let mut expressions = vec![&function.body];
495            while let Some(expr) = expressions.pop() {
496                pending.push(expr.ty.clone());
497                if let ir::ExprKind::Actor(ir::ActorExpr::Receive { view, mailbox, .. }) =
498                    &expr.kind
499                {
500                    pending.push(view.item(mailbox));
501                }
502                expressions.extend(children(expr));
503            }
504        }
505        let mut seen = HashSet::new();
506        let mut layouts = Vec::new();
507        while let Some(ty) = pending.pop() {
508            if !seen.insert(ty.clone()) {
509                continue;
510            }
511            if seen.len() > MAX_TYPE_NODES {
512                return Err(Diagnostic::new(
513                    Span::default(),
514                    "concrete type layout expansion limit exceeded",
515                ));
516            }
517            match &ty {
518                Type::Named(_, args) => {
519                    let layout = self.layout(&ty, Span::default())?;
520                    pending.extend(args.iter().cloned());
521                    pending.extend(layout.variants.iter().flatten().cloned());
522                    layouts.push(layout);
523                }
524                Type::Function(args, result) => {
525                    pending.extend(args.iter().cloned());
526                    pending.push((**result).clone());
527                }
528                Type::Union(fields) | Type::Tuple(fields) => pending.extend(fields.iter().cloned()),
529                Type::Pid(a) | Type::List(a) | Type::Option(a) => pending.push((**a).clone()),
530                Type::ActorFunction(a, b) | Type::Result(a, b) | Type::Map(a, b) => {
531                    pending.push((**a).clone());
532                    pending.push((**b).clone());
533                }
534                _ => {}
535            }
536        }
537        Ok(layouts)
538    }
539}

546/// Collect generic parameter names in deterministic order from bounded annotations.
547pub(super) fn generics(types: impl IntoIterator<Item = Type>) -> Vec<String> {
548    let mut names = BTreeSet::new();
549    let mut pending: Vec<_> = types.into_iter().collect();
550    while let Some(ty) = pending.pop() {
551        match ty {
552            Type::Generic(n) => {
553                names.insert(n);
554            }
555            Type::Function(args, result) => {
556                pending.extend(args);
557                pending.push(*result);
558            }
559            Type::Union(args) | Type::Tuple(args) | Type::Named(_, args) => pending.extend(args),
560            Type::Pid(a) | Type::List(a) | Type::Option(a) => pending.push(*a),
561            Type::ActorFunction(a, b) | Type::Result(a, b) | Type::Map(a, b) => {
562                pending.push(*a);
563                pending.push(*b);
564            }
565            _ => {}
566        }
567    }
568    names.into_iter().collect()
569}

571/// Substitute generic annotations with a strict bound on the expanded result.
572pub(super) fn substitute(ty: &Type, values: &HashMap<String, Type>) -> Checked<Type> {
573    let mut budget = MAX_TYPE_NODES;
574    substitute_inner(ty, values, 0, &mut budget, true)
575}

577/// Charge every expanded type node, including repeated generic substitutions.
578fn substitute_inner(
579    ty: &Type,
580    values: &HashMap<String, Type>,
581    depth: usize,
582    budget: &mut usize,
583    expand: bool,
584) -> Checked<Type> {
585    if depth >= MAX_TYPE_DEPTH || *budget == 0 {
586        return Err(Diagnostic::new(
587            Span::default(),
588            "generic type expansion limit exceeded",
589        ));
590    }
591    *budget -= 1;
592    Ok(match ty {
593        Type::Generic(n) if expand => match values.get(n) {
594            Some(value) => substitute_inner(value, values, depth, budget, false)?,
595            None => ty.clone(),
596        },
597        Type::Function(args, result) => Type::Function(
598            substitute_fields(args, values, depth, budget, expand)?,
599            Box::new(substitute_inner(result, values, depth + 1, budget, expand)?),
600        ),
601        Type::Union(args) => crate::unions::make(
602            substitute_fields(args, values, depth, budget, expand)?,
603            Span::default(),
604        )?,
605        Type::Tuple(args) => Type::Tuple(substitute_fields(args, values, depth, budget, expand)?),
606        Type::Named(n, args) => Type::Named(
607            n.clone(),
608            substitute_fields(args, values, depth, budget, expand)?,
609        ),
610        Type::Pid(a) => Type::Pid(Box::new(substitute_inner(
611            a,
612            values,
613            depth + 1,
614            budget,
615            expand,
616        )?)),
617        Type::ActorFunction(a, b) => Type::ActorFunction(
618            Box::new(substitute_inner(a, values, depth + 1, budget, expand)?),
619            Box::new(substitute_inner(b, values, depth + 1, budget, expand)?),
620        ),
621        Type::List(a) => Type::List(Box::new(substitute_inner(
622            a,
623            values,
624            depth + 1,
625            budget,
626            expand,
627        )?)),
628        Type::Option(a) => Type::Option(Box::new(substitute_inner(
629            a,
630            values,
631            depth + 1,
632            budget,
633            expand,
634        )?)),
635        Type::Map(a, b) => Type::Map(
636            Box::new(substitute_inner(a, values, depth + 1, budget, expand)?),
637            Box::new(substitute_inner(b, values, depth + 1, budget, expand)?),
638        ),
639        Type::Result(a, b) => Type::Result(
640            Box::new(substitute_inner(a, values, depth + 1, budget, expand)?),
641            Box::new(substitute_inner(b, values, depth + 1, budget, expand)?),
642        ),
643        _ => ty.clone(),
644    })
645}

647/// Substitute ordered fields without resetting the enclosing expansion budget.
648fn substitute_fields(
649    args: &[Type],
650    values: &HashMap<String, Type>,
651    depth: usize,
652    budget: &mut usize,
653    expand: bool,
654) -> Checked<Vec<Type>> {
655    args.iter()
656        .map(|a| substitute_inner(a, values, depth + 1, budget, expand))
657        .collect()
658}

662    template: &Type,
663    actual: &Type,
664    values: &mut HashMap<String, Type>,
665    depth: usize,
666) -> Checked<()> {
667    if depth >= MAX_TYPE_DEPTH {
668        return Err(Diagnostic::new(
669            Span::default(),
670            "generic specialization type depth exceeded",
671        ));
672    }
673    if matches!(template, Type::Union(_)) {
674        return super::unions::capture(template, actual, values, depth);
675    }
676    match (template, actual) {
677        (Type::Generic(n), ty) => {
678            if let Some(previous) = values.insert(n.clone(), ty.clone())
679                && previous != *ty
680            {
681                return Err(Diagnostic::new(
682                    Span::default(),
683                    "inconsistent generic specialization",
684                ));
685            }
686        }
687        (Type::Named(n, a), Type::Named(m, b)) if n == m && a.len() == b.len() => {
688            for (a, b) in a.iter().zip(b) {
689                capture(a, b, values, depth + 1)?;
690            }
691        }
692        (Type::Tuple(a), Type::Tuple(b)) if a.len() == b.len() => {
693            for (a, b) in a.iter().zip(b) {
694                capture(a, b, values, depth + 1)?;
695            }
696        }
697        (Type::Function(a, ar), Type::Function(b, br)) if a.len() == b.len() => {
698            for (a, b) in a.iter().zip(b) {
699                capture(a, b, values, depth + 1)?;
700            }
701            capture(ar, br, values, depth + 1)?;
702        }
703        (Type::Pid(a), Type::Pid(b))
704        | (Type::List(a), Type::List(b))
705        | (Type::Option(a), Type::Option(b)) => capture(a, b, values, depth + 1)?,
706        (Type::ActorFunction(a, b), Type::ActorFunction(c, d))
707        | (Type::Result(a, b), Type::Result(c, d))
708        | (Type::Map(a, b), Type::Map(c, d)) => {
709            capture(a, c, values, depth + 1)?;
710            capture(b, d, values, depth + 1)?;
711        }
712        _ if template == actual => {}
713        _ => {
714            return Err(Diagnostic::new(
715                Span::default(),
716                "invalid concrete generic specialization",
717            ));
718        }
719    }
720    Ok(())
721}

723/// Bound concrete/inferred layout arguments before substituting recursive fields.
724fn validate_layout_type(ty: &Type, span: Span) -> Checked<()> {
725    let mut pending = vec![(ty, 0)];
726    let mut nodes = 0;
727    while let Some((ty, depth)) = pending.pop() {
728        nodes += 1;
729        if depth >= MAX_TYPE_DEPTH || nodes > MAX_TYPE_NODES {
730            return Err(Diagnostic::new(
731                span,
732                "nominal type expansion limit exceeded",
733            ));
734        }
735        match ty {
736            Type::Union(args) | Type::Tuple(args) | Type::Named(_, args) => {
737                pending.extend(args.iter().map(|t| (t, depth + 1)))
738            }
739            Type::Pid(a) | Type::List(a) | Type::Option(a) => pending.push((a, depth + 1)),
740            Type::ActorFunction(a, b) | Type::Result(a, b) | Type::Map(a, b) => {
741                pending.push((a, depth + 1));
742                pending.push((b, depth + 1));
743            }
744            _ => {}
745        }
746    }
747    Ok(())
748}
