5pub(super) const WAITING: &str = "return inference is waiting for a dependency";
6/// A call-site result stays independent until its callee's generic scheme is known.
7pub(super) struct DeferredCall {
8    target: Type,
9    substitutions: HashMap<String, Type>,
10    result: Type,
11    template: bool,
12    span: Span,
13}

15type Prepared<'a> = (Cow<'a, ast::Program>, HashMap<String, Signature>, usize);

17/// Reserve one shared result variable per private definition lacking an annotation.
18pub(super) fn initial_result(function: &ast::Function, inference: &mut Inference) -> Checked<Type> {
19    if function.name != "main" && !function.public && function.return_type.is_none() {
20        Ok(inference.fresh())
21    } else {
22        function_result(function)
23    }
24}

26/// Accept the documented entry contracts without allowing polymorphic process results.
27pub(super) fn main_result(ty: &Type) -> bool {
28    match ty {
29        Type::Int | Type::Unit => true,
30        Type::Result(ok, error) => {
31            **ok == Type::Unit
32                && nominal::generics([*error.clone()]).is_empty()
33                && !has_infer(error)
34        }
35        _ => false,
36    }
37}

39/// Solve shared return constraints, then publish annotations for the existing monomorphizer.
40pub(super) fn resolve<'a>(
41    program: &'a ast::Program,
42    registry: &nominal::Registry,
43    dispatch: &HashSet<String>,
44) -> Checked<Prepared<'a>> {
45    let mut inference = Inference {
46        newtypes: registry.newtype_definitions(),
47        newtype_work: registry.newtype_budget(),
48        ..Inference::default()
49    };
50    let mut signatures = super::signatures(program, registry, &mut inference, dispatch)?;
51    let missing: Vec<_> = program
52        .functions
53        .iter()
54        .enumerate()
55        .filter(|(_, f)| f.return_type.is_none() && f.name != "main")
56        .map(|(i, _)| i)
57        .collect();
58    if missing.is_empty() {
59        return Ok((
60            Cow::Borrowed(program),
61            signatures,
62            inference.probe_work.get(),
63        ));
64    }
65    solve(program, registry, &mut signatures, &mut inference, &missing)?;
66    let mut source = program.clone();
67    for index in missing {
68        let function = &mut source.functions[index];
69        let result = inference.resolve(&signatures[&function.name].result, function.span)?;
70        if has_infer(&result) {
71            return Err(unresolved(function));
72        }
73        validate_type(&result, function.span)?;
74        let allowed = signatures[&function.name]
75            .generics
76            .iter()
77            .cloned()
78            .collect();
79        registry.validate(&result, &allowed, function.span)?;
80        signatures
81            .get_mut(&function.name)
82            .expect("collected function")
83            .result = result.clone();
84        function.return_type = Some(result);
85    }
86    Ok((Cow::Owned(source), signatures, inference.probe_work.get()))
87}

89/// Retry only shape-dependent inference, stopping after each definition can succeed once.
90pub(super) fn solve(
91    program: &ast::Program,
92    registry: &nominal::Registry,
93    signatures: &mut HashMap<String, Signature>,
94    inference: &mut Inference,
95    missing: &[usize],
96) -> Checked<()> {
97    let mut pending = missing.to_vec();
98    for _ in 0..=missing.len() {
99        if pending.is_empty() {
100            settle_calls(inference)?;
101            return Ok(());
102        }
103        let previous = pending.len();
104        let mut waiting = Vec::new();
105        for index in pending {
106            let function = &program.functions[index];
107            match probe(function, registry, signatures, inference) {
108                Ok(_) => {}
109                Err(error) if error.message.contains(WAITING) => waiting.push(index),
110                Err(error) => return Err(error),
111            }
112        }
113        let settled = settle_calls(inference)?;
114        if waiting.len() == previous && !settled {
115            return Err(unresolved(&program.functions[waiting[0]]));
116        }
117        pending = waiting;
118    }
119    Err(Diagnostic::new(
120        Span::default(),
121        "return inference work limit exceeded",
122    ))
123}

125/// Check one body with shared recursive result slots, postponing concrete IR validation.
126pub(super) fn probe(
127    function: &ast::Function,
128    registry: &nominal::Registry,
129    signatures: &HashMap<String, Signature>,
130    inference: &mut Inference,
131) -> Checked<ir::Expr> {
132    let signature = &signatures[&function.name];
133    let mut checker = Checker {
134        mailbox: signature.mailbox.clone(),
135        editor: None,
136        recovery: None,
137        signatures,
138        registry,
139        scopes: vec![HashMap::new()],
140        local_count: 0,
141        expr_count: 0,
142        inference: std::mem::take(inference),
143        function_return: signature.result.clone(),
144        deferred: false,
145        loop_depth: 0,
146    };
147    checker.inference.probing = true;
148    checker.inference.template = !signature.generics.is_empty();
149    checker.inference.template_names = signature.generics.iter().cloned().collect();
150    for param in &function.params {
151        checker.bind(
152            clauses::parameter_name(param),
153            clauses::parameter_type(param).clone(),
154        );
155    }
156    let expected = if function.name == "main" && signature.result == Type::Unit {
157        None
158    } else {
159        Some(&signature.result)
160    };
161    let result = checker
162        .expression_expected(&function.body, expected, 0)
163        .map_err(|error| closures::context(error, "function return"));
164    *inference = checker.inference;
165    inference.probing = false;
166    inference.template = false;
167    inference.template_names.clear();
168    result
169}

171/// Instantiate a known scheme or defer a fresh call result without leaking generic names.
172pub(super) fn call_result(
173    inference: &mut Inference,
174    signature: &Signature,
175    substitutions: &HashMap<String, Type>,
176    span: Span,
177) -> Checked<Type> {
178    let result = inference.resolve(&signature.result, span)?;
179    if inference.probing
180        && (inference.template || !signature.generics.is_empty())
181        && has_infer(&result)
182    {
183        let result = inference.fresh();
184        inference.pending_returns.push(DeferredCall {
185            target: signature.result.clone(),
186            substitutions: substitutions.clone(),
187            result: result.clone(),
188            template: inference.template,
189            span,
190        });
191        Ok(result)
192    } else {
193        nominal::substitute(&result, substitutions)
194    }
195}

197/// Publish newly anchored schemes into independent call variables within the aggregate budget.
198pub(super) fn settle_calls(inference: &mut Inference) -> Checked<bool> {
199    let previous = std::mem::replace(&mut inference.settling, true);
200    let result = settle_inner(inference);
201    inference.settling = previous;
202    result
203}
204
205/// Iterate only deferred obligations whose signatures gained concrete structure.
206fn settle_inner(inference: &mut Inference) -> Checked<bool> {
207    let mut changed = false;
208    for _ in 0..=inference.pending_returns.len() {
209        let calls = std::mem::take(&mut inference.pending_returns);
210        let previous = calls.len();
211        for call in calls {
212            charge_work(inference, call.span)?;
213            let target = inference.resolve(&call.target, call.span)?;
214            if has_infer(&target) {
215                inference.pending_returns.push(call);
216                continue;
217            }
218            let target = nominal::substitute(&target, &call.substitutions)?;
219            let previous_template = std::mem::replace(&mut inference.template, call.template);
220            let result = inference.unify(&call.result, &target, call.span, "inferred call return");
221            inference.template = previous_template;
222            result?;
223        }
224        if inference.pending_returns.len() == previous {
225            return Ok(changed);
226        }
227        changed = true;
228    }
229    Ok(changed)
230}

232/// Delay projections whose receiver shape may be established by a later definition.
233pub(super) fn shape_ready(inference: &Inference, ty: &Type, span: Span) -> Checked<()> {
234    if inference.probing && matches!(inference.resolve(ty, span)?, Type::Infer(_)) {
235        Err(Diagnostic::new(span, WAITING))
236    } else {
237        Ok(())
238    }
239}

241/// Distinguish unresolved payload variables from valid declared generic names.
242pub(super) fn has_infer(ty: &Type) -> bool {
243    let mut pending = vec![ty];
244    while let Some(ty) = pending.pop() {
245        match ty {
246            Type::Infer(_) => return true,
247            Type::List(a) | Type::Option(a) => pending.push(a),
248            Type::ActorFunction(a, b) | Type::Result(a, b) | Type::Map(a, b) => {
249                pending.push(a);
250                pending.push(b);
251            }
252            Type::Function(args, result) => {
253                pending.extend(args);
254                pending.push(result);
255            }
256            Type::Union(args) | Type::Tuple(args) | Type::Named(_, args) => pending.extend(args),
257            _ => {}
258        }
259    }
260    false
261}

263/// Explain a missing inference anchor without inventing a default payload type.
264fn unresolved(function: &ast::Function) -> Diagnostic {
265    Diagnostic::new(
266        function.span,
267        format!(
268            "cannot infer return type of '{}'; add a return type annotation to resolve recursive or generic dependencies",
269            function.name
270        ),
271    )
272}

274impl Checker<'_> {
275    /// Preserve symbolic numeric domains while inferring a private return scheme.
276    pub(super) fn numeric_domain(&self, op: ast::BinaryOp, ty: &Type, span: Span) -> Checked<Type> {
277        let ty = self.inference.resolve(ty, span)?;
278        if op != ast::BinaryOp::Remainder {
279            if ty == Type::Float {
280                return Ok(Type::Float);
281            }
282            if self.inference.probing && matches!(ty, Type::Infer(_) | Type::Generic(_)) {
283                return Ok(ty);
284            }
285        }
286        Ok(Type::Int)
287    }
288}

290/// Bound total work and fresh-variable storage across all dependency retries.
291pub(super) fn charge(inference: &Inference, span: Span) -> Checked<()> {
292    if inference.probing || inference.settling || inference.whole_signature {
293        charge_work(inference, span)?;
294    }
295    Ok(())
296}

298/// Charge delayed constraints as well as expression visits to avoid quadratic unbounded work.
299fn charge_work(inference: &Inference, span: Span) -> Checked<()> {
300    inference.probe_work.set(inference.probe_work.get() + 1);
301    if inference.probe_work.get() > MAX_EXPR_COUNT * 4
302        || inference.bindings.len() > MAX_EXPR_COUNT * 4
303    {
304        return Err(Diagnostic::new(
305            span,
306            "return inference work limit exceeded; add return type annotations to dependency chains",
307        ));
308    }
309    Ok(())
310}

312/// Charge expanded expression types, including nominal field projections, before retaining IR.
313pub(super) fn charge_output(inference: &Inference, ty: &Type, span: Span) -> Checked<()> {
314    if !inference.probing && !inference.whole_signature {
315        return Ok(());
316    }
317    let mut pending = vec![ty];
318    while let Some(ty) = pending.pop() {
319        charge_work(inference, span)?;
320        match ty {
321            Type::List(a) | Type::Option(a) => pending.push(a),
322            Type::ActorFunction(a, b) | Type::Result(a, b) | Type::Map(a, b) => {
323                pending.push(a);
324                pending.push(b);
325            }
326            Type::Function(args, result) => {
327                pending.extend(args);
328                pending.push(result);
329            }
330            Type::Union(args) | Type::Tuple(args) | Type::Named(_, args) => pending.extend(args),
331            _ => {}
332        }
333    }
334    Ok(())
335}
