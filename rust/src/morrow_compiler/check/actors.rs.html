4/// Infer mailbox schemes from owned receive patterns, never from an arbitrary scalar witness.
5pub(super) fn attach(
6    program: &ast::Program,
7    registry: &nominal::Registry,
8    signatures: &mut HashMap<String, Signature>,
9    inference: &mut Inference,
10) -> Checked<()> {
11    for function in &program.functions {
12        let Some(mailbox) = mailbox(&function.body, registry)? else {
13            continue;
14        };
15        let signature = signatures
16            .get_mut(&function.name)
17            .ok_or_else(|| Diagnostic::new(function.span, "missing actor signature"))?;
18        signature.mailbox = Some(mailbox.clone());
19        for generic in nominal::generics([mailbox]) {
20            if !signature.generics.contains(&generic) {
21                signature.generics.push(generic);
22            }
23        }
24    }
25    if !signatures
26        .values()
27        .any(|signature| signature.mailbox.is_some())
28    {
29        return Ok(());
30    }
31    let graph = dependencies::owned_calls(program)?;
32    inference
33        .probe_work
34        .set(inference.probe_work.get().saturating_add(graph.work));
35    if inference.probe_work.get() > MAX_EXPR_COUNT * 4 {
36        return Err(Diagnostic::new(
37            Span::default(),
38            "actor effect inference work limit exceeded",
39        ));
40    }
41    // Propagate effect presence once along reverse lexical call edges. A reference to a
42    // receiving function, or creation of a receiving closure, does not suspend its owner.
43    let mut callers = vec![Vec::new(); graph.groups.len()];
44    let mut pending = Vec::new();
45    for (index, group) in graph.groups.iter().enumerate() {
46        if signatures[&group.name].mailbox.is_some() {
47            pending.push(index);
48        }
49        for &callee in &group.callees {
50            callers[callee].push(index);
51        }
52    }
53    while let Some(callee) = pending.pop() {
54        for &caller in &callers[callee] {
55            let signature = signatures
56                .get_mut(&graph.groups[caller].name)
57                .expect("registered caller");
58            if signature.mailbox.is_none() {
59                let generic = format!("$mailbox_call{caller}");
60                signature.mailbox = Some(Type::Generic(generic.clone()));
61                signature.generics.push(generic);
62                pending.push(caller);
63            }
64        }
65    }
66    // Infer callees before callers. Recursive components share existential mailbox slots
67    // until every body has contributed its constraints, then publish independent schemes.
68    for component in &graph.components {
69        let mut candidates = Vec::new();
70        for &index in component {
71            let group = &graph.groups[index];
72            let signature = signatures.get_mut(&group.name).expect("registered actor");
73            let Some(mailbox) = signature.mailbox.clone() else {
74                continue;
75            };
76            let variables: HashMap<_, _> = nominal::generics([mailbox.clone()])
77                .into_iter()
78                .filter(|name| name.starts_with("$mailbox"))
79                .map(|name| (name, inference.fresh()))
80                .collect();
81            if variables.is_empty() {
82                continue;
83            }
84            let candidate = nominal::substitute(&mailbox, &variables)?;
85            signature.mailbox = Some(candidate.clone());
86            signature
87                .generics
88                .retain(|name| !variables.contains_key(name));
89            candidates.push((index, candidate));
90        }
91        for &(index, _) in &candidates {
92            for function in &program.functions[graph.groups[index].clauses.clone()] {
93                let previous = (
94                    inference.whole_signature,
95                    inference.probing,
96                    inference.template,
97                );
98                inference.whole_signature = true;
99                let result = returns::probe(function, registry, signatures, inference);
100                inference.whole_signature = previous.0;
101                inference.probing = previous.1;
102                inference.template = previous.2;
103                if let Err(error) = result
104                    && !error.message.contains(returns::WAITING)
105                {
106                    return Err(error);
107                }
108            }
109        }
110        for (index, candidate) in candidates {
111            let group = &graph.groups[index];
112            let mailbox = generalize(&inference.resolve(&candidate, group.span)?);
113            let signature = signatures.get_mut(&group.name).expect("registered actor");
114            signature.mailbox = Some(mailbox.clone());
115            for generic in nominal::generics([mailbox]) {
116                if !signature.generics.contains(&generic) {
117                    signature.generics.push(generic);
118                }
119            }
120        }
121    }
122    Ok(())
123}

125/// Constrain all selective patterns together while excluding independently lifted lambda bodies.
126fn mailbox(body: &ast::Expr, registry: &nominal::Registry) -> Checked<Option<Type>> {
127    let mut pending = vec![body];
128    let mut patterns = Vec::new();
129    let mut actor_context = false;
130    let mut work = 0;
131    while let Some(expr) = pending.pop() {
132        work += 1;
133        if work > MAX_EXPR_COUNT {
134            return Err(Diagnostic::new(
135                expr.span,
136                "actor effect work limit exceeded",
137            ));
138        }
139        if matches!(expr.kind, ast::ExprKind::Lambda { .. }) {
140            continue;
141        }
142        if let ast::ExprKind::Receive { view, arms, .. } = &expr.kind {
143            actor_context = true;
144            patterns.extend(arms.iter().map(|a| (&a.pattern, *view)));
145        }
146        actor_context |= process_context(expr);
147        pending.extend(crate::actors::source_children(expr));
148    }
149    if !actor_context {
150        return Ok(None);
151    }
152    let signatures = HashMap::new();
153    let mut checker = Checker {
154        mailbox: None,
155        editor: None,
156        recovery: None,
157        signatures: &signatures,
158        registry,
159        scopes: vec![HashMap::new()],
160        local_count: 0,
161        expr_count: 0,
162        inference: Inference::default(),
163        function_return: Type::Unit,
164        deferred: false,
165        loop_depth: 0,
166    };
167    let ty = checker.inference.fresh();
168    parameters::constrain(
169        &mut checker,
170        patterns
171            .into_iter()
172            .map(|(p, view)| (p, view.item(&ty)))
173            .collect(),
174    )?;
175    let ty = checker.inference.resolve(&ty, body.span)?;
176    Ok(Some(generalize(&ty)))
177}

179/// Unconstrained mailbox components remain quantified identities until spawn supplies a type.
180fn generalize(ty: &Type) -> Type {
181    match ty {
182        Type::Infer(id) => Type::Generic(format!("$mailbox{id}")),
183        Type::Pid(t) => Type::Pid(Box::new(generalize(t))),
184        Type::List(t) => Type::List(Box::new(generalize(t))),
185        Type::Option(t) => Type::Option(Box::new(generalize(t))),
186        Type::Result(a, b) => Type::Result(Box::new(generalize(a)), Box::new(generalize(b))),
187        Type::Map(a, b) => Type::Map(Box::new(generalize(a)), Box::new(generalize(b))),
188        Type::Tuple(ts) => Type::Tuple(ts.iter().map(generalize).collect()),
189        Type::Union(ts) => Type::Union(ts.iter().map(generalize).collect()),
190        Type::Named(n, ts) => Type::Named(n.clone(), ts.iter().map(generalize).collect()),
191        _ => ty.clone(),
192    }
193}

196    /// Instantiate one receiving function value with a fresh mailbox scheme and ordinary signature.
197    pub(super) fn actor_name(&mut self, name: &str, span: Span) -> Checked<Option<TypedKind>> {
198        let Some(signature) = self.signatures.get(name) else {
199            return Ok(None);
200        };
201        let Some(mailbox) = &signature.mailbox else {
202            return Ok(None);
203        };
204        let values: HashMap<_, _> = signature
205            .generics
206            .iter()
207            .map(|n| (n.clone(), self.inference.fresh()))
208            .collect();
209        let params = signature
210            .params
211            .iter()
212            .map(|t| nominal::substitute(t, &values))
213            .collect::<Checked<Vec<_>>>()?;
214        let result = returns::call_result(&mut self.inference, signature, &values, span)?;
215        let mailbox = nominal::substitute(mailbox, &values)?;
216        self.inference
217            .call_names
218            .insert(signature.id.0, name.into());
219        Ok(Some((
220            ir::ExprKind::FunctionValue {
221                target: ir::CallTarget::Function(signature.id),
222            },
223            Type::ActorFunction(
224                Box::new(mailbox),
225                Box::new(Type::Function(params, Box::new(result))),
226            ),
227        )))
228    }

230    /// Resolve only the two managed source primitives; legacy actors.* remains a separate API.
231    pub(super) fn actor_call(
232        &mut self,
233        name: &str,
234        args: &[ast::Argument],
235        expected: Option<&Type>,
236        span: Span,
237        depth: usize,
238    ) -> Checked<TypedKind> {
239        if self.deferred {
240            return Err(Diagnostic::new(
241                span,
242                "actor operations are unsupported in deferred cleanup",
243            ));
244        }
245        labels::positional(args)?;
246        if name == "spawn" || name == "supervise" {
247            return self.spawn(args, expected, span, depth, name == "supervise");
248        }
249        if name == "supervised_current" {
250            if args.len() != 1 {
251                return Err(Diagnostic::new(
252                    span,
253                    "supervised_current expects one typed Pid",
254                ));
255            }
256            let pid = self.expression(&args[0], depth)?;
257            let ty = self.inference.resolve(&pid.ty, span)?;
258            if !matches!(ty, Type::Pid(_)) {
259                return Err(Diagnostic::new(
260                    span,
261                    "supervised_current requires a typed Pid",
262                ));
263            }
264            return Ok((
265                ir::ExprKind::Actor(ir::ActorExpr::SupervisedCurrent { pid: Box::new(pid) }),
266                Type::Result(Box::new(ty), Box::new(Type::Int)),
267            ));
268        }
269        if args.len() != 2 {
270            return Err(Diagnostic::new(span, "send expects pid and message"));
271        }
272        let pid = self.expression(&args[0], depth)?;
273        let Type::Pid(mailbox) = self.inference.resolve(&pid.ty, span)? else {
274            return Err(Diagnostic::new(span, "send requires a typed Pid"));
275        };
276        let message = self.expression_expected(&args[1], Some(&mailbox), depth)?;
277        Ok((
278            ir::ExprKind::Actor(ir::ActorExpr::Send {
279                pid: Box::new(pid),
280                message: Box::new(message),
281            }),
282            Type::Result(Box::new(Type::Unit), Box::new(Type::Int)),
283        ))
284    }

286    /// Spawn context unifies mailbox evidence before a receiving lambda is checked.
287    fn spawn(
288        &mut self,
289        args: &[ast::Argument],
290        expected: Option<&Type>,
291        span: Span,
292        depth: usize,
293        supervised: bool,
294    ) -> Checked<TypedKind> {
295        if args.len() != if supervised { 2 } else { 1 } {
296            return Err(Diagnostic::new(
297                span,
298                if supervised {
299                    "supervise expects a zero-argument Unit function and integer restart budget"
300                } else {
301                    "spawn expects one zero-argument Unit function"
302                },
303            ));
304        }
305        let mailbox = self.inference.fresh();
306        let ty = Type::Pid(Box::new(mailbox.clone()));
307        self.constrain_result(&ty, expected, span)?;
308        let entry = self.spawn_entry(&args[0], &mailbox, span, depth)?;
309        let max_restarts = if supervised {
310            Some(Box::new(self.expression_expected(
311                &args[1],
312                Some(&Type::Int),
313                depth,
314            )?))
315        } else {
316            None
317        };
318        Ok((
319            ir::ExprKind::Actor(ir::ActorExpr::Spawn {
320                entry: Box::new(entry),
321                mailbox,
322                max_restarts,
323            }),
324            ty,
325        ))
326    }

328    /// Share entry checking between legacy and isolated admission APIs.
329    pub(super) fn spawn_entry(
330        &mut self,
331        argument: &ast::Argument,
332        mailbox: &Type,
333        span: Span,
334        depth: usize,
335    ) -> Checked<ir::Expr> {
336        let function = Type::Function(Vec::new(), Box::new(Type::Unit));
337        let context = Type::ActorFunction(Box::new(mailbox.clone()), Box::new(function));
338        let entry = if let ast::ExprKind::Lambda { body, .. } = &argument.value.kind {
339            let ordinary = Type::Function(Vec::new(), Box::new(Type::Unit));
340            let expected = if self.actor_body(body) {
341                &context
342            } else {
343                &ordinary
344            };
345            self.expression_expected(argument, Some(expected), depth)?
346        } else {
347            self.expression(argument, depth)?
348        };
349        let resolved = self.inference.resolve(&entry.ty, span)?;
350        if resolved == Type::Never {
351            return Ok(entry);
352        }
353        let Some((effect, params, result)) = crate::actors::function(&resolved) else {
354            return Err(Diagnostic::new(span, "spawn requires a function value"));
355        };
356        if !params.is_empty() {
357            return Err(Diagnostic::new(
358                span,
359                "spawn requires a zero-argument Unit function",
360            ));
361        }
362        self.inference
363            .unify(result, &Type::Unit, span, "spawn entry return")?;
364        if let Some(effect) = effect {
365            self.inference
366                .unify(effect, mailbox, span, "spawn mailbox")?;
367        }
368        Ok(entry)
369    }

371    /// Type selective arms in isolated scopes without falsely requiring exhaustive message coverage.
372    pub(super) fn receive(
373        &mut self,
374        view: crate::processes::ReceiveView,
375        arms: &[ast::MatchArm],
376        timeout: Option<&(Box<ast::Expr>, Box<ast::Expr>)>,
377        expected: Option<&Type>,
378        span: Span,
379        depth: usize,
380    ) -> Checked<TypedKind> {
381        let mailbox = self
382            .mailbox
383            .clone()
384            .ok_or_else(|| Diagnostic::new(span, "receive requires an actor context"))?;
385        if self.deferred {
386            return Err(Diagnostic::new(span, "actor receive cannot occur in defer"));
387        }
388        let result = expected.cloned().unwrap_or_else(|| self.inference.fresh());
389        let timeout = timeout
390            .map(|(duration, body)| {
391                let duration = self.expression_equal(duration, &Type::Int, depth)?;
392                let body = self.expression_equal(body, &result, depth)?;
393                Ok((Box::new(duration), Box::new(body)))
394            })
395            .transpose()?;
396        let mut checked = Vec::new();
397        for arm in arms {
398            if let Some(guard) = &arm.guard {
399                guard_source(guard)?;
400            }
401            self.scopes.push(HashMap::new());
402            let pattern =
403                self.pattern(&arm.pattern, &view.item(&mailbox), &mut HashSet::new(), 0)?;
404            let guard = arm
405                .guard
406                .as_ref()
407                .map(|g| self.expression_equal(g, &Type::Bool, depth))
408                .transpose()?;
409            let body = self.expression_equal(&arm.body, &result, depth)?;
410            self.scopes.pop();
411            checked.push(ir::MatchArm {
412                pattern,
413                guard,
414                body,
415                span: arm.span,
416            });
417        }
418        Ok((
419            ir::ExprKind::Actor(ir::ActorExpr::Receive {
420                view,
421                mailbox,
422                arms: checked,
423                timeout,
424            }),
425            result,
426        ))
427    }
428}

430/// Repeated mailbox selection may evaluate only scalar expressions with no hidden calls or effects.
431fn guard_source(expr: &ast::Expr) -> Checked<()> {
432    let mut pending = vec![expr];
433    let mut work = 0;
434    while let Some(expr) = pending.pop() {
435        work += 1;
436        if work > MAX_EXPR_COUNT {
437            return Err(Diagnostic::new(
438                expr.span,
439                "receive guard work limit exceeded",
440            ));
441        }
442        if matches!(
443            expr.kind,
444            ast::ExprKind::Binary {
445                op: ast::BinaryOp::Divide | ast::BinaryOp::Remainder | ast::BinaryOp::Power,
446                ..
447            }
448        ) {
449            return Err(Diagnostic::new(
450                expr.span,
451                "receive guard must be non-failing",
452            ));
453        }
454        if !matches!(
455            expr.kind,
456            ast::ExprKind::Name(_)
457                | ast::ExprKind::Int(_)
458                | ast::ExprKind::Float(_)
459                | ast::ExprKind::Bool(_)
460                | ast::ExprKind::String(_)
461                | ast::ExprKind::Unit
462                | ast::ExprKind::Unary { .. }
463                | ast::ExprKind::Binary { .. }
464                | ast::ExprKind::Field { .. }
465        ) {
466            return Err(Diagnostic::new(
467                expr.span,
468                "receive guard must be a pure scalar expression without calls",
469            ));
470        }
471        pending.extend(crate::actors::source_children(expr));
472    }
473    Ok(())
474}

489    /// Finalize each effect's children and reject unproved transfer/accountability boundaries.
490    pub(super) fn finalize_actor(&self, actor: &mut ir::ActorExpr, span: Span) -> Checked<()> {
491        match actor {
492            ir::ActorExpr::Process(operation) => self.finalize_process(operation, span)?,
493            ir::ActorExpr::Lowered(_) => {
494                return Err(Diagnostic::new(
495                    span,
496                    "private actor continuation reached checker",
497                ));
498            }
499            ir::ActorExpr::Spawn { mailbox, .. }
500            | ir::ActorExpr::Receive { mailbox, .. }
501            | ir::ActorExpr::Call { mailbox, .. } => {
502                *mailbox = self.inference.concrete(mailbox, span).map_err(|e| {
503                    Diagnostic::new(e.span, format!("cannot infer actor mailbox: {}", e.message))
504                })?;
505                self.actor_sendable(mailbox, span)?;
506                if self.registry.contains_result(mailbox)? {
507                    return Err(Diagnostic::new(
508                        span,
509                        "Result-bearing actor messages are unsupported until suspension accountability is proved",
510                    ));
511                }
512            }
513            ir::ActorExpr::Send { message, .. } => {
514                let ty = self.inference.resolve(&message.ty, span)?;
515                self.actor_sendable(&ty, span)?;
516                if self.registry.contains_result(&ty)? {
517                    return Err(Diagnostic::new(
518                        span,
519                        "Result-bearing actor messages are unsupported; send does not handle the sender's Result",
520                    ));
521                }
522            }
523            ir::ActorExpr::SupervisedCurrent { .. } => {}
524        }
525        for child in crate::actors::children_mut(actor) {
526            self.finalize(child)?;
527        }
528        if let ir::ActorExpr::Receive {
529            view,
530            mailbox,
531            arms,
532            ..
533        } = actor
534        {
535            for arm in arms.iter() {
536                if let Some(guard) = &arm.guard {
537                    crate::actors::contracts::guard(guard)?;
538                }
539            }
540            let retained = coverage::selective(
541                &view.item(mailbox),
542                arms,
543                self.registry,
544                span,
545                self.inference.specializing,
546            )?;
547            let mut index = 0;
548            arms.retain(|_| {
549                let keep = retained[index];
550                index += 1;
551                keep
552            });
553        }
554        Ok(())
555    }
556}

558impl Checker<'_> {
559    /// Preserve source argument order while marking receiving calls for actor-tail conversion.
560    pub(super) fn actor_named_call(
561        &mut self,
562        name: &str,
563        args: &[ast::Argument],
564        expected: Option<&Type>,
565        span: Span,
566        depth: usize,
567    ) -> Checked<TypedKind> {
568        let owner = self.mailbox.clone().ok_or_else(|| {
569            Diagnostic::new(span, "receiving function call requires an actor context")
570        })?;
571        let (kind, ty) = self
572            .actor_name(name, span)?
573            .ok_or_else(|| Diagnostic::new(span, "missing actor callable"))?;
574        let Type::ActorFunction(mailbox, function) = ty else {
575            unreachable!()
576        };
577        let Type::Function(params, result) = *function else {
578            unreachable!()
579        };
580        self.inference
581            .unify(&mailbox, &owner, span, "actor call mailbox")?;
582        self.constrain_result(&result, expected, span)?;
583        let ir::ExprKind::FunctionValue {
584            target: ir::CallTarget::Function(id),
585        } = kind
586        else {
587            unreachable!()
588        };
589        let signature = &self.signatures[name];
590        let order = labels::order(args, &signature.labels, span)?;
591        labels::required(args, signature, &order)?;
592        let written: Vec<_> = order.iter().map(|i| params[*i].clone()).collect();
593        let args = self.call_arguments(args, &written, span, depth)?;
594        let (mut kind, result) =
595            self.ordered_call(ir::CallTarget::Function(id), args, &order, *result, span);
596        let call = match &mut kind {
597            ir::ExprKind::Block(stmts) => match stmts.last_mut() {
598                Some(ir::Stmt::Expr(expr)) => &mut expr.kind,
599                _ => return Err(Diagnostic::new(span, "invalid ordered actor call")),
600            },
601            kind => kind,
602        };
603        if let ir::ExprKind::Call { args, .. } = call {
604            *call = ir::ExprKind::Actor(ir::ActorExpr::Call {
605                function: id,
606                args: std::mem::take(args),
607                mailbox: owner,
608            });
609        }
610        Ok((kind, result))
611    }
612}

614impl Checker<'_> {
615    /// Detect owned suspension effects without attributing a nested closure's body to its creator.
616    fn actor_body(&self, body: &ast::Expr) -> bool {
617        let mut pending = vec![body];
618        while let Some(expr) = pending.pop() {
619            if matches!(expr.kind, ast::ExprKind::Lambda { .. }) {
620                continue;
621            }
622            if matches!(expr.kind, ast::ExprKind::Receive { .. }) || process_context(expr) {
623                return true;
624            }
625            let name = match &expr.kind {
626                ast::ExprKind::Call { name, .. } | ast::ExprKind::Pipe { name, .. } => Some(name),
627                ast::ExprKind::GlobalCall { resolved, .. }
628                | ast::ExprKind::GlobalPipe { resolved, .. } => Some(resolved),
629                _ => None,
630            };
631            if name.is_some_and(|name| {
632                self.signatures
633                    .get(name)
634                    .is_some_and(|f| f.mailbox.is_some())
635            }) {
636                return true;
637            }
638            pending.extend(crate::actors::source_children(expr));
639        }
640        false
641    }

643    /// Require recursively immutable, accounted message layouts while keeping generic obligations open.
644    pub(super) fn actor_sendable(&self, ty: &Type, span: Span) -> Checked<()> {
645        let mut pending = vec![ty.clone()];
646        let mut seen = HashSet::new();
647        let mut work = 0usize;
648        while let Some(ty) = pending.pop() {
649            work = work.saturating_add(crate::unions::cost(&ty, span)?);
650            if work > 400_000 || pending.len() > 4096 {
651                return Err(Diagnostic::new(
652                    span,
653                    "actor message type work limit exceeded",
654                ));
655            }
656            if !seen.insert(ty.clone()) {
657                continue;
658            }
659            match ty {
660                Type::Int
661                | Type::Bool
662                | Type::Unit
663                | Type::Float
664                | Type::String
665                | Type::Range
666                | Type::Native(
667                    crate::runtime::NativeType::JsonValue
668                    | crate::runtime::NativeType::ProcessId
669                    | crate::runtime::NativeType::MonitorRef,
670                )
671                | Type::Pid(_)
672                | Type::Generic(_)
673                | Type::Infer(_) => {}
674                Type::List(item) | Type::Option(item) => pending.push(*item),
675                Type::Tuple(fields) | Type::Union(fields) => pending.extend(fields),
676                Type::Map(key, value) => pending.extend([*key, *value]),
677                Type::Named(ref name, _) if name == "Ptr" => {
678                    return Err(Diagnostic::new(
679                        span,
680                        "foreign pointers cannot cross actor boundaries",
681                    ));
682                }
683                Type::Named(_, _) => pending.extend(
684                    self.registry
685                        .layout(&ty, span)?
686                        .variants
687                        .into_iter()
688                        .flatten(),
689                ),
690                Type::Result(_, _) => {
691                    return Err(Diagnostic::new(
692                        span,
693                        "Result-bearing actor messages are unsupported; send does not handle the sender's Result",
694                    ));
695                }
696                _ => {
697                    return Err(Diagnostic::new(
698                        span,
699                        "function or native handle actor messages are unsupported",
700                    ));
701                }
702            }
703        }
704        Ok(())
705    }
706}

708/// Actor-only primitives establish context even without a receive in this body.
709fn process_context(expr: &ast::Expr) -> bool {
710    match &expr.kind {
711        ast::ExprKind::Call { name, .. } | ast::ExprKind::Pipe { name, .. } => {
712            crate::processes::requires_actor(name)
713        }
714        ast::ExprKind::GlobalCall { resolved, .. } | ast::ExprKind::GlobalPipe { resolved, .. } => {
715            crate::processes::requires_actor(resolved)
716        }
717        _ => false,
718    }
719}
