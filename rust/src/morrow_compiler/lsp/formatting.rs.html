1//! Canonical whole-document edits from the accepted open buffer, without effects.
2use super::*;
3
4impl Server {
5    /// Return no edit for clean source or one syntax-preserving UTF-16 replacement.
6    /// Invalid parameters and formatting failures have distinct protocol error codes.
7    pub(super) fn formatting(&self, params: &Json) -> std::result::Result<Json, (i64, String)> {
8        let invalid = |message: String| (-32602, message);
9        options(field(params, "options").map_err(invalid)?).map_err(invalid)?;
10        let uri = document_uri(params).map_err(invalid)?;
11        let document = self
12            .documents
13            .get(uri)
14            .ok_or_else(|| invalid("request document is not open".into()))?;
15        let text = crate::format::format(&document.source)
16            .map_err(|error| (-32803, format!("cannot format {uri}: {}", error.message)))?;
17        if text == document.source {
18            return Ok(Json::Array(Vec::new()));
19        }
20        Ok(Json::Array(vec![object([
21            (
22                "range",
23                navigation::source_range(
24                    &document.source,
25                    Span {
26                        start: 0,
27                        end: document.source.len(),
28                    },
29                ),
30            ),
31            ("newText", string(text)),
32        ])]))
33    }
34}

36/// Validate standard protocol options while retaining Morrow's fixed canonical style.
37/// Additional options may be scalar strings, booleans or signed 32-bit integers.
38fn options(value: &Json) -> Result<()> {
39    let Json::Object(fields) = value else {
40        return Err("formatting options must be an object".into());
41    };
42    let tab_size = field(value, "tabSize")?.integer()?;
43    if !(1..=i64::from(i32::MAX)).contains(&tab_size) {
44        return Err("formatting tabSize must be a positive protocol uinteger".into());
45    }
46    if !matches!(field(value, "insertSpaces")?, Json::Bool(_)) {
47        return Err("formatting insertSpaces must be boolean".into());
48    }
49    for (name, value) in fields {
50        if matches!(
51            name.as_str(),
52            "trimTrailingWhitespace" | "insertFinalNewline" | "trimFinalNewlines"
53        ) && !matches!(value, Json::Bool(_))
54        {
55            return Err(format!("formatting {name} must be boolean"));
56        }
57        let valid = match value {
58            Json::Bool(_) | Json::String(_) => true,
59            Json::Number(_) => value
60                .integer()
61                .is_ok_and(|value| i32::try_from(value).is_ok()),
62            _ => false,
63        };
64        if !valid {
65            return Err(format!("invalid scalar formatting option {name}"));
66        }
67    }
68    Ok(())
69}
