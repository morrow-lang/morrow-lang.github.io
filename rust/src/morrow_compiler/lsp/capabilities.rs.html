1//! Negotiate edit payloads instead of assuming every client implements newer LSP forms.
2use super::*;
3
4#[derive(Default)]
5pub(super) struct ClientEdits {
6    pub document_changes: bool,
7    pub prepare_rename: bool,
8    pub literal_actions: bool,
9}
10impl ClientEdits {
11    /// Missing, false or malformed declarations never enable an incompatible payload.
12    pub fn initialize(params: &Json) -> Self {
13        let document_changes = matches!(
14            at(
15                params,
16                &[
17                    "capabilities",
18                    "workspace",
19                    "workspaceEdit",
20                    "documentChanges"
21                ]
22            ),
23            Some(Json::Bool(true))
24        );
25        let prepare_rename = document_changes
26            && matches!(
27                at(
28                    params,
29                    &["capabilities", "textDocument", "rename", "prepareSupport"]
30                ),
31                Some(Json::Bool(true))
32            );
33        // LSP guarantees clients gracefully handle kinds outside their declared valueSet.
34        let literal_actions = document_changes
35            && at(
36                params,
37                &[
38                    "capabilities",
39                    "textDocument",
40                    "codeAction",
41                    "codeActionLiteralSupport",
42                    "codeActionKind",
43                    "valueSet",
44                ],
45            )
46            .and_then(|value| value.array().ok())
47            .is_some_and(|values| values.iter().all(|value| value.string().is_ok()));
48        Self {
49            document_changes,
50            prepare_rename,
51            literal_actions,
52        }
53    }
54    /// A boolean rename provider retains compatibility when prepareRename is not supported.
55    pub fn rename_provider(&self) -> Json {
56        if self.prepare_rename {
57            object([("prepareProvider", Json::Bool(true))])
58        } else {
59            Json::Bool(self.document_changes)
60        }
61    }
62    /// Offer source actions only when literal actions and their versioned edits are supported.
63    pub fn action_provider(&self) -> Json {
64        if self.literal_actions {
65            object([(
66                "codeActionKinds",
67                Json::Array(vec![string("source.fixAll.morrow")]),
68            )])
69        } else {
70            Json::Bool(false)
71        }
72    }
73}
74/// Walk only object members; malformed or absent intermediate shapes mean no support.
75fn at<'a>(value: &'a Json, path: &[&str]) -> Option<&'a Json> {
76    path.iter().try_fold(value, |value, key| value.get(key))
77}
