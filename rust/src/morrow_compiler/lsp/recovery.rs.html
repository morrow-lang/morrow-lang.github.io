1//! Completion-only recovery over one current token-level member hole, never cached type evidence.
2use super::*;
3impl Server {
4    /// Recheck current overlays; a failed proof simply retains ordinary lexical fallback.
5    pub(super) fn member_completion(&self, uri: &str, source: &str, cursor: usize) -> Option<Json> {
6        let (program, site) = if let Some(path) = file_path(uri).ok()? {
7            let loaded = modules::load_member_sources(&path, &self.overlays(), cursor).ok()?;
8            (loaded.program, loaded.recovery?)
9        } else {
10            parse::recover_member(source, cursor).ok()?
11        };
12        let facts = check::recovery::analyze(&program, &site).ok()?;
13        let candidates = facts
14            .members
15            .iter()
16            .filter(|member| member.name.starts_with(site.prefix()))
17            .map(|member| (member.name.clone(), 5))
18            .collect();
19        Some(navigation::completion_items(
20            candidates,
21            navigation::source_range(source, site.selector()),
22            Some(&facts),
23        ))
24    }
25}

27impl Server {
28    /// Suggest source names only after canonical lexical resolution of the current argument site.
29    pub(super) fn label_completion(&self, uri: &str, source: &str, cursor: usize) -> Option<Json> {
30        let (labels, site) = if let Some(path) = file_path(uri).ok()? {
31            let identity = modules::source_identity(&path).ok()?;
32            let loaded = modules::load_label_sources(&path, &self.overlays(), cursor).ok()?;
33            let site = loaded.label_recovery.clone()?;
34            let index =
35                index::Index::loaded_labels(&loaded, &identity, cursor, Some(site.clone()))?;
36            (index.call_labels?, site)
37        } else {
38            let (program, site) = parse::recover_labels(source, cursor).ok()?;
39            if !program.imports.is_empty() {
40                return None;
41            }
42            let path = PathBuf::from(uri);
43            let index =
44                index::Index::single_labels(&program, source, &path, cursor, Some(site.clone()))?;
45            (index.call_labels?, site)
46        };
47        Some(label_items(labels, source, &site))
48    }
49}

51/// Bound names and serialized edits independently of semantic checking or recovery provenance.
52fn label_items(labels: Vec<String>, source: &str, site: &parse::LabelSite) -> Json {
53    let range = navigation::source_range(source, site.selector());
54    let mut items = Vec::new();
55    let mut bytes = 0usize;
56    for name in labels {
57        bytes = bytes
58            .saturating_add(name.len().saturating_mul(2))
59            .saturating_add(256);
60        if items.len() >= 256 || bytes > 1024 * 1024 {
61            return navigation::completion_list(items, true);
62        }
63        let text = if site.colon() {
64            name.clone()
65        } else {
66            format!("{name}: ")
67        };
68        items.push(object([
69            ("label", string(name)),
70            ("kind", number(5)),
71            ("detail", string("Source parameter name")),
72            (
73                "textEdit",
74                object([("range", range.clone()), ("newText", string(text))]),
75            ),
76        ]));
77    }
78    navigation::completion_list(items, false)
79}
