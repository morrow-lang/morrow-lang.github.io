1//! Typed callback execution without invoking the legacy native callback ABI.
2use super::*;
3
4impl Machine {
5    /// Dispatch checked higher-order operations, evaluating only the selected callbacks.
6    pub(super) fn higher_order(&mut self, builtin: ir::Builtin, args: &[Value]) -> Eval<Value> {
7        use ir::Builtin::*;
8        match (builtin, args) {
9            (ListEnumerate, [Value::List(xs)]) => self.enumerate(xs),
10            (ListMap | ListFilter | ListFind | ListAny | ListAll, [Value::List(xs), callback]) => {
11                self.list_callback(builtin, xs, callback)
12            }
13            (ListSortBy, [Value::List(xs), callback]) => self.sort_by(xs, callback),
14            (ListFold, [Value::List(xs), initial, callback]) => {
15                let mut value = initial.clone();
16                for item in xs.iter() {
17                    value = self.invoke(callback, vec![value, item.clone()])?;
18                }
19                Ok(value)
20            }
21            (
22                OptionMap | ResultMap | ResultAndThen | ResultUnwrapOrElse,
23                [value @ Value::Sum(tag, fields), callback],
24            ) => self.sum_callback(builtin, value, *tag, fields, callback),
25            _ => self.map_builtin(builtin, args),
26        }
27    }
28    /// Stable bottom-up merge sort; only a `Greater` comparator result moves the right element first.
29    /// This matches the native sort driver's comparison contract, not merely its output.
30    fn sort_by(&mut self, xs: &[Value], callback: &Value) -> Eval<Value> {
31        if xs.len() > 65_536 {
32            return Err(fault("interactive list limit exceeded"));
33        }
34        let mut items = xs.to_vec();
35        let mut width = 1;
36        while width < items.len() {
37            let mut merged = Vec::with_capacity(items.len());
38            for run in items.chunks(2 * width) {
39                let (left, right) = run.split_at(width.min(run.len()));
40                let (mut i, mut j) = (0, 0);
41                while i < left.len() && j < right.len() {
42                    let ordering =
43                        self.invoke(callback, vec![left[i].clone(), right[j].clone()])?;
44                    let Value::Sum(tag, _) = ordering else {
45                        return Err(fault("invalid comparator result"));
46                    };
47                    if tag == 2 {
48                        merged.push(right[j].clone());
49                        j += 1;
50                    } else {
51                        merged.push(left[i].clone());
52                        i += 1;
53                    }
54                }
55                merged.extend_from_slice(&left[i..]);
56                merged.extend_from_slice(&right[j..]);
57            }
58            items = merged;
59            width *= 2;
60        }
61        Ok(Value::List(Rc::new(items)))
62    }
63    /// Preserve input order and terminate predicates at their first deciding element.
64    fn list_callback(
65        &mut self,
66        builtin: ir::Builtin,
67        xs: &[Value],
68        callback: &Value,
69    ) -> Eval<Value> {
70        use ir::Builtin::*;
71        if xs.len() > 65_536 {
72            return Err(fault("interactive list limit exceeded"));
73        }
74        let mut mapped = Vec::new();
75        for item in xs {
76            let value = self.invoke(callback, vec![item.clone()])?;
77            if builtin == ListMap {
78                mapped.push(value);
79                continue;
80            }
81            let Value::Bool(selected) = value else {
82                return Err(fault("invalid callback predicate"));
83            };
84            match builtin {
85                ListFilter if selected => mapped.push(item.clone()),
86                ListFind if selected => return Ok(Value::Sum(0, Rc::new(vec![item.clone()]))),
87                ListAny if selected => return Ok(Value::Bool(true)),
88                ListAll if !selected => return Ok(Value::Bool(false)),
89                _ => {}
90            }
91        }
92        Ok(match builtin {
93            ListMap | ListFilter => Value::List(Rc::new(mapped)),
94            ListFind => Value::Sum(1, Rc::new(Vec::new())),
95            ListAll => Value::Bool(true),
96            ListAny => Value::Bool(false),
97            _ => return Err(fault("invalid list callback")),
98        })
99    }
100    /// Preserve absent/error values and catch callback Result propagation at its own function.
101    fn sum_callback(
102        &mut self,
103        builtin: ir::Builtin,
104        original: &Value,
105        tag: usize,
106        fields: &[Value],
107        callback: &Value,
108    ) -> Eval<Value> {
109        use ir::Builtin::*;
110        if builtin == ResultUnwrapOrElse {
111            let field = fields
112                .first()
113                .ok_or_else(|| fault("missing Result payload"))?;
114            return if tag == 0 {
115                Ok(field.clone())
116            } else {
117                self.invoke(callback, vec![field.clone()])
118            };
119        }
120        if tag == 1 {
121            return Ok(original.clone());
122        }
123        let field = fields.first().ok_or_else(|| fault("missing sum payload"))?;
124        let value = self.invoke(callback, vec![field.clone()])?;
125        if builtin == ResultAndThen {
126            Ok(value)
127        } else {
128            Ok(Value::Sum(0, Rc::new(vec![value])))
129        }
130    }
