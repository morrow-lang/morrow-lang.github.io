4const MAX_LIST: usize = 65_536;

6impl Machine {
7    /// Apply checked builtins to immutable shared values, bounding allocations first.
8    pub(super) fn builtin(&mut self, builtin: ir::Builtin, args: Vec<Value>) -> Eval<Value> {
9        use ir::Builtin::*;
10        if self.comptime && matches!(builtin, Print | Println) {
11            return Err(fault("output is not available during comptime evaluation"));
12        }
13        match (builtin, args.as_slice()) {
14            (Print | Println, [value]) => {
15                let text = match value {
16                    Value::String(s) => s.to_string(),
17                    other => display(other),
18                };
19                let extra = usize::from(builtin == Println);
20                bounded(
21                    self.output
22                        .len()
23                        .checked_add(text.len())
24                        .and_then(|n| n.checked_add(extra)),
25                    MAX_STRING,
26                    "output",
27                )?;
28                self.output.push_str(&text);
29                if builtin == Println {
30                    self.output.push('\n');
31                }
32                Ok(Value::Unit)
33            }
34            (StringConcat, [Value::String(a), Value::String(b)]) => concat_strings(a, b),
35            (StringEq, [Value::String(a), Value::String(b)]) => Ok(Value::Bool(a == b)),
36            (StringLen, [Value::String(s)]) => Ok(Value::Int(s.len() as i64)),
37            (ListLen, [Value::List(xs)]) => Ok(Value::Int(xs.len() as i64)),
38            (ListIsEmpty, [Value::List(xs)]) => Ok(Value::Bool(xs.is_empty())),
39            (ListGet, [Value::List(xs), Value::Int(i)]) => usize::try_from(*i)
40                .ok()
41                .and_then(|i| xs.get(i))
42                .cloned()
43                .ok_or_else(|| fault("list index out of bounds")),
44            (ListHead, [Value::List(xs)]) => xs
45                .first()
46                .cloned()
47                .ok_or_else(|| fault("head of empty list")),
48            (ListTail, [Value::List(xs)]) => {
49                list(xs.len().saturating_sub(1), xs.iter().skip(1).cloned())
50            }
51            (ListReverse, [Value::List(xs)]) => list(xs.len(), xs.iter().rev().cloned()),
52            (ListPush, [Value::List(xs), value]) => list(
53                xs.len() + 1,
54                xs.iter().cloned().chain(std::iter::once(value.clone())),
55            ),
56            (ListConcat, [Value::List(a), Value::List(b)]) => {
57                list(a.len() + b.len(), a.iter().chain(b.iter()).cloned())
58            }
59            (ListContains, [Value::List(xs), value]) => Ok(Value::Bool(xs.contains(value))),
60            (OptionIsSome | ResultIsOk, [Value::Sum(tag, _)]) => Ok(Value::Bool(*tag == 0)),
61            (OptionIsNone | ResultIsErr, [Value::Sum(tag, _)]) => Ok(Value::Bool(*tag == 1)),
62            (OptionUnwrapOr | ResultUnwrapOr, [Value::Sum(tag, fields), fallback]) => {
63                Ok(if *tag == 0 {
64                    fields
65                        .first()
66                        .cloned()
67                        .ok_or_else(|| fault("missing sum payload"))?
68                } else {
69                    fallback.clone()
70                })
71            }
72            _ => self.higher_order(builtin, &args),
73        }
74    }
75    /// Dispatch stable registry identities and name unsupported operations by their source API.
76    pub(super) fn runtime(&mut self, id: usize, args: Vec<Value>) -> Eval<Value> {
77        let signature = runtime::signature(id).ok_or_else(|| fault("unknown runtime function"))?;
78        if matches!(
79            signature.symbol,
80            "morrow_ffi_borrow_string" | "morrow_ffi_read_string"
81        ) {
82            return Err(fault("foreign pointer operations require native execution"));
83        }
84        if signature.symbol == "morrow_ffi_float32" {
85            let [Value::Float(value)] = args.as_slice() else {
86                return Err(fault("invalid CFloat32 conversion"));
87            };
88            let narrow = *value as f32;
89            return if value.is_finite() && !narrow.is_finite() {
90                Ok(Value::Sum(
91                    1,
92                    vec![Value::String(Rc::new(
93                        "foreign scalar conversion is outside its representable range".into(),
94                    ))]
95                    .into(),
96                ))
97            } else {
98                Ok(Value::Sum(0, vec![Value::Float(f64::from(narrow))].into()))
99            };
100        }
101        if let Some(builtin) = core_builtin(signature.symbol, signature.operation) {
102            return self.builtin(builtin, args);
103        }
104        if let Some(value) = self.json(signature.symbol, &args) {
105            return value;
106        }
107        if signature.operation == runtime::Operation::DecimalPredicate {
108            let [Value::String(text)] = args.as_slice() else {
109                return Err(fault("invalid decimal predicate arguments"));
110            };
111            if text.len() > 16 * 1024 * 1024 {
112                return Err(fault("string size limit exceeded"));
113            }
114            self.charge_steps(text.len().div_ceil(64).max(1))?;
115            return decimal(text);
116        }
117        if let Some(value) = strings(signature.symbol, &args) {
118            return value;
119        }
120        if let Some(value) = positional(signature.symbol, &args) {
121            return value;
122        }
123        if self.comptime {
124            return Err(fault(
125                "this host API is not available during comptime evaluation",
126            ));
127        }
128        if self.simulation {
129            return Err(fault("host APIs are not available during actor simulation"));
130        }
131        if matches!(
132            signature.symbol,
133            "morrow_read_file"
134                | "morrow_write_file"
135                | "morrow_append_file"
136                | "morrow_delete_file"
137                | "morrow_file_size"
138                | "morrow_file_exists"
139                | "morrow_is_dir"
140        ) {
141            return files(signature.symbol, &args);
142        }
143        let name = runtime::names()
144            .into_iter()
145            .find(|name| runtime::resolve(name) == Some(id))
146            .unwrap_or("requested API");
147        Err(fault(format!(
148            "{name} is available in native programs; interactive support is not implemented"
149        )))
150    }
151}

153/// Map native aliases to the same semantic implementation, including inverted predicates.
154fn core_builtin(symbol: &str, operation: runtime::Operation) -> Option<ir::Builtin> {
155    use ir::Builtin::*;
156    Some(match symbol {
157        "morrow_str_concat" => StringConcat,
158        "morrow_str_eq" => StringEq,
159        "morrow_str_len" => StringLen,
160        "morrow_list_len" => ListLen,
161        "morrow_list_get" => ListGet,
162        "morrow_list_head" => ListHead,
163        "morrow_list_tail" => ListTail,
164        "morrow_list_push" => ListPush,
165        "morrow_list_reverse" => ListReverse,
166        "morrow_list_concat" => ListConcat,
167        "morrow_list_is_empty" => ListIsEmpty,
168        "morrow_list_contains" => ListContains,
169        "morrow_result_is_ok" if operation == runtime::Operation::InvertBool => ResultIsErr,
170        "morrow_result_is_ok" => ResultIsOk,
171        "morrow_result_unwrap_or" => ResultUnwrapOr,
172        _ => return None,
173    })
174}

176/// Check arithmetic and allocation budgets before constructing the backing storage.
177fn bounded(size: Option<usize>, maximum: usize, kind: &str) -> Eval<usize> {
178    size.filter(|size| *size <= maximum)
179        .ok_or_else(|| fault(format!("interactive {kind} limit exceeded")))
180}
181fn string(value: &str) -> Eval<Value> {
182    bounded(Some(value.len()), MAX_STRING, "string")?;
183    Ok(Value::String(Rc::new(value.into())))
184}
185fn list(length: usize, values: impl Iterator<Item = Value>) -> Eval<Value> {
186    bounded(Some(length), MAX_LIST, "list")?;
187    Ok(Value::List(Rc::new(values.collect())))
188}
189/// Both source concatenation spellings share the same preallocation bound.
190pub(super) fn concat_strings(a: &str, b: &str) -> Eval<Value> {
191    let length = bounded(a.len().checked_add(b.len()), MAX_STRING, "string")?;
192    let mut text = String::with_capacity(length);
193    text.push_str(a);
194    text.push_str(b);
195    Ok(Value::String(Rc::new(text)))
196}
197fn sum(tag: usize, value: Value) -> Value {
198    Value::Sum(tag, Rc::new(vec![value]))
199}
200fn file_result(value: Result<Value, i64>) -> Value {
201    match value {
202        Ok(value) => sum(0, value),
203        Err(code) => sum(1, Value::Int(code)),
204    }
205}
206fn option(value: Option<usize>) -> Value {
207    value.map_or_else(
208        || Value::Sum(1, Rc::new(Vec::new())),
209        |n| sum(0, Value::Int(n as i64)),
210    )
211}
212fn optional(value: Option<Value>) -> Value {
213    value.map_or_else(|| Value::Sum(1, Rc::new(Vec::new())), |value| sum(0, value))
214}

216/// Mirror the native `String.quote` literal spelling exactly.
217fn quote(text: &str) -> String {
218    let mut out = String::with_capacity(text.len() + 2);
219    out.push('"');
220    for c in text.chars() {
221        match c {
222            '"' => out.push_str("\\\""),
223            '\\' => out.push_str("\\\\"),
224            '\n' => out.push_str("\\n"),
225            '\r' => out.push_str("\\r"),
226            '\t' => out.push_str("\\t"),
227            c => out.push(c),
228        }
229    }
230    out.push('"');
231    out
232}

234fn checked(value: Option<i64>) -> Value {
235    optional(value.map(Value::Int))

237/// Mirror the native element-directed sort: Int/Bool numeric, Float total order, String bytes.
238fn order(a: &Value, b: &Value) -> std::cmp::Ordering {
239    match (a, b) {
240        (Value::Int(a), Value::Int(b)) => a.cmp(b),
241        (Value::Bool(a), Value::Bool(b)) => a.cmp(b),
242        (Value::Float(a), Value::Float(b)) => a.total_cmp(b),
243        (Value::String(a), Value::String(b)) => a.as_bytes().cmp(b.as_bytes()),
244        _ => std::cmp::Ordering::Equal,
245    }
246}

248/// Non-faulting positional access mirrors the native clamping and `None` contracts exactly.
249fn positional(symbol: &str, args: &[Value]) -> Option<Eval<Value>> {
250    let clamp = |count: &i64, len: usize| usize::try_from(*count).map_or(0, |n| n.min(len));
251    Some(match (symbol, args) {
252        ("morrow_list_at", [Value::List(xs), Value::Int(i)]) => Ok(optional(
253            usize::try_from(*i).ok().and_then(|i| xs.get(i)).cloned(),
254        )),
255        ("morrow_list_first", [Value::List(xs)]) => Ok(optional(xs.first().cloned())),
256        ("morrow_list_last", [Value::List(xs)]) => Ok(optional(xs.last().cloned())),
257        ("morrow_list_take", [Value::List(xs), Value::Int(n)]) => {
258            let count = clamp(n, xs.len());
259            list(count, xs.iter().take(count).cloned())
260        }
261        ("morrow_list_drop", [Value::List(xs), Value::Int(n)]) => {
262            let count = clamp(n, xs.len());
263            list(xs.len() - count, xs.iter().skip(count).cloned())
264        }
265        ("morrow_list_sum", [Value::List(xs)]) => {
266            Ok(Value::Int(xs.iter().fold(0, |total, x| match x {
267                Value::Int(n) => total.wrapping_add(*n),
268                _ => total,
269            })))
270        }
271        ("morrow_list_range", [Value::Int(start), Value::Int(end)]) => {
272            match usize::try_from((i128::from(*end) - i128::from(*start)).max(0)) {
273                Ok(count) => list(
274                    count,
275                    (0..count).map(|offset| Value::Int(start + offset as i64)),
276                ),
277                Err(_) => Err(fault("interactive list limit exceeded")),
278            }
279        }
280        ("morrow_list_zip", [Value::List(a), Value::List(b)]) => list(
281            a.len().min(b.len()),
282            a.iter()
283                .zip(b.iter())
284                .map(|(x, y)| Value::Sum(0, Rc::new(vec![x.clone(), y.clone()]))),
285        ),
286        ("morrow_list_sort", [Value::List(xs)]) => {
287            let mut sorted = xs.as_ref().clone();
288            sorted.sort_by(order);
289            list(sorted.len(), sorted.into_iter())
290        }
291        ("morrow_int_checked_add", [Value::Int(a), Value::Int(b)]) => {
292            Ok(checked(a.checked_add(*b)))
293        }
294        ("morrow_int_checked_sub", [Value::Int(a), Value::Int(b)]) => {
295            Ok(checked(a.checked_sub(*b)))
296        }
297        ("morrow_int_checked_mul", [Value::Int(a), Value::Int(b)]) => {
298            Ok(checked(a.checked_mul(*b)))
299        }
300        ("morrow_int_checked_div", [Value::Int(a), Value::Int(b)]) => {
301            Ok(checked(a.checked_div(*b)))
302        }
303        ("morrow_int_checked_rem", [Value::Int(a), Value::Int(b)]) => {
304            Ok(checked(a.checked_rem(*b)))
305        }
306        ("morrow_int_checked_neg", [Value::Int(a)]) => Ok(checked(a.checked_neg())),
307        ("morrow_int_parse", [Value::String(text)]) => {
308            let digits = text.strip_prefix(['+', '-']).unwrap_or(text);
309            let exact = !digits.is_empty() && digits.bytes().all(|byte| byte.is_ascii_digit());
310            Ok(optional(
311                exact
312                    .then(|| text.parse::<i64>().ok())
313                    .flatten()
314                    .map(Value::Int),
315            ))
316        }
317        _ => return None,
318    })
319}

321/// Preserve the runtime's ASCII case conversion and exact four-character trim set.
322fn strings(symbol: &str, args: &[Value]) -> Option<Eval<Value>> {
323    let whitespace = |c| matches!(c, ' ' | '\t' | '\n' | '\r');
324    Some(match (symbol, args) {
325        ("morrow_str_compare", [Value::String(a), Value::String(b)]) => {
326            Ok(Value::Int(match a.cmp(b) {
327                std::cmp::Ordering::Less => -1,
328                std::cmp::Ordering::Equal => 0,
329                std::cmp::Ordering::Greater => 1,
330            }))
331        }
332        ("morrow_str_to_upper", [Value::String(s)]) => ascii_case(s, true),
333        ("morrow_str_to_lower", [Value::String(s)]) => ascii_case(s, false),
334        ("morrow_str_quote", [Value::String(s)]) => string(&quote(s)),
335        ("morrow_str_trim", [Value::String(s)]) => string(s.trim_matches(whitespace)),
336        ("morrow_str_trim_start", [Value::String(s)]) => string(s.trim_start_matches(whitespace)),
337        ("morrow_str_trim_end", [Value::String(s)]) => string(s.trim_end_matches(whitespace)),
338        ("morrow_str_contains", [Value::String(s), Value::String(p)]) => {
339            Ok(Value::Bool(s.contains(p.as_str())))
340        }
341        ("morrow_str_starts_with", [Value::String(s), Value::String(p)]) => {
342            Ok(Value::Bool(s.starts_with(p.as_str())))
343        }
344        ("morrow_str_ends_with", [Value::String(s), Value::String(p)]) => {
345            Ok(Value::Bool(s.ends_with(p.as_str())))
346        }
347        ("morrow_str_is_empty", [Value::String(s)]) => Ok(Value::Bool(s.is_empty())),
348        ("morrow_str_index_of", [Value::String(s), Value::String(p)]) => {
349            Ok(option(s.find(p.as_str())))
350        }
351        ("morrow_str_char_at", [Value::String(s), Value::Int(i)]) => Ok(option(
352            usize::try_from(*i)
353                .ok()
354                .and_then(|i| s.as_bytes().get(i))
355                .map(|b| *b as usize),
356        )),
357        ("morrow_str_slice", [Value::String(s), Value::Int(start), Value::Int(end)]) => {
358            slice(s, *start, *end)
359        }
360        ("morrow_str_repeat", [Value::String(s), Value::Int(n)]) => repeat(s, *n),
361        ("morrow_str_replace", [Value::String(s), Value::String(old), Value::String(new)]) => {
362            replace(s, old, new)
363        }
364        ("morrow_str_split", [Value::String(s), Value::String(p)]) => split(s, p),
365        ("morrow_str_lines", [Value::String(s)]) => lines(s),
366        ("morrow_str_join", [Value::List(xs), Value::String(separator)]) => join(xs, separator),
367        _ => return None,
368    })
369}

371/// Case conversion preserves byte length and checks its allocation before copying.
372fn ascii_case(text: &str, upper: bool) -> Eval<Value> {
373    bounded(Some(text.len()), MAX_STRING, "string")?;
374    let mut value = text.to_owned();
375    if upper {
376        value.make_ascii_uppercase();
377    } else {
378        value.make_ascii_lowercase();
379    }
380    Ok(Value::String(Rc::new(value)))
381}

383/// Native indices count bytes; unsupported invalid UTF-8 results are explicit, never lossy.
384fn slice(text: &str, start: i64, end: i64) -> Eval<Value> {
385    let start = start.max(0);
386    let end = end.max(start);
387    let start = (start as u64).min(text.len() as u64) as usize;
388    let end = (end as u64).min(text.len() as u64) as usize;
389    string(
390        text.get(start..end)
391            .ok_or_else(|| fault("String.slice indices must be UTF-8 character boundaries"))?,
392    )
393}
394fn repeat(text: &str, count: i64) -> Eval<Value> {
395    if count <= 0 || text.is_empty() {
396        return string("");
397    }
398    if count as u64 > (16 * 1024 * 1024 / text.len()) as u64 {
399        return Err(fault("string size limit exceeded"));
400    }
401    let count = usize::try_from(count).map_err(|_| fault("interactive string limit exceeded"))?;
402    bounded(text.len().checked_mul(count), MAX_STRING, "string")?;
403    Ok(Value::String(Rc::new(text.repeat(count))))
404}
405fn replace(text: &str, old: &str, new: &str) -> Eval<Value> {
406    if old.is_empty() {
407        return string(text);
408    }
409    let count = text.matches(old).count();
410    let retained = text.len() - count * old.len();
411    bounded(
412        count
413            .checked_mul(new.len())
414            .and_then(|n| retained.checked_add(n)),
415        MAX_STRING,
416        "string",
417    )?;
418    Ok(Value::String(Rc::new(text.replace(old, new))))
419}
420fn split(text: &str, separator: &str) -> Eval<Value> {
421    if separator.is_empty() {
422        return list(
423            text.chars().count(),
424            text.chars()
425                .map(|character| Value::String(Rc::new(character.to_string()))),
426        );
427    }
428    let parts = text.split(separator);
429    let length = parts.clone().count();
430    list(
431        length,
432        parts.map(|part| Value::String(Rc::new(part.into()))),
433    )
434}
435fn lines(text: &str) -> Eval<Value> {
436    let mut values = Vec::new();
437    let mut rest = text;
438    while let Some(index) = rest.find(['\r', '\n']) {
439        bounded(Some(values.len() + 1), MAX_LIST, "list")?;
440        values.push(Value::String(Rc::new(rest[..index].into())));
441        rest = &rest[index
442            + if rest[index..].starts_with("\r\n") {
443                2
444            } else {
445                1
446            }..];
447    }
448    if !rest.is_empty() || values.is_empty() {
449        bounded(Some(values.len() + 1), MAX_LIST, "list")?;
450        values.push(Value::String(Rc::new(rest.into())));
451    }
452    Ok(Value::List(Rc::new(values)))
453}
454fn join(values: &[Value], separator: &str) -> Eval<Value> {
455    let mut length = bounded(
456        separator.len().checked_mul(values.len().saturating_sub(1)),
457        MAX_STRING,
458        "string",
459    )?;
460    for value in values {
461        let Value::String(text) = value else {
462            return Err(fault("String.join requires strings"));
463        };
464        length = bounded(length.checked_add(text.len()), MAX_STRING, "string")?;
465    }
466    let mut text = String::with_capacity(length);
467    for (index, value) in values.iter().enumerate() {
468        if index > 0 {
469            text.push_str(separator);
470        }
471        if let Value::String(value) = value {
472            text.push_str(value);
473        }
474    }
475    Ok(Value::String(Rc::new(text)))
476}

478/// Morrow file errors are fixed public codes, never platform errno values.
479fn files(symbol: &str, args: &[Value]) -> Eval<Value> {
480    let Some(Value::String(path)) = args.first() else {
481        return Err(fault("file path must be String"));
482    };
483    let path = std::path::Path::new(path.as_str());
484    match symbol {
485        "morrow_file_exists" => Ok(Value::Bool(std::fs::File::open(path).is_ok())),
486        "morrow_is_dir" => Ok(Value::Bool(path.is_dir())),
487        "morrow_read_file" => read_file(path),
488        "morrow_file_size" => Ok(file_result(file_size(path).map(Value::Int))),
489        "morrow_write_file" | "morrow_append_file" => {
490            let Some(Value::String(text)) = args.get(1) else {
491                return Err(fault("file contents must be String"));
492            };
493            Ok(file_result(
494                write_file(path, text, symbol == "morrow_append_file").map(Value::Int),
495            ))
496        }
497        "morrow_delete_file" => {
498            let result = std::fs::remove_file(path).or_else(|_| std::fs::remove_dir(path));
499            Ok(file_result(result.map(|()| Value::Int(0)).map_err(|_| 1)))
500        }
501        _ => Err(fault("unsupported file operation")),
502    }
503}
504fn file_size(path: &std::path::Path) -> Result<i64, i64> {
505    use std::io::{Seek, SeekFrom};
506    let mut file = std::fs::File::open(path).map_err(|_| 1)?;
507    file.seek(SeekFrom::End(0))
508        .map_err(|_| 3)
509        .and_then(|n| i64::try_from(n).map_err(|_| 3))
510}
511/// Publish complete UTF8/NUL-free text within the native profile and stricter session budget.
512/// Safe std File drop cannot observe a late OS close error; all explicit reads are checked.
513fn read_file(path: &std::path::Path) -> Eval<Value> {
514    use std::io::{Read, Seek, SeekFrom};
515    let mut file = match std::fs::File::open(path) {
516        Ok(file) => file,
517        Err(_) => return Ok(file_result(Err(1))),
518    };
519    let length = match file.seek(SeekFrom::End(0)) {
520        Ok(n) if n <= 16 * 1024 * 1024 => n,
521        _ => return Ok(file_result(Err(3))),
522    };
523    let length = bounded(usize::try_from(length).ok(), MAX_STRING, "string")?;
524    if file.seek(SeekFrom::Start(0)).is_err() {
525        return Ok(file_result(Err(3)));
526    }
527    let mut bytes = vec![0; length + 1];
528    if file.read_exact(&mut bytes[..length]).is_err()
529        || !matches!(file.read(&mut bytes[length..]), Ok(0))
530    {
531        return Ok(file_result(Err(3)));
532    }
533    bytes.truncate(length);
534    if bytes.contains(&0) {
535        return Ok(file_result(Err(3)));
536    }
537    let Ok(text) = String::from_utf8(bytes) else {
538        return Ok(file_result(Err(3)));
539    };
540    Ok(file_result(Ok(Value::String(Rc::new(text)))))
541}

543/// Validate before target side effects, then complete unbuffered writes without claiming durability.
544fn write_file(path: &std::path::Path, text: &str, append: bool) -> Result<i64, i64> {
545    use std::io::Write;
546    if text.len() > 16 * 1024 * 1024 || text.contains('\0') {
547        return Err(3);
548    }
549    let mut file = std::fs::OpenOptions::new()
550        .write(true)
551        .create(true)
552        .append(append)
553        .truncate(!append)
554        .open(path)
555        .map_err(|_| 2)?;
556    file.write_all(text.as_bytes()).map_err(|_| 3)?;
557    Ok(text.len() as i64)
558}

591/// Apply the native decimal content ceiling before scanning; retained interactive limits remain stricter.
592fn decimal(text: &str) -> Eval<Value> {
593    if text.len() > 16 * 1024 * 1024 {
594        return Err(fault("string size limit exceeded"));
595    }
596    Ok(Value::Bool(crate::decimal::is_decimal(text)))
597}
