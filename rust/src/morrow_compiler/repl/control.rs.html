1//! Function-owned abrupt exits and bounded, dynamically registered cleanup.
2use super::*;
3
4impl Machine {
5    /// Reserve a separate entry-wide cleanup budget, including nested cleanup calls.
6    pub(super) fn charge_step(&mut self) -> Eval<()> {
7        self.charge_steps(1)
8    }
9
10    /// Reserve aggregate work before copying pattern tails or executing a node.
11    pub(super) fn charge_steps(&mut self, count: usize) -> Eval<()> {
12        let exhausted = if self.cleanup_depth == 0 {
13            self.steps = self.steps.saturating_add(count);
14            self.steps > 100_000
15        } else {
16            self.cleanup_steps = self.cleanup_steps.saturating_add(count);
17            self.cleanup_steps > 10_000
18        };
19        if exhausted || self.depth >= 128 {
20            Err(fault("interactive evaluation limit exceeded"))
21        } else {
22            Ok(())
23        }
24    }
25
26    /// Capture immutable locals now; execute the captured expression only at function exit.
27    pub(super) fn defer(&mut self, value: &ir::Expr) -> Eval<Value> {
28        if self.defers.len() >= 4096 {
29            return Err(fault("interactive deferred cleanup limit exceeded"));
30        }
31        let value = self.expression(value)?;
32        if !matches!(value, Value::Closure(_)) {
33            return Err(fault("invalid deferred cleanup"));
34        }
35        self.defers.push(value);
36        Ok(Value::Unit)
37    }
38
39    /// Drain in reverse registration order, preserving the first evaluation failure.
40    pub(super) fn finish(&mut self, mut result: Eval<Value>) -> Eval<Value> {
41        self.cleanup_depth += 1;
42        while let Some(cleanup) = self.defers.pop() {
43            let outcome = self.invoke(&cleanup, Vec::new());
44            let failure = match outcome {
45                Ok(Value::Unit) => None,
46                Ok(_) => Some(fault("deferred cleanup must return Unit")),
47                Err(error) => Some(error),
48            };
49            if let Some(error) = failure
50                && !matches!(result, Err(Failure::Message(_) | Failure::JsonLimit))
51            {
52                result = Err(error);
53            }
54        }
55        self.cleanup_depth -= 1;
56        result
57    }
58
59    /// Keep success binders in the current scope; failed matching discards partial binders.
60    pub(super) fn let_else(
61        &mut self,
62        binding: &ir::Pattern,
63        value: &ir::Expr,
64        alternative: &ir::Expr,
65    ) -> Eval<Value> {
66        let value = self.expression(value)?;
67        let previous = self.locals.clone();
68        if self.pattern(binding, &value)? {
69            Ok(Value::Unit)
70        } else {
71            self.locals = previous;
72            self.expression(alternative)?;
73            Err(fault("let-else alternative must leave the function"))
74        }
75    }
76}
