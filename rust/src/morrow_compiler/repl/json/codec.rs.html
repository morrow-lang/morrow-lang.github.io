1//! Concrete codecs share one budget across parsing, conversion, path growth and output.

7type Callback<'a> = dyn FnMut(ir::FunctionId, Value, &mut Budget<'_>) -> Result<Value> + 'a;
8struct Execution<'p, 'b> {
9    callback: Option<&'p mut Callback<'p>>,
10    plan: &'p Plan,
11    budget: Budget<'b>,
12    path: Rc<String>,
13}
14impl Machine {
15    /// Evaluate the sole operand once, preserving normal propagation and cleanup behavior.
16    pub(in crate::repl) fn json_codec(
17        &mut self,
18        direction: Direction,
19        input: &ir::Expr,
20        plan: &Plan,
21    ) -> Eval<Value> {
22        let input = self.expression(input)?;
23        let cleanup = self.cleanup_depth != 0;
24        let limits = if cleanup {
25            &mut self.json_cleanup
26        } else {
27            &mut self.json_limits
28        };
29        let mut limits = std::mem::replace(limits, Limits::new(0));
30        let mut callback_fault = None;
31        let mut callback = |id, args, budget: &mut Budget<'_>| {
32            let scope = morrow_json::scope::Scope::enter(
33                budget.work,
34                ALLOC.saturating_sub(budget.allocated),
35                NODES.saturating_sub(budget.nodes),
36            )?;
37            let limits = if cleanup {
38                &mut self.json_cleanup
39            } else {
40                &mut self.json_limits
41            };
42            std::mem::swap(limits, budget.limits);
43            let result = self.function(self.program.clone(), id, &[], vec![args]);
44            let limits = if cleanup {
45                &mut self.json_cleanup
46            } else {
47                &mut self.json_limits
48            };
49            std::mem::swap(limits, budget.limits);
50            let used = scope.spent();
51            drop(scope);
52            budget.work = budget.work.saturating_sub(used.work);
53            budget.allocated = budget.allocated.saturating_add(used.allocated);
54            budget.nodes = budget.nodes.saturating_add(used.nodes);
55            let nodes = morrow_json::scope::charge_nodes(used.nodes);
56            if matches!(result, Err(Failure::JsonLimit)) {
57                morrow_json::scope::mark_exhausted();
58                return Err(error(4, -1));
59            }
60            if let Err(failure) = result {
61                callback_fault = Some(failure);
62                return Err(error(0, -1));
63            }
64            if used.exhausted {
65                morrow_json::scope::mark_exhausted();
66                return Err(error(4, -1));
67            }
68            nodes?;
69            match result {
70                Ok(Value::Sum(0, values)) if values.len() == 1 => Ok(values[0].clone()),
71                Ok(Value::Sum(1, values)) if values.len() == 1 => match &values[0] {
72                    Value::JsonError(e) => Err(e.clone()),
73                    _ => Err(error(5, -1)),
74                },
75                Ok(_) => Err(error(5, -1)),
76                Err(failure) => {
77                    callback_fault = Some(failure);
78                    Err(error(0, -1))
79                }
80            }
81        };
82        let result = scoped_operation(&mut limits, |limits| {
83            execute(direction, &input, plan, limits, Some(&mut callback))
84        });
85        if cleanup {
86            self.json_cleanup = limits;
87        } else {
88            self.json_limits = limits;
89        }
90        if let Some(failure) = callback_fault {
91            return Err(failure);
92        }
93        match result {
94            Ok(value) => Ok(Value::Sum(0, Rc::new(vec![value]))),
95            Err(Error { code: 0, .. }) => Err(fault("interactive evaluation limit exceeded")),
96            Err(error) => Ok(Value::Sum(1, Rc::new(vec![Value::JsonError(error)]))),
97        }
98    }
99}
100/// Reserve failure publication before descent; no failure path allocates additional path text.
101fn execute<'p>(
102    direction: Direction,
103    input: &Value,
104    plan: &'p Plan,
105    limits: &mut Limits,
106    callback: Option<&'p mut Callback<'p>>,
107) -> Result<Value> {
108    let mut budget = Budget {
109        limits,
110        work: 64 * 1024 * 1024,
111        allocated: 0,
112        nodes: 0,
113        at: 0,
114    };
115    budget.allocate(128)?;
116    let actual_publication = std::mem::size_of::<Value>() + 80;
117    Limits::charge(
118        &mut budget.limits.allocated,
119        actual_publication.saturating_sub(128),
120    )?;
121    let mut execution = Execution {
122        callback,
123        plan,
124        budget,
125        path: Rc::new(String::new()),
126    };
127    match direction {
128        Direction::Encode => {
129            let node = execution.encode(plan.root, input, 0)?;
130            let result = (|| {
131                execution.budget.work(node.encoded + node.nodes)?;
132                execution.budget.allocate(node.encoded + 1)?;
133                let mut text = String::with_capacity(node.encoded);
134                value::encode(&node, &mut text);
135                Ok(Value::String(Rc::new(text)))
136            })();
137            execution.locate(result)
138        }
139        Direction::Decode => {
140            let Value::String(text) = input else {
141                return Err(error(5, -1));
142            };
143            let text = source_text(text, &mut execution.budget).map_err(|e| Error {
144                offset: INPUT as i64,
145                ..e
146            })?;
147            execution.budget.work(8 * text.len())?;
148            let node = parse::document_in(text, &mut execution.budget)?;
149            execution.budget.at = 0;
150            execution.decode(plan.root, &node, 0)
151        }
152    }
153}
154/// Charge every source byte before scanning for the native String terminator.
155fn source_text<'s>(text: &'s str, budget: &mut Budget<'_>) -> Result<&'s str> {
156    for (i, byte) in text.bytes().take(INPUT + 1).enumerate() {
157        budget.work(1)?;
158        if byte == 0 {
159            return Ok(&text[..i]);
160        }
161    }
162    if text.len() > INPUT {
163        return Err(error(4, -1));
164    }
165    budget.work(1)?;
166    Ok(text)
167}
168impl Execution<'_, '_> {
169    /// Charge depth/work before dispatch and attach the already retained failing location once.
170    fn encode(&mut self, id: usize, input: &Value, depth: usize) -> Result<Json> {
171        self.step(depth)?;
172        let result = self.encode_kind(id, input, depth);
173        self.locate(result)
174    }
175    fn decode(&mut self, id: usize, input: &Json, depth: usize) -> Result<Value> {
176        self.step(depth)?;
177        let result = self.decode_kind(id, input, depth);
178        self.locate(result)
179    }
180    /// Operation limits are conversion failures at the current path, not parser offsets.
181    fn step(&mut self, depth: usize) -> Result<()> {
182        if depth >= DEPTH {
183            return self.locate(Err(error(4, -1)));
184        }
185        let result = self.budget.work(1).map_err(|e| Error { offset: -1, ..e });
186        self.locate(result)
187    }
188    fn locate<T>(&self, result: Result<T>) -> Result<T> {
189        result.map_err(|mut error| {
190            if error.path.is_none() {
191                if error.code == 4 {
192                    error.offset = -1;
193                }
194                error.path = Some(self.path.clone());
195            }
196            error
197        })
198    }
199    /// Reuse decimal conversion policy; known primitive nodes require no new parser or child allowance.
200    fn encode_kind(&mut self, id: usize, input: &Value, depth: usize) -> Result<Json> {
201        match (&self.plan.entries[id].kind, input) {
202            (Wire::Union(children), Value::Union(value)) => {
203                self.encode_union(children, value, depth)
204            }
205            (Wire::Custom { encode, .. }, _) => {
206                let value = self.custom(encode, input.clone())?;
207                let Value::Json(node) = value else {
208                    return Err(error(5, -1));
209                };
210                self.budget.work(node.nodes)?;
211                if node.height.saturating_add(depth) > DEPTH
212                    || node.nodes > NODES
213                    || node.encoded > OUTPUT
214                {
215                    return Err(error(4, -1));
216                }
217                Ok(node)
218            }
219            (Wire::Newtype(child), _) => self.encode(*child, input, depth + 1),
220            (Wire::Dynamic, Value::Json(node)) => {
221                self.budget.work(node.nodes)?;
222                Ok(node.clone())
223            }
224            (Wire::Unit, Value::Unit)
225            | (Wire::Bool, Value::Bool(_))
226            | (Wire::Int, Value::Int(_))
227            | (Wire::Float, Value::Float(_)) => self.scalar(input),
228            (Wire::String, Value::String(v)) => self.text(v),
229            _ => self.encode_container(id, input, depth),
230        }
231    }
232    /// Preserve exact dynamic integer/Float/text error semantics under the shared codec budget.
233    fn decode_kind(&mut self, id: usize, input: &Json, depth: usize) -> Result<Value> {
234        if matches!(
235            self.plan.entries[id].kind,
236            Wire::Int | Wire::Float | Wire::Bool | Wire::String
237        ) {
238            let work = match &input.kind {
239                Kind::Number(v) | Kind::String(v) => v.len() * 2,
240                _ => 1,
241            };
242            self.budget.work(work)?;
243            self.budget.allocate(64)?;
244        }
245        match (&self.plan.entries[id].kind, &input.kind) {
246            (Wire::Union(children), _) => self.decode_union(children, input, depth),
247            (Wire::Custom { decode, .. }, _) => self.custom(decode, Value::Json(input.clone())),
248            (Wire::Newtype(child), _) => self.decode(*child, input, depth + 1),
249            (Wire::Dynamic, _) => {
250                self.budget.work(input.nodes)?;
251                Ok(Value::Json(input.clone()))
252            }
253            (Wire::Unit, Kind::Null) => Ok(Value::Unit),
254            (Wire::Bool, Kind::Bool(v)) => Ok(Value::Bool(*v)),
255            (Wire::Int, Kind::Number(v)) => convert::integer(v).map(Value::Int),
256            (Wire::Float, Kind::Number(v)) => convert::float(v).map(Value::Float),
257            (Wire::String, Kind::String(v)) => self.string(v),
258            _ => self.decode_container(id, input, depth),
259        }
260    }
261    fn custom(&mut self, callback: &crate::json_codec::Callback, input: Value) -> Result<Value> {
262        let crate::json_codec::Callback::Function(id) = callback else {
263            return Err(error(5, -1));
264        };
265        let call = self.callback.as_deref_mut().ok_or_else(|| error(5, -1))?;
266        match call(*id, input, &mut self.budget) {
267            Err(mut failure) if failure.code != 0 => {
268                let child = failure.path.as_deref().map(String::as_str).unwrap_or("");
269                if child.len() > OUTPUT.saturating_sub(self.path.len()) {
270                    return Err(error(4, -1));
271                }
272                self.budget
273                    .work(child.len().saturating_mul(2) + self.path.len())?;
274                self.budget.allocate(self.path.len() + child.len() + 40)?;
275                failure.path = Some(Rc::new(format!("{}{child}", self.path)));
276                Err(failure)
277            }
278            result => result,
279        }
280    }
281    /// Reserve the same fixed native adapter envelope before formatting or allocation.
282    fn scalar(&mut self, input: &Value) -> Result<Json> {
283        self.budget.work(256)?;
284        self.budget.allocate(256)?;
285        if self.budget.nodes == NODES {
286            return Err(error(4, -1));
287        }
288        morrow_json::scope::charge_nodes(1)?;
289        self.budget.nodes += 1;
290        let (kind, encoded) = match input {
291            Value::Unit => (Kind::Null, 4),
292            Value::Bool(value) => (Kind::Bool(*value), if *value { 4 } else { 5 }),
293            Value::Int(value) => {
294                let text = value.to_string();
295                let n = text.len();
296                (Kind::Number(text), n)
297            }
298            Value::Float(value) => {
299                Limits::charge(&mut self.budget.limits.allocated, 512)?;
300                let text = convert::format(*value)?;
301                let n = text.len();
302                (Kind::Number(text), n)
303            }
304            _ => return Err(error(5, -1)),
305        };
306        Ok(Rc::new(Node {
307            kind,
308            encoded,
309            offset: 0,
310            height: 1,
311            nodes: 1,
312        }))
313    }
314    fn text(&mut self, text: &str) -> Result<Json> {
315        let text = source_text(text, &mut self.budget)?;
316        self.budget.work(text.len() * 2)?;
317        self.budget.node()?;
318        self.budget.allocate(text.len() + 1)?;
319        Ok(value::text_node(text.to_owned(), 0))
320    }
321    fn string(&mut self, text: &str) -> Result<Value> {
322        if text.contains('\0') {
323            return Err(error(10, -1));
324        }
325        // Native borrows the immutable DOM text; Rust requires a separate String allocation.
326        Limits::charge(&mut self.budget.limits.allocated, text.len() + 41)?;
327        Ok(Value::String(Rc::new(text.into())))
328    }
329    /// Reserve the full immutable pointer path before descent; failed growth reports the parent path.
330    fn at<T>(&mut self, key: &str, call: impl FnOnce(&mut Self) -> Result<T>) -> Result<T> {
331        if self.path.len() >= OUTPUT || key.len() > (OUTPUT - self.path.len() - 1) / 2 {
332            return self.locate(Err(error(4, -1)));
333        }
334        let length = self
335            .path
336            .len()
337            .saturating_add(1)
338            .saturating_add(key.len().saturating_mul(2));
339        let reservation = (|| {
340            self.budget.work(length)?;
341            self.budget.allocate(length + 41)
342        })();
343        self.locate(reservation.map_err(|e| Error { offset: -1, ..e }))?;
344        let mut path = String::with_capacity(length);
345        path.push_str(&self.path);
346        path.push('/');
347        for ch in key.chars() {
348            match ch {
349                '~' => path.push_str("~0"),
350                '/' => path.push_str("~1"),
351                _ => path.push(ch),
352            }
353        }
354        let previous = std::mem::replace(&mut self.path, Rc::new(path));
355        let result = call(self);
356        let result = self.locate(result);
357        self.path = previous;
358        result
359    }
360    /// Charge both native payload slots and larger interactive storage before allocating a container.
361    fn slots(&mut self, count: usize) -> Result<()> {
362        if count > NODES {
363            return Err(error(4, -1));
364        }
365        self.budget.work(count)?;
366        let native = count * 8 + 8;
367        self.budget.allocate(native)?;
368        let actual = count * std::mem::size_of::<Value>() + 40;
369        Limits::charge(
370            &mut self.budget.limits.allocated,
371            actual.saturating_sub(native),
372        )
373    }
374}

380    fn scalar_adapter_reservations_match_native_before_conversion() {
381        for (ty, kind, value) in [
382            (Type::Int, Wire::Int, Value::Int(42)),
383            (Type::Bool, Wire::Bool, Value::Bool(true)),
384            (Type::Unit, Wire::Unit, Value::Unit),
385        ] {
386            let plan = Plan {
387                root: 0,
388                entries: vec![crate::json_codec::Entry { ty, kind }],
389            };
390            let mut limits = Limits::new(64 * 1024 * 1024);
391            let budget = Budget {
392                limits: &mut limits,
393                work: 64 * 1024 * 1024,
394                allocated: 128,
395                nodes: 0,
396                at: 0,
397            };
398            let mut execution = Execution {
399                callback: None,
400                plan: &plan,
401                budget,
402                path: Rc::new(String::new()),
403            };
404            execution.encode(0, &value, 0).unwrap();

406            assert_eq!(execution.budget.allocated, 128 + 256);
