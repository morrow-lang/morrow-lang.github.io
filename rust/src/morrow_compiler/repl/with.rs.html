1//! Sequential Result bindings with concretely typed, outer-scope error handlers.
2use super::*;
3
4impl Machine {
5    /// Restore all success and error bindings even when the block returns or propagates.
6    pub(super) fn with_block(
7        &mut self,
8        steps: &[ir::WithStep],
9        body: &ir::Expr,
10        handlers: &[ir::WithHandler],
11    ) -> Eval<Value> {
12        let outer = self.locals.clone();
13        let result = self.with_steps(steps, body, handlers, &outer);
14        self.locals = outer;
15        result
16    }
17
18    /// Stop at the first failed step; no later initializer or success body executes.
19    fn with_steps(
20        &mut self,
21        steps: &[ir::WithStep],
22        body: &ir::Expr,
23        handlers: &[ir::WithHandler],
24        outer: &HashMap<usize, Value>,
25    ) -> Eval<Value> {
26        for step in steps {
27            let value = self.expression(&step.value)?;
28            let Value::Sum(tag, fields) = &value else {
29                return Err(fault("with step must produce Result"));
30            };
31            if fields.len() != 1 {
32                return Err(fault("invalid Result payload"));
33            }
34            match tag {
35                0 => {
36                    if !self.pattern(&step.pattern, &fields[0])? {
37                        return Err(fault("with binding must match every success"));
38                    }
39                }
40                1 => {
41                    let Some(index) = step.error_handler else {
42                        return Err(Failure::Return(value));
43                    };
44                    let handler = handlers
45                        .get(index)
46                        .ok_or_else(|| fault("invalid with error handler"))?;
47                    self.locals = outer.clone();
48                    self.locals.insert(handler.error.id.0, fields[0].clone());
49                    return self.expression(&handler.body);
50                }
51                _ => return Err(fault("invalid Result tag")),
52            }
53        }
54        self.expression(body)
55    }
56}
