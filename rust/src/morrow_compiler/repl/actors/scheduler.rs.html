1//! FIFO dispatch promotes virtual deadlines only after runnable work is exhausted.
2use super::*;
3impl Machine {
4    pub(in crate::repl) fn run_actors(&mut self) -> Eval<()> {
5        loop {
6            self.charge_step()?;
7            let mut due: Vec<_> = self
8                .actors
9                .actors
10                .iter()
11                .filter_map(|(id, a)| {
12                    a.waiting
13                        .as_ref()?
14                        .deadline
15                        .filter(|d| *d <= self.actors.time)
16                        .map(|d| (d, *id))
17                })
18                .collect();
19            due.sort_unstable();
20            for (_, id) in due {
21                self.actors.enqueue(id);
22            }
23            if let Some(id) = self.actors.ready.pop_front() {
24                let Some(actor) = self.actors.actors.get_mut(&id) else {
25                    continue;
26                };
27                actor.queued = false;
28                self.current_actor = Some(id);
29                let result = self.actor_turn(id);
30                self.current_actor = None;
31                self.actors.time = self
32                    .actors
33                    .time
34                    .checked_add(1)
35                    .ok_or_else(|| fault("virtual actor clock overflow"))?;
36                let result = result.and_then(|()| self.actors.storage().map_err(fault));
37                if let Err(error) = result {
38                    self.current_actor = Some(id);
39                    let _ = self.drain_actor_scopes(id);
40                    self.current_actor = None;
41                    let lineage = self.actors.retire(id);
42                    if let Some(lineage) = lineage {
43                        if let Some(supervisor) = self.actors.supervisors.get_mut(&lineage)
44                            && supervisor.remaining > 0
45                        {
46                            supervisor.remaining -= 1;
47                            let initializer = supervisor.initializer.clone();
48                            let pid = self.actors.spawn(initializer, Some(lineage))?;
49                            self.actors.supervisors.get_mut(&lineage).unwrap().current = pid.id;
50                        } else {
51                            self.actors.supervisors.remove(&lineage);
52                        }
53                    } else {
54                        return Err(error);
55                    }
56                }
57                continue;
58            }
59            let next = self
60                .actors
61                .actors
62                .values()
63                .filter_map(|a| a.waiting.as_ref()?.deadline)
64                .min();
65            let Some(deadline) = next else {
66                break;
67            };
68            self.actors.time = deadline;
69            let due: Vec<_> = self
70                .actors
71                .actors
72                .iter()
73                .filter_map(|(id, a)| {
74                    a.waiting
75                        .as_ref()?
76                        .deadline
77                        .filter(|d| *d <= deadline)
78                        .map(|d| (d, *id))
79                })
80                .collect();
81            for (_, id) in due {
82                self.actors.enqueue(id);
83            }
84        }
85        self.actors.storage().map_err(fault)
86    }
87    fn actor_turn(&mut self, id: u64) -> Eval<()> {
88        self.actors.turns = self
89            .actors
90            .turns
91            .checked_add(1)
92            .ok_or_else(|| fault("actor turn counter overflow"))?;
93        let entry = self.actors.actors.get_mut(&id).unwrap().entry.take();
94        if let Some(entry) = entry {
95            let status = self.invoke(&entry, vec![])?;
96            match status {
97                Value::Int(0) => self.actors.enqueue(id),
98                Value::Int(1) => self.poll_actor(id)?,
99                Value::Int(2) | Value::Unit => {
100                    self.drain_actor_scopes(id)?;
101                    if let Some(lineage) = self.actors.retire(id) {
102                        self.actors.supervisors.remove(&lineage);
103                    }
104                }
105                _ => return Err(fault("invalid actor continuation status")),
106            }
107        } else {
108            self.poll_actor(id)?;
109        }
110        Ok(())
111    }
112    fn poll_actor(&mut self, id: u64) -> Eval<()> {
113        let Some(waiting) = self.actors.actors.get(&id).and_then(|a| a.waiting.as_ref()) else {
114            return Ok(());
115        };
116        let selector = waiting.selector.clone();
117        let messages: Vec<_> = self.actors.actors[&id].mailbox.iter().cloned().collect();
118        for (index, message) in messages.into_iter().enumerate() {
119            let selected = self.invoke(&selector, vec![message])?;
120            if matches!(selected, Value::Closure(_)) {
121                let actor = self.actors.actors.get_mut(&id).unwrap();
122                actor.mailbox.remove(index);
123                self.actors.messages -= 1;
124                actor.waiting = None;
125                actor.entry = Some(selected);
126                self.actors.enqueue(id);
127                return Ok(());
128            }
129            if selected != Value::Int(0) {
130                return Err(fault("invalid actor selector result"));
131            }
132        }
133        let actor = self.actors.actors.get_mut(&id).unwrap();
134        if actor
135            .waiting
136            .as_ref()
137            .and_then(|w| w.deadline)
138            .is_some_and(|d| d <= self.actors.time)
139        {
140            actor.entry = actor.waiting.take().and_then(|w| w.timeout);
141            if actor.entry.is_none() {
142                return Err(fault("missing actor timeout continuation"));
143            }
144            self.actors.enqueue(id);
145        }
146        Ok(())
147    }
