21impl Completer for Completion {
22    type Candidate = Pair;
23    /// Complete the token before the cursor without changing any trailing source.
24    fn complete(
25        &self,
26        line: &str,
27        position: usize,
28        _: &Context<'_>,
29    ) -> rustyline::Result<(usize, Vec<Pair>)> {
30        let Some(prefix) = line.get(..position) else {
31            return Ok((position, vec![]));
32        };
33        let start = prefix
34            .rfind(|c: char| !c.is_alphanumeric() && !matches!(c, '_' | '.' | ':'))
35            .map_or(0, |at| {
36                at + prefix[at..].chars().next().map_or(0, char::len_utf8)
37            });
38        let word = &prefix[start..];
39        let mut words = crate::runtime::names();
40        words.extend([
41            "print", "println", "fn", "let", "if", "else", "match", "for", "while", "loop",
42            "return", "break", "continue", "true", "false", "and", "or", "not", "type", "newtype",
43            "pub", "import", "module", "defer", "with", "in", "as", "Ok", "Err", "Some", "None",
44            ":help", ":type", ":clear", ":reset", ":quit", ":paste", ":end", ":actors", ":stop",
45        ]);
46        words.sort_unstable();
47        words.dedup();
48        Ok((
49            start,
50            words
51                .into_iter()
52                .filter(|candidate| !word.is_empty() && candidate.starts_with(word))
53                .map(|candidate| Pair {
54                    display: candidate.into(),
55                    replacement: candidate.into(),
56                })
57                .collect(),
58        ))
59    }
60}

62/// Retain at most 1000 history entries; unavailable history never prevents evaluation.
63pub(super) fn serve(quiet: bool) -> Result<(), String> {
64    let config = Config::builder()
65        .max_history_size(1000)
66        .map_err(|error| error.to_string())?
67        .build();
68    let mut editor = Editor::<Completion, DefaultHistory>::with_config(config)
69        .map_err(|error| error.to_string())?;
70    editor.set_helper(Some(Completion));
71    let history = history_path();
72    if let Some(path) = &history
73        && let Ok(file) = std::fs::File::open(path)
74        && file.metadata().is_ok_and(|metadata| metadata.is_file())
75    {
76        let mut text = String::new();
77        if file
78            .take(8 * 1024 * 1024 + 1)
79            .read_to_string(&mut text)
80            .is_ok()
81        {
82            for entry in history_entries(&text) {
83                let _ = editor.add_history_entry(entry);
84            }
85        }
86    }
87    let result = super::serve_lines(std::io::stdout().lock(), !quiet, |ready, _| {
88        let prompt = if quiet {
89            ""
90        } else if ready {
91            "morrow> "
92        } else {
93            "...   "
94        };
95        match editor.readline(prompt) {
96            Ok(line) => {
97                if line.len() > 1024 * 1024 {
98                    return Err("interactive input limit exceeded".into());
99                }
100                if line.len() <= 4096 && !line.starts_with(':') && !line.trim().is_empty() {
101                    let _ = editor.add_history_entry(&line);
102                }
103                Ok(Input::Line(line))
104            }
105            Err(ReadlineError::Interrupted) => Ok(Input::Interrupted),
106            Err(ReadlineError::Eof) => Ok(Input::End),
107            Err(error) => Err(error.to_string()),
108        }
109    });
110    if let Some(path) = history {
111        let _ = editor.save_history(&path);
112    }
113    result
114}

116/// Preserve the C history location while allowing isolated/custom editor sessions.
117fn history_path() -> Option<PathBuf> {
118    std::env::var_os("MORROW_REPL_HISTORY")
119        .filter(|path| !path.is_empty())
120        .map(PathBuf::from)
121        .or_else(|| {
122            std::env::var_os("HOME").map(|home| PathBuf::from(home).join(".morrow_history"))
123        })
124        .filter(|path| std::fs::metadata(path).map_or(true, |metadata| metadata.is_file()))
125}

127/// Bound history bytes, decoded entry size and retained count before exposing recall.
128fn history_entries(text: &str) -> Vec<String> {
129    if text.len() > 8 * 1024 * 1024 {
130        return Vec::new();
131    }
132    let (escaped, text) = text
133        .strip_prefix("#V2\n")
134        .map_or((false, text), |text| (true, text));
135    let mut entries: Vec<_> = text
136        .lines()
137        .rev()
138        .filter_map(|line| {
139            if line.len() > 8192 || line.trim().is_empty() {
140                return None;
141            }
142            let entry = if escaped {
143                unescape_history(line)?
144            } else {
145                line.to_owned()
146            };
147            (entry.len() <= 4096).then_some(entry)
148        })
149        .take(1000)
150        .collect();
151    entries.reverse();
152    entries
153}

155/// Rustyline v2 history escapes backslashes and embedded newlines; C history stays literal.
156fn unescape_history(line: &str) -> Option<String> {
157    let mut result = String::new();
158    let mut characters = line.chars();
159    while let Some(character) = characters.next() {
160        if character == '\\' {
161            match characters.next()? {
162                'n' => result.push('\n'),
163                '\\' => result.push('\\'),
164                _ => return None,
165            }
166        } else {
167            result.push(character);
168        }
169    }
170    Some(result)
171}

173#[cfg(test)]
174mod tests {
175    #[test]
176    fn bounded_history_import_accepts_legacy_and_escaped_rustyline_records() {
177        assert_eq!(
178            super::history_entries("40 + 2\nString.len(\"a\\n\")\n"),
179            vec!["40 + 2", "String.len(\"a\\n\")"]
180        );
181        assert_eq!(
182            super::history_entries("#V2\nString.len(\"a\\\\n\")\n"),
183            vec!["String.len(\"a\\n\")"]
184        );
185        let oversized = format!("40 + 2\n{}\n", "x".repeat(4097));
186        assert_eq!(super::history_entries(&oversized), vec!["40 + 2"]);
187        let excessive = format!("40 + 2\n{}", "x".repeat(8 * 1024 * 1024));
188        assert!(super::history_entries(&excessive).is_empty());
189    }
