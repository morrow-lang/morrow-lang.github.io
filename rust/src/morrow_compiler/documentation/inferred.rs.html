10/// Typecheck a standalone source library once and render its final reusable signatures.
11/// Imports must be module-resolved by callers using render_with_schemes instead.
12pub fn render_inferred(source: &str, title: &str, output: Output) -> Result<String, Diagnostic> {
13    let program = parse::parse(source)?;
14    let schemes = editor::function_schemes(&program)?;
15    render_with_schemes(source, title, output, Some(&schemes))
16}

18/// Render source with optional externally checked schemes at original source-local spans.
19/// Callers must supply unmodified schemes from the same source snapshot.
20/// Shape/identity/size checks do not attest caller-supplied semantic types.
21pub fn render_with_schemes(
22    source: &str,
23    title: &str,
24    output: Output,
25    schemes: Option<&[FunctionInfo]>,
26) -> Result<String, Diagnostic> {
27    if title.len() > 4096 {
28        return Err(limit("documentation title exceeds 4096 bytes"));
29    }
30    let program = parse::parse(source)?;
31    let mut items = declarations(source, &program)?;
32    if let Some(schemes) = schemes {
33        attach(&program, &mut items, schemes)?;
34    }
35    let module_doc = program.module_doc.as_ref().map(|doc| doc.text.as_str());
36    render_declarations(title, output, module_doc, &items)
37}

39/// Join exact declaration anchors, rejecting duplicate or unmatched metadata rather than guessing names.
40pub(super) fn attach(
41    program: &ast::Program,
42    items: &mut [Declaration<'_>],
43    schemes: &[FunctionInfo],
44) -> Result<(), Diagnostic> {
45    if schemes.len() > MAX_DECLARATIONS {
46        return Err(limit("documentation scheme count exceeds limit"));
47    }
48    editor::validate_schemes(schemes)?;
49    let mut bytes = 0;
50    let mut by_anchor = HashMap::new();
51    for info in schemes {
52        if info
53            .name
54            .rsplit('.')
55            .next()
56            .is_some_and(|name| name.starts_with('$'))
57        {
58            // Generated trait-implementation methods are not documented declarations.
59            continue;
60        }
61        if by_anchor.insert(info.origin.start, info).is_some() {
62            return Err(limit("duplicate checked signature anchor"));
63        }
64    }
65    let originals: HashMap<_, _> = program
66        .functions
67        .iter()
68        .map(|f| (f.span.start, f))
69        .collect();
70    let mut clauses = HashMap::new();
71    for function in &program.functions {
72        *clauses.entry(function.name.as_str()).or_insert(0) += 1;
73    }
74    for item in items {
75        if let Some(function) = originals.get(&item.span.start) {
76            let info = by_anchor
77                .remove(&item.span.start)
78                .ok_or_else(|| limit("missing checked signature anchor"))?;
79            identity(function, info, clauses[function.name.as_str()])?;
80            let text = signature(function, info)?;
81            bytes += text.len();
82            if bytes > MAX_OUTPUT {
83                return Err(limit("checked signatures exceed 8 MiB"));
84            }
85            item.checked = Some(text);
86        }
87    }
88    if !by_anchor.is_empty() {
89        return Err(limit("unmatched checked signature anchor"));
90    }
91    Ok(())
92}

94/// Reject mismatched source identities; semantic authenticity remains the caller's obligation.
95fn identity(
96    function: &ast::Function,
97    info: &FunctionInfo,
98    clauses: usize,
99) -> Result<(), Diagnostic> {
100    let name_matches = info.name == function.name
101        || info
102            .name
103            .strip_suffix(&function.name)
104            .is_some_and(|prefix| prefix.ends_with('.'));
105    if info.origin != function.span
106        || !name_matches
107        || info.parameters.len() != function.params.len()
108        || info.clauses != clauses
109    {
110        return Err(limit("checked signature source identity mismatch"));
111    }
112    Ok(())
113}

115/// Preserve one generic-name assignment across a signature and all intrinsic requirements.
116fn signature(function: &ast::Function, info: &FunctionInfo) -> Result<String, Diagnostic> {
117    let generated: Vec<_> = info
118        .generics
119        .iter()
120        .filter(|name| name.starts_with('$'))
121        .cloned()
122        .collect();
123    let limits = presentation::Limits {
124        bytes: 32_768,
125        ..Default::default()
126    };
127    let mut text = presentation::resolved_signature(
128        function,
129        &info.parameters,
130        &info.result,
131        &generated,
132        limits,
133    )?;
134    let context = Type::Function(info.parameters.clone(), Box::new(info.result.clone()));
135    let mut seen = BTreeSet::new();
136    for requirement in &info.requirements {
137        let ty =
138            presentation::render_type_in_context(&requirement.ty, &context, &generated, limits)?;
139        if requirement.message.len() > 32_768 {
140            return Err(limit("checked requirement exceeds 32 KiB"));
141        }
142        let line = format!("\nrequires {ty}: {}", requirement.message);
143        if line.len() > 32_768_usize.saturating_sub(text.len()) {
144            return Err(limit("checked signature exceeds 32 KiB"));
145        }
146        if seen.insert(line.clone()) {
147            text.push_str(&line);
148        }
149    }
150    Ok(text)
151}
