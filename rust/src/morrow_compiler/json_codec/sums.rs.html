1//! Sum metadata is checked against independently retained source constructor identities.
2use super::*;
3impl Plan {
4    /// Validate native tag order and every child reference, including unused alternatives.
5    pub(super) fn sum(
6        &self,
7        index: usize,
8        variants: &[Variant],
9        entry: &Entry,
10        layouts: &HashMap<&Type, &ir::TypeLayout>,
11        audit: &mut Validation,
12    ) -> Result<bool, Diagnostic> {
13        let layout = layouts
14            .get(&entry.ty)
15            .ok_or_else(|| audit.error("JSON sum codec is missing its checked layout"))?;
16        if variants.is_empty()
17            || variants.len() > 255
18            || layout.storage != ir::LayoutStorage::Tagged
19            || !layout.fields.is_empty()
20            || layout.variants.len() != variants.len()
21            || layout.variant_names.len() != variants.len()
22        {
23            return Ok(false);
24        }
25        let mut names = std::collections::HashSet::new();
26        for (tag, variant) in variants.iter().enumerate() {
27            audit.charge(variant.wire_tag.len())?;
28            if !source_name(&variant.wire_tag)
29                || !names.insert(&variant.wire_tag)
30                || variant.wire_tag != layout.variant_names[tag]
31                || variant.fields.len() != layout.variants[tag].len()
32            {
33                return Ok(false);
34            }
35            for (id, ty) in variant.fields.iter().zip(&layout.variants[tag]) {
36                if self.child(*id, index, audit)?.ty != *ty {
37                    return Ok(false);
38                }
39            }
40        }
41        Ok(true)
42    }
43}
44/// Unqualified constructor tokens cannot contain whitespace, punctuation or embedded terminators.
45pub(crate) fn source_name(name: &str) -> bool {
46    let mut chars = name.chars();
47    let Some(first) = chars.next() else {
48        return false;
49    };
50    crate::parse::identifier_char(first, true)
51        && chars.all(|c| crate::parse::identifier_char(c, false))
52}
