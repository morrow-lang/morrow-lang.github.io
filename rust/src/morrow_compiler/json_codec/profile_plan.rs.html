1//! Public concrete plans have no unknown profile leaves or unvalidated member identities.
2use super::*;
3use profiles::{Key, Node, Shape};
4impl Plan {
5    /// Build borrowed-name profiles after the complete plan/layout audit, with no type witnesses.
6    pub(super) fn profiles(&self, audit: &mut Validation) -> Result<(), Diagnostic> {
7        audit.charge(self.entries.len())?;
8        let mut nodes = Vec::with_capacity(self.entries.len());
9        for entry in &self.entries {
10            let shape = match &entry.kind {
11                Kind::Int | Kind::Float => Shape::Number,
12                Kind::Bool => Shape::Bool,
13                Kind::String => Shape::String,
14                Kind::Unit => Shape::Null,
15                Kind::Dynamic | Kind::Custom { .. } => Shape::Any,
16                Kind::List(_) => Shape::Array(None),
17                Kind::Tuple(fields) => Shape::Array(Some(fields.len())),
18                Kind::Map(_) => Shape::Map,
19                Kind::Record(fields) => {
20                    audit.charge(fields.len())?;
21                    Shape::Object(
22                        fields
23                            .iter()
24                            .map(|f| Key {
25                                name: &f.name,
26                                required: !f.optional,
27                            })
28                            .collect(),
29                    )
30                }
31                Kind::Sum(variants) => {
32                    audit.charge(variants.len())?;
33                    Shape::Sum(variants.iter().map(|v| v.wire_tag.as_str()).collect())
34                }
35                Kind::Newtype(id) | Kind::Option(id) => {
36                    audit.charge(1)?;
37                    nodes.push(Node::follow(
38                        vec![*id],
39                        matches!(entry.kind, Kind::Option(_)),
40                        false,
41                    ));
42                    continue;
43                }
44                Kind::Union(ids) => {
45                    audit.charge(ids.len())?;
46                    nodes.push(Node::follow(ids.clone(), false, true));
47                    continue;
48                }
49            };
50            let mut node = Node::leaf(shape);
51            node.children = match &entry.kind {
52                Kind::Tuple(fields) => {
53                    audit.charge(fields.len())?;
54                    fields.clone()
55                }
56                Kind::Record(fields) => {
57                    audit.charge(fields.len())?;
58                    fields.iter().map(|f| f.codec).collect()
59                }
60                _ => Vec::new(),
61            };
62            if let Kind::Sum(variants) = &entry.kind {
63                audit.charge(variants.len())?;
64                for variant in variants {
65                    audit.charge(variant.fields.len())?;
66                    node.variants.push(variant.fields.clone());
67                }
68            }
69            nodes.push(node);
70        }
71        profiles::validate(&nodes, &mut audit.work, audit.span)
72    }
73    /// Source union canonical order and exact child identity are independently rechecked.
74    pub(super) fn union_members(
75        &self,
76        index: usize,
77        ids: &[usize],
78        types: &[Type],
79        audit: &mut Validation,
80    ) -> Result<bool, Diagnostic> {
81        if ids.len() != types.len() || ids.len() < 2 || ids.len() > 128 {
82            return Ok(false);
83        }
84        for (at, (id, ty)) in ids.iter().zip(types).enumerate() {
85            audit.charge(type_size(ty))?;
86            if matches!(ty, Type::Union(_))
87                || (at > 0 && types[at - 1] >= *ty)
88                || self.child(*id, index, audit)?.ty != *ty
89            {
90                return Ok(false);
91            }
92        }
93        Ok(true)
94    }
95}
