1//! A charged least fixed point proves finite values without recursive graph walks.
2use crate::{Diagnostic, Span};
3
4pub(crate) struct Proof<'a> {
5    reverse: Vec<Vec<usize>>,
6    remaining: Vec<usize>,
7    disjunction: Vec<bool>,
8    required: usize,
9    work: &'a mut usize,
10    span: Span,
11}
12impl<'a> Proof<'a> {
13    /// Reserve bounded graph storage before creating per-entry slots.
14    pub(crate) fn new(count: usize, work: &'a mut usize, span: Span) -> Result<Self, Diagnostic> {
15        charge(work, count.saturating_mul(4), span)?;
16        if count > super::MAX_ENTRIES {
17            return Err(Diagnostic::new(
18                span,
19                "JSON codec plan count limit exceeded",
20            ));
21        }
22        Ok(Self {
23            reverse: vec![Vec::new(); count],
24            remaining: vec![0; count],
25            disjunction: vec![false; count],
26            required: count,
27            work,
28            span,
29        })
30    }
31    /// Strict constructors require every child; List, Map and Option supply independent bases.
32    pub(crate) fn edge(&mut self, parent: usize, child: usize) -> Result<(), Diagnostic> {
33        charge(self.work, 2, self.span)?;
34        if parent >= self.remaining.len() || child >= self.remaining.len() {
35            return Err(Diagnostic::new(self.span, "invalid JSON codec child slot"));
36        }
37        self.remaining[parent] += 1;
38        self.reverse[child].push(parent);
39        Ok(())
40    }
41    /// A sum has no implicit empty base; one explicit constructor product must complete.
42    pub(crate) fn disjunction(&mut self, id: usize) -> Result<(), Diagnostic> {
43        charge(self.work, 1, self.span)?;
44        if id >= self.required || self.remaining[id] != 0 {
45            return Err(Diagnostic::new(self.span, "invalid JSON sum proof vertex"));
46        }
47        self.disjunction[id] = true;
48        self.remaining[id] = 1;
49        Ok(())
50    }
51    /// Auxiliary product clauses are proof metadata, never fabricated executable plan entries.
52    pub(crate) fn alternative(&mut self, parent: usize) -> Result<usize, Diagnostic> {
53        charge(self.work, 6, self.span)?;
54        if parent >= self.required || !self.disjunction[parent] {
55            return Err(Diagnostic::new(self.span, "invalid JSON sum proof parent"));
56        }
57        let id = self.remaining.len();
58        self.remaining.push(0);
59        self.disjunction.push(false);
60        self.reverse.push(vec![parent]);
61        Ok(id)
62    }
63    /// Each ready vertex and dependency is visited once; OR completion never double-enqueues.
64    pub(crate) fn finish(mut self) -> Result<(), Diagnostic> {
65        charge(self.work, self.remaining.len(), self.span)?;
66        let mut ready: Vec<_> = self
67            .remaining
68            .iter()
69            .enumerate()
70            .filter_map(|(id, count)| (*count == 0).then_some(id))
71            .collect();
72        while let Some(id) = ready.pop() {
73            charge(
74                self.work,
75                self.reverse[id].len().saturating_add(1),
76                self.span,
77            )?;
78            for parent in &self.reverse[id] {
79                if self.remaining[*parent] == 0 {
80                    continue;
81                }
82                if self.disjunction[*parent] {
83                    self.remaining[*parent] = 0;
84                } else {
85                    self.remaining[*parent] -= 1;
86                }
87                if self.remaining[*parent] == 0 {
88                    ready.push(*parent);
89                }
90            }
91        }
92        if self.remaining[..self.required].iter().any(|n| *n != 0) {
93            return Err(Diagnostic::new(
94                self.span,
95                "recursive JSON schema has no finite value",
96            ));
97        }
98        Ok(())
99    }
100}
101/// All proof storage, edges and scans spend the caller's existing aggregate allowance.
102fn charge(work: &mut usize, amount: usize, span: Span) -> Result<(), Diagnostic> {
103    *work = work.saturating_add(amount);
104    if *work > super::MAX_WORK {
105        return Err(Diagnostic::new(span, "JSON codec plan work limit exceeded"));
106    }
107    Ok(())
108}

113    #[test]
114    fn all_four_vertex_strict_graphs_match_independent_fixed_point() {
115        for bits in 0u32..65536 {
116            let mut dependencies = [0u32; 4];
117            let mut work = 0;
118            let mut proof = Proof::new(4, &mut work, Span::default()).unwrap();
119            for (parent, row) in dependencies.iter_mut().enumerate() {
120                *row = (bits >> (parent * 4)) & 15;
121                for child in 0..4 {
122                    if *row & (1 << child) != 0 {
123                        proof.edge(parent, child).unwrap();
124                    }
125                }
126            }
127            let mut known = 0u32;
128            for _ in 0..4 {
129                let mut next = known;
130                for (id, row) in dependencies.iter().enumerate() {
131                    if row & !known == 0 {
132                        next |= 1 << id;
133                    }
134                }
135                known = next;
136            }
137            assert_eq!(proof.finish().is_ok(), known == 15, "graph {bits}");
138        }
139    }
140    #[test]
141    fn finite_sum_products_match_independent_least_fixed_point() {
142        let mut seed = 71u64;
143        for _ in 0..16384 {
144            let mut equations = [[0u8; 2]; 4];
145            let mut work = 0;
146            let mut proof = Proof::new(4, &mut work, Span::default()).unwrap();
147            for (id, variants) in equations.iter_mut().enumerate() {
148                proof.disjunction(id).unwrap();
149                for fields in variants {
150                    seed = seed.wrapping_mul(6364136223846793005).wrapping_add(1);
151                    *fields = (seed >> 32) as u8 & 15;
152                    let product = proof.alternative(id).unwrap();
153                    for child in 0..4 {
154                        if *fields & (1 << child) != 0 {
155                            proof.edge(product, child).unwrap();
156                        }
157                    }
158                }
159            }
160            let mut known = 0u8;
161            for _ in 0..4 {
162                let mut next = known;
163                for (id, variants) in equations.iter().enumerate() {
164                    if variants.iter().any(|fields| fields & !known == 0) {
165                        next |= 1 << id;
166                    }
167                }
168                known = next;
169            }
170            assert_eq!(proof.finish().is_ok(), known == 15, "{equations:?}");
171        }
172    }
