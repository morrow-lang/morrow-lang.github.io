144/// Recover only the requested parser operation; dependencies always parse normally.
145fn entry_syntax(text: &str, request: Option<RecoveryRequest>) -> Result<EntrySyntax, Diagnostic> {
146    match request {
147        Some(RecoveryRequest::Member(cursor)) => {
148            parse::recover_member(text, cursor).map(|(program, site)| (program, Some(site), None))
149        }
150        Some(RecoveryRequest::Label(cursor)) => {
151            parse::recover_labels(text, cursor).map(|(program, site)| (program, None, Some(site)))
152        }
153        None => parse::parse(text).map(|program| (program, None, None)),
154    }
155}

166/// Share identical import resolution while making editor-only metadata an explicit bounded opt-in.
167fn load_sources(
168    entry: &Path,
169    sources: &HashMap<PathBuf, String>,
170    editor: bool,
171    cursor: Option<RecoveryRequest>,
172    snapshot_limit: usize,
173) -> Result<Loaded, Error> {
174    let snapshots = snapshots(sources, snapshot_limit)?;
175    let entry = source_identity(entry)?;
176    let text = read_source(&entry, &snapshots)?;
177    let (syntax, recovery, label_recovery) =
178        entry_syntax(&text, cursor).map_err(|e| located(&entry, &text, e))?;
179    let name = syntax.module.clone().unwrap_or_else(|| {
180        let candidate = entry
181            .file_stem()
182            .unwrap_or_default()
183            .to_string_lossy()
184            .into_owned();
185        if valid_module(&candidate).is_ok() {
186            candidate
187        } else {
188            "__entry".into()
189        }
190    });
191    valid_module(&name)?;
192    let mut root = entry.parent().unwrap_or_else(|| Path::new(".")).to_owned();
193    let levels =
194        name.split('.').count() - usize::from(entry.file_stem().is_some_and(|s| s != "mod"));
195    for _ in 0..levels {
196        if !root.pop() {
197            return Err(failure("module name does not match entry path"));
198        }
199    }
200    if syntax.module.is_some() {
201        let stem = root.join(name.replace('.', "/"));
202        if entry != stem.with_extension("mr") && entry != stem.join("mod.mr") {
203            return Err(failure(format!(
204                "{}: module declaration {name} does not match file path",
205                entry.display()
206            )));
207        }
208    }
209    let mut loader = Loader {
210        snapshots,
211        root,
212        modules: Vec::new(),
213        seen: BTreeMap::new(),
214        visiting: BTreeSet::new(),
215        bytes: 0,
216        editor,
217        entry_syntax: Some((entry.clone(), syntax, text)),
218        recovery,
219        label_recovery,
220    };
221    let entry_id = loader.visit(&entry, Some(&name), 0)?;
222    loader.resolve(entry_id)
223}

226    /// Borrow every original source and its offset in the flattened syntax.
227    pub fn sources(&self) -> impl Iterator<Item = SourceView<'_>> {
228        self.sources.iter().map(|source| SourceView {

235    /// Locate a source span without modifying or allocating source contents.
236    pub fn locate_span(&self, span: Span) -> Option<(&Path, Span)> {
237        let source = self.sources.iter().find(|source| {
238            span.start <= span.end
239                && span.start >= source.start
240                && span.end <= source.start + source.text.len()
241        })?;
242        Some((
243            &source.path,
244            Span {
245                start: span.start - source.start,
246                end: span.end - source.start,
247            },
248        ))
249    }
250
251    /// Locate a checker diagnostic without losing its original source or byte range.
252    pub fn locate(&self, mut diagnostic: Diagnostic) -> Option<SourceDiagnostic> {
253        let source = self.sources.iter().find(|s| {
254            diagnostic.span.start >= s.start && diagnostic.span.start <= s.start + s.text.len()
255        })?;
256        diagnostic.span.start -= source.start;
257        diagnostic.span.end = diagnostic.span.end.saturating_sub(source.start);
258        Some(SourceDiagnostic {
259            path: source.path.clone(),
260            source: source.text.clone(),
261            diagnostic,
262        })
263    }
264    /// Render a checker/emitter diagnostic against the original imported source file.
265    pub fn render(&self, error: Diagnostic) -> String {
266        let fallback = format!("error: {}", error.message);
267        self.locate(error)
268            .map_or(fallback, |e| render(&e.path, &e.source, e.diagnostic))
269    }
270}

272/// Format errors consistently with the single-file CLI, using Unicode-aware columns.
273fn render(path: &Path, source: &str, error: Diagnostic) -> String {
274    let mut offset = error.span.start.min(source.len());
275    while !source.is_char_boundary(offset) {
276        offset -= 1;
277    }
278    let before = &source[..offset];
279    let line = before.bytes().filter(|b| *b == b'\n').count() + 1;
280    let start = before.rfind('\n').map_or(0, |n| n + 1);
281    let column = source[start..offset].chars().count() + 1;
282    format!(
283        "{}:{line}:{column}: error: {}\n  {}",
284        path.display(),
285        error
286            .message
287            .strip_prefix("error: ")
288            .unwrap_or(&error.message),
289        source[start..].lines().next().unwrap_or("")
290    )
291}
292/// Attach the stable CLI marker without duplicating already located diagnostics.
293fn failure(message: impl Into<String>) -> Error {
294    let message = message.into();
295    let message = if message.starts_with("error:") || message.contains(": error:") {
296        message
297    } else {
298        format!("error: {message}")
299    };
300    Error {
301        message,
302        location: None,
303        span: None,
304    }
305}

307/// Canonicalize existing files and new unsaved files whose parent directory exists.
308pub fn source_identity(path: &Path) -> Result<PathBuf, Error> {
309    if fs::symlink_metadata(path).is_ok() {
310        return path
311            .canonicalize()
312            .map_err(|e| failure(format!("{}: {e}", path.display())));
313    }
314    let parent = path
315        .parent()
316        .filter(|p| !p.as_os_str().is_empty())
317        .unwrap_or_else(|| Path::new("."));
318    let name = path
319        .file_name()
320        .ok_or_else(|| failure("source path has no file name"))?;
321    Ok(parent
322        .canonicalize()
323        .map_err(|e| failure(format!("{}: {e}", path.display())))?
324        .join(name))
325}

327/// Normalize aliases once, rejecting conflicting snapshots and oversized editor inputs.
328fn snapshots(
329    sources: &HashMap<PathBuf, String>,
330    maximum: usize,
331) -> Result<HashMap<PathBuf, &str>, Error> {
332    if sources.len() > maximum || sources.values().map(String::len).sum::<usize>() > 2 * MAX_BYTES {
333        return Err(failure(format!(
334            "source snapshots exceed {maximum}-file or 16 MiB limit"
335        )));
336    }
337    let mut result = HashMap::new();
338    for (path, text) in sources {
339        if text.len() > 1024 * 1024 {
340            return Err(failure("source exceeds 1 MiB limit"));
341        }
342        let path = source_identity(path)?;
343        if result.get(&path).is_some_and(|old| *old != text.as_str()) {
344            return Err(failure("conflicting snapshots for the same source file"));
345        }
346        result.insert(path, text.as_str());
347    }
348    Ok(result)
349}

389/// Read one bounded UTF-8 source, never allocating an unbounded file into memory.
390fn read(path: &Path) -> Result<String, Error> {
391    let file = fs::File::open(path).map_err(|e| failure(format!("{}: {e}", path.display())))?;
392    let mut text = String::new();
393    file.take(1024 * 1024 + 1)
394        .read_to_string(&mut text)
395        .map_err(|e| failure(format!("{}: {e}", path.display())))?;
396    if text.len() > 1024 * 1024 {
397        return Err(failure(format!(
398            "{}: source exceeds 1 MiB limit",
399            path.display()
400        )));
401    }
402    Ok(text)
403}

405/// Restrict module paths to dotted identifiers so imports cannot express path traversal.
406fn valid_module(name: &str) -> Result<(), Error> {
407    if name.split('.').any(|p| {
408        p.is_empty()
409            || !p
410                .chars()
411                .enumerate()
412                .all(|(i, c)| crate::parse::identifier_char(c, i == 0))
413    }) {
414        return Err(failure(format!("invalid module name: {name}")));
415    }
416    Ok(())
417}

419impl Loader<'_> {
420    /// Visit one dependency after recording the current DFS stack to reject cycles.
421    fn visit(&mut self, path: &Path, expected: Option<&str>, depth: usize) -> Result<usize, Error> {
422        let path = source_identity(path)?;
423        if self.visiting.contains(&path) {
424            return Err(failure(format!("import cycle at {}", path.display())));
425        }
426        if let Some(id) = self.seen.get(&path) {
427            if expected.is_some_and(|e| e != self.modules[*id].name) {
428                return Err(failure("one file imported under inconsistent module names"));
429            }
430            return Ok(*id);
431        }
432        if depth >= MAX_FILES || self.modules.len() + self.visiting.len() >= MAX_FILES {
433            return Err(failure("module graph exceeds 128-file limit"));
434        }
435        let text = self.source_text(&path)?;
436        let start = self.bytes;
437        self.bytes += text.len() + 1;
438        if self.bytes > MAX_BYTES {
439            return Err(failure("module sources exceed 8 MiB limit"));
440        }
441        let syntax = self.source_syntax(&path, &text)?;
442        let name = syntax
443            .module
444            .clone()
445            .or_else(|| expected.map(str::to_owned))
446            .unwrap_or_default();
447        valid_module(&name)?;
448        if expected.is_some_and(|e| e != name) {
449            return Err(located(
450                &path,
451                &text,
452                Diagnostic::new(
453                    Span::default(),
454                    format!(
455                        "declared module {name} does not match imported path {}",
456                        expected.unwrap_or("")
457                    ),
458                ),
459            ));
460        }
461        self.visiting.insert(path.clone());
462        let mut dependencies = Vec::new();
463        for import in &syntax.imports {
464            let imported = self
465                .import_path(&import.module)
466                .map_err(|e| located(&path, &text, Diagnostic::new(import.span, e.message)))?;
467            dependencies.push(
468                self.visit(&imported, Some(&import.module), depth + 1)
469                    .map_err(|e| {
470                        if e.location.is_some() {
471                            e
472                        } else {
473                            located(&path, &text, Diagnostic::new(import.span, e.message))
474                        }
475                    })?,
476            );
477        }
478        self.visiting.remove(&path);
479        let id = self.modules.len();
480        self.seen.insert(path.clone(), id);
481        self.modules.push(Module {
482            source: Source { path, text, start },
483            name,
484            syntax,
485            dependencies,
486        });
487        Ok(id)
488    }

490    /// Consume the exact entry bytes paired with cached syntax, or copy only the selected dependency.
491    fn source_text(&mut self, path: &Path) -> Result<String, Error> {
492        match &mut self.entry_syntax {
493            Some((entry, _, text)) if entry == path => Ok(std::mem::take(text)),
494            _ => read_source(path, &self.snapshots),
495        }
496    }
497
498    /// Reuse the entry's proven syntax once; dependencies always parse their current snapshots.
499    fn source_syntax(&mut self, path: &Path, text: &str) -> Result<ast::Program, Error> {
500        if self
501            .entry_syntax
502            .as_ref()
503            .is_some_and(|(entry, _, _)| entry == path)
504        {
505            return Ok(self.entry_syntax.take().expect("matching entry syntax").1);
506        }
507        parse::parse(text).map_err(|e| located(path, text, e))
508    }

510    /// Resolve module.mr or module/mod.mr within the project, rejecting ambiguity.
511    fn import_path(&self, name: &str) -> Result<PathBuf, Error> {
512        valid_module(name)?;
513        let stem = self.root.join(name.replace('.', "/"));
514        let candidates: Vec<_> = [stem.with_extension("mr"), stem.join("mod.mr")]
515            .into_iter()
516            .filter(|p| {
517                p.is_file() || source_identity(p).is_ok_and(|id| self.snapshots.contains_key(&id))
518            })
519            .collect();
520        if candidates.len() != 1 {
521            return Err(failure(format!(
522                "module {name}: expected exactly one module file; found {}",
523                candidates.len()
524            )));
525        }
526        let path = source_identity(&candidates[0])?;
527        if !path.starts_with(&self.root) {
528            return Err(failure(format!(
529                "module {name} resolves outside project root"
530            )));
531        }
532        Ok(path)
533    }

535    /// Keep the private member operation aligned with ordinary flattened source spans.
536    fn relocate_recovery(&mut self, entry: usize) {
537        if let Some(site) = &mut self.recovery {
538            site.shift(self.modules[entry].source.start);
539        }
540        if let Some(site) = &mut self.label_recovery {
541            site.shift(self.modules[entry].source.start);
542        }
543    }
544
545    /// Qualify declarations in dependency order, then merge their concrete namespaces.
546    fn resolve(mut self, entry: usize) -> Result<Loaded, Error> {
547        self.relocate_recovery(entry);
548        let mut exports: Vec<Names> = Vec::new();
549        let mut program = ast::Program::default();
550        let mut sources = Vec::new();
551        let mut symbols = Vec::new();
552        let (mut symbol_count, mut symbol_bytes) = (0, 0);
553        for (id, mut module) in self.modules.into_iter().enumerate() {
554            let public = (|| {
555                let mut visible = own_names(&module, id == entry)?;
556                let own = visible.clone();
557                let mut public = exported_names(&module, &own)?;
558                let mut imported_prefixes = BTreeSet::new();
559                for (import, dependency) in module.syntax.imports.iter().zip(&module.dependencies) {
560                    import_names(
561                        import,
562                        &exports[*dependency],
563                        &mut visible,
564                        &mut public,
565                        &mut imported_prefixes,
566                    )
567                    .map_err(|e| at_span(e, import.span))?;
568                }
569                if self.editor {
570                    record_symbols(
571                        &mut symbols,
572                        &mut symbol_count,
573                        &mut symbol_bytes,
574                        &module.source.path,
575                        &visible,
576                    )?;
577                }
578                qualify_docs(&mut module.syntax, &own, module.source.start)?;
579                for function in &mut module.syntax.functions {
580                    function.name = own.values[&function.name].clone();
581                    qualify_function(function, &visible, &imported_prefixes, module.source.start)?;
582                    shift(&mut function.span, module.source.start);
583                }
584                qualify_source_types(&mut module.syntax, &own, &visible, module.source.start)?;
585                Ok(public)
586            })()
587            .map_err(|e| at_source(e, &module.source))?;
588            program.docs.extend(module.syntax.docs);
589            program.functions.extend(module.syntax.functions);
590            program.types.extend(module.syntax.types);
591            program.aliases.extend(module.syntax.aliases);
592            program.newtypes.extend(module.syntax.newtypes);
593            program.traits.extend(module.syntax.traits);
594            program
595                .implementations
596                .extend(module.syntax.implementations);
597            exports.push(public);
598            sources.push(module.source);
599        }
600        Ok(Loaded {
601            recovery: self.recovery,
602            label_recovery: self.label_recovery,
603            program,
604            sources,
605            symbols,
606        })
607    }
608}

610/// Qualify documentation by its following declaration's namespace, not its spelling.
611fn qualify_docs(program: &mut ast::Program, names: &Names, offset: usize) -> Result<(), Error> {
612    let mut roles: Vec<_> = program
613        .functions
614        .iter()
615        .map(|decl| (decl.span.start, true))
616        .chain(program.types.iter().map(|decl| (decl.span.start, false)))
617        .chain(program.aliases.iter().map(|decl| (decl.span.start, false)))
618        .chain(program.newtypes.iter().map(|decl| (decl.span.start, false)))
619        .chain(program.traits.iter().map(|decl| (decl.span.start, false)))
620        .collect();
621    roles.sort_unstable_by_key(|role| role.0);
622    for doc in &mut program.docs {
623        let position = roles.partition_point(|role| role.0 < doc.span.end);
624        let (_, value) = roles
625            .get(position)
626            .ok_or_else(|| failure("missing documentation declaration"))?;
627        let namespace = if *value { &names.values } else { &names.types };
628        doc.target = namespace
629            .get(&doc.target)
630            .ok_or_else(|| failure("unknown documentation target"))?
631            .clone();
632        shift(&mut doc.span, offset);
633    }
634    Ok(())
635}

637/// Bound aliases to 100,000 entries and 8 MiB of name bytes before cloning across the graph.
638/// The navigation index's one active visibility-table clone is bounded by this same aggregate cap.
639fn record_symbols(
640    symbols: &mut Vec<ModuleSymbols>,
641    count: &mut usize,
642    bytes: &mut usize,
643    path: &Path,
644    names: &Names,
645) -> Result<(), Error> {
646    *count = count.saturating_add(names.types.len() + names.values.len());
647    if *count > 100_000 {
648        return Err(failure("editor symbol index limit exceeded"));
649    }
650    for (name, target) in names.types.iter().chain(&names.values) {
651        *bytes = bytes
652            .saturating_add(name.len())
653            .saturating_add(target.len());
654        if *bytes > MAX_EDITOR_SYMBOL_BYTES {
655            return Err(failure("editor symbol byte limit exceeded"));
656        }
657    }
658    symbols.push(ModuleSymbols {
659        path: path.to_owned(),
660        types: names.types.clone(),
661        values: names.values.clone(),
662    });
663    Ok(())
664}

666/// Expand exported types to include only their own constructors.
667fn exported_names(module: &Module, own: &Names) -> Result<Names, Error> {
668    let mut public = Names::default();
669    for declaration in module.syntax.traits.iter().filter(|t| t.public) {
670        public.types.insert(
671            declaration.name.clone(),
672            own.types[&declaration.name].clone(),
673        );
674    }
675    for function in module.syntax.functions.iter().filter(|f| f.public) {
676        public
677            .values
678            .insert(function.name.clone(), own.values[&function.name].clone());
679    }
680    for alias in module.syntax.aliases.iter().filter(|d| d.public) {
681        public
682            .types
683            .insert(alias.name.clone(), own.types[&alias.name].clone());
684    }
685    for decl in module.syntax.newtypes.iter().filter(|d| d.public) {
686        public
687            .types
688            .insert(decl.name.clone(), own.types[&decl.name].clone());
689        export_constructor(&mut public.values, own, &decl.name, &decl.constructor);
690    }
691    for decl in module.syntax.types.iter().filter(|d| d.public) {
692        public
693            .types
694            .insert(decl.name.clone(), own.types[&decl.name].clone());
695        for variant in &decl.variants {
696            export_constructor(&mut public.values, own, &decl.name, &variant.name);
697        }
698    }
699    Ok(public)
700}

704    public.insert(name.into(), own.values[name].clone());

708/// Qualify source type declarations while keeping distinct owner and constructor identities.
709fn qualify_source_types(
710    program: &mut ast::Program,
711    own: &Names,
712    visible: &Names,
713    offset: usize,
714) -> Result<(), Error> {
715    qualify_aliases(&mut program.aliases, own, visible, offset)?;
716    for declaration in &mut program.traits {
717        declaration.name = own.types[&declaration.name].clone();
718        for parent in &mut declaration.parents {
719            qualify_trait_bound(parent, visible, offset)?;
720        }
721        for method in &mut declaration.methods {
722            method.function = own.values[&method.function].clone();
723            if let Some(default) = &mut method.default {
724                *default = own.values[default].clone();
725            }
726        }
727        shift(&mut declaration.span, offset);
728    }
729    for implementation in &mut program.implementations {
730        qualify_trait_bound(&mut implementation.bound, visible, offset)?;
731        for bound in &mut implementation.constraints {
732            qualify_trait_bound(bound, visible, offset)?;
733        }
734        for (_, function) in &mut implementation.methods {
735            *function = own.values[function].clone();
736        }
737        shift(&mut implementation.span, offset);
738    }
739    qualify_declarations(&mut program.types, own, visible, offset)?;
740    for decl in &mut program.newtypes {
741        decl.name = own.types[&decl.name].clone();
742        decl.constructor = own.values[&decl.constructor].clone();
743        qualify_type(&mut decl.inner, visible).map_err(|e| at_span(e, decl.inner_span))?;
744        shift(&mut decl.span, offset);
745        shift(&mut decl.constructor_span, offset);
746        shift(&mut decl.inner_span, offset);
747        for derive in &mut decl.derives {
748            shift(&mut derive.span, offset);
749        }
750    }
751    Ok(())
752}

769/// Qualify owned type identities and field annotations with the module's visibility map.
770fn qualify_declarations(
771    declarations: &mut [ast::TypeDecl],
772    own: &Names,
773    visible: &Names,
774    offset: usize,
775) -> Result<(), Error> {
776    for decl in declarations {
777        decl.name = own.types[&decl.name].clone();
778        shift(&mut decl.span, offset);
779        for derive in &mut decl.derives {
780            shift(&mut derive.span, offset);
781        }
782        for variant in &mut decl.variants {
783            variant.name = own.values[&variant.name].clone();
784            shift(&mut variant.span, offset);
785            for field in &mut variant.fields {
786                qualify_type(&mut field.ty, visible).map_err(|e| at_span(e, field.span))?;
787                shift(&mut field.span, offset);
788            }
789        }
790    }
791    Ok(())
792}

794/// Resolve each clause's parameter bindings before its guard and body, retaining group identity.
795fn qualify_trait_bound(
796    bound: &mut ast::TraitBound,
797    visible: &Names,
798    offset: usize,
799) -> Result<(), Error> {
800    bound.name = visible
801        .types
802        .get(&bound.name)
803        .cloned()
804        .or_else(|| {
805            matches!(
806                bound.name.as_str(),
807                "Show" | "Eq" | "Ord" | "Clone" | "Json"
808            )
809            .then(|| bound.name.clone())
810        })
811        .ok_or_else(|| {
812            at_span(
813                failure(format!("unknown or private trait '{}'", bound.name)),
814                bound.span,
815            )
816        })?;
817    qualify_type(&mut bound.ty, visible).map_err(|e| at_span(e, bound.span))?;
818    shift(&mut bound.span, offset);
819    Ok(())
820}

822fn qualify_function(
823    function: &mut ast::Function,
824    visible: &Names,
825    prefixes: &BTreeSet<String>,
826    offset: usize,
827) -> Result<(), Error> {
828    for bound in &mut function.constraints {
829        qualify_trait_bound(bound, visible, offset)?;
830    }
831    let mut bound = BTreeSet::new();
832    for param in &mut function.params {
833        if let Some(ty) = &mut param.annotation {
834            qualify_type(ty, visible).map_err(|e| at_span(e, param.span))?;
835        }
836        if let Some(label) = &mut param.label {
837            shift(&mut label.span, offset);
838        }
839        pattern(&mut param.pattern, visible, &mut bound, offset)?;
840        shift(&mut param.span, offset);
841    }
842    if let Some(ty) = &mut function.return_type {
843        qualify_type(ty, visible).map_err(|e| at_span(e, function.span))?;
844    }
845    let mut scopes = vec![bound];
846    if let Some(guard) = &mut function.guard {
847        rewrite(guard, visible, prefixes, &mut scopes, offset)?;
848    }
849    rewrite(&mut function.body, visible, prefixes, &mut scopes, offset)?;
850    function.group_start += offset;
851    Ok(())
852}

854/// Map local function/type/constructor identities into a module-qualified namespace.
855fn own_names(module: &Module, entry: bool) -> Result<Names, Error> {
856    let mut names = Names::default();
857    let mut groups = BTreeMap::new();
858    for declaration in &module.syntax.traits {
859        register_type(
860            module,
861            &mut names.types,
862            &declaration.name,
863            declaration.span,
864        )?;
865    }
866    for function in &module.syntax.functions {
867        if groups.get(&function.name) == Some(&function.group_start) {
868            continue;
869        }
870        groups.insert(function.name.clone(), function.group_start);
871        if reserved_declaration(&function.name) {
872            return Err(failure(format!(
873                "{} is reserved for a builtin",
874                function.name
875            )));
876        }
877        let qualified = if entry && function.name == "main" {
878            "main".into()
879        } else {
880            format!("{}.{}", module.name, function.name)
881        };
882        insert(&mut names.values, function.name.clone(), qualified)?;
883    }
884    for decl in &module.syntax.types {
885        register_type(module, &mut names.types, &decl.name, decl.span)?;
886        for variant in &decl.variants {
887            register_constructor(
888                module,
889                &mut names.values,
890                &decl.name,
891                &variant.name,
892                variant.span,
893            )?;
894        }
895    }
896    for alias in &module.syntax.aliases {
897        register_type(module, &mut names.types, &alias.name, alias.span)?;
898    }
899    for decl in &module.syntax.newtypes {
900        register_type(module, &mut names.types, &decl.name, decl.span)?;
901        register_constructor(
902            module,
903            &mut names.values,
904            &decl.name,
905            &decl.constructor,
906            decl.constructor_span,
907        )?;
908    }
909    for namespace in [&mut names.types, &mut names.values] {
910        let aliases: Vec<_> = namespace
911            .iter()
912            .map(|(name, value)| (format!("{}.{name}", module.name), value.clone()))
913            .collect();
914        for (name, value) in aliases {
915            insert(namespace, name, value)?;
916        }
917    }
918    Ok(names)
919}

921/// Register one type identity, rejecting duplicates only in the type namespace.
922fn register_type(
923    module: &Module,
924    names: &mut NameMap,
925    name: &str,
926    span: Span,
927) -> Result<(), Error> {
928    if reserved_declaration(name) {
929        return Err(at_span(
930            failure(format!("type name {name} is reserved")),
931            span,
932        ));
933    }
934    insert(names, name.into(), format!("{}.{name}", module.name)).map_err(|e| at_span(e, span))
935}

938fn register_constructor(
939    module: &Module,
940    names: &mut NameMap,
941    owner: &str,

958        || crate::ffi::reserved_type(name)
959        || crate::ffi::is_api(name)
960        || crate::check::sets::is_api(name)
961        || crate::codec_syntax::is_codec(name)
962        || crate::runtime::native_type(name).is_some()
963        || crate::runtime::lookup(name).is_some()

965        || name.starts_with("Process.")

967        || matches!(
968            name,

973                | "Float"

975                | "Unit"
976                | "List"

981                | "Some"

998/// Expose only exported dependency names, applying selected imports and aliases.
999fn import_names(
1000    import: &ast::Import,
1001    exported: &Names,
1002    visible: &mut Names,
1003    public: &mut Names,
1004    prefixes: &mut BTreeSet<String>,
1005) -> Result<(), Error> {
1006    let prefix = import.alias.as_ref().unwrap_or(&import.module);
1007    if import.items.is_none() {
1008        prefixes.insert(prefix.clone());
1009    }
1010    if let Some(items) = &import.items {
1011        let mut selected = BTreeSet::new();
1012        for item in items.iter().filter(|item| item.as_str() != "*") {
1013            if !selected.insert(item) {
1014                return Err(failure(format!("duplicate import selector: {item}")));
1015            }
1016            if !exported.types.contains_key(item) && !exported.values.contains_key(item) {
1017                return Err(failure(format!(
1018                    "{}: {item} is private or not exported",
1019                    import.module
1020                )));
1021            }
1022        }
1023    }
1024    import_namespace(
1025        import,
1026        &exported.types,
1027        &mut visible.types,
1028        &mut public.types,
1029    )?;
1030    import_namespace(
1031        import,
1032        &exported.values,
1033        &mut visible.values,
1034        &mut public.values,
1035    )
1036}

1038/// Apply identical import syntax independently to each public namespace.
1039fn import_namespace(
1040    import: &ast::Import,
1041    exported: &NameMap,
1042    visible: &mut NameMap,
1043    public: &mut NameMap,
1044) -> Result<(), Error> {
1045    let prefix = import.alias.as_ref().unwrap_or(&import.module);
1046    for (name, value) in exported {
1047        let selected = import.items.as_ref().is_none_or(|items| {
1048            items.iter().any(|item| {
1049                item == "*"
1050                    || name == item
1051                    || name
1052                        .strip_prefix(item)
1053                        .is_some_and(|tail| tail.starts_with('.'))
1054            })
1055        });
1056        if !selected {
1057            continue;
1058        }
1059        let key = if import.items.is_none() {
1060            format!("{prefix}.{name}")
1061        } else {
1062            name.clone()
1063        };
1064        insert(visible, key.clone(), value.clone())?;
1065        if import.public {
1066            insert(public, key, value.clone())?;
1067        }
1068    }
1069    Ok(())
1070}

1083/// Qualify nominal annotations while leaving declared lowercase generic variables intact.
1084fn qualify_type(ty: &mut Type, names: &Names) -> Result<(), Error> {
1085    match ty {
1086        Type::Named(name, args) => {
1087            if !crate::processes::nominal_name(name) {
1088                resolve_global(name, &names.types, false)?;
1089            }
1090            for arg in args {
1091                qualify_type(arg, names)?;
1092            }
1093        }
1094        Type::Function(params, result) => {
1095            for param in params {
1096                qualify_type(param, names)?;
1097            }
1098            qualify_type(result, names)?;
1099        }
1100        Type::Union(fields) | Type::Tuple(fields) => {
1101            for field in fields {
1102                qualify_type(field, names)?;
1103            }
1104        }
1105        Type::Pid(t) | Type::List(t) | Type::Option(t) => qualify_type(t, names)?,
1106        Type::ActorFunction(a, b) | Type::Result(a, b) | Type::Map(a, b) => {
1107            qualify_type(a, names)?;
1108            qualify_type(b, names)?;
1109        }
1110        _ => {}
1111    }
1112    Ok(())
1113}

1115/// Resolve names without rewriting lexical locals or accepting private qualified calls.
1116fn resolve_name(
1117    name: &mut String,
1118    names: &Names,
1119    prefixes: &BTreeSet<String>,
1120    scopes: &[BTreeSet<String>],
1121) -> Result<(), Error> {
1122    if local(scopes, name) {
1123        return Ok(());
1124    }
1125    if !names.values.contains_key(name)
1126        && prefixes.iter().any(|p| name.starts_with(&format!("{p}.")))
1127    {
1128        return Err(failure(format!(
1129            "{name} is private or not exported by its imported module"
1130        )));
1131    }
1132    resolve_global(name, &names.values, true)?;
1133    Ok(())
1134}

1136/// Resolve only visible global identities; merged dependency declarations are not implicit imports.
1137fn resolve_global(name: &mut String, names: &NameMap, allow_builtin: bool) -> Result<(), Error> {
1138    if let Some(value) = names.get(name) {
1139        *name = value.clone();
1140    } else if (name.contains('.') || name == "main") && !(allow_builtin && builtin_path(name)) {
1141        return Err(failure(unresolved_global(name, names, allow_builtin)));
1142    }
1143    Ok(())
1144}

1146/// Describe an unresolved qualified name: builtin modules name their missing member and the
1147/// nearest existing API; other prefixes keep the import guidance with a nearest visible name.
1148fn unresolved_global(name: &str, names: &NameMap, allow_builtin: bool) -> String {
1149    let builtins = crate::check::builtin_api_names();
1150    if allow_builtin
1151        && let Some((prefix, member)) = name.rsplit_once('.')
1152        && builtins.iter().any(|api| {
1153            api.strip_prefix(prefix)
1154                .is_some_and(|rest| rest.starts_with('.'))
1155        })
1156    {
1157        let members = builtins.iter().copied().filter(|api| {
1158            api.strip_prefix(prefix)
1159                .is_some_and(|rest| rest.starts_with('.'))
1160        });
1161        let suggestion = crate::suggest::nearest_member(prefix, member, members)
1162            .map_or_else(String::new, |found| format!("; did you mean '{found}'?"));
1163        return format!("{prefix} has no function '{member}'{suggestion}");
1164    }
1165    format!(
1166        "{name} is private, not exported, or not imported{}",
1167        crate::suggest::hint(name, names.keys().map(String::as_str))
1168    )
1169}

1171/// Builtin-qualified calls need no source import; arbitrary module prefixes do.
1172fn builtin_path(name: &str) -> bool {
1173    if crate::processes::is_api(name) || crate::processes::constructor_path(name) {
1174        return true;
1175    }
1176    crate::check::builtin(name).is_some()
1177        || crate::check::sets::is_api(name)
1178        || crate::ffi::is_api(name)
1179        || crate::codec_syntax::is_codec(name)
1180        || crate::runtime::lookup(name).is_some()
1181        || crate::runtime::omissions()
1182            .iter()
1183            .any(|entry| entry.names.contains(&name))
1184}

1186/// Resolve a codec's callee first, so its static target never enters the value namespace.
1187fn prepare_codec(
1188    expr: &mut ast::Expr,
1189    names: &Names,
1190    scopes: &[BTreeSet<String>],
1191) -> Result<(), Error> {
1192    if let ast::ExprKind::TypeTarget(ty) = &mut expr.kind {
1193        return qualify_type(ty, names);
1194    }
1195    let canonical = match &expr.kind {
1196        ast::ExprKind::Call { name, .. } | ast::ExprKind::Pipe { name, .. }
1197            if !local(scopes, name) =>
1198        {
1199            names.values.get(name).unwrap_or(name).clone()
1200        }
1201        ast::ExprKind::GlobalCall { resolved, .. } | ast::ExprKind::GlobalPipe { resolved, .. } => {
1202            resolved.clone()
1203        }
1204        _ => return Ok(()),
1205    };
1206    crate::codec_syntax::prepare(expr, &canonical).map_err(|e| at_span(failure(e.message), e.span))
1207}

1209/// Rewrite expression identities and relocate byte spans; block-local bindings stay local.
1210fn rewrite(
1211    expr: &mut ast::Expr,
1212    names: &Names,
1213    prefixes: &BTreeSet<String>,
1214    scopes: &mut Vec<BTreeSet<String>>,
1215    offset: usize,
1216) -> Result<(), Error> {
1217    let original = expr.span;
1218    prepare_codec(expr, names, scopes).map_err(|e| at_span(e, original))?;
1219    shift(&mut expr.span, offset);
1220    mark_global(&mut expr.kind, names, prefixes, scopes).map_err(|e| at_span(e, original))?;
1221    match &mut expr.kind {
1222        ast::ExprKind::ForeignCall { declaration, args } => {
1223            for abi in declaration
1224                .params
1225                .iter_mut()
1226                .chain([&mut declaration.result])
1227            {
1228                if let crate::ffi::AbiType::Pointer(ty) = abi {
1229                    qualify_type(ty, names)?;
1230                }
1231            }
1232            rewrite_values(args, names, prefixes, scopes, offset)?;
1233        }
1234        ast::ExprKind::Name(_) | ast::ExprKind::GlobalName { .. } => {}
1235        ast::ExprKind::Pipe {
1236            value, args, label, ..
1237        }
1238        | ast::ExprKind::GlobalPipe {
1239            value, args, label, ..
1240        } => {
1241            if let Some(label) = label {
1242                shift(&mut label.span, offset);
1243            }
1244            rewrite(value, names, prefixes, scopes, offset)?;
1245            rewrite_arguments(args, names, prefixes, scopes, offset)?;
1246        }
1247        ast::ExprKind::Lambda { params, body } => {
1248            rewrite_lambda(params, body, names, prefixes, scopes, offset)?
1249        }
1250        ast::ExprKind::Apply { callee, args } => {
1251            rewrite(callee, names, prefixes, scopes, offset)?;
1252            rewrite_arguments(args, names, prefixes, scopes, offset)?;
1253        }
1254        ast::ExprKind::Interpolate(parts) | ast::ExprKind::MultilineString(parts) => {
1255            rewrite_string(parts, names, prefixes, scopes, offset)?
1256        }
1257        ast::ExprKind::Call { args, .. } | ast::ExprKind::GlobalCall { args, .. } => {
1258            rewrite_arguments(args, names, prefixes, scopes, offset)?;
1259        }
1260        ast::ExprKind::Tuple(values) | ast::ExprKind::List(values) => {
1261            rewrite_values(values, names, prefixes, scopes, offset)?;
1262        }
1263        ast::ExprKind::Try(value)
1264        | ast::ExprKind::Return(value)
1265        | ast::ExprKind::Defer(value)
1266        | ast::ExprKind::Unary { value, .. }
1267        | ast::ExprKind::Field { value, .. } => rewrite(value, names, prefixes, scopes, offset)?,
1268        kind @ (ast::ExprKind::Binary { .. }
1269        | ast::ExprKind::Range { .. }
1270        | ast::ExprKind::If { .. }
1271        | ast::ExprKind::PostfixIf { .. }
1272        | ast::ExprKind::ConditionMatch(_)) => {
1273            rewrite_control(kind, names, prefixes, scopes, offset)?
1274        }
1275        ast::ExprKind::Block(stmts) => rewrite_block(stmts, names, prefixes, scopes, offset)?,
1276        ast::ExprKind::Match { value, arms } => {
1277            rewrite_match(value, arms, names, prefixes, scopes, offset)?
1278        }
1279        ast::ExprKind::Map(pairs) => rewrite_pairs(pairs, names, prefixes, scopes, offset)?,
1280        ast::ExprKind::RecordUpdate { value, fields } => {
1281            rewrite_update(value, fields, names, prefixes, scopes, offset)?
1282        }
1283        kind @ (ast::ExprKind::Receive { .. }
1284        | ast::ExprKind::For { .. }
1285        | ast::ExprKind::With { .. }) => {
1286            rewrite_binding_flow(kind, names, prefixes, scopes, offset)?
1287        }
1288        _ => {}
1289    }
1290    Ok(())
1291}

1293/// Dispatch expressions whose bindings have success-only or per-iteration scope.
1294fn rewrite_binding_flow(
1295    kind: &mut ast::ExprKind,
1296    names: &Names,
1297    prefixes: &BTreeSet<String>,
1298    scopes: &mut Vec<BTreeSet<String>>,
1299    offset: usize,
1300) -> Result<(), Error> {
1301    match kind {
1302        ast::ExprKind::Receive { .. } => rewrite_receive(kind, names, prefixes, scopes, offset),
1303        ast::ExprKind::For {
1304            pattern,
1305            iterable,
1306            body,
1307        } => rewrite_for(pattern, iterable, body, names, prefixes, scopes, offset),
1308        ast::ExprKind::With {
1309            bindings,
1310            body,
1311            arms,
1312        } => rewrite_with(bindings, body, arms, names, prefixes, scopes, offset),
1313        _ => Ok(()),
1314    }
1315}

1317/// Loop iterable names resolve before its per-iteration pattern bindings become visible.
1318fn rewrite_for(
1319    binding: &mut ast::Pattern,
1320    iterable: &mut ast::Expr,
1321    body: &mut ast::Expr,
1322    names: &Names,
1323    prefixes: &BTreeSet<String>,
1324    scopes: &mut Vec<BTreeSet<String>>,
1325    offset: usize,
1326) -> Result<(), Error> {
1327    rewrite(iterable, names, prefixes, scopes, offset)?;
1328    scopes.push(BTreeSet::new());
1329    pattern(binding, names, scopes.last_mut().unwrap(), offset)?;
1330    let result = rewrite(body, names, prefixes, scopes, offset);
1331    scopes.pop();
1332    result
1333}

1335/// With success scopes accumulate sequentially; error arms see only the outer scope.
1336fn rewrite_with(
1337    bindings: &mut [ast::WithBinding],
1338    body: &mut ast::Expr,
1339    arms: &mut Option<Vec<ast::MatchArm>>,
1340    names: &Names,
1341    prefixes: &BTreeSet<String>,
1342    scopes: &mut Vec<BTreeSet<String>>,
1343    offset: usize,
1344) -> Result<(), Error> {
1345    scopes.push(BTreeSet::new());
1346    for binding in bindings {
1347        rewrite(&mut binding.value, names, prefixes, scopes, offset)?;
1348        pattern(
1349            &mut binding.pattern,
1350            names,
1351            scopes.last_mut().unwrap(),
1352            offset,
1353        )?;
1354        shift(&mut binding.span, offset);
1355    }
1356    rewrite(body, names, prefixes, scopes, offset)?;
1357    scopes.pop();
1358    if let Some(arms) = arms {
1359        for arm in arms {
1360            rewrite_arm(arm, names, prefixes, scopes, offset)?;
1361        }
1362    }
1363    Ok(())
1364}

1366/// Resolve conditions and branch expressions without exposing branch-local bindings.
1367fn rewrite_control(
1368    kind: &mut ast::ExprKind,
1369    names: &Names,
1370    prefixes: &BTreeSet<String>,
1371    scopes: &mut Vec<BTreeSet<String>>,
1372    offset: usize,
1373) -> Result<(), Error> {
1374    match kind {
1375        ast::ExprKind::Binary { left, right, .. }
1376        | ast::ExprKind::Range {
1377            start: left,
1378            end: right,
1379            ..
1380        } => {
1381            rewrite(left, names, prefixes, scopes, offset)?;
1382            rewrite(right, names, prefixes, scopes, offset)?;
1383        }
1384        ast::ExprKind::If {
1385            condition,
1386            then_branch,
1387            else_branch,
1388        } => {
1389            rewrite(condition, names, prefixes, scopes, offset)?;
1390            rewrite(then_branch, names, prefixes, scopes, offset)?;
1391            if let Some(value) = else_branch {
1392                rewrite(value, names, prefixes, scopes, offset)?;
1393            }
1394        }
1395        ast::ExprKind::PostfixIf { value, condition } => {
1396            rewrite(condition, names, prefixes, scopes, offset)?;
1397            rewrite(value, names, prefixes, scopes, offset)?;
1398        }
1399        ast::ExprKind::ConditionMatch(arms) => {
1400            for arm in arms {
1401                shift(&mut arm.span, offset);
1402                if let Some(condition) = &mut arm.condition {
1403                    rewrite(condition, names, prefixes, scopes, offset)?;
1404                }
1405                rewrite(&mut arm.body, names, prefixes, scopes, offset)?;
1406            }
1407        }
1408        _ => {}
1409    }
1410    Ok(())
1411}

1493/// Keep pattern bindings local to one arm while relocating its guard and body spans.
1494fn rewrite_arm(
1495    arm: &mut ast::MatchArm,
1496    names: &Names,
1497    prefixes: &BTreeSet<String>,
1498    scopes: &mut Vec<BTreeSet<String>>,
1499    offset: usize,
1500) -> Result<(), Error> {
1501    scopes.push(BTreeSet::new());
1502    pattern(&mut arm.pattern, names, scopes.last_mut().unwrap(), offset)?;
1503    if let Some(guard) = &mut arm.guard {
1504        rewrite(guard, names, prefixes, scopes, offset)?;
1505    }
1506    rewrite(&mut arm.body, names, prefixes, scopes, offset)?;
1507    shift(&mut arm.span, offset);
1508    scopes.pop();
1509    Ok(())
1510}

1512/// Resolve callback annotations before introducing its parameter scope for the body.
1513fn rewrite_lambda(
1514    params: &mut [ast::LambdaParam],
1515    body: &mut ast::Expr,
1516    names: &Names,
1517    prefixes: &BTreeSet<String>,
1518    scopes: &mut Vec<BTreeSet<String>>,
1519    offset: usize,
1520) -> Result<(), Error> {
1521    let mut bound = BTreeSet::new();
1522    for param in params {
1523        if let Some(ty) = &mut param.annotation {
1524            qualify_type(ty, names).map_err(|e| at_span(e, param.span))?;
1525        }
1526        shift(&mut param.span, offset);
1527        bound.insert(param.name.clone());
1528    }
1529    scopes.push(bound);
1530    let result = rewrite(body, names, prefixes, scopes, offset);
1531    scopes.pop();
1532    result
1533}

1551/// Preserve lexical block scopes while resolving annotations and binding initializers.
1552fn rewrite_block(
1553    stmts: &mut [ast::Stmt],
1554    names: &Names,
1555    prefixes: &BTreeSet<String>,
1556    scopes: &mut Vec<BTreeSet<String>>,
1557    offset: usize,
1558) -> Result<(), Error> {
1559    scopes.push(BTreeSet::new());
1560    for stmt in stmts {
1561        match stmt {
1562            ast::Stmt::LetElse { .. } => rewrite_let_else(stmt, names, prefixes, scopes, offset)?,
1563            ast::Stmt::Let {
1564                name,
1565                annotation,
1566                value,
1567                span,
1568            } => {
1569                rewrite(value, names, prefixes, scopes, offset)?;
1570                if let Some(ty) = annotation {
1571                    qualify_type(ty, names).map_err(|e| at_span(e, *span))?;
1572                }
1573                shift(span, offset);
1574                scopes.last_mut().unwrap().insert(name.clone());
1575            }
1576            ast::Stmt::LetPattern {
1577                pattern: binding,
1578                annotation,
1579                value,
1580                span,
1581            } => {
1582                rewrite(value, names, prefixes, scopes, offset)?;
1583                if let Some(ty) = annotation {
1584                    qualify_type(ty, names).map_err(|e| at_span(e, *span))?;
1585                }
1586                pattern(binding, names, scopes.last_mut().unwrap(), offset)?;
1587                shift(span, offset);
1588            }
1589            ast::Stmt::Expr(value) => rewrite(value, names, prefixes, scopes, offset)?,
1590        }
1591    }
1592    scopes.pop();
1593
1594    Ok(())
1595}

1597/// The failure branch sees the outer scope; successful bindings begin after its initializer.
1598fn rewrite_let_else(
1599    stmt: &mut ast::Stmt,
1600    names: &Names,
1601    prefixes: &BTreeSet<String>,
1602    scopes: &mut Vec<BTreeSet<String>>,
1603    offset: usize,
1604) -> Result<(), Error> {
1605    if let ast::Stmt::LetElse {
1606        pattern: binding,
1607        annotation,
1608        value,
1609        else_branch,
1610        span,
1611    } = stmt
1612    {
1613        rewrite(value, names, prefixes, scopes, offset)?;
1614        rewrite(else_branch, names, prefixes, scopes, offset)?;
1615        if let Some(ty) = annotation {
1616            qualify_type(ty, names).map_err(|e| at_span(e, *span))?;
1617        }
1618        pattern(binding, names, scopes.last_mut().unwrap(), offset)?;
1619        shift(span, offset);
1620    }
1621    Ok(())
1622}

1624/// Qualify constructor patterns and gather all nested lexical payload bindings.
1625fn pattern(
1626    pattern: &mut ast::Pattern,
1627    names: &Names,
1628    bound: &mut BTreeSet<String>,
1629    offset: usize,
1630) -> Result<(), Error> {
1631    let original = pattern.span;
1632    shift(&mut pattern.span, offset);
1633    match &mut pattern.kind {
1634        ast::PatternKind::Typed {
1635            pattern: inner,
1636            annotation,
1637        } => {
1638            qualify_type(annotation, names).map_err(|e| at_span(e, original))?;
1639            pattern_binding(inner, names, bound, offset)?;
1640        }
1641        ast::PatternKind::List { prefix, rest } => {
1642            for field in prefix {
1643                pattern_binding(field, names, bound, offset)?;
1644            }
1645            if let Some(rest) = rest {
1646                pattern_binding(rest, names, bound, offset)?;
1647            }
1648        }
1649        ast::PatternKind::TupleRest { prefix, rest } => {
1650            for field in prefix {
1651                pattern_binding(field, names, bound, offset)?;
1652            }
1653            pattern_binding(rest, names, bound, offset)?;
1654        }
1655        ast::PatternKind::Tuple(fields) => {
1656            for field in fields {
1657                pattern_binding(field, names, bound, offset)?;
1658            }
1659        }
1660        ast::PatternKind::Bind(name) => {
1661            bound.insert(name.clone());
1662        }
1663        ast::PatternKind::Constructor {
1664            binding: Some(name),
1665            ..
1666        } => {
1667            bound.insert(name.clone());
1668        }
1669        ast::PatternKind::NamedConstructor { name, fields } => {
1670            resolve_global(
1671                name,
1672                &names.values,
1673                crate::processes::constructor_path(name),
1674            )
1675            .map_err(|e| at_span(e, original))?;
1676            for field in fields {
1677                pattern_binding(field, names, bound, offset)?;
1678            }
1679        }
1680        _ => {}
1681    }
1682    Ok(())
1683}
1684fn pattern_binding(
1685    value: &mut ast::Pattern,
1686    names: &Names,
1687    bound: &mut BTreeSet<String>,
1688    offset: usize,
1689) -> Result<(), Error> {
1690    pattern(value, names, bound, offset)
1691}

1693/// Mark a proven global before canonical qualification can collide with a lexical root.
1694fn mark_global(
1695    kind: &mut ast::ExprKind,
1696    names: &Names,
1697    prefixes: &BTreeSet<String>,
1698    scopes: &[BTreeSet<String>],
1699) -> Result<(), Error> {
1700    // A visible constant can own record fields; only its visible declaration
1701    // prefix is qualified, so a field cannot bypass module export filtering.
1702    if let ast::ExprKind::Name(name) = kind
1703        && !local(scopes, name)
1704        && !names.values.contains_key(name)
1705    {
1706        for (offset, _) in name.match_indices('.').rev() {
1707            if let Some(root) = names.values.get(&name[..offset]) {
1708                let resolved = format!("{root}{}", &name[offset..]);
1709                *kind = ast::ExprKind::GlobalName {
1710                    name: name.clone(),
1711                    resolved,
1712                };
1713                return Ok(());
1714            }
1715        }
1716    }
1717    let name = match kind {
1718        ast::ExprKind::Name(name)
1719        | ast::ExprKind::Call { name, .. }
1720        | ast::ExprKind::Pipe { name, .. } => name,
1721        _ => return Ok(()),
1722    };
1723    if local(scopes, name) {
1724        return Ok(());
1725    }
1726    let mut resolved = name.clone();
1727    resolve_name(&mut resolved, names, prefixes, scopes)?;
1728    if !names.values.contains_key(name) {
1729        return Ok(());
1730    }
1731    *kind = match std::mem::replace(kind, ast::ExprKind::Unit) {
1732        ast::ExprKind::Name(name) => ast::ExprKind::GlobalName { name, resolved },
1733        ast::ExprKind::Call { name, args } => ast::ExprKind::GlobalCall {
1734            name,
1735            resolved,
1736            args,
1737        },
1738        ast::ExprKind::Pipe {
1739            value,
1740            name,
1741            args,
1742            position,
1743            label,
1744        } => ast::ExprKind::GlobalPipe {
1745            value,
1746            name,
1747            resolved,
1748            args,
1749            position,
1750            label,
1751        },
1752        _ => unreachable!("only named references reach global marking"),
1753    };
1754    Ok(())
1755}

1757/// Rewrite each receive arm in its own binder scope; timeout expressions use the outer scope.
1758fn rewrite_receive(
1759    kind: &mut ast::ExprKind,
1760    names: &Names,
1761    prefixes: &BTreeSet<String>,
1762    scopes: &mut Vec<BTreeSet<String>>,
1763    offset: usize,
1764) -> Result<(), Error> {
1765    if let ast::ExprKind::Receive { arms, timeout, .. } = kind {
1766        for arm in arms {
1767            rewrite_arm(arm, names, prefixes, scopes, offset)?;
1768        }
1769        if let Some((duration, body)) = timeout {
1770            rewrite(duration, names, prefixes, scopes, offset)?;
1771            rewrite(body, names, prefixes, scopes, offset)?;
1772        }
1773    }
1774    Ok(())
