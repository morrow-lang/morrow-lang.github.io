1//! Heap closures use a code pointer followed by full-width lexical captures.
2use super::*;

4impl Emitter<'_> {
5    /// Bind checked capture identities from the uniform environment parameter.
6    pub(super) fn load_captures(
7        &mut self,
8        function: &Function,
9        locals: &mut Locals,
10    ) -> Lowering<()> {
11        for (index, capture) in function.captures.iter().enumerate() {
12            let address = self.assign(
13                locals,
14                Type::Int,
15                NativeOperation::Binary(
16                    MachineBinary::Add,
17                    native_operand("%env"),
18                    native_operand(&(8 * (index + 1)).to_string()),
19                ),
20            );
21            let raw = self.assign(
22                locals,
23                Type::Int,
24                NativeOperation::Load(LoadKind::I64, native_operand(&(address))),
25            );
26            let value = self.unpack(locals, &capture.ty, raw);
27            locals.define(capture.id.0, capture.ty.clone(), value, function.body.span)?;
28        }
29        Ok(())
30    }

32    /// Validate capture arity/types before allocating an escaping lexical environment.
33    pub(super) fn closure(
34        &mut self,
35        id: ir::FunctionId,
36        captures: &[Expr],
37        span: Span,
38        locals: &mut Locals,
39        depth: usize,
40    ) -> Lowering<(Type, String)> {
41        let function = self
42            .functions
43            .get(&id.0)
44            .ok_or_else(|| invalid(span, "unknown closure function identity"))?;
45        if captures.len() != function.captures.len() || captures.len() > MAX_NODES {
46            return Err(invalid(
47                span,
48                "closure capture count differs from lifted function",
49            ));
50        }
51        let mut ty = Type::Function(
52            function.params.iter().map(|p| p.ty.clone()).collect(),
53            Box::new(function.return_type.clone()),
54        );
55        if let Some(mailbox) = &function.mailbox {
56            ty = Type::ActorFunction(Box::new(mailbox.clone()), Box::new(ty));
57        }
58        let expected: Vec<_> = function.captures.iter().map(|p| p.ty.clone()).collect();
59        let mut values = Vec::new();
60        for (capture, expected) in captures.iter().zip(expected) {
61            expect_type(capture.ty.clone(), expected, capture.span)?;
62            let value = self.expr(capture, locals, depth)?;
63            values.push(self.payload(locals, &capture.ty, value));
64        }
65        let object = self.assign(
66            locals,
67            ty.clone(),
68            NativeOperation::Call {
69                callee: native_operand("$morrow_alloc"),
70                args: vec![(
71                    Scalar::I64,
72                    native_operand(&(8 * (values.len() + 1)).to_string()),
73                )],
74                variadic: None,
75            },
76        );
77        let identity = if self.actors.entries.contains_key(&id.0) {
78            format!("actor_identity{}", id.0)
79        } else {
80            format!("f{}", id.0)
81        };
82        self.output.statement(Statement::Store {
83            kind: LoadKind::I64,
84            value: native_operand(&format!("${}", identity)),
85            address: native_operand(&(object)),
86        });
87        for (index, value) in values.iter().enumerate() {
88            let address = self.assign(
89                locals,
90                Type::Int,
91                NativeOperation::Binary(
92                    MachineBinary::Add,
93                    native_operand(&(object).to_string()),
94                    native_operand(&(8 * (index + 1)).to_string()),
95                ),
96            );
97            self.output.statement(Statement::Store {
98                kind: LoadKind::I64,
99                value: native_operand(&(value).to_string()),
100                address: native_operand(&(address)),
101            });
102        }
103        Ok((ty, object))
104    }

106    /// Evaluate the callee first and each argument once before indirect invocation.
107    pub(super) fn invoke(
108        &mut self,
109        callee: &Expr,
110        args: &[Expr],
111        span: Span,
112        locals: &mut Locals,
113        depth: usize,
114    ) -> Lowering<(Type, String)> {
115        let Type::Function(params, result) = &callee.ty else {
116            return Err(invalid(span, "invocation requires function value"));
117        };
118        if params.len() != args.len() {
119            return Err(invalid(
120                span,
121                "invocation argument count differs from signature",
122            ));
123        }
124        let closure = self.expr(callee, locals, depth)?;
125        let mut values = Vec::new();
126        for (arg, param) in args.iter().zip(params) {
127            expect_type(arg.ty.clone(), param.clone(), arg.span)?;
128            values.push((param.clone(), self.expr(arg, locals, depth)?));
129        }
130        let value = self.invoke_values(&closure, &values, result, locals);
131        Ok((*result.clone(), value))
132    }
133
134    /// Invoke an already evaluated, signature-checked callback through its stored code.
135    pub(super) fn invoke_values(
136        &mut self,
137        closure: &str,
138        args: &[(Type, String)],
139        result: &Type,
140        locals: &mut Locals,
141    ) -> String {
142        let code = self.assign(
143            locals,
144            Type::Int,
145            NativeOperation::Load(LoadKind::I64, native_operand(closure)),
146        );
147        let mut arguments = vec![
148            (Scalar::I64, native_operand(closure)),
149            (Scalar::I64, native_operand("%fault")),
150        ];
151        arguments.extend(
152            args.iter()
153                .map(|(ty, value)| (machine_width(self.width(ty.clone())), native_operand(value))),
154        );
155        let instruction = NativeOperation::Call {
156            callee: native_operand(&(code)),
157            args: arguments.clone(),
158            variadic: None,
159        };
160        let value = if *result == Type::Unit {
161            self.output.statement(Statement::Effect(instruction));
162            "0".into()
163        } else {
164            self.assign(locals, result.clone(), instruction)
165        };
166        self.guard_fault(locals);
167        value
168    }
