1//! Value-return continuations use pure frame factories, avoiding retained native stacks.
2use super::*;

4impl Builder<'_> {
5    /// Completion stages a typed return value into a fresh caller frame.
6    pub(super) fn finish(&mut self, value: Expr, next: Option<&Continuation>) -> Lowering<Expr> {
7        if let Some(next) = next {
8            // Branches share the continuation identity, but every defining occurrence
9            // requires its own SSA source local in the generated physical function.
10            let id = self.local(value.span)?;
11            let mut next = next.clone();
12            let ExprKind::Closure { captures, .. } = &mut next.entry.kind else {
13                return Err(invalid(value.span, "invalid actor continuation recipe"));
14            };
15            for capture in captures {
16                if matches!(capture.kind, ExprKind::Local(found) if found == next.value.id) {
17                    capture.kind = ExprKind::Local(id);
18                }
19            }
20            next.value.id = id;
21            return Ok(finish(value, Some(&next)));
22        }
23        let span = value.span;
24        let mut prefix = Vec::new();
25        let value = if self.scoped {
26            let id = self.local(span)?;
27            let ty = value.ty.clone();
28            prefix.push(Stmt::Let { id, value });
29            prefix.push(Stmt::Expr(operation(Operation::ScopeLeave, span)));
30            node(ExprKind::Local(id), ty, span)
31        } else {
32            value
33        };
34        let completion = if let Some(target) = &self.return_to {
35            let frame = node(
36                ExprKind::Invoke {
37                    callee: Box::new(node(ExprKind::Local(target.id), target.ty.clone(), span)),
38                    args: vec![value],
39                },
40                frame_type(),
41                span,
42            );
43            operation(Operation::Continue(Box::new(frame)), span)
44        } else {
45            finish(value, None)
46        };
47        if prefix.is_empty() {
48            Ok(completion)
49        } else {
50            prefix.push(Stmt::Expr(completion));
51            Ok(node(ExprKind::Block(prefix), Type::Int, span))
52        }
53    }

55    /// Only source function entry starts a scope; later callbacks retain that same activation.
56    pub(super) fn enter_scope(&self, body: Expr) -> Expr {
57        if !self.scoped {
58            return body;
59        }
60        let span = body.span;
61        node(
62            ExprKind::Block(vec![
63                Stmt::Expr(operation(Operation::ScopeEnter, span)),
64                Stmt::Expr(body),
65            ]),
66            Type::Int,
67            span,
68        )
69    }

71    /// Actor copies receive a return factory in addition to their ordinary lexical arguments.
72    pub(super) fn returning_function(&mut self, original: &Function, body: &Expr) -> Lowering<()> {
73        let target = ir::Param {
74            id: self.local(body.span)?,
75            ty: Type::Function(vec![original.return_type.clone()], Box::new(frame_type())),
76        };
77        self.return_to = Some(target.clone());
78        let body = self.expression(body, None, 0);
79        self.return_to = None;
80        let body = self.enter_scope(body?);
81        let mut function = self.function(body, vec![], false)?;
82        self.plan.steps.remove(&function.id.0);
83        self.plan.generic_steps.remove(&function.id.0);
84        function.id = ir::FunctionId(self.returning[&original.id.0]);
85        function.captures = original
86            .captures
87            .iter()
88            .chain(&original.params)
89            .cloned()
90            .chain([target])
91            .collect();
92        self.plan.steps.insert(function.id.0, self.mailbox.clone());
93        if self.generic {
94            self.plan.generic_steps.insert(function.id.0);
95        }
96        self.plan.functions.push(function);
97        Ok(())
98    }

100    /// A pure factory only packs values; the next source operation runs in a later callback.
101    pub(super) fn returning_call(
102        &mut self,
103        id: ir::FunctionId,
104        args: &[Expr],
105        next: Option<&Continuation>,
106        span: Span,
107    ) -> Lowering<Expr> {
108        self.returning_closure_call(id, &[], args, next, span)
109    }

111    pub(super) fn returning_closure_call(
112        &mut self,
113        id: ir::FunctionId,
114        environment: &[Expr],
115        args: &[Expr],
116        next: Option<&Continuation>,
117        span: Span,
118    ) -> Lowering<Expr> {
119        let target = self.targets[&id.0].clone();
120        if environment.is_empty()
121            && next.is_none()
122            && self.return_to.is_none()
123            && !self.scoped
124            && target.mailbox.is_none()
125            && target.return_type == Type::Unit
126        {
127            return self.helper_call(id, args, span);
128        }
129        if environment.len() != target.captures.len() || args.len() != target.params.len() {
130            return Err(invalid(span, "invalid returning helper signature"));
131        }
132        for (arg, param) in environment
133            .iter()
134            .chain(args)
135            .zip(target.captures.iter().chain(&target.params))
136        {
137            atomic(arg)?;
138            expect_type(arg.ty.clone(), param.ty.clone(), arg.span)?;
139        }
140        let factory = if let Some(target) = self
141            .return_to
142            .as_ref()
143            .filter(|_| next.is_none() && !self.scoped)
144        {
145            node(ExprKind::Local(target.id), target.ty.clone(), span)
146        } else {
147            let (value, entry) = if let Some(next) = next {
148                (next.value.clone(), next.entry.clone())
149            } else {
150                let value = ir::Param {
151                    id: self.local(span)?,
152                    ty: target.return_type.clone(),
153                };
154                let completion = self.finish(
155                    node(ExprKind::Local(value.id), value.ty.clone(), span),
156                    None,
157                )?;
158                let entry = self.closure(completion, vec![], false)?;
159                (value, entry)
160            };
161            expect_type(value.ty.clone(), target.return_type.clone(), span)?;
162            let mut factory = self.function(entry, vec![value], false)?;
163            self.plan.steps.remove(&factory.id.0);
164            self.plan.generic_steps.remove(&factory.id.0);
165            factory.return_type = frame_type();
166            let closure = node(
167                ExprKind::Closure {
168                    function: factory.id,
169                    captures: factory
170                        .captures
171                        .iter()
172                        .map(|p| node(ExprKind::Local(p.id), p.ty.clone(), span))
173                        .collect(),
174                },
175                Type::Function(vec![target.return_type], Box::new(frame_type())),
176                span,
177            );
178            self.plan.functions.push(factory);
179            closure
180        };
181        let mut captures: Vec<_> = environment.iter().chain(args).cloned().collect();
182        captures.push(factory);
183        let entry = node(
184            ExprKind::Closure {
185                function: ir::FunctionId(self.returning[&id.0]),
186                captures,
187            },
188            frame_type(),
189            span,
190        );
191        Ok(operation(Operation::Continue(Box::new(entry)), span))
192    }

194    pub(super) fn needs(&self, expr: &Expr) -> bool {
195        needs(expr)
196            || tail_helpers::expression_work(expr, &self.inline_work) > tail_helpers::STEP_WORK
197            || matches!(expr.kind, ExprKind::Invoke { .. })
198            || matches!(expr.kind, ExprKind::Call { target: CallTarget::Builtin(b), .. } if crate::lowering::higher_order::is_higher_order(b))
199            || matches!(&expr.kind, ExprKind::Call { target: CallTarget::Function(id), .. } if self.returning.contains_key(&id.0))
200            || ir::children(expr).into_iter().any(|e| self.needs(e))
201    }
202}
203fn frame_type() -> Type {
204    Type::Function(vec![], Box::new(Type::Int))
