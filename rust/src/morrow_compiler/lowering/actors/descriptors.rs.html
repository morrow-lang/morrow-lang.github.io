5    /// Describe every known closure identity before runtime registration validates capture layouts.
6    pub(in crate::lowering) fn actor_descriptors(&mut self) -> Lowering<()> {
7        if !self.actors.active {
8            return Ok(());
9        }
10        let functions: Vec<_> = self.functions.values().copied().collect();
11        let mut table = Vec::new();
12        for function in functions {
13            let id = function.id.0;
14            let mut captures = Vec::new();
15            for capture in &function.captures {
16                captures.push(self.actor_type(&capture.ty)?);
17            }
18            let captures = self.actor_array(&format!("actor_captures{id}"), &captures);
19            let mailbox = function
20                .mailbox
21                .as_ref()
22                .or_else(|| {
23                    (!self.actors.generic_steps.contains(&id))
24                        .then(|| self.actors.steps.get(&id))
25                        .flatten()
26                })
27                .or_else(|| self.actors.selectors.get(&id))
28                .cloned();
29            let mailbox = mailbox
30                .as_ref()
31                .map(|ty| self.actor_type(ty))
32                .transpose()?
33                .unwrap_or_else(|| "0".into());
34            let identity = if self.actors.entries.contains_key(&id) {
35                self.data.data(
36                    &format!("$actor_identity{}", id),
37                    vec![DataValue::Word(native_operand(&(id).to_string()))],
38                );
39                format!("$actor_identity{id}")
40            } else {
41                format!("$f{id}")
42            };
43            let selector = self.actors.selectors.contains_key(&id);
44            let (step, select) = if selector {
45                ("0".into(), format!("$actor_callback{id}"))
46            } else {
47                (format!("$actor_callback{id}"), "0".into())
48            };
49            self.data.data(
50                &format!("$actor_descriptor{}", id),
51                vec![
52                    DataValue::Word(native_operand(&(identity))),
53                    DataValue::Word(native_operand(&(step))),
54                    DataValue::Word(native_operand(&(select))),
55                    DataValue::Word(native_operand(&(function.captures.len()).to_string())),
56                    DataValue::Word(native_operand(&(captures))),
57                    DataValue::Word(native_operand(&(mailbox))),
58                ],
59            );
60            table.push(format!("$actor_descriptor{id}"));
61            self.actor_callback(function, selector);
62        }
63        self.actor_array("actor_functions", &table);
64        Ok(())
65    }

67    /// Bridge only validated step/selector signatures; unrelated callable entries cannot be spawned.
68    fn actor_callback(&mut self, function: &Function, selector: bool) {
69        let id = function.id.0;
70        let target = self
71            .actors
72            .entries
73            .get(&id)
74            .or_else(|| self.actors.helpers.get(&id))
75            .copied()
76            .unwrap_or(id);
77        let generated = self.actors.steps.contains_key(&target);
78        let mut params = vec![(Scalar::I64, "%exec".into()), (Scalar::I64, "%env".into())];
79        if selector {
80            params.push((Scalar::I64, "%payload".into()));
81        }
82        self.output.begin(
83            &format!("$actor_callback{id}"),
84            Some(Scalar::I64),
85            params,
86            false,
87        );
88        self.output.statement(Statement::Label("@start".to_owned()));
89        if !selector
90            && (!function.params.is_empty() || (!generated && function.return_type != Type::Unit))
91        {
92            self.output
93                .statement(Statement::Return(Some(native_operand("3"))));
94            self.output.end();
95            return;
96        }
97        self.output.statement(Statement::Assign {
98            destination: "%fault".to_owned(),
99            ty: Scalar::I64,
100            operation: NativeOperation::Call {
101                callee: native_operand("$morrow_managed_fault"),
102                args: vec![(Scalar::I64, native_operand("%exec"))],
103                variadic: None,
104            },
105        });
106        let mut args = vec![
107            (Scalar::I64, native_operand("%env")),
108            (Scalar::I64, native_operand("%fault")),
109        ];
110        if self.actors.managed.contains(&target) {
111            args.push((Scalar::I64, native_operand("%exec")));
112        }
113        if selector {
114            let ty = &function.params[0].ty;
115            let width = self.width(ty.clone());
116            let value = if width == 'l' {
117                "%payload"
118            } else {
119                self.output.statement(Statement::Assign {
120                    destination: "%decoded".into(),
121                    ty: machine_width(width),
122                    operation: NativeOperation::Unary(
123                        if width == 'd' {
124                            MachineUnary::Cast
125                        } else {
126                            MachineUnary::Copy
127                        },
128                        native_operand("%payload"),
129                    ),
130                });
131                "%decoded"
132            };
133            args.push((machine_width(width), native_operand(value)));
134        }
135        if generated || selector {
136            self.output.statement(Statement::Assign {
137                destination: "%status".to_owned(),
138                ty: Scalar::I64,
139                operation: NativeOperation::Call {
140                    callee: native_operand(&format!("$f{}", target)),
141                    args: args.clone(),
142                    variadic: None,
143                },
144            });
145            self.output
146                .statement(Statement::Return(Some(native_operand("%status"))));
147            self.output.end();
148        } else {
149            self.output
150                .statement(Statement::Effect(NativeOperation::Call {
151                    callee: native_operand(&format!("$f{}", target)),
152                    args: args.clone(),
153                    variadic: None,
154                }));
155            self.output
156                .statement(Statement::Return(Some(native_operand("2"))));
157            self.output.end();
158        }
159    }

161    /// Deduplicate by semantic type, reserving identity before descending recursive nominal fields.
162    pub(super) fn actor_type(&mut self, ty: &Type) -> Lowering<String> {
163        if let Some(id) = self.actors.types.get(ty) {
164            return Ok(format!("$actor_type{id}"));
165        }
166        if self.actors.types.len() >= 4096 {
167            return Err(invalid(
168                Span::default(),
169                "actor type descriptor limit exceeded",
170            ));
171        }
172        let id = self.actors.types.len();
173        self.actors.types.insert(ty.clone(), id);
174        let (kind, groups) = self.actor_shape(ty)?;
175        let mut children = Vec::new();
176        for group in &groups {
177            for child in group {
178                children.push(self.actor_type(child)?);
179            }
180        }
181        let child_array = self.actor_array(&format!("actor_children{id}"), &children);
182        let count = if kind == 4 {
183            groups.len()
184        } else {
185            children.len()
186        };
187        let arities = if kind == 4 {
188            let lengths: Vec<_> = groups.iter().map(|group| group.len().to_string()).collect();
189            self.actor_array(&format!("actor_arities{id}"), &lengths)
190        } else {
191            "0".into()
192        };
193        self.data.data(
194            &format!("$actor_type{}", id),
195            vec![
196                DataValue::Word(native_operand(&(kind).to_string())),
197                DataValue::Word(native_operand(&(count).to_string())),
198                DataValue::Word(native_operand(&(child_array))),
199                DataValue::Word(native_operand(&(arities))),
200            ],
201        );
202        Ok(format!("$actor_type{id}"))
203    }

205    /// Translate approved immutable language layouts, retaining tags and full-width payload identity.
206    fn actor_shape(&self, ty: &Type) -> Lowering<(usize, Vec<Vec<Type>>)> {
207        Ok(match ty {
208            Type::Int | Type::Bool | Type::Unit | Type::Float => (0, vec![]),
209            Type::String => (1, vec![]),
210            Type::List(item) => (2, vec![vec![*item.clone()]]),
211            Type::Tuple(fields) => (3, vec![fields.clone()]),
212            Type::Option(item) => (4, vec![vec![*item.clone()], vec![]]),
213            Type::Result(ok, err) => (4, vec![vec![*ok.clone()], vec![*err.clone()]]),
214            Type::Union(members) => (4, members.iter().map(|ty| vec![ty.clone()]).collect()),
215            // A foreign address cannot be relocated with its owning string graph.
216            Type::Named(name, _) if name == "Ptr" => (11, vec![]),
217            Type::Named(_, _) => {
218                let layout = self
219                    .layouts
220                    .get(ty)
221                    .ok_or_else(|| invalid(Span::default(), "unknown actor nominal layout"))?;
222                (
223                    if layout.storage == ir::LayoutStorage::Unboxed {
224                        5
225                    } else {
226                        4
227                    },
228                    layout.variants.clone(),
229                )
230            }
231            Type::Pid(mailbox) => (6, vec![vec![*mailbox.clone()]]),
232            Type::Function(_, _) | Type::ActorFunction(_, _) => (7, vec![]),
233            Type::Map(key, value) => (9, vec![vec![*key.clone(), *value.clone()]]),
234            Type::Range => (10, vec![]),
235            Type::Native(crate::runtime::NativeType::JsonValue) => (12, vec![]),
236            Type::Native(crate::runtime::NativeType::ProcessId) => (13, vec![]),
237            Type::Native(crate::runtime::NativeType::MonitorRef) => (14, vec![]),
238            Type::Native(_) => (11, vec![]),
239            _ => {
240                return Err(invalid(
241                    Span::default(),
242                    "native handle is not supported in actor frame accounting",
243                ));
244            }
245        })
246    }

248    /// Empty descriptor vectors use null; every nonempty vector has an immutable fixed-length object.
249    fn actor_array(&mut self, name: &str, values: &[String]) -> String {
250        if values.is_empty() {
251            return "0".into();
252        }
253        let entries = values
254            .iter()
255            .map(|value| DataValue::Word(native_operand(value)))
256            .collect();
257        self.data.data(&format!("${name}"), entries);
258        format!("${name}")
259    }
