1//! Fuse small pure self-tail paths into a bounded actor callback. Ordinary ABIs stay intact.

4impl Builder<'_> {
5    pub(super) fn batch_body(&mut self, function: &Function) -> Lowering<Expr> {
6        let cost = tail_helpers::expression_work(&function.body, &self.inline_work).max(1);
7        if function.mailbox.is_some()
8            || !function.captures.is_empty()
9            || cost > tail_helpers::STEP_WORK / 2
10        {
11            return Ok(function.body.clone());
12        }
13        // Counting both branches overestimates each executed path. Also cap
14        // zero-cost syntax growth and leave room for normalization
15        // and CPS depth: an optimization must not exhaust their recursion budget.
16        let copies = copies(&function.body, cost);
17        if copies < 2 || eligible(&function.body, function.id, true, 0) != Some((1, false)) {
18            return Ok(function.body.clone());
19        }
20        let mut body = function.body.clone();
21        self.expand_tail(&mut body, function, copies - 1, 0)?;
22        Ok(body)
23    }
24
25    fn expand_tail(
26        &mut self,
27        expr: &mut Expr,
28        function: &Function,
29        remaining: usize,
30        depth: usize,
31    ) -> Lowering<()> {
32        self.work += 1;
33        if depth >= MAX_DEPTH || self.work > MAX_NODES {
34            return Err(invalid(expr.span, "actor tail batch work limit exceeded"));
35        }
36        if remaining == 0 {
37            return Ok(());
38        }
39        if let ExprKind::Call {
40            target: CallTarget::Function(id),
41            args,
42        } = &expr.kind
43            && *id == function.id
44        {
45            let offset = self.next_local;
46            self.next_local = offset
47                .checked_add(function.local_count)
48                .ok_or_else(|| invalid(expr.span, "actor tail batch local limit exceeded"))?;
49            let mut body = function.body.clone();
50            self.rename_batch(&mut body, offset, depth + 1)?;
51            self.expand_tail(&mut body, function, remaining - 1, depth + 1)?;
52            // The checked IR already stages labeled arguments in written order.
53            // Bind every parameter before the next body so swaps and faults retain
54            // simultaneous full-width assignment and exactly-once evaluation.
55            let mut statements: Vec<_> = args
56                .iter()
57                .zip(&function.params)
58                .map(|(value, param)| Stmt::Let {
59                    id: ir::LocalId(param.id.0 + offset),
60                    value: value.clone(),
61                })
62                .collect();
63            statements.push(Stmt::Expr(body));
64            expr.kind = ExprKind::Block(statements);
65        } else {
66            for child in children_mut(expr) {
67                self.expand_tail(child, function, remaining, depth + 1)?;
68            }
69        }
70        Ok(())
71    }
72
73    fn rename_batch(&mut self, expr: &mut Expr, offset: usize, depth: usize) -> Lowering<()> {
74        self.work += 1;
75        if depth >= MAX_DEPTH || self.work > MAX_NODES {
76            return Err(invalid(expr.span, "actor tail batch rename limit exceeded"));
77        }
78        match &mut expr.kind {
79            ExprKind::Local(id) => id.0 += offset,
80            ExprKind::Block(stmts) => {
81                for stmt in stmts {
82                    if let Stmt::Let { id, .. } = stmt {
83                        id.0 += offset;
84                    }
85                }
86            }
87            ExprKind::Match { arms, .. } => {
88                for arm in arms {
89                    rename_pattern(&mut arm.pattern, offset);
90                }
91            }
92            _ => {}
93        }
94        for child in children_mut(expr) {
95            self.rename_batch(child, offset, depth + 1)?;
96        }
97        Ok(())
98    }
99}

101/// A small, fixed source-growth bound also limits eligibility and cloning work.
102fn copies(body: &Expr, cost: usize) -> usize {
103    let mut pending = vec![(body, 1)];
104    let mut nodes = 0;
105    let mut depth = 0;
106    while let Some((expr, level)) = pending.pop() {
107        nodes += 1;
108        depth = depth.max(level);
109        if nodes > 1024 || depth >= MAX_DEPTH / 4 {
110            return 1;
111        }
112        pending.extend(
113            ir::children(expr)
114                .into_iter()
115                .map(|child| (child, level + 1)),
116        );
117    }
118    // Each expansion replaces one self-call (charge 1) with the copied body;
119    // argument computations are retained exactly once. Recursive cycles never
120    // enter inline_work's finite-sink cost map, so expanded work is
121    // 1 + copies * (cost - 1). A cost 1 body is still bounded by syntax growth.
122    let work_copies = if cost == 1 {
123        8
124    } else {
125        (tail_helpers::STEP_WORK - 1) / (cost - 1)
126    };
127    work_copies
128        .min(8)
129        .min(1024 / nodes)
130        .min(MAX_DEPTH / (4 * (depth + 1)))
131}

133/// Return (self calls, pure). An effect may occur only on a terminating path;
134/// every computation preceding the unique self-tail call is scalar and finite.
135fn eligible(
136    expr: &Expr,
137    self_id: ir::FunctionId,
138    tail: bool,
139    depth: usize,
140) -> Option<(usize, bool)> {
141    if depth >= MAX_DEPTH {
142        return None;
143    }
144    match &expr.kind {
145        ExprKind::Local(_)
146        | ExprKind::Int(_)
147        | ExprKind::Float(_)
148        | ExprKind::Bool(_)
149        | ExprKind::Unit => Some((0, true)),
150        ExprKind::Binary { left, right, .. } => {
151            if !matches!(left.ty, Type::Int | Type::Float | Type::Bool)
152                || !matches!(right.ty, Type::Int | Type::Float | Type::Bool)
153            {
154                return None;
155            }
156            let a = eligible(left, self_id, false, depth + 1)?;
157            let b = eligible(right, self_id, false, depth + 1)?;
158            Some((0, a.1 && b.1))
159        }
160        ExprKind::Unary { value, .. } => eligible(value, self_id, false, depth + 1),
161        ExprKind::Call {
162            target: CallTarget::Function(id),
163            ..
164        } if *id == self_id && tail => pure_children(expr, self_id, depth).then_some((1, false)),
165        ExprKind::Block(stmts) => {
166            let mut calls = 0;
167            let mut pure = true;
168            for (index, stmt) in stmts.iter().enumerate() {
169                let (value, tail) = match stmt {
170                    Stmt::Expr(value) => (value, tail && index + 1 == stmts.len()),
171                    Stmt::Let { value, .. } => (value, false),
172                    _ => return None,
173                };
174                let (found, child_pure) = eligible(value, self_id, tail, depth + 1)?;
175                if found != 0 && !pure {
176                    return None;
177                }
178                calls += found;
179                pure &= child_pure;
180            }
181            (calls <= 1).then_some((calls, pure))
182        }
183        ExprKind::If {
184            condition,
185            then_branch,
186            else_branch,
187        } => {
188            if eligible(condition, self_id, false, depth + 1)? != (0, true) {
189                return None;
190            }
191            let (a, ap) = eligible(then_branch, self_id, tail, depth + 1)?;
192            let (b, bp) = if let Some(branch) = else_branch {
193                eligible(branch, self_id, tail, depth + 1)?
194            } else {
195                (0, true)
196            };
197            (a + b <= 1).then_some((a + b, ap && bp))
198        }
199        // Terminal effects retain their normal normalization and CPS boundaries.
200        ExprKind::String(_)
201        | ExprKind::Interpolate(_)
202        | ExprKind::Tuple(_)
203        | ExprKind::Construct { .. }
204        | ExprKind::CustomConstruct { .. }
205        | ExprKind::Match { .. }
206        | ExprKind::Actor(ir::ActorExpr::Send { .. })
207        | ExprKind::Call {
208            target: CallTarget::Builtin(_) | CallTarget::Runtime(_),
209            ..
210        } => {
211            if matches!(expr.kind, ExprKind::Call { target: CallTarget::Builtin(b), .. }
212                if crate::lowering::higher_order::is_higher_order(b))
213            {
214                return None;
215            }
216            for child in ir::children(expr) {
217                if eligible(child, self_id, false, depth + 1)?.0 != 0 {
218                    return None;
219                }
220            }
221            Some((0, false))
222        }
223        _ => None,
224    }
225}

227fn pure_children(expr: &Expr, self_id: ir::FunctionId, depth: usize) -> bool {
228    ir::children(expr)
229        .into_iter()
230        .all(|child| eligible(child, self_id, false, depth + 1) == Some((0, true)))
231}

233/// Deliberately matches the eligibility whitelist; unsupported forms never reach expansion.
234fn children_mut(expr: &mut Expr) -> Vec<&mut Expr> {
235    match &mut expr.kind {
236        ExprKind::Binary { left, right, .. } => vec![left, right],
237        ExprKind::Unary { value, .. } => vec![value],
238        ExprKind::Block(stmts) => stmts
239            .iter_mut()
240            .filter_map(|stmt| match stmt {
241                Stmt::Let { value, .. } | Stmt::Expr(value) => Some(value),
242                _ => None,
243            })
244            .collect(),
245        ExprKind::If {
246            condition,
247            then_branch,
248            else_branch,
249        } => {
250            let mut children = vec![condition.as_mut(), then_branch.as_mut()];
251            children.extend(else_branch.as_deref_mut());
252            children
253        }
254        ExprKind::Match { value, arms } => {
255            let mut children = vec![value.as_mut()];
256            for arm in arms {
257                children.extend(arm.guard.as_mut());
258                children.push(&mut arm.body);
259            }
260            children
261        }
262        ExprKind::Call { args, .. }
263        | ExprKind::Interpolate(args)
264        | ExprKind::Tuple(args)
265        | ExprKind::CustomConstruct { fields: args, .. } => args.iter_mut().collect(),
266        ExprKind::Construct { value, .. } => value.iter_mut().map(|value| value.as_mut()).collect(),
267        ExprKind::Actor(ir::ActorExpr::Send { pid, message }) => vec![pid, message],
268        _ => vec![],
269    }
270}

272fn rename_pattern(pattern: &mut Pattern, offset: usize) {
273    match pattern {
274        Pattern::Bind(id)
275        | Pattern::Constructor {
276            binding: Some(id), ..
277        } => id.0 += offset,
278        Pattern::UnionSelect {
279            binding: Some(param),
280            ..
281        } => param.id.0 += offset,
282        Pattern::Newtype(inner) => rename_pattern(inner, offset),
283        Pattern::Tuple(fields) | Pattern::Variant { fields, .. } => {
284            for field in fields {
285                rename_pattern(field, offset);
286            }
287        }
288        Pattern::List { prefix, rest } => {
289            for field in prefix {
290                rename_pattern(field, offset);
291            }
292            if let Some(rest) = rest {
293                rename_pattern(rest, offset);
294            }
295        }
296        Pattern::TupleRest { prefix, rest } => {
297            for field in prefix {
298                rename_pattern(field, offset);
299            }
300            rename_pattern(rest, offset);
301        }
302        _ => {}
303    }
304}

311    fn zero_work_syntax_and_cost_one_keep_growth_and_depth_limits() {
312        let span = Span::default();
313        let leaf = node(ExprKind::Unit, Type::Unit, span);
314        assert_eq!(copies(&leaf, 1), 8);
315        let wide = node(
316            ExprKind::Block(vec![Stmt::Expr(leaf.clone()); 1024]),
317            Type::Unit,
318            span,
319        );
320        assert_eq!(copies(&wide, 1), 1);
321        let mut deep = leaf;
322        for _ in 0..MAX_DEPTH / 4 {
323            deep = node(ExprKind::Block(vec![Stmt::Expr(deep)]), Type::Unit, span);
324        }
325        assert_eq!(copies(&deep, 1), 1);
326    }
