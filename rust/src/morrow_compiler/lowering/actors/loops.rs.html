1//! Iteration continuations own immutable collection/index frames, never native stacks.
2use super::*;
3use crate::ast::BinaryOp;

5impl Builder<'_> {
6    /// Evaluate the collection once, then publish at most one iteration per callback.
7    pub(super) fn iteration(
8        &mut self,
9        pattern: &Pattern,
10        iterable: &Expr,
11        body: &Expr,
12        next: Option<&Continuation>,
13        span: Span,
14        depth: usize,
15    ) -> Lowering<Expr> {
16        atomic(iterable)?;
17        let item = match &iterable.ty {
18            Type::Range => Type::Int,
19            Type::List(item) => *item.clone(),
20            Type::Map(key, value) => Type::Tuple(vec![*key.clone(), *value.clone()]),
21            _ => return Err(invalid(span, "for requires List, Map, or Range")),
22        };
23        let collection = ir::Param {
24            id: self.local(span)?,
25            ty: iterable.ty.clone(),
26        };
27        let index = ir::Param {
28            id: self.local(span)?,
29            ty: Type::Int,
30        };
31        let local = |p: &ir::Param| node(ExprKind::Local(p.id), p.ty.clone(), span);
32        let field = |field| {
33            operation(
34                Operation::IterateField {
35                    value: Box::new(local(&collection)),
36                    field,
37                },
38                span,
39            )
40        };
41        let completed = self.finish(unit(span), next)?;
42        let done = operation(
43            Operation::Continue(Box::new(self.closure(completed, vec![], false)?)),
44            span,
45        );
46        // Include lexical loop exits when discovering the frame; an inner break/continue
47        // can schedule an outer frame whose captures are absent from source local reads.
48        let mut evidence = vec![
49            Stmt::Expr(node(
50                ExprKind::For {
51                    pattern: pattern.clone(),
52                    iterable: Box::new(local(&collection)),
53                    body: Box::new(body.clone()),
54                },
55                Type::Unit,
56                span,
57            )),
58            Stmt::Expr(done.clone()),
59            Stmt::Expr(local(&index)),
60        ];
61        if let Some(target) = &self.return_to {
62            evidence.push(Stmt::Expr(local(target)));
63        }
64        for (advance, exit) in &self.loops {
65            evidence.extend([Stmt::Expr(advance.clone()), Stmt::Expr(exit.clone())]);
66        }
67        let captures = free(
68            &node(ExprKind::Block(evidence), Type::Unit, span),
69            &[],
70            &mut self.work,
71        )?;
72        let id = ir::FunctionId(self.identity(span)?);
73        let entry = |increment: bool| {
74            node(
75                ExprKind::Closure {
76                    function: id,
77                    captures: captures
78                        .iter()
79                        .map(|p| {
80                            if p.id == index.id && increment {
81                                binary(BinaryOp::Add, local(p), integer(1, span), Type::Int)
82                            } else {
83                                local(p)
84                            }
85                        })
86                        .collect(),
87                },
88                Type::Function(vec![], Box::new(Type::Int)),
89                span,
90            )
91        };
92        // Test before increment: inclusive i64::MAX completes without wrapping.
93        let advance = node(
94            ExprKind::If {
95                condition: Box::new(binary(BinaryOp::Eq, local(&index), field(1), Type::Bool)),
96                then_branch: Box::new(done.clone()),
97                else_branch: Some(Box::new(operation(
98                    Operation::Continue(Box::new(entry(true))),
99                    span,
100                ))),
101            },
102            Type::Int,
103            span,
104        );
105        let continuation = Continuation {
106            value: ir::Param {
107                id: self.local(span)?,
108                ty: Type::Unit,
109            },
110            entry: self.closure(advance.clone(), vec![], false)?,
111        };
112        self.loops.push((advance, done.clone()));
113        let converted = self.expression(body, Some(&continuation), depth);
114        self.loops.pop();
115        let converted = converted?;
116        let value = node(
117            ExprKind::Actor(ir::ActorExpr::Lowered(Lowered {
118                operation: Operation::IterateItem {
119                    value: Box::new(local(&collection)),
120                    index: Box::new(local(&index)),
121                    item: item.clone(),
122                },
123            })),
124            item,
125            span,
126        );
127        let selected = node(
128            ExprKind::Match {
129                value: Box::new(value),
130                arms: vec![MatchArm {
131                    pattern: pattern.clone(),
132                    guard: None,
133                    body: converted,
134                    span,
135                }],
136            },
137            Type::Int,
138            span,
139        );
140        let before = binary(BinaryOp::Lt, local(&index), field(1), Type::Bool);
141        let last = binary(
142            BinaryOp::And,
143            binary(BinaryOp::Eq, local(&index), field(1), Type::Bool),
144            binary(BinaryOp::Ne, field(2), integer(0, span), Type::Bool),
145            Type::Bool,
146        );
147        let step_body = node(
148            ExprKind::If {
149                condition: Box::new(binary(BinaryOp::Or, before, last, Type::Bool)),
150                then_branch: Box::new(selected),
151                else_branch: Some(Box::new(done)),
152            },
153            Type::Int,
154            span,
155        );
156        let mut step = self.function(step_body, vec![], false)?;
157        self.plan.steps.remove(&step.id.0);
158        self.plan.generic_steps.remove(&step.id.0);
159        step.id = id;
160        // Extra captures are permitted; every actual read must have a typed frame slot.
161        for actual in &step.captures {
162            if !captures
163                .iter()
164                .any(|p| p.id == actual.id && p.ty == actual.ty)
165            {
166                return Err(invalid(
167                    span,
168                    "actor loop frame is missing a lexical capture",
169                ));
170            }
171        }
172        step.captures = captures.clone();
173        self.plan.steps.insert(id.0, self.mailbox.clone());
174        if self.generic {
175            self.plan.generic_steps.insert(id.0);
176        }
177        self.plan.functions.push(step);
178        Ok(node(
179            ExprKind::Block(vec![
180                Stmt::Let {
181                    id: collection.id,
182                    value: iterable.clone(),
183                },
184                Stmt::Let {
185                    id: index.id,
186                    value: field(0),
187                },
188                Stmt::Expr(operation(Operation::Continue(Box::new(entry(false))), span)),
189            ]),
190            Type::Int,
191            span,
192        ))
193    }
194}

196/// Preserve ordinary checked operator semantics in generated induction and guards.
197pub(super) fn binary(op: BinaryOp, left: Expr, right: Expr, ty: Type) -> Expr {
198    let span = left.span;
199    node(
200        ExprKind::Binary {
201            op,
202            left: Box::new(left),
203            right: Box::new(right),
204        },
205        ty,
206        span,
207    )
