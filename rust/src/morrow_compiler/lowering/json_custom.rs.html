1//! Uniform native callback thunks preserve source widths and explicit fault propagation.
2use super::*;
3use crate::json_codec::Callback;
4
5impl Emitter<'_> {
6    pub(super) fn custom_codec(
7        &mut self,
8        prefix: &str,
9        encode: &Callback,
10        decode: &Callback,
11        span: Span,
12    ) -> Lowering<()> {
13        let Callback::Function(id) = encode else {
14            return Err(invalid(span, "unresolved custom JSON encoder"));
15        };
16        let function = self
17            .functions
18            .get(&id.0)
19            .ok_or_else(|| invalid(span, "missing custom JSON encoder"))?;
20        let Some(parameter) = function.params.first() else {
21            return Err(invalid(span, "custom JSON encoder requires one argument"));
22        };
23        let managed = !matches!(
24            self.representation(&parameter.ty),
25            Type::Int | Type::Float | Type::Bool | Type::Unit | Type::Never
26        );
27        let encode = self.codec_callback(&format!("{prefix}_encode"), encode, span)?;
28        let decode = self.codec_callback(&format!("{prefix}_decode"), decode, span)?;
29        let callbacks = format!("{prefix}_callbacks");
30        self.data.data(
31            &callbacks,
32            vec![
33                DataValue::Word(native_operand(&encode)),
34                DataValue::Word(native_operand(&decode)),
35                DataValue::Word(Operand::Int(i64::from(managed))),
36            ],
37        );
38        self.data.data(
39            prefix,
40            vec![
41                DataValue::Word(Operand::Int(14)),
42                DataValue::Word(Operand::Int(2)),
43                DataValue::Word(native_operand(&callbacks)),
44                DataValue::Word(Operand::Int(0)),
45            ],
46        );
47        Ok(())
48    }

50    fn codec_callback(&mut self, name: &str, callback: &Callback, span: Span) -> Lowering<String> {
51        let Callback::Function(id) = callback else {
52            return Err(invalid(span, "unresolved custom JSON callback"));
53        };
54        let function = self
55            .functions
56            .get(&id.0)
57            .ok_or_else(|| invalid(span, "missing custom JSON callback"))?;
58        let [parameter] = function.params.as_slice() else {
59            return Err(invalid(span, "custom JSON callback requires one argument"));
60        };
61        if !function.captures.is_empty() || function.mailbox.is_some() {
62            return Err(invalid(
63                span,
64                "custom JSON callback must be a closed ordinary function",
65            ));
66        }
67        let width = machine_width(self.width(parameter.ty.clone()));
68        self.data.begin(
69            name,
70            Some(Scalar::I64),
71            vec![
72                (Scalar::I64, "%fault".into()),
73                (Scalar::I64, "%bits".into()),
74            ],
75            false,
76        );
77        self.data.statement(Statement::Label("entry".into()));
78        let argument = if width == Scalar::I64 {
79            native_operand("%bits")
80        } else {
81            self.data.statement(Statement::Assign {
82                destination: "%argument".into(),
83                ty: width,
84                operation: NativeOperation::Unary(
85                    if width == Scalar::F64 {
86                        MachineUnary::Cast
87                    } else {
88                        MachineUnary::Copy
89                    },
90                    native_operand("%bits"),
91                ),
92            });
93            native_operand("%argument")
94        };
95        // No allocation occurs before the checked callee establishes its typed
96        // roots. The caller keeps the incoming value rooted across this thunk.
97        self.data.statement(Statement::Assign {
98            destination: "%result".into(),
99            ty: Scalar::I64,
100            operation: NativeOperation::Call {
101                callee: native_operand(&format!("$f{}", id.0)),
102                args: vec![
103                    (Scalar::I64, Operand::Int(0)),
104                    (Scalar::I64, native_operand("%fault")),
105                    (width, argument),
106                ],
107                variadic: None,
108            },
109        });
110        self.data
111            .statement(Statement::Return(Some(native_operand("%result"))));
112        self.data.end();
113        Ok(name.into())
114    }
