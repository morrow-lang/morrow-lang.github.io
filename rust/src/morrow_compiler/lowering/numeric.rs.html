4/// Compute an operator result from concrete operand types, independently of source checking.
5pub(super) fn binary_type(op: BinaryOp, operand: &Type, span: Span) -> Lowering<Type> {
6    use BinaryOp::*;
7    let valid = match op {
8        Add if *operand == Type::String => return Ok(Type::String),
9        Add | Subtract | Multiply | Divide | Power => matches!(operand, Type::Int | Type::Float),
10        Remainder | BitAnd | BitOr | BitXor | ShiftLeft | ShiftRight => *operand == Type::Int,
11        Lt | Le | Gt | Ge => matches!(operand, Type::Int | Type::Float),
12        Eq | Ne => {
13            crate::processes::opaque(operand)
14                || matches!(
15                    operand,
16                    Type::Int | Type::Float | Type::Bool | Type::String | Type::Pid(_)
17                )
18        }
19        In | And | Or => {
20            return Err(invalid(
21                span,
22                "logical operation requires short-circuit lowering",
23            ));
24        }
25    };
26    if !valid {
27        return Err(invalid(span, "operator has incompatible operand type"));
28    }
29    Ok(if matches!(op, Eq | Ne | Lt | Le | Gt | Ge) {
30        Type::Bool
31    } else {
32        operand.clone()
33    })
34}

37    /// Guard integer domains and normalize shift counts before target-specific instructions.
38    pub(super) fn numeric_value(
39        &mut self,
40        op: BinaryOp,
41        operand: &Type,
42        result: &Type,
43        lhs: &str,
44        rhs: &str,
45        locals: &mut Locals,
46    ) -> String {
47        if crate::processes::opaque(operand) {
48            let symbol = if *operand == crate::processes::identity() {
49                "$morrow_process_id_equal"
50            } else {
51                "$morrow_process_monitor_equal"
52            };
53            let value = self.assign(
54                locals,
55                Type::Int,
56                NativeOperation::Call {
57                    callee: native_operand(symbol),
58                    args: vec![
59                        (Scalar::I64, native_operand(lhs)),
60                        (Scalar::I64, native_operand(rhs)),
61                    ],
62                    variadic: None,
63                },
64            );
65            return self.assign(
66                locals,
67                Type::Bool,
68                NativeOperation::Binary(
69                    MachineBinary::Compare(
70                        if op == BinaryOp::Eq {
71                            Comparison::Ne
72                        } else {
73                            Comparison::Eq
74                        },
75                        Scalar::I64,
76                    ),
77                    native_operand(&value),
78                    native_operand("0"),
79                ),
80            );
81        }
82        // Keep proven-safe literal divisors visible to native optimization.
83        // Operands have already been evaluated in source order. Division by a
84        // nonzero value other than -1 cannot fault or overflow for any i64 lhs,
85        // so this path needs neither a helper call nor a new fault-slot check.
86        // Zero, -1 and dynamic divisors retain the wrapping/fault-aware helper.
87        if *operand == Type::Int
88            && matches!(op, BinaryOp::Divide | BinaryOp::Remainder)
89            && rhs
90                .parse::<i64>()
91                .is_ok_and(|value| !matches!(value, 0 | -1))
92        {
93            return self.assign(
94                locals,
95                Type::Int,
96                NativeOperation::Binary(
97                    binary_instruction(op, Type::Int),
98                    native_operand(lhs),
99                    native_operand(rhs),
100                ),
101            );
102        }
103        if *operand == Type::Int
104            && matches!(op, BinaryOp::Divide | BinaryOp::Remainder | BinaryOp::Power)
105        {
106            self.numeric_used = true;
107            let instruction = if op == BinaryOp::Power {
108                NativeOperation::Call {
109                    callee: native_operand("$morrow_rs_int_pow"),
110                    args: vec![
111                        (Scalar::I64, native_operand("%fault")),
112                        (Scalar::I64, native_operand(lhs)),
113                        (Scalar::I64, native_operand(rhs)),
114                    ],
115                    variadic: None,
116                }
117            } else {
118                NativeOperation::Call {
119                    callee: native_operand("$morrow_rs_int_div"),
120                    args: vec![
121                        (Scalar::I64, native_operand("%fault")),
122                        (Scalar::I64, native_operand(lhs)),
123                        (Scalar::I64, native_operand(rhs)),
124                        (
125                            Scalar::I32,
126                            native_operand(&(u8::from(op == BinaryOp::Remainder)).to_string()),
127                        ),
128                    ],
129                    variadic: None,
130                }
131            };
132            let value = self.assign(locals, Type::Int, instruction);
133            self.guard_fault(locals);
134            return value;
135        }
136        if op == BinaryOp::Power {
137            return self.assign(
138                locals,
139                Type::Float,
140                NativeOperation::Call {
141                    callee: native_operand("$pow"),
142                    args: vec![
143                        (Scalar::F64, native_operand(lhs)),
144                        (Scalar::F64, native_operand(rhs)),
145                    ],
146                    variadic: None,
147                },
148            );
149        }
150        let rhs = if matches!(op, BinaryOp::ShiftLeft | BinaryOp::ShiftRight) {
151            self.assign(
152                locals,
153                Type::Int,
154                NativeOperation::Binary(
155                    MachineBinary::And,
156                    native_operand(rhs),
157                    native_operand("63"),
158                ),
159            )
160        } else {
161            rhs.into()
162        };
163        let instruction = binary_instruction(op, operand.clone());
164        self.assign(
165            locals,
166            result.clone(),
167            NativeOperation::Binary(instruction, native_operand(lhs), native_operand(&(rhs))),
168        )
169    }

171    /// Share one value-aware contains path between intrinsic and registry call identities.
172    pub(super) fn scalar_contains(
173        &mut self,
174        args: &[Expr],
175        span: Span,
176        locals: &mut Locals,
177        depth: usize,
178    ) -> Lowering<(Type, String)> {
179        let [list, value] = args else {
180            return Err(invalid(span, "List.contains requires two arguments"));
181        };
182        self.scalar_contains_ordered(list, value, locals, depth, false)
183    }

185    /// Select the element-directed sort helper; Float words sort by bit-decoded total order.
186    pub(super) fn scalar_sort(
187        &mut self,
188        args: &[Expr],
189        span: Span,
190        locals: &mut Locals,
191        depth: usize,
192    ) -> Lowering<(Type, String)> {
193        let [list] = args else {
194            return Err(invalid(span, "List.sort requires one argument"));
195        };
196        let Type::List(item) = &list.ty else {
197            return Err(invalid(span, "List.sort requires List"));
198        };
199        nominal::resolved(item, &self.layouts, span, 0)?;
200        let symbol = match self.representation(item) {
201            Type::Int | Type::Bool => "morrow_list_sort",
202            Type::Float => "morrow_list_sort_float",
203            Type::String => "morrow_list_sort_str",
204            _ => {
205                return Err(invalid(
206                    span,
207                    "List.sort requires Int, Float, Bool, or String elements",
208                ));
209            }
210        };
211        let value = self.expr(list, locals, depth)?;
212        let sorted = self.assign(
213            locals,
214            list.ty.clone(),
215            NativeOperation::Call {
216                callee: native_operand(&format!("${symbol}")),
217                args: vec![(Scalar::I64, native_operand(&(value)))],
218                variadic: None,
219            },
220        );
221        Ok((list.ty.clone(), sorted))
222    }

224    /// Preserve caller operand order while sharing scalar equality and ABI validation.
225    pub(super) fn scalar_contains_ordered(
226        &mut self,
227        list: &Expr,
228        value: &Expr,
229        locals: &mut Locals,
230        depth: usize,
231        item_first: bool,
232    ) -> Lowering<(Type, String)> {
233        let span = list.span;
234        let Type::List(item) = &list.ty else {
235            return Err(invalid(span, "List.contains requires List"));
236        };
237        expect_type(value.ty.clone(), *item.clone(), value.span)?;
238        nominal::resolved(item, &self.layouts, span, 0)?;
239        let primitive = self.representation(item).clone();
240        if !matches!(
241            primitive,
242            Type::Int | Type::Bool | Type::String | Type::Float
243        ) {
244            return Err(invalid(span, "List.contains requires scalar equality"));
245        }
246        let (list, raw) = if item_first {
247            let raw = self.expr(value, locals, depth)?;
248            (self.expr(list, locals, depth)?, raw)
249        } else {
250            let list = self.expr(list, locals, depth)?;
251            (list, self.expr(value, locals, depth)?)
252        };
253        let found = if primitive == Type::Float {
254            self.float_contains_used = true;
255            self.assign(
256                locals,
257                Type::Bool,
258                NativeOperation::Call {
259                    callee: native_operand("$morrow_rs_list_contains_float"),
260                    args: vec![
261                        (Scalar::I64, native_operand(&(list))),
262                        (Scalar::F64, native_operand(&(raw))),
263                    ],
264                    variadic: None,
265                },
266            )
267        } else {
268            let payload = self.payload(locals, item, raw);
269            let symbol = if primitive == Type::String {
270                "morrow_list_contains_str"
271            } else {
272                "morrow_list_contains"
273            };
274            let raw = self.assign(
275                locals,
276                Type::Int,
277                NativeOperation::Call {
278                    callee: native_operand(&format!("${}", symbol)),
279                    args: vec![
280                        (Scalar::I64, native_operand(&(list))),
281                        (Scalar::I64, native_operand(&(payload))),
282                    ],
283                    variadic: None,
284                },
285            );
286            self.unpack(locals, &Type::Bool, raw)
287        };
288        Ok((Type::Bool, found))
289    }
