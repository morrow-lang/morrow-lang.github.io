1//! Direct self-tail transfers reuse typed parameter state and their invocation context.
2use super::*;
3
4pub(super) struct TailLoop {
5    function: ir::FunctionId,
6    params: Vec<ir::Param>,
7    state: Parameters,
8}
9
10enum Parameters {
11    /// Managed and mixed representations retain private invocation-local storage.
12    Slots(Vec<String>),
13    /// Plain Int parameters need no collector roots and can remain SSA loop values.
14    Phis(Vec<TailPhi>),
15}
16
17struct TailPhi {
18    instruction: usize,
19    incoming: Vec<(String, Operand)>,
20}
21
22impl Locals {
23    /// Reserve fixed scratch storage in the entry block, not on a repeated dynamic edge.
24    pub(super) fn stack_slot(&mut self) -> String {
25        let name = self.temporary();
26        self.stack_allocations.statement(Statement::Assign {
27            destination: (name).to_string(),
28            ty: Scalar::I64,
29            operation: NativeOperation::StackAlloc { bytes: 8, align: 8 },
30        });
31        name
32    }
33}

35impl Emitter<'_> {
36    /// Initialize parameter storage once, then enter the reusable body through a fresh block.
37    pub(super) fn start_tail(&mut self, function: &Function, locals: &mut Locals) -> Lowering<()> {
38        if !eligible(function)? {
39            return Ok(());
40        }
41        if function.params.iter().all(|param| param.ty == Type::Int) {
42            let predecessor = locals.current.clone();
43            // Snapshot original parameter identities before rebinding them to loop values.
44            let initial = function
45                .params
46                .iter()
47                .map(|param| {
48                    Self::local(param.id, function.body.span, locals).map(|(_, value)| value)
49                })
50                .collect::<Lowering<Vec<_>>>()?;
51            self.output.statement(Statement::Jump("@recur".into()));
52            self.start_block(locals, "@recur");
53            let mut phis = Vec::new();
54            for (param, initial) in function.params.iter().zip(initial) {
55                let incoming = vec![(predecessor.clone(), native_operand(&initial))];
56                let instruction = self.output.len();
57                let value = self.assign(locals, Type::Int, NativeOperation::Phi(incoming.clone()));
58                locals.values.insert(param.id.0, (Type::Int, value));
59                phis.push(TailPhi {
60                    instruction,
61                    incoming,
62                });
63            }
64            locals.tail = Some(TailLoop {
65                function: function.id,
66                params: function.params.clone(),
67                state: Parameters::Phis(phis),
68            });
69            return Ok(());
70        }
71        let mut slots = Vec::new();
72        for param in &function.params {
73            let slot = locals.stack_slot();
74            let (_, value) = Self::local(param.id, function.body.span, locals)?;
75            let value = self.payload(locals, &param.ty, value);
76            self.output.statement(Statement::Store {
77                kind: LoadKind::I64,
78                value: native_operand(&(value)),
79                address: native_operand(&(slot)),
80            });
81            slots.push(slot);
82        }
83        self.output.statement(Statement::Jump("@recur".to_owned()));
84        self.start_block(locals, "@recur");
85        for (param, slot) in function.params.iter().zip(&slots) {
86            let raw = self.assign(
87                locals,
88                Type::Int,
89                NativeOperation::Load(LoadKind::I64, native_operand(&(slot).to_string())),
90            );
91            let value = self.unpack(locals, &param.ty, raw);
92            locals.values.insert(param.id.0, (param.ty.clone(), value));
93        }
94        locals.tail = Some(TailLoop {
95            function: function.id,
96            params: function.params.clone(),
97            state: Parameters::Slots(slots),
98        });
99        Ok(())
100    }

102    /// Complete forward phi inputs before inserting any hoisted allocations.
103    pub(super) fn finish_tail(&mut self, locals: &mut Locals, span: Span) -> Lowering<()> {
104        let Some(TailLoop {
105            state: Parameters::Phis(phis),
106            ..
107        }) = locals.tail.take()
108        else {
109            return Ok(());
110        };
111        for phi in phis {
112            let Some(machine::Item::Statement(Statement::Assign {
113                operation: NativeOperation::Phi(incoming),
114                ..
115            })) = self.output.items.get_mut(phi.instruction)
116            else {
117                return Err(invalid(span, "missing tail parameter phi"));
118            };
119            *incoming = phi.incoming;
120        }
121        Ok(())
122    }

124    /// Keep argument/condition evaluation in value mode while propagating explicit tail positions.
125    pub(super) fn position_expr(
126        &mut self,
127        expr: &Expr,
128        locals: &mut Locals,
129        depth: usize,
130        tail: bool,
131    ) -> Lowering<String> {
132        if tail {
133            self.tail_expr(expr, locals, depth)
134        } else {
135            self.expr(expr, locals, depth)
136        }
137    }

139    /// Rewrite only expressions whose successful result is the enclosing function's result.
140    pub(super) fn tail_expr(
141        &mut self,
142        expr: &Expr,
143        locals: &mut Locals,
144        depth: usize,
145    ) -> Lowering<String> {
146        if locals.tail.is_none() || (expr.ty != locals.return_type && expr.ty != Type::Never) {
147            return self.expr(expr, locals, depth);
148        }
149        let supported = matches!(
150            expr.kind,
151            ExprKind::Call {
152                target: CallTarget::Function(_),
153                ..
154            } | ExprKind::Block(_)
155                | ExprKind::If { .. }
156                | ExprKind::Match { .. }
157                | ExprKind::With { .. }
158        );
159        if !supported {
160            return self.expr(expr, locals, depth);
161        }
162        self.validate_expr(expr, depth)?;
163        self.strict_termination(expr, locals, depth + 1)?;
164        let (actual, value) = match &expr.kind {
165            ExprKind::Call { target, args } => {
166                if matches!(target, CallTarget::Function(id) if Some(*id) == locals.tail.as_ref().map(|t|t.function))
167                {
168                    return self.tail_call(*target, args, &expr.ty, expr.span, locals, depth + 1);
169                }
170                self.call(*target, args, &expr.ty, expr.span, locals, depth + 1)?
171            }
172            ExprKind::Block(stmts) => self.block(stmts, locals, depth + 1, true)?,
173            ExprKind::If {
174                condition,
175                then_branch,
176                else_branch,
177            } => self.conditional(
178                condition,
179                then_branch,
180                else_branch.as_deref(),
181                locals,
182                depth + 1,
183                true,
184            )?,
185            ExprKind::Match { value, arms } => {
186                self.matching(value, arms, locals, depth + 1, true)?
187            }
188            ExprKind::With {
189                steps,
190                body,
191                handlers,
192            } => self.with(steps, body, handlers, locals, depth + 1, true)?,
193            _ => unreachable!("tail dispatch was checked before lowering"),
194        };
195        expect_type(actual, expr.ty.clone(), expr.span)?;
196        Ok(value)
197    }

199    /// Evaluate every new argument before changing any old parameter, retaining fault checks.
200    fn tail_call(
201        &mut self,
202        target: CallTarget,
203        args: &[Expr],
204        result: &Type,
205        span: Span,
206        locals: &mut Locals,
207        depth: usize,
208    ) -> Lowering<String> {
209        let (_, params, actual) = self.signature(target, args, span)?;
210        expect_type(actual.clone(), result.clone(), span)?;
211        expect_type(actual, locals.return_type.clone(), span)?;
212        if args.len() != params.len() {
213            return Err(invalid(
214                span,
215                "call argument count differs from resolved signature",
216            ));
217        }
218        let mut values = Vec::new();
219        for (arg, expected) in args.iter().zip(params) {
220            expect_type(arg.ty.clone(), expected.clone(), arg.span)?;
221            let value = self.expr(arg, locals, depth)?;
222            values.push(self.payload(locals, &expected, value));
223        }
224        let tail = locals
225            .tail
226            .as_mut()
227            .ok_or_else(|| invalid(span, "missing self-tail target"))?;
228        if tail.params.len() != values.len() {
229            return Err(invalid(span, "self-tail parameter count mismatch"));
230        }
231        match &mut tail.state {
232            Parameters::Slots(slots) => {
233                for (slot, value) in slots.iter().zip(values) {
234                    self.output.statement(Statement::Store {
235                        kind: LoadKind::I64,
236                        value: native_operand(&value),
237                        address: native_operand(slot),
238                    });
239                }
240            }
241            Parameters::Phis(phis) => {
242                // Only successful argument evaluation reaches this predecessor.
243                // Phi assignment is simultaneous, including swaps of old parameters.
244                for (phi, value) in phis.iter_mut().zip(values) {
245                    phi.incoming
246                        .push((locals.current.clone(), native_operand(&value)));
247                }
248            }
249        }
250        self.output.statement(Statement::Jump("@recur".to_owned()));
251        Err(Exit::Terminated)
252    }
253}

255/// An owned defer anywhere disables reuse; lifted closure bodies own independent frames.
256fn eligible(function: &Function) -> Lowering<bool> {
257    if !function.captures.is_empty() {
258        return Ok(false);
259    }
260    let mut pending = vec![(&function.body, 0)];
261    let mut count = 0;
262    let mut recursive = false;
263    while let Some((expr, depth)) = pending.pop() {
264        count += 1;
265        if count > MAX_NODES || depth > MAX_DEPTH {
266            return Err(invalid(
267                expr.span,
268                "tail analysis complexity limit exceeded",
269            ));
270        }
271        match &expr.kind {
272            ExprKind::Defer(_) => return Ok(false),
273            ExprKind::Call {
274                target: CallTarget::Function(id),
275                ..
276            } if *id == function.id => recursive = true,
277            _ => {}
278        }
279        pending.extend(children(expr).into_iter().map(|child| (child, depth + 1)));
280    }
281    Ok(recursive)
282}

284/// Walk only expressions executed in this function, including closure capture expressions.
285fn children(expr: &Expr) -> Vec<&Expr> {
286    match &expr.kind {
287        ExprKind::Closure { captures, .. } => captures.iter().collect(),
288        ExprKind::Invoke { callee, args } => std::iter::once(callee.as_ref()).chain(args).collect(),
289        ExprKind::Return(value)
290        | ExprKind::Defer(value)
291        | ExprKind::Unary { value, .. }
292        | ExprKind::Try(value)
293        | ExprKind::Field { value, .. }
294        | ExprKind::Wrap(value)
295        | ExprKind::JsonCodecTemplate { input: value, .. }
296        | ExprKind::JsonCodec { input: value, .. }
297        | ExprKind::UnionInject { value }
298        | ExprKind::UnionWiden { value }
299        | ExprKind::Unwrap(value) => vec![value],
300        ExprKind::Binary { left, right, .. }
301        | ExprKind::Range {
302            start: left,
303            end: right,
304            ..
305        } => vec![left, right],
306        ExprKind::Call { args, .. }
307        | ExprKind::List(args)
308        | ExprKind::Tuple(args)
309        | ExprKind::Interpolate(args)
310        | ExprKind::CustomConstruct { fields: args, .. } => args.iter().collect(),
311        ExprKind::Map(entries) => entries.iter().flat_map(|(k, v)| [k, v]).collect(),
312        ExprKind::Construct { value, .. } => value.iter().map(|v| v.as_ref()).collect(),
313        ExprKind::If {
314            condition,
315            then_branch,
316            else_branch,
317        } => std::iter::once(condition.as_ref())
318            .chain(std::iter::once(then_branch.as_ref()))
319            .chain(else_branch.as_deref())
320            .collect(),
321        ExprKind::Match { value, arms } => std::iter::once(value.as_ref())
322            .chain(
323                arms.iter()
324                    .flat_map(|a| a.guard.iter().chain(std::iter::once(&a.body))),
325            )
326            .collect(),
327        ExprKind::Block(stmts) => stmts
328            .iter()
329            .flat_map(|s| match s {
330                Stmt::Expr(v) | Stmt::Let { value: v, .. } => vec![v],
331                Stmt::LetElse {
332                    value, else_branch, ..
333                } => vec![value, else_branch],
334            })
335            .collect(),
336        ExprKind::For { iterable, body, .. } => vec![iterable, body],
337        ExprKind::With {
338            steps,
339            body,
340            handlers,
341        } => steps
342            .iter()
343            .map(|s| &s.value)
344            .chain(std::iter::once(body.as_ref()))
345            .chain(handlers.iter().map(|h| &h.body))
346            .collect(),
347        _ => vec![],
348    }
