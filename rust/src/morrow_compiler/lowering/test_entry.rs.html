1//! Test execution cannot count an explicit process exit as a returned test result.
2use super::*;
3
4impl Emitter<'_> {
5    /// Redirect the audited runtime exit identity only in test mode, including lifted callables.
6    pub(super) fn test_runtime_symbol<'s>(&self, symbol: &'s str) -> &'s str {
7        if self.test_mode && symbol == "morrow_exit" {
8            "morrow_rs_test_exit"
9        } else {
10            symbol
11        }
12    }
13
14    /// Emit a diagnostic and failing process exit; unused application exit calls remain inert.
15    /// The source status is intentionally ignored. Normal compilation emits no helper or message.
16    pub(super) fn test_exit_helper(&mut self) {
17        if !self.test_mode {
18            return;
19        }
20        const MESSAGE: &str = "morrow: System.exit cannot terminate a test";
21        let bytes = MESSAGE.len() + 1;
22        self.data.data(
23            "$morrow_rs_test_exit_message",
24            vec![
25                DataValue::Bytes(MESSAGE.as_bytes().to_vec()),
26                DataValue::Bytes(vec![10]),
27            ],
28        );
29        self.output.begin(
30            "$morrow_rs_test_exit",
31            None,
32            vec![(Scalar::I64, "%status".to_owned())],
33            false,
34        );
35        self.output.statement(Statement::Label("@start".to_owned()));
36        self.output
37            .statement(Statement::Effect(NativeOperation::Call {
38                callee: native_operand("$write"),
39                args: vec![
40                    (Scalar::I32, native_operand("2")),
41                    (Scalar::I64, native_operand("$morrow_rs_test_exit_message")),
42                    (Scalar::I64, native_operand(&(bytes).to_string())),
43                ],
44                variadic: None,
45            }));
46        self.output
47            .statement(Statement::Effect(NativeOperation::Call {
48                callee: native_operand("$morrow_exit"),
49                args: vec![(Scalar::I64, native_operand("1"))],
50                variadic: None,
51            }));
52        self.output.statement(Statement::Return(None));
53        self.output.end();
54    }
55}
