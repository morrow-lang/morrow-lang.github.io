1//! Inline small, statically identified callbacks without changing control scopes.
2use super::*;
3
4// No recursive call expansion: eligible bodies contain no calls or closures.
5// The normal emitter node/root limits additionally bound total generated work.
6const MAX_INLINE_NODES: usize = 64;
7
8/// The whitelist excludes function-scoped exits, cleanup, suspension and loops.
9/// Arithmetic/aggregate lowering can still fault or allocate; it keeps the
10/// caller's fault handler and precise roots exactly as ordinary expressions do.
11fn eligible(expr: &Expr, remaining: &mut usize) -> bool {
12    let Some(next) = remaining.checked_sub(1) else {
13        return false;
14    };
15    *remaining = next;
16    match &expr.kind {
17        ExprKind::Int(_)
18        | ExprKind::Float(_)
19        | ExprKind::Bool(_)
20        | ExprKind::String(_)
21        | ExprKind::Local(_)
22        | ExprKind::Unit => true,
23        ExprKind::Field { value, .. } | ExprKind::Unary { value, .. } => eligible(value, remaining),
24        ExprKind::Binary { left, right, .. } => {
25            eligible(left, remaining) && eligible(right, remaining)
26        }
27        ExprKind::CustomConstruct { fields, .. } | ExprKind::Tuple(fields) => {
28            fields.iter().all(|field| eligible(field, remaining))
29        }
30        ExprKind::Construct { value, .. } => value
31            .as_ref()
32            .is_none_or(|value| eligible(value, remaining)),
33        ExprKind::If {
34            condition,
35            then_branch,
36            else_branch,
37        } => {
38            eligible(condition, remaining)
39                && eligible(then_branch, remaining)
40                && else_branch
41                    .as_ref()
42                    .is_none_or(|branch| eligible(branch, remaining))
43        }
44        ExprKind::Block(statements) => statements.iter().all(|statement| match statement {
45            Stmt::Let { value, .. } | Stmt::Expr(value) => eligible(value, remaining),
46            Stmt::LetElse { .. } => false,
47        }),
48        _ => false,
49    }
50}
51
52impl Emitter<'_> {
53    /// Reuse the already evaluated environment so capture evaluation and source
54    /// argument order never move into the loop. Unknown/effectful functions keep
55    /// their ordinary callback ABI, cleanup activation and suspension behavior.
56    pub(super) fn invoke_callback_values(
57        &mut self,
58        expression: &Expr,
59        closure: &str,
60        args: &[(Type, String)],
61        result: &Type,
62        locals: &mut Locals,
63    ) -> Lowering<String> {
64        let function = match &expression.kind {
65            ExprKind::Closure { function, .. } => self.functions.get(&function.0).copied(),
66            _ => None,
67        };
68        let Some(function) = function.filter(|function| {
69            let mut remaining = MAX_INLINE_NODES;
70            function.mailbox.is_none()
71                && function.captures.len() <= MAX_INLINE_NODES
72                && function.params.len() <= MAX_INLINE_NODES
73                && !self.actors.entries.contains_key(&function.id.0)
74                && !self.actors.managed.contains(&function.id.0)
75                && eligible(&function.body, &mut remaining)
76        }) else {
77            return Ok(self.invoke_values(closure, args, result, locals));
78        };
79        if function.params.len() != args.len() {
80            return Err(invalid(
81                expression.span,
82                "inlined callback argument count differs from signature",
83            ));
84        }
85        expect_type(
86            function.return_type.clone(),
87            result.clone(),
88            expression.span,
89        )?;
90
91        // Local identities belong to each lifted function. Only the lexical
92        // identity tables change: SSA names, control blocks, fault handler and
93        // root slots remain owned by the enclosing physical function.
94        let values = std::mem::take(&mut locals.values);
95        let defined = std::mem::take(&mut locals.defined);
96        let count = std::mem::replace(&mut locals.count, function.local_count);
97        let outcome = (|| {
98            for (param, (ty, value)) in function.params.iter().zip(args) {
99                expect_type(ty.clone(), param.ty.clone(), expression.span)?;
100                locals.define(param.id.0, ty.clone(), value.clone(), expression.span)?;
101            }
102            for (index, capture) in function.captures.iter().enumerate() {
103                let address = self.assign(
104                    locals,
105                    Type::Int,
106                    NativeOperation::Binary(
107                        MachineBinary::Add,
108                        native_operand(closure),
109                        Operand::Int((8 * (index + 1)) as i64),
110                    ),
111                );
112                let raw = self.assign(
113                    locals,
114                    Type::Int,
115                    NativeOperation::Load(LoadKind::I64, native_operand(&address)),
116                );
117                let value = self.unpack(locals, &capture.ty, raw);
118                self.root_value(locals, &capture.ty, &value);
119                locals.define(capture.id.0, capture.ty.clone(), value, expression.span)?;
120            }
121            self.expr(&function.body, locals, 0)
122        })();
123        locals.values = values;
124        locals.defined = defined;
125        locals.count = count;
126        outcome
127    }
128}
