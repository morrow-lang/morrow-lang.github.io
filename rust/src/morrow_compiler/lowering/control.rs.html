4impl Emitter<'_> {
5    /// A strict Never expression evaluates only its operands, stopping at termination.
6    pub(super) fn strict_termination(
7        &mut self,
8        expr: &Expr,
9        locals: &mut Locals,
10        depth: usize,
11    ) -> Lowering<()> {
12        if expr.ty != Type::Never {
13            return Ok(());
14        }
15        let children: Vec<&Expr> = match &expr.kind {
16            ExprKind::List(items) | ExprKind::Tuple(items) | ExprKind::Interpolate(items) => {
17                items.iter().collect()
18            }
19            ExprKind::Range { start, end, .. } => vec![start, end],
20            ExprKind::For { iterable, .. } if iterable.ty == Type::Never => vec![iterable],
21            ExprKind::Map(entries) => entries.iter().flat_map(|(k, v)| [k, v]).collect(),
22            ExprKind::CustomConstruct { fields, .. } => fields.iter().collect(),
23            ExprKind::Construct { value, .. } => value.iter().map(|v| v.as_ref()).collect(),
24            ExprKind::Field { value, .. }
25            | ExprKind::Try(value)
26            | ExprKind::Wrap(value)
27            | ExprKind::JsonCodecTemplate { input: value, .. }
28            | ExprKind::JsonCodec { input: value, .. }
29            | ExprKind::UnionInject { value }
30            | ExprKind::UnionWiden { value }
31            | ExprKind::Unwrap(value)
32            | ExprKind::Unary { value, .. } => vec![value],
33            ExprKind::Call { args, .. } => args.iter().collect(),
34            ExprKind::Invoke { callee, args } => {
35                std::iter::once(callee.as_ref()).chain(args).collect()
36            }
37            ExprKind::Closure { captures, .. } => captures.iter().collect(),
38            ExprKind::Binary { op, left, right } if !matches!(op, BinaryOp::And | BinaryOp::Or) => {
39                vec![left, right]
40            }
41            ExprKind::If { condition, .. } if condition.ty == Type::Never => vec![condition],
42            ExprKind::Match { value, .. } if value.ty == Type::Never => vec![value],
43            _ => return Ok(()),
44        };
45        for child in children {
46            self.expr(child, locals, depth)?;
47        }
48        Err(invalid(
49            expr.span,
50            "Never expression has no terminating operand",
51        ))
52    }

54    /// Save a full-width return payload before any deferred callback can allocate or run.
55    pub(super) fn save_return(&mut self, value: &str, locals: &mut Locals) {
56        let payload = if locals.return_type == Type::Unit {
57            "0".into()
58        } else {
59            self.payload(locals, &locals.return_type.clone(), value.into())
60        };
61        self.output.statement(Statement::Store {
62            kind: LoadKind::I64,
63            value: native_operand(&(payload)),
64            address: native_operand("%return_slot"),
65        });
66        self.output.statement(Statement::Jump("@return".to_owned()));
67    }

69    /// Flush only this function's dynamic cleanup stack and restore its typed return ABI.
70    pub(super) fn finish_function(&mut self, locals: &mut Locals) {
71        self.start_block(locals, "@return");
72        self.output
73            .statement(Statement::Effect(NativeOperation::Call {
74                callee: native_operand("$morrow_rs_run_defers"),
75                args: vec![
76                    (Scalar::I64, native_operand("%defer_head")),
77                    (Scalar::I64, native_operand("%fault")),
78                ],
79                variadic: None,
80            }));
81        let raw = self.assign(
82            locals,
83            Type::Int,
84            NativeOperation::Load(LoadKind::I64, native_operand("%return_slot")),
85        );
86        let value = self.unpack(locals, &locals.return_type.clone(), raw);
87        self.root_exit(locals);
88        self.output
89            .statement(Statement::Return(Some(native_operand(&(value)))));
90        self.output.end();
91    }

93    /// Route explicit returns through the same cleanup path as implicit return and Try.
94    pub(super) fn returned(
95        &mut self,
96        value: &Expr,
97        locals: &mut Locals,
98        depth: usize,
99    ) -> Lowering<(Type, String)> {
100        expect_type(value.ty.clone(), locals.return_type.clone(), value.span)?;
101        let value = self.tail_expr(value, locals, depth)?;
102        self.save_return(&value, locals);
103        Err(Exit::Terminated)
104    }

106    /// Register a captured Unit callback without running its deferred body now.
107    pub(super) fn defer(
108        &mut self,
109        value: &Expr,
110        locals: &mut Locals,
111        depth: usize,
112    ) -> Lowering<(Type, String)> {
113        expect_type(
114            value.ty.clone(),
115            Type::Function(vec![], Box::new(Type::Unit)),
116            value.span,
117        )?;
118        let closure = self.expr(value, locals, depth)?;
119        let previous = self.assign(
120            locals,
121            Type::Int,
122            NativeOperation::Load(LoadKind::I64, native_operand("%defer_head")),
123        );
124        let node = self.assign(
125            locals,
126            Type::Int,
127            NativeOperation::Call {
128                callee: native_operand("$morrow_alloc"),
129                args: vec![(Scalar::I64, native_operand("16"))],
130                variadic: None,
131            },
132        );
133        let link = self.assign(
134            locals,
135            Type::Int,
136            NativeOperation::Binary(
137                MachineBinary::Add,
138                native_operand(&(node)),
139                native_operand("8"),
140            ),
141        );
142        self.output.statement(Statement::Store {
143            kind: LoadKind::I64,
144            value: native_operand(&(closure)),
145            address: native_operand(&(node)),
146        });
147        self.output.statement(Statement::Store {
148            kind: LoadKind::I64,
149            value: native_operand(&(previous)),
150            address: native_operand(&(link)),
151        });
152        self.output.statement(Statement::Store {
153            kind: LoadKind::I64,
154            value: native_operand(&(node)),
155            address: native_operand("%defer_head"),
156        });
157        Ok((Type::Unit, "0".into()))
158    }

160    /// Record a branch only if it reaches the merge block with an actual value.
161    pub(super) fn incoming(
162        &mut self,
163        outcome: Lowering<String>,
164        incoming: &mut Vec<(String, String)>,
165        merge: &str,
166        locals: &mut Locals,
167    ) -> Lowering<()> {
168        match outcome {
169            Ok(value) => {
170                incoming.push((locals.current.clone(), value));
171                self.output.statement(Statement::Jump((merge).to_string()));
172            }
173            Err(Exit::Terminated) => {}
174            Err(error) => return Err(error),
175        }
176        Ok(())
177    }

179    /// Merge continuing branches; a single predecessor needs no artificial phi.
180    pub(super) fn join(
181        &mut self,
182        ty: Type,
183        incoming: Vec<(String, String)>,
184        merge: &str,
185        locals: &mut Locals,
186    ) -> Lowering<(Type, String)> {
187        if incoming.is_empty() {
188            return Err(Exit::Terminated);
189        }
190        self.start_block(locals, merge);
191        let value = if ty == Type::Unit {
192            "0".into()
193        } else if incoming.len() == 1 {
194            incoming[0].1.clone()
195        } else {
196            let operands = incoming
197                .into_iter()
198                .map(|(label, value)| (label, native_operand(&value)))
199                .collect();
200            self.assign(locals, ty.clone(), NativeOperation::Phi(operands))
201        };
202        Ok((ty, value))
203    }

206/// Join branch types through bottom without permitting incompatible live results.
207pub(super) fn joined(left: &Type, right: &Type, span: Span) -> Lowering<Type> {
208    if *left == Type::Never {
209        return Ok(right.clone());
210    }
211    expect_type(right.clone(), left.clone(), span)?;
212    Ok(left.clone())
