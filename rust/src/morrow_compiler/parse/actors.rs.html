1//! Indented selective receive syntax retains timeout expressions separately from messages.
2use super::*;
3impl Parser {
4    /// Parse a bounded receive block; timeout is one final wildcard arm, never a message pattern.
5    pub(super) fn receive_expression(&mut self) -> ParseResult<Parsed> {
6        let view = if self.word("receive_event") {
7            crate::processes::ReceiveView::Events
8        } else {
9            crate::processes::ReceiveView::Messages
10        };
11        let start = self.take().span.start;
12        self.expect(Kind::Colon, "expected ':' after receive")?;
13        self.expect(Kind::Newline, "receive requires an indented arm block")?;
14        self.expect(Kind::Indent, "receive requires an indented arm block")?;
15        let mut arms = Vec::new();
16        let mut timeout = None;
17        let mut depth = 1;
18        let mut end = start;
19        for _ in 0..self.tokens.len() {
20            if self.eat(&Kind::Dedent) {
21                break;
22            }
23            if timeout.is_some() {
24                return Err(self.error("receive timeout must be the final arm"));
25            }
26            if arms.len() >= 128 {
27                return Err(self.error("receive arm limit exceeded (128)"));
28            }
29            let pattern = if view == crate::processes::ReceiveView::Events && self.word("after") {
30                Pattern {
31                    kind: PatternKind::Wildcard,
32                    span: self.current().span,
33                }
34            } else {
35                self.typed_pattern()?
36            };
37            if self.word("after") {
38                if !matches!(pattern.kind, PatternKind::Wildcard) {
39                    return Err(Diagnostic::new(
40                        pattern.span,
41                        "receive timeout requires '_'",
42                    ));
43                }
44                self.take();
45                let duration = self.guard_expression()?;
46                self.expect(Kind::Arrow, "expected '->' after receive timeout")?;
47                let body = self.suite()?;
48                end = body.node.span.end;
49                depth = depth.max(duration.depth + 1).max(body.depth + 1);
50                timeout = Some((Box::new(duration.node), Box::new(body.node)));
51            } else {
52                let (arm, arm_depth) = self.receive_arm(pattern)?;
53                end = arm.span.end;
54                depth = depth.max(arm_depth);
55                arms.push(arm);
56            }
57            if !self.eat(&Kind::Newline)
58                && self.current().kind != Kind::Dedent
59                && !self.previous_dedent()
60            {
61                return Err(self.error("expected end of line after receive arm"));
62            }
63        }
64        if arms.is_empty() {
65            return Err(self.error("receive requires at least one message arm"));
66        }
67        expression(
68            ExprKind::Receive {
69                view,
70                arms,
71                timeout,
72            },
73            Span { start, end },
74            depth,
75        )
76    }
77    /// Parse one message arm while retaining the maximum child nesting and its independent scope.
78    fn receive_arm(&mut self, pattern: Pattern) -> ParseResult<(MatchArm, usize)> {
79        let guard = if self.word("if") {
80            self.take();
81            Some(self.guard_expression()?)
82        } else {
83            None
84        };
85        self.expect(Kind::Arrow, "expected '->' after receive pattern")?;
86        let body = self.suite()?;
87        let end = body.node.span.end;
88        let mut depth = body.depth + 1;
89        if let Some(guard) = &guard {
90            depth = depth.max(guard.depth + 1);
91        }
92        let span = Span {
93            start: pattern.span.start,
94            end,
95        };
96        Ok((
97            MatchArm {
98                pattern,
99                guard: guard.map(|v| v.node),
100                body: body.node,
101                span,
102            },
103            depth,
104        ))
105    }
