1//! Union-bearing argument syntax is a type candidate, never a runtime type witness.
2use super::*;
3
4const LOOKAHEAD_LIMIT: usize = 400_000;
5
6impl Parser {
7    /// Preserve ordinary expression syntax; only a new complete type-only union takes this path.
8    pub(super) fn argument_value(&mut self) -> ParseResult<Parsed> {
9        if !self.union_argument()? {
10            return self.expr(0);
11        }
12        let start = self.current().span.start;
13        let ty = self.ty()?;
14        if !matches!(self.current().kind, Kind::Comma | Kind::Right) {
15            return Err(self.error("expected ',' or ')' after static type argument"));
16        }
17        let end = self.tokens[self.position - 1].span.end;
18        expression(ExprKind::TypeTarget(ty), Span { start, end }, 1)
19    }
20
21    /// Build linear token metadata once, after editor recovery finishes changing its token stream.
22    fn union_argument(&mut self) -> ParseResult<bool> {
23        let span = self.current().span;
24        charge(&mut self.type_argument_work, 1, span)?;
25        if self.type_argument_shapes.is_none() {
26            self.type_argument_shapes = Some(Shapes::build(
27                &self.tokens,
28                &mut self.type_argument_work,
29                span,
30            )?);
31        }
32        Ok(self
33            .type_argument_shapes
34            .as_ref()
35            .is_some_and(|shape| shape.candidate(&self.tokens, self.position)))
36    }
37}

39#[derive(Clone, Copy, Default)]
40struct Counts {
41    unions: usize,
42    invalid: usize,
43}
44
45/// Prefix counts and argument boundaries make every subsequent recognition query constant time.
46pub(super) struct Shapes {
47    ends: Vec<usize>,
48    counts: Vec<Counts>,
49}
50impl Shapes {
51    /// Precharge metadata arrays, delimiter stack and both linear passes before any allocation or token read.
52    fn build(tokens: &[Token], work: &mut usize, span: Span) -> ParseResult<Self> {
53        charge(work, tokens.len().saturating_mul(5).saturating_add(1), span)?;
54        let mut counts = Vec::with_capacity(tokens.len() + 1);
55        let mut count = Counts::default();
56        counts.push(count);
57        for (index, token) in tokens.iter().enumerate() {
58            count.unions += usize::from(token.kind == Kind::Bar);
59            count.invalid += usize::from(!type_token(tokens, index));
60            counts.push(count);
61        }
62        let mut ends = vec![tokens.len(); tokens.len()];
63        let mut outer = Vec::new();
64        let mut end = tokens.len();
65        for (index, token) in tokens.iter().enumerate().rev() {
66            match token.kind {
67                Kind::Right => {
68                    outer.push(end);
69                    end = index;
70                }
71                Kind::Left => end = outer.pop().unwrap_or(tokens.len()),
72                Kind::Comma => end = index,
73                _ => {}
74            }
75            ends[index] = end;
76        }
77        Ok(Self { ends, counts })
78    }
79
80    /// Syntax is merely a candidate: canonical decoder-slot checking alone establishes type context.
81    fn candidate(&self, tokens: &[Token], start: usize) -> bool {
82        let Some(&end) = self.ends.get(start) else {
83            return false;
84        };
85        let Some(token) = tokens.get(end) else {
86            return false;
87        };
88        matches!(token.kind, Kind::Right | Kind::Comma)
89            && self.counts[end].unions > self.counts[start].unions
90            && self.counts[end].invalid == self.counts[start].invalid
91    }
92}
93
94/// Exclude executable calls and arrows without treating uppercase spelling as semantic authority.
95fn type_token(tokens: &[Token], index: usize) -> bool {
96    match &tokens[index].kind {
97        Kind::Name(name) => {
98            !tokens.get(index + 1).is_some_and(|t| t.kind == Kind::Left)
99                || name.starts_with(char::is_uppercase)
100        }
101        Kind::Left | Kind::Right | Kind::Comma | Kind::Dot | Kind::Bar => true,
102        _ => false,
103    }
104}
105
106/// Retain one shared preinspection budget; cached queries cannot restart it.
107fn charge(work: &mut usize, amount: usize, span: Span) -> ParseResult<()> {
108    *work = work.saturating_add(amount);
109    if *work > LOOKAHEAD_LIMIT {
110        Err(Diagnostic::new(
111            span,
112            "static type argument lookahead limit exceeded",
113        ))
114    } else {
115        Ok(())
116    }
117}

119#[cfg(test)]
120mod tests {
121    use super::*;
122    #[test]
123    fn scanning_is_read_only_and_shares_one_preinspection_budget() {
124        let mut parser = source_parser("f(Int | String)", true).unwrap();
125        parser.position = 2;
126        assert!(parser.union_argument().unwrap());
127        assert_eq!(parser.position, 2);
128        assert!(parser.type_spans.as_ref().unwrap().is_empty());
129        parser.type_argument_work = LOOKAHEAD_LIMIT;
130        let error = parser.union_argument().unwrap_err();
131        assert!(error.message.contains("lookahead limit"));
132        assert_eq!(parser.position, 2);
133        assert!(parser.type_spans.unwrap().is_empty());
134    }
135    /// Every token can be queried without repeating any subtree walk or exhausting the shared cap.
136    #[test]
137    fn shape_metadata_matches_a_separate_balanced_scan_with_linear_total_work() {
138        let mut parser=source_parser("f(List(Int | String), (Int | Bool,String), json.decode(x,Int | String), (x)->json.decode(x,Int | String), %{ x: y })", false).unwrap();
139        for position in 0..parser.tokens.len() {
140            parser.position = position;
141            let expected = linear_candidate(&parser.tokens[position..]);
142            assert_eq!(
143                parser.union_argument().unwrap(),
144                expected,
145                "token {position}"
146            );
147        }
148        assert_eq!(parser.type_argument_work, 6 * parser.tokens.len() + 1);
149    }
150    /// The test oracle directly follows nesting instead of reusing cached ranges or prefix counts.
151    fn linear_candidate(tokens: &[Token]) -> bool {
152        let mut nesting = 0;
153        let mut found = false;
154        for (index, token) in tokens.iter().enumerate() {
155            match &token.kind {
156                Kind::Right | Kind::Comma if nesting == 0 => return found,
157                Kind::Left => nesting += 1,
158                Kind::Right => nesting -= 1,
159                Kind::Bar => found = true,
160                Kind::Name(name) => {
161                    if tokens.get(index + 1).is_some_and(|t| t.kind == Kind::Left)
162                        && !name.chars().next().is_some_and(char::is_uppercase)
163                    {
164                        return false;
165                    }
166                }
167                Kind::Comma | Kind::Dot => {}
168                _ => return false,
169            }
170        }
171        false
172    }
