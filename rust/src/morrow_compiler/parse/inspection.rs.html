1//! Bounded source-only inspection; private tokens remain an implementation detail.
2use super::{ParseResult, parse, source_parser};
3use crate::{Diagnostic, Span};
4use std::fmt::{self, Write};
5
6const MAX_DUMP: usize = 16 * 1024 * 1024;
7
8/// Render the real layout-aware lexer with source byte spans and escaped token payloads.
9/// Uses ordinary source/token/depth limits; no recovered token or partial text is published.
10pub fn debug_tokens(source: &str) -> ParseResult<String> {
11    let parser = source_parser(source, false)?;
12    let mut output = Dump::default();
13    for token in parser.tokens {
14        writeln!(
15            output,
16            "{}..{} {:?}",
17            token.span.start, token.span.end, token.kind
18        )
19        .map_err(|_| limit())?;
20    }
21    Ok(output.text)
22}
23
24/// Render the parsed source AST without loading imports, checking types or executing code.
25/// The format is for compiler inspection, not stable AST serialization.
26pub fn debug_ast(source: &str) -> ParseResult<String> {
27    let program = parse(source)?;
28    let mut output = Dump::default();
29    writeln!(output, "{program:#?}").map_err(|_| limit())?;
30    Ok(output.text)
31}
32
33/// Own a bounded complete dump before the CLI is allowed to publish it.
34#[derive(Default)]
35struct Dump {
36    text: String,
37}
38
39impl fmt::Write for Dump {
40    /// Reject each append before allocation when its complete contents exceed the budget.
41    fn write_str(&mut self, text: &str) -> fmt::Result {
42        if !self
43            .text
44            .len()
45            .checked_add(text.len())
46            .is_some_and(|size| size <= MAX_DUMP)
47        {
48            return Err(fmt::Error);
49        }
50        self.text.push_str(text);
51        Ok(())
52    }
53}
54
55/// Report a stable bounded-output failure without exposing an incomplete dump.
56fn limit() -> Diagnostic {
57    Diagnostic::new(Span::default(), "syntax dump exceeds 16 MiB output limit")
58}

60#[cfg(test)]
61mod tests {
62    use super::*;
63
64    #[test]
65    fn output_limit_rejects_before_appending_partial_text() {
66        let mut output = Dump {
67            text: "x".repeat(MAX_DUMP - 2),
68        };
69        output.write_str("é").unwrap();
70        assert_eq!(output.text.len(), MAX_DUMP);
71        assert!(output.write_str("x").is_err());
72        assert_eq!(output.text.len(), MAX_DUMP);
73        assert!(output.text.ends_with('é'));
74    }
