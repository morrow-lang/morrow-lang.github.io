1//! Constants reuse the ordinary function body grammar and checked value representation.
2use super::*;
3
4impl Parser {
5    pub(super) fn constant(&mut self, public: bool) -> ParseResult<Function> {
6        let start = self.take().span.start;
7        let (name, _) = self.name()?;
8        if name == "main" {
9            return Err(self.error("the program entry must be a function, not a constant"));
10        }
11        let return_type = if self.eat(&Kind::Colon) {
12            Some(self.ty()?)
13        } else {
14            None
15        };
16        self.expect(Kind::Assign, "expected '=' after constant name or type")?;
17        if !self.word("comptime") {
18            return Err(self.error("constant initializer requires 'comptime:'"));
19        }
20        self.take();
21        self.expect(Kind::Colon, "expected ':' after comptime")?;
22        let body = self.suite()?;
23        Ok(Function {
24            constraints: Vec::new(),
25            public,
26            name,
27            params: Vec::new(),
28            return_type,
29            guard: None,
30            syntax: FunctionSyntax::Constant,
31            group_start: start,
32            span: Span {
33                start,
34                end: body.node.span.end,
35            },
36            body: body.node,
37        })
38    }
39}
