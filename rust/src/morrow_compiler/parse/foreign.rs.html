1//! Trusted C declarations synthesize ordinary source wrappers with explicit ABI metadata.
2use super::*;
3use crate::ffi::Declaration;
4impl Parser {
5    pub(super) fn foreign_function(&mut self, public: bool) -> ParseResult<Function> {
6        let start = self.take().span.start;
7        let abi = self.foreign_text("expected foreign ABI string \"C\"")?;
8        if abi != "C" {
9            return Err(self.error("only the explicit C foreign ABI is supported"));
10        }
11        if !self.word("fn") {
12            return Err(self.error("expected fn after foreign ABI"));
13        }
14        self.take();
15        let (name, _) = self.name()?;
16        self.expect(Kind::Left, "expected '(' after foreign function name")?;
17        let params = self.function_parameters()?;
18        let mut arguments = Vec::new();
19        let mut physical = Vec::new();
20        for param in &params {
21            let PatternKind::Bind(name) = &param.pattern.kind else {
22                return Err(Diagnostic::new(
23                    param.span,
24                    "foreign parameters require simple named bindings",
25                ));
26            };
27            let ty = param.annotation.as_ref().ok_or_else(|| {
28                Diagnostic::new(param.span, "foreign parameters require explicit ABI types")
29            })?;
30            physical.push(crate::ffi::abi_type(ty, param.span)?);
31            arguments.push(Expr {
32                kind: ExprKind::Name(name.clone()),
33                span: param.span,
34            });
35        }
36        self.expect(
37            Kind::Arrow,
38            "foreign function requires explicit return type",
39        )?;
40        let result = self.ty()?;
41        let result_abi = crate::ffi::abi_type(&result, self.current().span)?;
42        let symbol = if self.word("as") {
43            self.take();
44            self.foreign_text("expected C symbol string after as")?
45        } else {
46            name.clone()
47        };
48        let library = if self.word("from") {
49            self.take();
50            Some(self.foreign_text("expected library string after from")?)
51        } else {
52            None
53        };
54        let span = Span {
55            start,
56            end: self.current().span.start,
57        };
58        self.line_end()?;
59        let declaration = Declaration {
60            symbol,
61            library,
62            params: physical,
63            result: result_abi,
64        };
65        declaration.validate(span)?;
66        Ok(Function {
67            constraints: vec![],
68            guard: None,
69            group_start: start,
70            syntax: FunctionSyntax::Foreign,
71            public,
72            name,
73            params,
74            return_type: Some(result),
75            body: Expr {
76                kind: ExprKind::ForeignCall {
77                    declaration,
78                    args: arguments,
79                },
80                span,
81            },
82            span,
83        })
84    }
85    fn foreign_text(&mut self, message: &str) -> ParseResult<String> {
86        match self.take().kind {
87            Kind::Text(value) => Ok(value),
88            _ => Err(self.error(message)),
89        }
90    }
91}
