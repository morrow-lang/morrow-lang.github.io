4/// Opaque parser-proven member site; callers cannot fabricate field or receiver identities.
5#[derive(Clone, Debug)]
6pub(crate) struct HoleSite {
7    selector: Span,
8    prefix: String,
9    original: String,
10    field: Option<(Span, Span)>,
11}
12impl HoleSite {
13    /// Record the original receiver and field once, excluding unrelated parser nodes.
14    pub(super) fn attach(&mut self, receiver: Span, field: Span) -> ParseResult<()> {
15        if self.field.replace((receiver, field)).is_some() {
16            return Err(Diagnostic::new(field, "multiple editor member holes"));
17        }
18        Ok(())
19    }
20    /// Retain original source replacement coordinates for UTF16 completion edits.
21    pub(crate) fn selector(&self) -> Span {
22        self.selector
23    }
24    /// Filter members by the source prefix before the caret, without interpreting it as evidence.
25    pub(crate) fn prefix(&self) -> &str {
26        &self.prefix
27    }
28    /// A complete known selector must not hide an unrelated current-source type error.
29    pub(crate) fn original(&self) -> &str {
30        &self.original
31    }
32    /// Return the structurally recorded field selection after a successful recovered parse.
33    pub(crate) fn field(&self) -> Span {
34        self.field.expect("successful recovery attaches field").1
35    }
36    /// Match only the private empty-selector Field and its exact original receiver.
37    pub(crate) fn matches(&self, expr: &crate::ast::Expr) -> bool {
38        matches!(&expr.kind,ExprKind::Field {value,name} if name.is_empty() && self.field == Some((value.span,expr.span)))
39    }
40    /// Relocate only semantic identities; source replacement coordinates stay local.
41    pub(crate) fn shift(&mut self, offset: usize) {
42        if let Some((receiver, field)) = &mut self.field {
43            receiver.start += offset;
44            receiver.end += offset;
45            field.start += offset;
46            field.end += offset;
47        }
48    }
49}

51/// Parse one member selector as an opaque hole; all other source must parse normally.
52pub(crate) fn recover_member(source: &str, cursor: usize) -> ParseResult<(Program, HoleSite)> {
53    if !source.is_char_boundary(cursor) {
54        return Err(Diagnostic::new(
55            Span::default(),
56            "invalid member completion cursor",
57        ));
58    }
59    let mut parser = source_parser(source, false)?;
60    let (index, replace, site) = selection(source, cursor, &parser.tokens)?;
61    let token = Token {
62        kind: Kind::MemberHole,
63        span: site.selector,
64    };
65    if replace {
66        parser.tokens[index] = token;
67    } else {
68        if parser.tokens.len() >= MAX_TOKENS {
69            return Err(Diagnostic::new(site.selector, "token limit exceeded"));
70        }
71        parser.tokens.insert(index, token);
72    }
73    parser.member_hole = Some(site);
74    let program = parser.program()?;
75    let site = parser
76        .member_hole
77        .filter(|site| site.field.is_some())
78        .ok_or_else(|| Diagnostic::new(Span::default(), "cursor is not an expression member"))?;
79    Ok((program, site))
80}

82/// Locate a real Dot followed by the caret's identifier/tuple slot, never float or literal text.
83fn selection(
84    source: &str,
85    cursor: usize,
86    tokens: &[Token],
87) -> ParseResult<(usize, bool, HoleSite)> {
88    for (index, pair) in tokens.windows(2).enumerate() {
89        if pair[0].kind != Kind::Dot {
90            continue;
91        }
92        let next = &pair[1];
93        let replace = matches!(next.kind, Kind::Name(_) | Kind::Number(_))
94            && next.span.start <= cursor
95            && cursor <= next.span.end;
96        if !replace && pair[0].span.end != cursor {
97            continue;
98        }
99        let selector = if replace {
100            next.span
101        } else {
102            Span {
103                start: cursor,
104                end: cursor,
105            }
106        };
107        let prefix = source
108            .get(selector.start..cursor)
109            .ok_or_else(|| Diagnostic::new(selector, "invalid member prefix"))?;
110        let original = source
111            .get(selector.start..selector.end)
112            .ok_or_else(|| Diagnostic::new(selector, "invalid member selector"))?;
113        return Ok((
114            index + 1,
115            replace,
116            HoleSite {
117                selector,
118                prefix: prefix.into(),
119                original: original.into(),
120                field: None,
121            },
122        ));
123    }
124    Err(Diagnostic::new(
125        Span {
126            start: cursor,
127            end: cursor,
128        },
129        "cursor is not a member selector",
130    ))
131}
