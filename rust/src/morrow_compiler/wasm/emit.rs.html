34pub(super) fn function(
35    function: &ir::Function,
36    functions: &Functions<'_>,
37    work: &mut usize,
38    strings: Option<&strings::Runtime>,
39) -> Result<Function> {
40    let mut emitter = Emitter {
41        instructions: Vec::new(),
42        locals: Vec::new(),
43        param_count: (function.params.len() + function.captures.len()) as u32,
44        functions,
45        return_type: &function.return_type,
46        work,
47        strings,
48        root_base: None,
49        control_depth: 0,
50        loops: Vec::new(),
51        exit_target: None,
52        return_slot: None,
53        defer_head: None,
54        defer_root: None,
55    };
56    if strings.is_some() {
57        let base = emitter.temp(ValType::I32, function.body.span)?;
58        emitter.root_base = Some(base);
59        emitter.emit(I::GlobalGet(0));
60        emitter.emit(I::LocalSet(base));
61    }
62    let return_slot = emitter.temp(
63        value_type(&function.return_type, function.body.span)?,
64        function.body.span,
65    )?;
66    emitter.return_slot = Some(return_slot);
67    emitter.emit(I::Block(BlockType::Empty));
68    emitter.exit_target = Some(emitter.control_depth);
69    let mut pending = vec![&function.body];
70    let mut has_defers = false;
71    while let Some(expr) = pending.pop() {
72        has_defers |= matches!(expr.kind, ExprKind::Defer(_));
73        pending.extend(ir::children(expr));
74    }
75    if has_defers && strings.is_some() {
76        let head = emitter.temp(ValType::I32, function.body.span)?;
77        let root = emitter.temp(ValType::I32, function.body.span)?;
78        emitter.defer_head = Some(head);
79        emitter.defer_root = Some(root);
80        emitter.emit(I::GlobalGet(0));
81        emitter.emit(I::LocalSet(root));
82        emitter.emit(I::I32Const(0));
83        emitter.root_string(function.body.span)?;
84        emitter.emit(I::Drop);
85    }
86    let mut locals = Locals::new();
87    for (index, param) in function.captures.iter().chain(&function.params).enumerate() {
88        if locals
89            .insert(param.id.0, (index as u32, param.ty.clone()))
90            .is_some()
91        {
92            return Err(invalid(function.body.span, "duplicate parameter identity"));
93        }
94        if managed(&param.ty) {
95            emitter.emit(I::LocalGet(index as u32));
96            emitter.root_string(function.body.span)?;
97            emitter.emit(I::Drop);
98        }
99    }
100    let actual = emitter.expr(&function.body, &mut locals, 0)?;
101    expect(&actual, &function.return_type, function.body.span)?;
102    if actual != Type::Never {
103        emitter.emit(I::LocalSet(return_slot));
104    }
105    emitter.emit(I::End);
106    emitter.exit_target = None;
107    if has_defers && managed(&function.return_type) {
108        // Reserve return rooting before running allocating callbacks. In normal code
109        // expr already rooted the result; failure paths leave its initialized zero.
110        emitter.emit(I::LocalGet(return_slot));
111        emitter.root_string(function.body.span)?;
112        emitter.emit(I::Drop);
113    }
114    emitter.cleanups(function.body.span)?;
115    emitter.restore_roots();
116    emitter.emit(I::LocalGet(return_slot));
117    emitter.emit(I::End);
118    let mut encoded = Function::new(emitter.locals.into_iter().map(|ty| (1, ty)));
119    for instruction in emitter.instructions {
120        encoded.instruction(&instruction);
121    }
122    Ok(encoded)
123}

125pub(super) struct Emitter<'a, 'b> {
126    instructions: Vec<I<'static>>,
127    locals: Vec<ValType>,
128    param_count: u32,
129    functions: &'a Functions<'a>,
130    return_type: &'a Type,
131    work: &'b mut usize,
132    strings: Option<&'a strings::Runtime>,
133    root_base: Option<u32>,
134    control_depth: u32,
135    loops: Vec<(u32, u32)>,
136    exit_target: Option<u32>,
137    return_slot: Option<u32>,
138    defer_head: Option<u32>,
139    defer_root: Option<u32>,
140}

142impl Emitter<'_, '_> {
143    fn restore_roots(&mut self) {
144        if let Some(base) = self.root_base {
145            self.emit(I::LocalGet(base));
146            self.emit(I::GlobalSet(0));
147        }
148    }
149
150    fn root_string(&mut self, span: Span) -> Result<()> {
151        let runtime = self
152            .strings
153            .ok_or_else(|| invalid(span, "missing string heap"))?;
154        self.emit(I::Call(runtime.first + strings::PUSH));
155        Ok(())
156    }
157    pub(super) fn emit(&mut self, instruction: I<'static>) {
158        if matches!(instruction, I::Unreachable)
159            && self.strings.is_some()
160            && let Some(target) = self.exit_target
161        {
162            self.instructions.extend([
163                I::I32Const(1),
164                I::GlobalSet(2),
165                I::Br(self.control_depth - target),
166            ]);
167            return;
168        }
169        let check_fault = matches!(instruction, I::Call(_))
170            && self.strings.is_some()
171            && self.exit_target.is_some();
172        match instruction {
173            I::Block(_) | I::Loop(_) | I::If(_) => self.control_depth += 1,
174            I::End => self.control_depth = self.control_depth.saturating_sub(1),
175            _ => {}
176        }
177        self.instructions.push(instruction);
178        if check_fault {
179            self.instructions.extend([
180                I::GlobalGet(2),
181                I::If(BlockType::Empty),
182                I::Br(self.control_depth + 1 - self.exit_target.unwrap()),
183                I::End,
184            ]);
185        }
186    }
187
188    pub(super) fn temp(&mut self, ty: ValType, span: Span) -> Result<u32> {
189        if self.locals.len() + self.param_count as usize >= 65_536 {
190            return Err(invalid(span, "generated local limit exceeded"));
191        }
192        let index = self.param_count + self.locals.len() as u32;
193        self.locals.push(ty);
194        Ok(index)
195    }
196
197    fn bind(&mut self, locals: &mut Locals, id: ir::LocalId, ty: &Type, span: Span) -> Result<u32> {
198        if locals.contains_key(&id.0) {
199            return Err(invalid(span, "duplicate local identity"));
200        }
201        let index = self.temp(value_type(ty, span)?, span)?;
202        locals.insert(id.0, (index, ty.clone()));
203        Ok(index)
204    }

206    fn expr(&mut self, expr: &Expr, locals: &mut Locals, depth: usize) -> Result<Type> {
207        *self.work += 1;
208        if *self.work > 100_000 || depth >= 128 {
209            return Err(invalid(expr.span, "expression complexity limit exceeded"));
210        }
211        if expr.ty != Type::Never {
212            value_type(&expr.ty, expr.span)?;
213        }
214        let depth = depth + 1;
215        let actual = match &expr.kind {
216            ExprKind::Int(n) => {
217                self.emit(I::I64Const(*n));
218                Type::Int
219            }
220            ExprKind::Float(n) => {
221                self.emit(I::F64Const((*n).into()));
222                Type::Float
223            }
224            ExprKind::Bool(value) => {
225                self.emit(I::I32Const(i32::from(*value)));
226                Type::Bool
227            }
228            ExprKind::Unit => {
229                self.emit(I::I32Const(0));
230                Type::Unit
231            }
232            ExprKind::String(value) => {
233                let pointer = self
234                    .strings
235                    .and_then(|runtime| runtime.literals.get(value))
236                    .ok_or_else(|| invalid(expr.span, "missing string literal allocation"))?;
237                self.emit(I::I32Const(*pointer));
238                Type::String
239            }
240            ExprKind::Interpolate(parts) => {
241                let runtime = self
242                    .strings
243                    .ok_or_else(|| invalid(expr.span, "missing string heap"))?;
244                let first = runtime.first;
245                self.emit(I::I32Const(runtime.literals[""]));
246                for part in parts {
247                    let ty = self.expr(part, locals, depth)?;
248                    match ty {
249                        Type::String => {}
250                        Type::Int => self.emit(I::Call(first + strings::INT_TEXT)),
251                        Type::Bool => {
252                            self.emit(I::If(BlockType::Result(ValType::I32)));
253                            self.emit(I::I32Const(runtime.literals["true"]));
254                            self.emit(I::Else);
255                            self.emit(I::I32Const(runtime.literals["false"]));
256                            self.emit(I::End);
257                        }
258                        Type::Unit => {
259                            self.emit(I::Drop);
260                            self.emit(I::I32Const(runtime.literals["()"]));
261                        }
262                        _ => {
263                            return Err(invalid(
264                                part.span,
265                                "interpolation of this value type is unavailable",
266                            ));
267                        }
268                    }
269                    self.root_string(part.span)?;
270                    self.emit(I::Call(first + strings::CONCAT));
271                    self.root_string(part.span)?;
272                }
273                Type::String
274            }
275            ExprKind::Local(id) => {
276                let (index, ty) = locals
277                    .get(&id.0)
278                    .ok_or_else(|| invalid(expr.span, "local is not in lexical scope"))?;
279                self.emit(I::LocalGet(*index));
280                ty.clone()
281            }
282            ExprKind::Closure { function, captures } => {
283                self.closure(*function, captures, &expr.ty, locals, depth, expr.span)?
284            }
285            ExprKind::Invoke { callee, args } => {
286                let actual = self.expr(callee, locals, depth)?;
287                let slot = self.temp(ValType::I32, expr.span)?;
288                self.emit(I::LocalSet(slot));
289                let mut values = Vec::new();
290                for arg in args {
291                    let ty = self.expr(arg, locals, depth)?;
292                    let value = self.temp(value_type(&ty, arg.span)?, arg.span)?;
293                    self.emit(I::LocalSet(value));
294                    values.push((value, ty));
295                }
296                self.invoke_values(slot, &actual, &values, expr.span)?
297            }
298            ExprKind::Range {
299                start,
300                end,
301                inclusive,
302            } => self.range(start, end, *inclusive, locals, depth, expr.span)?,
303            ExprKind::For {
304                pattern,
305                iterable,
306                body,
307            } => self.iteration(pattern, iterable, body, locals, depth, expr.span)?,
308            ExprKind::Break | ExprKind::Continue => {
309                let (done, next) = *self
310                    .loops
311                    .last()
312                    .ok_or_else(|| invalid(expr.span, "loop control outside loop"))?;
313                let target = if matches!(expr.kind, ExprKind::Break) {
314                    done
315                } else {
316                    next
317                };
318                self.emit(I::Br(self.control_depth - target));
319                Type::Never
320            }
321            ExprKind::Try(value) => self.try_result(value, locals, depth, expr.span)?,
322            ExprKind::With { .. } => self.with_result(expr, locals, depth)?,
323            ExprKind::Defer(value) => self.defer(value, locals, depth, expr.span)?,
324            ExprKind::Return(value) => {
325                let actual = self.expr(value, locals, depth)?;
326                expect(&actual, self.return_type, value.span)?;
327                self.emit(I::LocalSet(self.return_slot.unwrap()));
328                self.emit(I::Br(self.control_depth - self.exit_target.unwrap()));
329                Type::Never
330            }
331            ExprKind::Unary { op, value } => {
332                let ty = self.expr(value, locals, depth)?;
333                match (op, &ty) {
334                    (UnaryOp::Negate, Type::Int) => {
335                        self.emit(I::I64Const(-1));
336                        self.emit(I::I64Mul);
337                    }
338                    (UnaryOp::Negate, Type::Float) => self.emit(I::F64Neg),
339                    (UnaryOp::Not, Type::Bool) => self.emit(I::I32Eqz),
340                    (UnaryOp::BitNot, Type::Int) => {
341                        self.emit(I::I64Const(-1));
342                        self.emit(I::I64Xor);
343                    }
344                    (_, Type::Never) => {}
345                    _ => return Err(invalid(expr.span, "invalid unary operand type")),
346                }
347                ty
348            }
349            ExprKind::Binary { op, left, right } => {
350                self.binary(*op, left, right, locals, depth, expr.span)?
351            }
352            ExprKind::UnionInject { value } | ExprKind::UnionWiden { value } => {
353                self.union_conversion(expr, value, locals, depth)?
354            }
355            ExprKind::Map(entries) => {
356                self.map_literal(entries, &expr.ty, locals, depth, expr.span)?
357            }
358            ExprKind::ForeignCall { .. } => {
359                return Err(invalid(expr.span, "foreign calls require a native runtime"));
360            }
361            ExprKind::Call {
362                target: CallTarget::Builtin(builtin),
363                args,
364            } if matches!(
365                builtin,
366                ir::Builtin::MapNew
367                    | ir::Builtin::MapGet
368                    | ir::Builtin::MapPut
369                    | ir::Builtin::MapDelete
370                    | ir::Builtin::MapLen
371                    | ir::Builtin::MapIsEmpty
372                    | ir::Builtin::MapContains
373                    | ir::Builtin::MapKeys
374                    | ir::Builtin::MapValues
375            ) =>
376            {
377                self.map_call(*builtin, args, &expr.ty, locals, depth, expr.span)?
378            }
379            ExprKind::Call {
380                target: CallTarget::Builtin(builtin),
381                args,
382            } if matches!(
383                builtin,
384                ir::Builtin::ListMap
385                    | ir::Builtin::ListFilter
386                    | ir::Builtin::ListFold
387                    | ir::Builtin::ListFind
388                    | ir::Builtin::ListAny
389                    | ir::Builtin::ListAll
390                    | ir::Builtin::ListContains
391                    | ir::Builtin::ListEnumerate
392                    | ir::Builtin::OptionMap
393                    | ir::Builtin::ResultMap
394                    | ir::Builtin::ResultAndThen
395                    | ir::Builtin::ResultUnwrapOrElse
396            ) =>
397            {
398                self.higher_order(*builtin, args, &expr.ty, locals, depth, expr.span)?
399            }
400            ExprKind::Call { target, args } => {
401                let CallTarget::Function(id) = target else {
402                    let actual = self.string_call(*target, args, locals, depth, expr.span)?;
403                    if actual != expr.ty {
404                        return Err(invalid(expr.span, "string call result type mismatch"));
405                    }
406                    if managed(&actual) {
407                        self.root_string(expr.span)?;
408                    }
409                    return Ok(actual);
410                };
411                let (index, function) = self
412                    .functions
413                    .get(&id.0)
414                    .ok_or_else(|| invalid(expr.span, "unknown function identity"))?;
415                if !function.captures.is_empty() {
416                    return Err(invalid(
417                        expr.span,
418                        "captured function requires closure invocation",
419                    ));
420                }
421                if args.len() != function.params.len() {
422                    return Err(invalid(expr.span, "function argument count mismatch"));
423                }
424                let mut diverges = false;
425                for (arg, param) in args.iter().zip(&function.params) {
426                    let actual = self.expr(arg, locals, depth)?;
427                    expect(&actual, &param.ty, arg.span)?;
428                    diverges |= actual == Type::Never;
429                }
430                self.emit(I::Call(*index));
431                if diverges {
432                    Type::Never
433                } else {
434                    function.return_type.clone()
435                }
436            }
437            ExprKind::If {
438                condition,
439                then_branch,
440                else_branch,
441            } => {
442                let condition_type = self.expr(condition, locals, depth)?;
443                expect(&condition_type, &Type::Bool, condition.span)?;
444                self.emit(I::If(block_type(&expr.ty, expr.span)?));
445                let then_ty = self.expr(then_branch, &mut locals.clone(), depth)?;
446                self.emit(I::Else);
447                let else_ty = if let Some(branch) = else_branch {
448                    self.expr(branch, &mut locals.clone(), depth)?
449                } else {
450                    self.emit(I::I32Const(0));
451                    Type::Unit
452                };
453                self.emit(I::End);
454                if condition_type == Type::Never {
455                    Type::Never
456                } else {
457                    expect(&then_ty, &expr.ty, then_branch.span)?;
458                    expect(&else_ty, &expr.ty, expr.span)?;
459                    expr.ty.clone()
460                }
461            }
462            ExprKind::Block(statements) => {
463                self.block(statements, &mut locals.clone(), depth, &expr.ty)?
464            }
465            ExprKind::Match { value, arms } => {
466                if arms.is_empty() || arms.len() > 1024 {
467                    return Err(invalid(
468                        expr.span,
469                        "match arm count is empty or exceeds limit",
470                    ));
471                }
472                let value_ty = self.expr(value, locals, depth)?;
473                if value_ty == Type::Never {
474                    return Err(invalid(
475                        expr.span,
476                        "match on a nonreturning value is not supported",
477                    ));
478                }
479                let temp = self.temp(value_type(&value_ty, value.span)?, value.span)?;
480                self.emit(I::LocalSet(temp));
481                self.emit(I::Block(block_type(&expr.ty, expr.span)?));
482                for arm in arms {
483                    let mut scoped = locals.clone();
484                    self.pattern(&arm.pattern, temp, &value_ty, &mut scoped, arm.span)?;
485                    self.emit(I::If(BlockType::Result(ValType::I32)));
486                    if let Some(guard) = &arm.guard {
487                        let guard_type = self.expr(guard, &mut scoped, depth)?;
488                        expect(&guard_type, &Type::Bool, guard.span)?;
489                    } else {
490                        self.emit(I::I32Const(1));
491                    }
492                    self.emit(I::Else);
493                    self.emit(I::I32Const(0));
494                    self.emit(I::End);
495                    self.emit(I::If(BlockType::Empty));
496                    let actual = self.expr(&arm.body, &mut scoped, depth)?;
497                    expect(&actual, &expr.ty, arm.body.span)?;
498                    self.emit(I::Br(1));
499                    self.emit(I::End);
500                }
501                self.emit(I::Unreachable); // A malformed non-exhaustive IR cannot fall through.
502                self.emit(I::End);
503                expr.ty.clone()
504            }
505            ExprKind::Tuple(_)
506            | ExprKind::List(_)
507            | ExprKind::CustomConstruct { .. }
508            | ExprKind::Construct { .. }
509            | ExprKind::Field { .. }
510            | ExprKind::Wrap(_)
511            | ExprKind::Unwrap(_) => self.aggregate(expr, locals, depth)?,
512            _ => return Err(invalid(expr.span, unsupported(&expr.kind))),
513        };
514        if actual != expr.ty {
515            return Err(invalid(
516                expr.span,
517                format!(
518                    "expression type mismatch: declared {:?}, produced {actual:?}",
519                    expr.ty
520                ),
521            ));
522        }
523        if managed(&actual) {
524            self.root_string(expr.span)?;
525        }
526        Ok(actual)
527    }

529    fn string_call(
530        &mut self,
531        target: CallTarget,
532        args: &[Expr],
533        locals: &mut Locals,
534        depth: usize,
535        span: Span,
536    ) -> Result<Type> {
537        if let CallTarget::Runtime(id) = target {
538            match crate::runtime::signature(id).map(|signature| signature.symbol) {
539                Some("morrow_str_compare") => return self.compare_text(args, locals, depth, span),
540                Some("morrow_str_join") => return self.join_text(args, locals, depth, span),
541                _ => {}
542            }
543        }
544        use ir::Builtin;
545        if let CallTarget::Builtin(builtin) = target
546            && matches!(
547                builtin,
548                Builtin::ListLen
549                    | Builtin::ListGet
550                    | Builtin::ListHead
551                    | Builtin::ListTail
552                    | Builtin::ListIsEmpty
553                    | Builtin::ListPush
554                    | Builtin::ListConcat
555                    | Builtin::ListReverse
556                    | Builtin::OptionIsSome
557                    | Builtin::OptionIsNone
558                    | Builtin::OptionUnwrapOr
559                    | Builtin::ResultIsOk
560                    | Builtin::ResultIsErr
561                    | Builtin::ResultUnwrapOr
562            )
563        {
564            return self.collection_call(builtin, args, locals, depth, span);
565        }
566        let operation = match target {
567            CallTarget::Builtin(Builtin::StringLen) => Some((strings::LEN, 1, Type::Int)),
568            CallTarget::Builtin(Builtin::StringConcat) => Some((strings::CONCAT, 2, Type::String)),
569            CallTarget::Builtin(Builtin::StringEq) => Some((strings::EQ, 2, Type::Bool)),
570            CallTarget::Runtime(id) => {
571                crate::runtime::signature(id).and_then(|signature| match signature.symbol {
572                    "morrow_str_len" => Some((strings::LEN, 1, Type::Int)),
573                    "morrow_str_concat" => Some((strings::CONCAT, 2, Type::String)),
574                    "morrow_str_eq" => Some((strings::EQ, 2, Type::Bool)),
575                    "morrow_str_quote" => Some((strings::QUOTE, 1, Type::String)),
576                    _ => None,
577                })
578            }
579            _ => None,
580        };
581        let (operation, arity, result) = operation.ok_or_else(|| {
582            invalid(
583                span,
584                format!("host capability {target:?} is unavailable in the portable browser target"),
585            )
586        })?;
587        let runtime = self
588            .strings
589            .ok_or_else(|| invalid(span, "missing string heap"))?;
590        if args.len() != arity {
591            return Err(invalid(span, "string call argument count mismatch"));
592        }
593        let mut diverges = false;
594        for arg in args {
595            let actual = self.expr(arg, locals, depth)?;
596            expect(&actual, &Type::String, arg.span)?;
597            diverges |= actual == Type::Never;
598        }
599        self.emit(I::Call(runtime.first + operation));
600        Ok(if diverges { Type::Never } else { result })
601    }

603    fn block(
604        &mut self,
605        statements: &[Stmt],
606        locals: &mut Locals,
607        depth: usize,
608        result: &Type,
609    ) -> Result<Type> {
610        let mut last = Type::Unit;
611        let mut diverges = false;
612        for (position, statement) in statements.iter().enumerate() {
613            let is_last = position + 1 == statements.len();
614            let span = match statement {
615                Stmt::Let { value, .. } | Stmt::Expr(value) | Stmt::LetElse { value, .. } => {
616                    value.span
617                }
618            };
619            let root_mark = if self.strings.is_some() {
620                let mark = self.temp(ValType::I32, span)?;
621                self.emit(I::GlobalGet(0));
622                self.emit(I::LocalSet(mark));
623                Some(mark)
624            } else {
625                None
626            };
627            let mut retained_bindings = Vec::new();
628            match statement {
629                Stmt::Let { id, value } => {
630                    let ty = self.expr(value, locals, depth)?;
631                    if ty != Type::Never {
632                        let index = self.bind(locals, *id, &ty, value.span)?;
633                        self.emit(I::LocalSet(index));
634                        if managed(&ty) {
635                            retained_bindings.push(index);
636                        }
637                    }
638                    diverges |= ty == Type::Never;
639                    last = Type::Unit;
640                }
641                Stmt::Expr(value) => {
642                    last = self.expr(value, locals, depth)?;
643                    diverges |= last == Type::Never;
644                    if !is_last && last != Type::Never {
645                        self.emit(I::Drop);
646                    }
647                }
648                Stmt::LetElse {
649                    pattern,
650                    value,
651                    else_branch,
652                } => {
653                    let ty = self.expr(value, locals, depth)?;
654                    let temp = self.temp(value_type(&ty, value.span)?, value.span)?;
655                    self.emit(I::LocalSet(temp));
656                    let previous = locals
657                        .keys()
658                        .copied()
659                        .collect::<std::collections::BTreeSet<_>>();
660                    self.pattern(pattern, temp, &ty, locals, value.span)?;
661                    for (id, (slot, ty)) in locals.iter() {
662                        if !previous.contains(id) && managed(ty) {
663                            retained_bindings.push(*slot);
664                        }
665                    }
666                    self.emit(I::I32Eqz);
667                    self.emit(I::If(BlockType::Empty));
668                    let actual = self.expr(else_branch, &mut locals.clone(), depth)?;
669                    expect(&actual, &Type::Never, else_branch.span)?;
670                    self.emit(I::End);
671                    last = Type::Unit;
672                }
673            }
674            if let Some(mark) = root_mark {
675                // The statement's operands are dead. Keep only its escaping binding or
676                // block result, with no allocating operation between reset and re-root.
677                self.emit(I::LocalGet(mark));
678                self.emit(I::GlobalSet(0));
679                for binding in retained_bindings {
680                    self.emit(I::LocalGet(binding));
681                    self.root_string(span)?;
682                    self.emit(I::Drop);
683                }
684                if is_last && managed(&last) {
685                    self.root_string(span)?;
686                }
687            }
688        }
689        if statements.is_empty() || !matches!(statements.last(), Some(Stmt::Expr(_))) {
690            self.emit(I::I32Const(0));
691        }
692        if diverges {
693            self.emit(I::Unreachable);
694            Ok(Type::Never)
695        } else {
696            expect(&last, result, Span::default())?;
697            Ok(last)
698        }
699    }
700
701    fn pattern(
702        &mut self,
703        pattern: &Pattern,
704        value: u32,
705        ty: &Type,
706        locals: &mut Locals,
707        span: Span,
708    ) -> Result<()> {
709        match pattern {
710            Pattern::Wildcard => self.emit(I::I32Const(1)),
711            Pattern::Bind(id) => {
712                let binding = self.bind(locals, *id, ty, span)?;
713                self.emit(I::LocalGet(value));
714                self.emit(I::LocalSet(binding));
715                self.emit(I::I32Const(1));
716            }
717            Pattern::Int(n) if *ty == Type::Int => {
718                self.emit(I::LocalGet(value));
719                self.emit(I::I64Const(*n));
720                self.emit(I::I64Eq);
721            }
722            Pattern::Bool(value_pattern) if *ty == Type::Bool => {
723                self.emit(I::LocalGet(value));
724                self.emit(I::I32Const(i32::from(*value_pattern)));
725                self.emit(I::I32Eq);
726            }
727            _ => return self.aggregate_pattern(pattern, value, ty, locals, span),
728        }
729        Ok(())
730    }
731
732    fn binary(
733        &mut self,
734        op: BinaryOp,
735        left: &Expr,
736        right: &Expr,
737        locals: &mut Locals,
738        depth: usize,
739        span: Span,
740    ) -> Result<Type> {
741        let lhs = self.expr(left, locals, depth)?;
742        if matches!(op, BinaryOp::And | BinaryOp::Or) {
743            expect(&lhs, &Type::Bool, left.span)?;
744            self.emit(I::If(BlockType::Result(ValType::I32)));
745            let rhs;
746            if op == BinaryOp::And {
747                rhs = self.expr(right, &mut locals.clone(), depth)?;
748                self.emit(I::Else);
749                self.emit(I::I32Const(0));
750            } else {
751                self.emit(I::I32Const(1));
752                self.emit(I::Else);
753                rhs = self.expr(right, &mut locals.clone(), depth)?;
754            }
755            self.emit(I::End);
756            expect(&rhs, &Type::Bool, right.span)?;
757            return Ok(if lhs == Type::Never {
758                Type::Never
759            } else {
760                Type::Bool
761            });
762        }
763        let rhs = self.expr(right, locals, depth)?;
764        let operand = if lhs == Type::Never { &rhs } else { &lhs };
765        expect(&rhs, operand, right.span)?;
766        if lhs == Type::Never || rhs == Type::Never {
767            self.emit(I::Unreachable);
768            return Ok(Type::Never);
769        }
770        if *operand == Type::String {
771            let runtime = self
772                .strings
773                .ok_or_else(|| invalid(span, "missing string heap"))?;
774            match op {
775                BinaryOp::Add => {
776                    self.emit(I::Call(runtime.first + strings::CONCAT));
777                    Ok(Type::String)
778                }
779                BinaryOp::Eq | BinaryOp::Ne => {
780                    self.emit(I::Call(runtime.first + strings::EQ));
781                    if op == BinaryOp::Ne {
782                        self.emit(I::I32Eqz);
783                    }
784                    Ok(Type::Bool)
785                }
786                _ => Err(invalid(span, "unsupported string operator")),
787            }
788        } else {
789            self.numeric(op, operand, span)
790        }
791    }
792}
793
794fn block_type(ty: &Type, span: Span) -> Result<BlockType> {
795    if *ty == Type::Never {
796        Ok(BlockType::Empty)
797    } else {
798        Ok(BlockType::Result(value_type(ty, span)?))
799    }
800}

802fn unsupported(kind: &ExprKind) -> &'static str {
803    match kind {
804        ExprKind::Actor(_) => "actors are a native server capability",
805        ExprKind::FunctionValue { .. } | ExprKind::Lambda { .. } => {
806            "unlifted callable cannot enter executable IR"
807        }
808        _ => "operation requires an unsupported browser representation or host capability",
809    }
