1//! Unicode16 decimal membership using checksum-generated tables, independent of the host Rust UCD.
2use crate::decimal_table::RANGES;
3
4/// Classify nonempty valid text; the caller must charge/check its input-byte ceiling first.
5pub(crate) fn is_decimal(text: &str) -> bool {
6    !text.is_empty() && text.chars().all(member)
7}
8
9/// Test one scalar against sorted, disjoint pinned Nd intervals without allocating.
10fn member(scalar: char) -> bool {
11    let value = scalar as u32;
12    if value < 128 {
13        return scalar.is_ascii_digit();
14    }
15    RANGES
16        .binary_search_by(|(low, high)| {
17            if value < *low {
18                std::cmp::Ordering::Greater
19            } else if value > *high {
20                std::cmp::Ordering::Less
21            } else {
22                std::cmp::Ordering::Equal
23            }
24        })
25        .is_ok()
26}

28#[cfg(test)]
29mod tests {
30    use super::*;
31    #[test]
32    fn every_scalar_agrees_with_primary_ucd_categories() {
33        let input =
34            include_str!("../../../tests/fixtures/unicode/16.0.0/DerivedGeneralCategory.txt");
35        let mut expected = vec![false; 0x110000];
36        for line in input.lines() {
37            let text = line.split('#').next().unwrap().trim();
38            let Some((span, category)) = text.split_once(';') else {
39                continue;
40            };
41            if category.trim() != "Nd" {
42                continue;
43            }
44            let ends: Vec<_> = span
45                .trim()
46                .split("..")
47                .map(|s| usize::from_str_radix(s, 16).unwrap())
48                .collect();
49            for item in &mut expected[ends[0]..=ends[ends.len() - 1]] {
50                *item = true;
51            }
52        }
53        for (code, expected) in expected.into_iter().enumerate() {
54            if let Some(scalar) = char::from_u32(code as u32) {
55                assert_eq!(member(scalar), expected, "U+{code:04X}");
56            }
57        }
58    }
