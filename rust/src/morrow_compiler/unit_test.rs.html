1//! Source-owned test functions use ordinary checking and isolated native entries.
2use crate::{Diagnostic, Span, Type, ast, check, doctest, ir, parse};
3
4/// One original test declaration; adjacent clauses share the first source anchor.
5#[derive(Debug)]
6pub struct Case {
7    pub name: String,
8    pub span: Span,
9}
10
11/// Discover at most 256 source test_ groups from bounded parsed source.
12/// Comments, imported functions and benchmark helpers do not declare unit tests.
13pub fn discover(source: &str) -> Result<Vec<Case>, Diagnostic> {
14    let parsed = parse::parse(source)?;
15    let mut tests = Vec::new();
16    let mut previous = None;
17    for function in &parsed.functions {
18        if !function.name.starts_with("test_") || previous == Some(&function.name) {
19            continue;
20        }
21        previous = Some(&function.name);
22        if tests.len() == 256 {
23            return Err(Diagnostic::new(
24                function.span,
25                "unit test limit exceeds 256",
26            ));
27        }
28        tests.push(Case {
29            name: function.name.clone(),
30            span: function.span,
31        });
32    }
33    Ok(tests)
34}
35
36/// Check the complete source library and reject reusable generic tests before specialization.
37/// The selected name is an already resolved source declaration identity, not an import spelling.
38pub fn prepare(source: &ast::Program, name: &str) -> Result<ir::Program, Diagnostic> {
39    let mut program = check::check_test(source, name)?;
40    select_entry(&mut program, name)?;
41    Ok(program)
42}
43
44/// Select one fully checked Unit/Result(Unit,E) function without changing resolved calls.
45/// Invalid names, arguments, captures or result shapes leave the supplied IR untouched.
46fn select_entry(program: &mut ir::Program, name: &str) -> Result<(), Diagnostic> {
47    let mut matches = program
48        .functions
49        .iter()
50        .filter(|function| function.name == name);
51    let valid = matches.next().is_some_and(|function| {
52        function.params.is_empty()
53            && function.captures.is_empty()
54            && (function.return_type == Type::Unit
55                || matches!(&function.return_type,
56                Type::Result(ok, _) if **ok == Type::Unit))
57    });
58    if !valid || matches.next().is_some() {
59        return Err(Diagnostic::new(
60            Span::default(),
61            format!("test {name} requires zero arguments and Unit or Result(Unit, E) result"),
62        ));
63    }
64    doctest::rename_entry(program, name);
65    Ok(())
66}
