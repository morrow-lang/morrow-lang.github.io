18const QUEUED: usize = 4096;
19#[derive(Clone, Serialize)]
20enum Body {
21    ToServer(Command),
22    ToClient(ServerMessage),
23}
24#[derive(Clone, Serialize)]
25struct Packet {
26    client: usize,
27    link: u64,
28    connection: String,
29    body: Body,
30}
31enum Event {
32    Work(u32),
33    Packet(Packet),
34}
35struct Peer {

45pub(crate) struct Engine {
46    // Retire native sessions and checkpoint locks before removing private files.
47    hub: Option<Hub>,
48    directory: Option<Directory>,
49    pub oracle: Shared,
50    clock: Rc<Cell<u64>>,
51    config: Config,
52    counts: Counts,
53    random: Random,
54    recorder: Recorder,
55    peers: Vec<Peer>,
56    queue: BTreeMap<(u64, u64), Event>,
57    serial: u64,
58    boot: u64,
59    healing: bool,
60    committed_commands: BTreeSet<(String, i64)>,
61}
62impl Engine {
63    pub fn new(config: Config) -> Result<Self, String> {
64        config.validate()?;
65        let directory = config
66            .durable
67            .then(Directory::new)
68            .transpose()
69            .map_err(|e| e.to_string())?;
70        let peers = (0..config.clients)
71            .map(|id| Peer {
72                client: None,
73                principal: format!("client-{id}"),
74                room: format!("room-{}", id % config.rooms),
75                namespace: None,
76                connection: None,
77                link: 0,
78                draft: String::new(),
79            })
80            .collect();
81        let mut engine = Self {
82            hub: None,
83            directory,
84            oracle: Rc::new(RefCell::new(Oracle::default())),
85            clock: Rc::new(Cell::new(0)),
86            random: Random::new(config.seed),
87            recorder: Recorder::new(&config),
88            config,
89            counts: Counts::default(),
90            peers,
91            queue: BTreeMap::new(),
92            serial: 0,
93            boot: 0,
94            healing: false,
95            committed_commands: BTreeSet::new(),
96        };
97        engine.start_server()?;
98        for client in 0..engine.peers.len() {
99            engine.connect(client)?;
100        }
101        Ok(engine)
102    }
103    fn hub(&self) -> &Hub {
104        self.hub
105            .as_ref()
106            .expect("server is installed between transitions")
107    }
108    fn hub_mut(&mut self) -> &mut Hub {
109        self.hub
110            .as_mut()
111            .expect("server is installed between transitions")
112    }
113    fn record(&mut self, kind: &str, client: Option<usize>, detail: String) {
114        self.recorder.record(self.clock.get(), kind, client, detail);
115    }
116    fn start_server(&mut self) -> Result<(), String> {
117        let native = NativeDomain::simulated(
118            self.directory.as_ref().map(|d| d.path.as_path()),
119            self.clock.clone(),
120        )
121        .map_err(|e| e.to_string())?;
122        let limits = Limits {
123            max_tasks: 32,
124            ..Limits::default()
125        };
126        self.hub = Some(
127            Hub::with_domain(
128                format!("sim-{}-boot-{}", self.config.seed, self.boot),
129                limits,
130                CheckedDomain {
131                    native,
132                    oracle: self.oracle.clone(),
133                },
134            )
135            .map_err(|e| e.to_string())?,
136        );
137        Ok(())
138    }
139    fn check(&mut self) -> Result<(), String> {
140        if let Some(message) = self.oracle.borrow().failure.clone() {
141            return Err(message);
142        }
143        for peer in &self.peers {
144            if let Some(client) = &peer.client
145                && client.draft() != peer.draft
146            {
147                return Err("network activity changed an unsent local draft".into());
148            }
149        }
150        let counts = self.hub().counts();
151        if counts.0 > self.config.rooms as usize
152            || counts.1 > 1024
153            || counts.2 > self.config.clients as usize
154        {
155            return Err("production admission counters exceeded configured bounds".into());
156        }
157        Ok(())
158    }
159    fn checked_snapshot(&mut self, room: &str) -> Result<morrow_web_protocol::Snapshot, String> {
160        let snapshot = self.hub().snapshot(room).map_err(|e| e.to_string())?;
161        self.oracle.borrow().check_snapshot(&snapshot)?;
162        self.counts.snapshots_checked += 1;
163        Ok(snapshot)
164    }
165    fn put(&mut self, at: u64, event: Event) {
166        self.serial += 1;
167        self.queue.insert((at, self.serial), event);
168        self.counts.max_queued_events = self.counts.max_queued_events.max(self.queue.len());
169    }
170    fn send(&mut self, packet: Packet) {
171        if !self.healing && self.random.chance(self.config.faults.drop_per_mille) {
172            self.counts.drops += 1;
173            self.record("drop", Some(packet.client), digest(&packet));
174            return;
175        }
176        let duplicate = !self.healing && self.random.chance(self.config.faults.duplicate_per_mille);
177        if duplicate {
178            self.counts.duplicates += 1;
179            self.record("duplicate", Some(packet.client), digest(&packet));
180        }
181        for _ in 0..if duplicate { 2 } else { 1 } {
182            // Reserve one queue entry for the next workload decision.
183            if self.queue.len() >= QUEUED - 1 {
184                self.counts.overload_drops += 1;
185                self.record("overload", Some(packet.client), digest(&packet));
186                continue;
187            }
188            let delay = if !self.healing
189                && self.config.max_delay_ms != 0
190                && self.random.chance(self.config.faults.delay_per_mille)
191            {
192                self.counts.delays += 1;
193                1 + self.random.below(u64::from(self.config.max_delay_ms))
194            } else {
195                0
196            };
197            let at = self.clock.get() + delay;
198            self.record(
199                "schedule",
200                Some(packet.client),
201                format!("{at}:{}", digest(&packet)),
202            );
203            self.put(at, Event::Packet(packet.clone()));
204        }
205    }
206    fn command_packet(&mut self, client: usize, command: Command) -> Result<(), String> {
207        let peer = &self.peers[client];
208        let bytes = encode(&ClientMessage::Command(command)).map_err(|e| e.to_string())?;
209        let ClientMessage::Command(command) = decode(&bytes).map_err(|e| e.to_string())? else {
210            return Err("command wire roundtrip changed envelope".into());
211        };
212        let packet = Packet {
213            client,
214            link: peer.link,
215            connection: peer
216                .connection
217                .clone()
218                .ok_or("sending without a connection")?,
219            body: Body::ToServer(command),
220        };
221        self.send(packet);
222        Ok(())
223    }
224    fn reply_packet(&mut self, client: usize, message: ServerMessage) -> Result<(), String> {
225        let peer = &self.peers[client];
226        let Some(connection) = peer.connection.clone() else {
227            return Ok(());
228        };
229        let message =
230            decode(&encode(&message).map_err(|e| e.to_string())?).map_err(|e| e.to_string())?;
231        self.send(Packet {
232            client,
233            link: peer.link,
234            connection,
235            body: Body::ToClient(message),
236        });
237        Ok(())
238    }
239    fn connect(&mut self, id: usize) -> Result<(), String> {
240        let peer = &self.peers[id];
241        let (principal, room, namespace) = (
242            peer.principal.clone(),
243            peer.room.clone(),
244            peer.namespace.clone(),
245        );
246        let now = self.clock.get();
247        let connected = match self
248            .hub_mut()
249            .connect(&principal, &room, namespace.as_deref(), now)
250        {
251            Err(Error::NamespaceExpired) => {
252                self.counts.expired_namespaces += 1;
253                self.hub_mut().connect(&principal, &room, None, now)
254            }
255            result => result,
256        }
257        .map_err(|e| e.to_string())?;
258        self.oracle.borrow().check_snapshot(&connected.snapshot)?;
259        self.counts.snapshots_checked += 1;
260        let detail = digest(&connected);
261        let peer = &mut self.peers[id];
262        peer.link += 1;
263        peer.connection = Some(connected.connection.clone());
264        peer.namespace = Some(connected.namespace.clone());
265        let retry = if let Some(client) = &mut peer.client {
266            self.counts.reconnects += 1;
267            client.reconnect(connected).map_err(|e| e.to_string())?
268        } else {
269            peer.client = Some(Client::new(connected).map_err(|e| e.to_string())?);
270            None
271        };
272        self.record("connect", Some(id), detail);
273        if let Some(command) = retry {
274            self.command_packet(id, command)?;
275        }
276        self.check()
277    }
278    fn disconnect(&mut self, id: usize) {
279        if let Some(connection) = self.peers[id].connection.take() {
280            self.hub_mut().disconnect(&connection);
281        }
282        let peer = &mut self.peers[id];
283        peer.link += 1;
284        if let Some(client) = &mut peer.client {
285            client.set_online(false);
286        }
287        self.counts.disconnects += 1;
288        self.record("disconnect", Some(id), String::new());
289    }
290    fn restart(&mut self) -> Result<(), String> {
291        self.hub.take(); // Drop the real native sessions and release the checkpoint lock.
292        if !self.config.durable {
293            self.oracle.borrow_mut().rooms.clear();
294        }
295        self.committed_commands.clear();
296        self.boot += 1;
297        self.counts.restarts += 1;
298        for peer in &mut self.peers {
299            peer.connection = None;
300            peer.link += 1;
301            peer.client
302                .as_mut()
303                .expect("initial handshake")
304                .set_online(false);
305        }
306        self.start_server()?;
307        self.record(
308            "restart",
309            None,
310            format!("boot={};durable={}", self.boot, self.config.durable),
311        );
312        Ok(())
313    }
314    fn work(&mut self, step: u32) -> Result<(), String> {
315        if step + 1 < self.config.steps {
316            let at = (u128::from(step + 1) * u128::from(self.config.duration_ms)
317                / u128::from(self.config.steps)) as u64;
318            self.put(at, Event::Work(step + 1));
319        }
320        if self.random.chance(self.config.faults.restart_per_mille) {
321            self.restart()?;
322        }
323        let id = self.random.below(self.peers.len() as u64) as usize;
324        let draft = format!("draft-{step}-{id}-🌿");
325        self.peers[id]
326            .client
327            .as_mut()
328            .expect("initial handshake")
329            .set_draft(draft.clone())
330            .map_err(|e| e.to_string())?;
331        self.peers[id].draft = draft;
332        if self.random.chance(self.config.faults.disconnect_per_mille) {
333            self.disconnect(id);
334            return Ok(());
335        }
336        let live = self.peers[id]
337            .connection
338            .as_ref()
339            .is_some_and(|connection| self.hub().connection_is_live(connection, self.clock.get()));
340        if !live {
341            self.connect(id)?;
342        }
343        if let Some(command) = self.peers[id].client.as_ref().unwrap().pending().cloned() {
344            return self.command_packet(id, command);
345        }
346        let mutation = self.mutation(id, step);
347        match self.peers[id].client.as_mut().unwrap().submit(mutation) {
348            Ok(command) => {
349                self.counts.commands += 1;
350                self.command_packet(id, command)?;
351            }
352            Err(Error::ResyncRequired) => self.connect(id)?,
353            Err(error) => return Err(format!("client submission failed unexpectedly: {error}")),
354        }
355        Ok(())
356    }
357    fn mutation(&mut self, id: usize, step: u32) -> Mutation {
358        let tasks = &self.peers[id].client.as_ref().unwrap().snapshot().tasks;
359        let choice = self.random.below(4);
360        if tasks.is_empty() || choice <= 1 {
361            return Mutation::Add {
362                label: format!("task-{step}-{id}-🌱"),
363            };
364        }
365        let task = &tasks[self.random.below(tasks.len() as u64) as usize];
366        if choice == 2 {
367            Mutation::SetDone {
368                id: task.id,
369                done: !task.done,
370            }
371        } else {
372            Mutation::Remove { id: task.id }
373        }
374    }
375    fn deliver_server(
376        &mut self,
377        id: usize,
378        connection: String,
379        command: Command,
380    ) -> Result<(), String> {
381        let room = self.peers[id].room.clone();
382        let principal = self.peers[id].principal.clone();
383        let before = self.checked_snapshot(&room)?;
384        let applied = self.oracle.borrow().applied;
385        let now = self.clock.get();
386        let result = self
387            .hub_mut()
388            .command(&principal, &connection, command.clone(), now);
389        self.check()?;
390        let after = self.checked_snapshot(&room)?;
391        let committed = self.oracle.borrow().applied != applied;
392        if committed {
393            if !self
394                .committed_commands
395                .insert((command.namespace.clone(), command.sequence.0))
396            {
397                return Err("one command committed more than once in the same namespace".into());
398            }
399            if after.revision.0 != before.revision.0 + 1 {
400                return Err("commit did not advance revision exactly once".into());
401            }
402        } else if after != before {
403            return Err("rejected or duplicate command changed authoritative state".into());
404        }
405        match result {
406            Ok(outcome) => {
407                if committed && outcome.status != Status::Applied {
408                    return Err("committed mutation was not acknowledged as applied".into());
409                }
410                if outcome.status == Status::Conflict {
411                    self.counts.conflicts += 1;
412                }
413                self.record("server_outcome", Some(id), digest(&outcome));
414                self.reply_packet(id, ServerMessage::Outcome(outcome))?;
415                for other in 0..self.peers.len() {
416                    if self.peers[other].room == room
417                        && self.peers[other]
418                            .connection
419                            .as_ref()
420                            .is_some_and(|c| self.hub().connection_is_live(c, now))
421                    {
422                        self.reply_packet(other, ServerMessage::Snapshot(after.clone()))?;
423                    }
424                }
425            }
426            Err(
427                error @ (Error::ConnectionExpired
428                | Error::NamespaceExpired
429                | Error::IncarnationMismatch),
430            ) => self.reply_packet(id, ServerMessage::Error(error))?,
431            Err(error) => {
432                return Err(format!(
433                    "production gateway rejected a well-formed scenario: {error}"
434                ));
435            }
436        }
437        Ok(())
438    }
439    fn deliver_client(&mut self, id: usize, message: ServerMessage) -> Result<(), String> {
440        let client = self.peers[id].client.as_mut().unwrap();
441        match message {
442            ServerMessage::Outcome(outcome) => match client.accept_outcome(&outcome) {
443                Ok(()) | Err(Error::UnexpectedOutcome) => (),
444                Err(error) => return Err(format!("valid outcome failed: {error}")),
445            },
446            ServerMessage::Snapshot(snapshot) => {
447                let before = client.snapshot().clone();
448                let stale = snapshot.incarnation == before.incarnation
449                    && snapshot.revision < before.revision;
450                let accepted = client
451                    .accept_snapshot(snapshot.clone(), false)
452                    .map_err(|e| e.to_string())?;
453                if stale && (accepted || client.snapshot() != &before) {
454                    return Err("stale snapshot moved confirmed client state backwards".into());
455                }
456                if accepted && client.snapshot() != &snapshot {
457                    return Err("accepted snapshot changed in transit".into());
458                }
459                self.counts.snapshots_checked += 1;
460            }
461            ServerMessage::Error(_) => self.disconnect(id),
462            _ => return Err("unexpected server envelope in application packet queue".into()),
463        }
464        Ok(())
465    }
466    fn event_budget(&self) -> u64 {
467        let clients = u64::from(self.config.clients);
468        // A decision can reconnect and retry, then retry once more: two sends,
469        // each duplicated. Each of those four command deliveries emits one
470        // outcome and at most `clients` snapshots, each duplicated once.
471        let decision = 1 + 4 + 8 * (1 + clients);
472        // Healing disables faults. Each client can retry once, and each room
473        // submits one fresh command; each costs a command plus its replies.
474        let healing = (clients + u64::from(self.config.rooms)) * (clients + 2);
475        u64::from(self.config.steps) * decision + healing
476    }
477    fn drain(&mut self) -> Result<(), String> {
478        while let Some(((at, _), event)) = self.queue.pop_first() {
479            if at < self.clock.get() {
480                return Err("event queue regressed virtual time".into());
481            }
482            self.clock.set(at);
483            self.counts.processed_events += 1;
484            if self.counts.processed_events > self.event_budget() {
485                return Err("simulation event budget exhausted".into());
486            }
487            match event {
488                Event::Work(step) => {
489                    self.record("work", None, step.to_string());
490                    self.work(step)?;
491                }
492                Event::Packet(packet) => {
493                    self.record("deliver", Some(packet.client), digest(&packet));
494                    if self.peers[packet.client].link != packet.link {
495                        self.counts.stale_frames += 1;
496                    } else {
497                        match packet.body {
498                            Body::ToServer(command) => {
499                                self.deliver_server(packet.client, packet.connection, command)?
500                            }
501                            Body::ToClient(message) => {
502                                self.deliver_client(packet.client, message)?
503                            }
504                        }
505                    }
506                }
507            }
508            self.check()?;
509        }
510        Ok(())
511    }
512    fn heal(&mut self) -> Result<(), String> {
513        self.healing = true;
514        self.clock
515            .set(self.clock.get().max(self.config.duration_ms) + 1);
516        self.record(
517            "heal",
518            None,
519            "fault injection disabled; reconnect, resolve uncertainty, commit per room".into(),
520        );
521        for id in 0..self.peers.len() {
522            self.connect(id)?;
523        }
524        self.drain()?;
525        for room in 0..self.config.rooms as usize {
526            let id = room;
527            let snapshot = self.peers[id].client.as_ref().unwrap().snapshot();
528            let mutation = snapshot.tasks.first().map_or_else(
529                || Mutation::Add {
530                    label: format!("healed-room-{room}"),
531                },
532                |task| Mutation::SetDone {
533                    id: task.id,
534                    done: !task.done,
535                },
536            );
537            let command = self.peers[id]
538                .client
539                .as_mut()
540                .unwrap()
541                .submit(mutation)
542                .map_err(|e| format!("healed client cannot submit: {e}"))?;
543            let before = self.oracle.borrow().applied;
544            self.counts.commands += 1;
545            self.command_packet(id, command)?;
546            self.drain()?;
547            if self.oracle.borrow().applied != before + 1 {
548                return Err("healthy phase made no application progress".into());
549            }
550            self.counts.healing_commits += 1;
551        }
552        for id in 0..self.peers.len() {
553            let room = self.peers[id].room.clone();
554            let server = self.checked_snapshot(&room)?;
555            let client = self.peers[id].client.as_ref().unwrap();
556            if client.snapshot() != &server || client.pending().is_some() {
557                return Err("clients did not converge after fault-free delivery".into());
558            }
559        }
560        Ok(())
561    }
562    pub fn execute(&mut self) -> Result<Report, String> {
563        self.put(0, Event::Work(0));
564        self.drain()?;
565        self.heal()?;
566        self.check()?;
567        self.counts.applied = self.oracle.borrow().applied;
568        self.counts.recovered_rooms = self.oracle.borrow().restored;
569        let mut snapshots = Vec::new();
570        for room in 0..self.config.rooms {
571            snapshots.push(self.checked_snapshot(&format!("room-{room}"))?);
572        }
573        Ok(Report {
574            simulator_version: SIMULATOR_VERSION,
575            config: self.config.clone(),
576            config_digest: digest(&self.config),
577            virtual_duration_ms: self.clock.get(),
578            counts: self.counts.clone(),
579            trace_events: self.recorder.events,
580            trace_omitted: self.recorder.events - self.recorder.trace.len() as u64,
581            trace: self.recorder.trace.iter().cloned().collect(),
582            trace_digest: self.recorder.hash(),
583            final_state_digest: digest(&snapshots),
584        })
585    }
586    pub fn failure(&self, message: String) -> Failure {
587        Failure {
588            simulator_version: SIMULATOR_VERSION,
589            config: self.config.clone(),
590            event: self.recorder.events,
591            time_ms: self.clock.get(),
592            message,
593            trace: self.recorder.trace.iter().cloned().collect(),
594        }
595    }
596}
