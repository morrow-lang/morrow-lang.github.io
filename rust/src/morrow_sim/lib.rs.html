9/// Runs the real protocol and native Morrow actors under a bounded virtual event loop.
10/// Checkpoint I/O uses the local filesystem; the driver injects transport and restart faults.
11pub fn run(config: Config) -> Result<Report, Failure> {
12    let mut engine = engine::Engine::new(config.clone()).map_err(|message| Failure {
13        simulator_version: SIMULATOR_VERSION,
14        config,
15        event: 0,
16        time_ms: 0,
17        message,
18        trace: vec![],
19    })?;
20    engine.execute().map_err(|message| engine.failure(message))
21}
22
23fn replay_error(config: &Config, message: &str) -> Failure {
24    Failure {
25        simulator_version: SIMULATOR_VERSION,
26        config: config.clone(),
27        event: 0,
28        time_ms: 0,
29        message: message.into(),
30        trace: vec![],
31    }
32}

34/// Re-executes the recorded configuration and compares every report field.
35/// Digests detect divergence; they do not authenticate an untrusted report.
36pub fn replay(report: &Report) -> Result<Report, Failure> {
37    if report.simulator_version != SIMULATOR_VERSION
38        || report.trace.len() > MAX_TRACE as usize
39        || report.config_digest != support::digest(&report.config)
40    {
41        return Err(replay_error(
42            &report.config,
43            "unsupported or inconsistent replay format",
44        ));
45    }
46    let actual = run(report.config.clone())?;
47    if &actual == report {
48        return Ok(actual);
49    }
50    Err(replay_error(
51        &report.config,
52        "replay report does not match the deterministic execution",
53    ))
54}
55
56/// A matching failure is a successful reproduction, not a passing simulation.
57/// CLI callers preserve the failing exit status and original failure report.
58pub fn replay_failure(expected: &Failure) -> Result<Failure, Failure> {
59    if expected.simulator_version != SIMULATOR_VERSION || expected.trace.len() > MAX_TRACE as usize
60    {
61        return Err(replay_error(
62            &expected.config,
63            "unsupported failure replay format",
64        ));
65    }
66    match run(expected.config.clone()) {
67        Err(actual) if actual == *expected => Ok(actual),
68        _ => Err(replay_error(
69            &expected.config,
70            "recorded failure did not reproduce exactly",
71        )),
72    }
73}
