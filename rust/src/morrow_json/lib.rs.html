48/// Construct a stable code/byte-offset failure; code zero is reserved for evaluator exhaustion.

56/// Return the native NUL-terminated prefix, scanning no farther than the input limit plus one.
57pub fn input(text: &str) -> &str {
58    let end = text
59        .bytes()
60        .take(INPUT + 1)
61        .position(|b| b == 0)
62        .unwrap_or(text.len());
63    &text[..end]
64}

66#[derive(Debug)]
67pub struct Limits {
68    pub work: usize,
69    pub allocated: usize,
70}
71impl Limits {
72    /// Initialize independent work/allocation allowances of `bytes` for one evaluation phase.
73    pub fn new(bytes: usize) -> Self {
74        Self {
75            work: bytes,
76            allocated: bytes,
77        }
78    }
79    /// Consume `amount` before work or allocation; exhaustion empties the allowance and returns a fault.
80    pub fn charge(left: &mut usize, amount: usize) -> Result<()> {
81        if amount > *left {
82            *left = 0;
83            scope::mark_exhausted();
84            return Err(error(0, -1));
85        }
86        *left -= amount;
87        Ok(())
88    }
89}
90pub struct Budget<'a> {
91    pub limits: &'a mut Limits,
92    pub work: usize,
93    pub allocated: usize,
94    pub nodes: usize,
95    pub at: usize,
96}
97impl<'a> Budget<'a> {
98    /// Reserve bounded input-scan work in a native-profile operation linked to aggregate limits.
99    pub fn new(limits: &'a mut Limits, bytes: usize) -> Result<Self> {
100        let mut budget = Self {
101            limits,
102            work: 8 * bytes + 64 * NODES,
103            allocated: 0,
104            nodes: 0,
105            at: 0,
106        };
107        budget.work(8 * bytes)?;
108        Ok(budget)
109    }
110    /// Charge aggregate and operation work before executing `units` of bounded processing.
111    pub fn work(&mut self, units: usize) -> Result<()> {
112        Limits::charge(&mut self.limits.work, units)?;
113        if units > self.work {
114            return Err(error(4, self.at as i64));
115        }
116        self.work -= units;
117        Ok(())
118    }
119    /// Charge logical native bytes before allocation; zero-byte requests retain the native one-byte rule.
120    pub fn allocate(&mut self, bytes: usize) -> Result<()> {
121        let bytes = bytes.max(1);
122        Limits::charge(&mut self.limits.allocated, bytes)?;
123        if bytes > ALLOC - self.allocated {
124            return Err(error(4, self.at as i64));
125        }
126        self.allocated += bytes;
127        Ok(())
128    }
129    /// Reserve one native-profile node plus larger Rust storage before publishing its Rc allocation.
130    pub fn node(&mut self) -> Result<()> {
131        scope::charge_nodes(1)?;
132        if self.nodes == NODES {
133            return Err(error(4, self.at as i64));
134        }
135        self.allocate(72)?;
136        let actual = std::mem::size_of::<Node>() + 2 * std::mem::size_of::<usize>();
137        Limits::charge(&mut self.limits.allocated, actual.saturating_sub(72))?;
138        self.nodes += 1;
139        Ok(())
140    }
141}
