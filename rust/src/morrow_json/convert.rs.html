1//! Exact decimal integers and explicitly rounded binary64 conversions.
2use super::*;
3struct Decimal {
4    negative: bool,
5    first: Option<usize>,
6    last: usize,
7    digits: usize,
8    fraction: usize,
9    exponent: i64,
10}
11/// Analyze validated number text without allocation, retaining significant positions and a capped exponent.
12fn decimal(text: &str) -> Decimal {
13    let mut shape = Decimal {
14        negative: text.starts_with('-'),
15        first: None,
16        last: 0,
17        digits: 0,
18        fraction: 0,
19        exponent: 0,
20    };
21    let mut fraction = false;
22    for (i, byte) in text.bytes().enumerate().skip(usize::from(shape.negative)) {
23        if matches!(byte, b'e' | b'E') {
24            shape.exponent = exponent(&text[i + 1..]);
25            break;
26        }
27        if byte == b'.' {
28            fraction = true;
29            continue;
30        }
31        if byte != b'0' {
32            if shape.first.is_none() {
33                shape.first = Some(shape.digits);
34            }
35            shape.last = shape.digits;
36        }
37        shape.digits += 1;
38        shape.fraction += usize::from(fraction);
39    }
40    shape
41}
42/// Read validated exponent digits into a saturated signed value; magnitude never controls loop counts.
43fn exponent(text: &str) -> i64 {
44    let negative = text.starts_with('-');
45    let cap = INPUT as i64 + 100;
46    let mut value = 0;
47    for byte in text.bytes().skip(usize::from(text.starts_with(['+', '-']))) {
48        value = (value * 10 + i64::from(byte - b'0')).min(cap);
49    }
50    if negative { -value } else { value }
51}
52/// Convert a validated number lexeme exactly to signed64, distinguishing fractions from range errors.
53pub fn integer(text: &str) -> Result<i64> {
54    let shape = decimal(text);
55    let Some(first) = shape.first else {
56        return Ok(0);
57    };
58    let scale = shape.exponent - shape.fraction as i64 + (shape.digits - shape.last - 1) as i64;
59    if scale < 0 {
60        return Err(error(9, -1));
61    }
62    let significant = shape.last - first + 1;
63    if significant > 19 || scale > 19 - significant as i64 {
64        return Err(error(8, -1));
65    }
66    let mut magnitude = 0u64;
67    for byte in text
68        .bytes()
69        .skip(usize::from(shape.negative))
70        .filter(|b| *b != b'.')
71        .skip(first)
72        .take(significant)
73    {
74        magnitude = magnitude * 10 + u64::from(byte - b'0');
75    }
76    for _ in 0..scale {
77        magnitude *= 10;
78    }
79    let maximum = i64::MAX as u64 + u64::from(shape.negative);
80    if magnitude > maximum {
81        return Err(error(8, -1));
82    }
83    Ok(if magnitude == 1 << 63 {
84        i64::MIN
85    } else if shape.negative {
86        -(magnitude as i64)
87    } else {
88        magnitude as i64
89    })
90}
91/// Round validated decimal text once to binary64; reject overflow and nonzero underflow while preserving -0.
92pub fn float(text: &str) -> Result<f64> {
93    let value: f64 = text.parse().map_err(|_| error(1, -1))?;
94    if !value.is_finite() || (value == 0.0 && decimal(text).first.is_some()) {
95        return Err(error(8, -1));
96    }
97    Ok(value)
98}
99/// Match C %.17g using exact-precision standard formatting, never shortest display.
100pub fn format(value: f64) -> Result<String> {
101    if !value.is_finite() {
102        return Err(error(11, -1));
103    }
104    if value == 0.0 {
105        return Ok(if value.is_sign_negative() { "-0" } else { "0" }.into());
106    }
107    let scientific = format!("{:.16e}", value.abs());
108    let (mantissa, exponent) = scientific.split_once('e').ok_or_else(|| error(8, -1))?;
109    let exponent: i32 = exponent.parse().map_err(|_| error(8, -1))?;
110    let mut digits = mantissa.replace('.', "");
111    while digits.len() > 1 && digits.ends_with('0') {
112        digits.pop();
113    }
114    let mut output = if value.is_sign_negative() {
115        "-".into()
116    } else {
117        String::new()
118    };
119    if !(-4..17).contains(&exponent) {
120        output.push_str(&digits[..1]);
121        if digits.len() > 1 {
122            output.push('.');
123            output.push_str(&digits[1..]);
124        }
125        output.push_str(&format!("e{exponent:+03}"));
126    } else {
127        fixed(&digits, exponent, &mut output);
128    }
129    Ok(output)
130}
131/// Append already-rounded digits in fixed form; `exponent` is restricted to the native %.17g fixed range.
132fn fixed(digits: &str, exponent: i32, output: &mut String) {
133    let point = exponent + 1;
134    if point <= 0 {
135        output.push_str("0.");
136        for _ in 0..-point {
137            output.push('0');
138        }
139        output.push_str(digits);
140    } else if point as usize >= digits.len() {
141        output.push_str(digits);
142        for _ in digits.len()..point as usize {
143            output.push('0');
144        }
145    } else {
146        output.push_str(&digits[..point as usize]);
147        output.push('.');
148        output.push_str(&digits[point as usize..]);
149    }
150}
