1//! Standard certificate validation plus configured leaf pinning; no permissive verifier.
2use rustls::{
3    ClientConfig, RootCertStore, ServerConfig,
4    pki_types::{CertificateDer, PrivateKeyDer},
5};
6use sha2::{Digest, Sha256};
7use std::{io, sync::Arc};
8
9/// Explicit binary peer protocol; JSON peers cannot negotiate this transport.
10pub const ALPN: &[u8] = b"morrow.peer.protobuf.v1";
11#[derive(Clone)]
12pub struct Security {
13    pub(crate) client: Arc<ClientConfig>,
14    pub(crate) server: Arc<ServerConfig>,
15    pub(crate) leaf_fingerprint: [u8; 32],
16}
17impl Security {
18    /// Load one cluster trust root and this node's leaf/key from bounded DER records.
19    /// Certificates must authorize both server and client authentication.
20    pub fn from_der(ca: &[u8], certificate: &[u8], private_key: &[u8]) -> io::Result<Self> {
21        if ca.is_empty()
22            || ca.len() > 16_384
23            || certificate.is_empty()
24            || certificate.len() > 16_384
25            || private_key.is_empty()
26            || private_key.len() > 8192
27        {
28            return Err(io::Error::new(
29                io::ErrorKind::InvalidInput,
30                "invalid TLS credential size",
31            ));
32        }
33        let provider = Arc::new(rustls::crypto::ring::default_provider());
34        let mut roots = RootCertStore::empty();
35        roots
36            .add(CertificateDer::from(ca.to_vec()))
37            .map_err(io::Error::other)?;
38        let verifier = rustls::server::WebPkiClientVerifier::builder_with_provider(
39            Arc::new(roots.clone()),
40            provider.clone(),
41        )
42        .build()
43        .map_err(io::Error::other)?;
44        let chain = vec![CertificateDer::from(certificate.to_vec())];
45        let key = PrivateKeyDer::try_from(private_key.to_vec()).map_err(io::Error::other)?;
46        let mut server = ServerConfig::builder_with_provider(provider.clone())
47            .with_protocol_versions(&[&rustls::version::TLS13])
48            .map_err(io::Error::other)?
49            .with_client_cert_verifier(verifier)
50            .with_single_cert(chain.clone(), key.clone_key())
51            .map_err(io::Error::other)?;
52        server.alpn_protocols = vec![ALPN.to_vec()];
53        server.max_early_data_size = 0;
54        let mut client = ClientConfig::builder_with_provider(provider)
55            .with_protocol_versions(&[&rustls::version::TLS13])
56            .map_err(io::Error::other)?
57            .with_root_certificates(roots)
58            .with_client_auth_cert(chain, key)
59            .map_err(io::Error::other)?;
60        client.alpn_protocols = vec![ALPN.to_vec()];
61        client.enable_early_data = false;
62        Ok(Self {
63            client: Arc::new(client),
64            server: Arc::new(server),
65            leaf_fingerprint: fingerprint(certificate),
66        })
67    }
68}
69pub(crate) fn fingerprint(certificate: &[u8]) -> [u8; 32] {
70    Sha256::digest(certificate).into()
71}
