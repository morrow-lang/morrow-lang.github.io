5/// Exact initial peer schema version; incompatible peers fail before forwarding.
6pub const PROTOCOL_VERSION: u8 = 2;
7// Legacy hash domains are persisted protocol identifiers; the rename must not move rooms.
8const OWNER_ALGORITHM: &[u8] = b"fern-room-owner-v1\0";
9/// Operator-controlled member record. Endpoints are never learned from peer frames.
10#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
11#[serde(deny_unknown_fields)]
12pub struct Member {
13    /// Stable logical node name.
14    pub id: NodeId,
15    /// Configured TCP host:port; the transport performs bounded resolution/dialing.
16    pub endpoint: String,
17    /// DNS identity required during ordinary TLS certificate verification.
18    pub tls_name: String,
19    /// SHA-256 of the verified leaf DER certificate, binding this member's identity.
20    pub certificate_sha256: [u8; 32],
21}
22/// A validated, sorted, fixed routing configuration for at most sixteen nodes.
23#[derive(Clone, Debug)]
24pub struct Config {
25    cluster: ClusterId,
26    local: NodeId,
27    members: Vec<Member>,
28    manifest: ManifestId,
29    placement: ManifestId,
30}
31impl Config {
32    /// Validate membership and hash a canonical length-delimited manifest.
33    /// Local identity is excluded from the digest so all members obtain the same value.
34    pub fn new(cluster: ClusterId, local: NodeId, mut members: Vec<Member>) -> Result<Self, Error> {
35        if members.is_empty() || members.len() > 16 {
36            return Err(Error::InvalidConfig);
37        }
38        members.sort_by(|a, b| a.id.cmp(&b.id));
39        for (index, member) in members.iter().enumerate() {
40            validate_member(member)?;
41            if index > 0 && members[index - 1].id == member.id {
42                return Err(Error::DuplicateNode);
43            }
44            if members[..index]
45                .iter()
46                .any(|m| m.certificate_sha256 == member.certificate_sha256)
47            {
48                return Err(Error::DuplicateCertificate);
49            }
50        }
51        if !members.iter().any(|m| m.id == local) {
52            return Err(Error::UnknownNode);
53        }
54        let mut hash = Sha256::new();
55        hash.update(b"fern-routing-v1\0");
56        field(&mut hash, cluster.as_str().as_bytes());
57        hash.update((members.len() as u32).to_be_bytes());
58        for member in &members {
59            field(&mut hash, member.id.as_str().as_bytes());
60            field(&mut hash, member.endpoint.as_bytes());
61            field(&mut hash, member.tls_name.as_bytes());
62            hash.update(member.certificate_sha256);
63        }
64        let mut placement = Sha256::new();
65        placement.update(b"fern-room-placement-v1\0");
66        field(&mut placement, OWNER_ALGORITHM);
67        field(&mut placement, cluster.as_str().as_bytes());
68        placement.update((members.len() as u32).to_be_bytes());
69        for member in &members {
70            field(&mut placement, member.id.as_str().as_bytes());
71        }
72        Ok(Self {
73            cluster,
74            local,
75            members,
76            manifest: ManifestId(hash.finalize().into()),
77            placement: ManifestId(placement.finalize().into()),
78        })
79    }
80    /// Stable cluster identity used in peer handshakes.
81    pub fn cluster(&self) -> &ClusterId {
82        &self.cluster
83    }
84    /// Configured identity of this process.
85    pub fn local(&self) -> &NodeId {
86        &self.local
87    }
88    /// Exact routing configuration fingerprint. It supplies agreement, not fencing.
89    pub fn manifest(&self) -> ManifestId {
90        self.manifest
91    }
92    /// Checkpoint ownership fingerprint: routing algorithm, cluster, and sorted node IDs.
93    /// Credential renewal and address changes preserve this value; membership changes
94    /// do not. Peer authentication must still compare the complete [`Self::manifest`].
95    pub fn placement(&self) -> ManifestId {
96        self.placement
97    }
98    /// Canonically sorted membership records.
99    pub fn members(&self) -> &[Member] {
100        &self.members
101    }
102    /// Look up an allowlisted peer without allocating or accepting a remote endpoint.
103    pub fn member(&self, id: &NodeId) -> Option<&Member> {
104        self.members
105            .binary_search_by(|m| m.id.cmp(id))
106            .ok()
107            .map(|i| &self.members[i])
108    }
109    /// Stable rendezvous owner for a valid UTF-8 room ID, independent of liveness.
110    /// Highest SHA-256 score wins; sorted node ID breaks the theoretical digest tie.
111    pub fn owner(&self, room: &str) -> Result<&NodeId, Error> {
112        if room.is_empty() || room.len() > 128 || room.chars().any(char::is_control) {
113            return Err(Error::InvalidRoom);
114        }
115        let mut winner: Option<([u8; 32], &NodeId)> = None;
116        for member in &self.members {
117            let mut hash = Sha256::new();
118            hash.update(OWNER_ALGORITHM);
119            field(&mut hash, self.cluster.as_str().as_bytes());
120            field(&mut hash, room.as_bytes());
121            field(&mut hash, member.id.as_str().as_bytes());
122            let score: [u8; 32] = hash.finalize().into();
123            if winner.as_ref().is_none_or(|(best, _)| score > *best) {
124                winner = Some((score, &member.id));
125            }
126        }
127        Ok(winner.expect("validated nonempty membership").1)
128    }
129    /// Validate protocol and membership agreement; TLS identity binding remains mandatory.
130    /// This method does not authenticate a certificate or a claimed node by itself.
131    pub fn validate_hello(&self, hello: &Hello) -> Result<(), Error> {
132        if hello.version != PROTOCOL_VERSION {
133            return Err(Error::VersionMismatch);
134        }
135        if hello.cluster != self.cluster {
136            return Err(Error::ClusterMismatch);
137        }
138        if hello.manifest != self.manifest {
139            return Err(Error::ManifestMismatch);
140        }
141        if hello.node == self.local {
142            return Err(Error::SelfConnection);
143        }
144        if self.member(&hello.node).is_none() {
145            return Err(Error::UnknownNode);
146        }
147        Ok(())
148    }
149    /// Validate agreement and require the identity authenticated independently by TLS.
150    pub fn validate_authenticated_hello(&self, peer: &NodeId, hello: &Hello) -> Result<(), Error> {
151        self.validate_hello(hello)?;
152        if &hello.node != peer {
153            return Err(Error::UnknownNode);
154        }
155        Ok(())
156    }
157}
158fn field(hash: &mut Sha256, bytes: &[u8]) {
159    hash.update((bytes.len() as u32).to_be_bytes());
160    hash.update(bytes);
161}
162fn validate_member(member: &Member) -> Result<(), Error> {
163    if member.endpoint.is_empty()
164        || member.endpoint.len() > 320
165        || member
166            .endpoint
167            .bytes()
168            .any(|b| b.is_ascii_control() || b.is_ascii_whitespace())
169        || member.certificate_sha256 == [0; 32]
170    {
171        return Err(Error::InvalidConfig);
172    }
173    let Some((host, port)) = member.endpoint.rsplit_once(':') else {
174        return Err(Error::InvalidConfig);
175    };
176    if host.is_empty() || port.parse::<u16>().ok().filter(|port| *port != 0).is_none() {
177        return Err(Error::InvalidConfig);
178    }
179    if member.tls_name.is_empty()
180        || member.tls_name.len() > 253
181        || !member.tls_name.split('.').all(|label| {
182            !label.is_empty()
183                && label.len() <= 63
184                && !label.starts_with('-')
185                && !label.ends_with('-')
186                && label
187                    .bytes()
188                    .all(|b| b.is_ascii_alphanumeric() || b == b'-')
189        })
190    {
191        return Err(Error::InvalidConfig);
192    }
193    Ok(())
194}
