8use morrow_web_protocol::{
9    Client, Connected, Decimal, Error, MAX_FRAME_BYTES, MAX_LABEL_BYTES, Snapshot, VERSION,
10};
11use serde::{Deserialize, Serialize};

13/// Maximum persisted offline record size. Records contain no authentication material.
14pub const MAX_SAVED_BYTES: usize = MAX_FRAME_BYTES;
15/// Browser socket output admission threshold; commands are never queued indefinitely.
16pub const MAX_OUTPUT_BYTES: usize = 262_144;
17
18/// Bounded offline state: cached confirmed data, editable draft and uncertainty notice.
19#[derive(Clone, Debug, Serialize, Deserialize)]
20#[serde(deny_unknown_fields)]
21pub struct Saved {
22    pub draft: String,
23    pub snapshot: Option<Snapshot>,
24    pub had_pending: bool,
25}
26impl Saved {
27    /// Restore untrusted origin-local storage without restoring mutation replay authority.
28    pub fn decode(text: &str) -> Result<Self, Error> {
29        let saved: Self = morrow_web_protocol::decode(text.as_bytes())?;
30        saved.validate()?;
31        Ok(saved)
32    }
33    /// Encode a validated bounded record for persistence.
34    pub fn encode(&self) -> Result<String, Error> {
35        self.validate()?;
36        String::from_utf8(morrow_web_protocol::encode(self)?).map_err(|_| Error::Malformed)
37    }
38    /// Origin-local storage is shared by every tab. A snapshot persist from a tab
39    /// whose draft is empty must not erase another tab's in-progress draft.
40    pub fn keep_existing_draft(&mut self, stored: Option<&str>) {
41        if !self.draft.is_empty() {
42            return;
43        }
44        if let Some(text) = stored
45            && let Ok(existing) = Self::decode(text)
46            && !existing.draft.is_empty()
47        {
48            self.draft = existing.draft;
49        }
50    }
51    fn validate(&self) -> Result<(), Error> {
52        if self.draft.len() > MAX_LABEL_BYTES {
53            return Err(Error::InvalidLabel);
54        }
55        if let Some(snapshot) = &self.snapshot {
56            Client::new(Connected {
57                version: VERSION,
58                connection: "offline".into(),
59                namespace: "offline".into(),
60                next_sequence: Decimal(1),
61                snapshot: snapshot.clone(),
62                resumed: false,
63            })?;
64        }
65        Ok(())
66    }
67}

69/// Capped exponential reconnect delay with injected jitter for deterministic tests.
70pub fn reconnect_delay_ms(attempt: u32, jitter_ms: u32) -> u32 {
71    500_u32.saturating_mul(1_u32 << attempt.min(6)).min(30_000) + jitter_ms.min(999)
72}
73
74/// Reject before appending to an already full browser WebSocket output buffer.
75pub fn can_send(buffered_bytes: usize, frame_bytes: usize) -> bool {
76    frame_bytes <= MAX_FRAME_BYTES && buffered_bytes <= MAX_OUTPUT_BYTES.saturating_sub(frame_bytes)
77}
