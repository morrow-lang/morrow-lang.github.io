1//! Immutable Rust launcher fixture for literal-path, failure and timeout tests.
2use std::{env, fs, os::unix::ffi::OsStrExt, path::Path, process::ExitCode, time::Duration};
3
4fn main() -> ExitCode {
5    let arguments: Vec<_> = env::args_os().collect();
6    let executable = Path::new(&arguments[0]);
7    if executable
8        .file_name()
9        .is_some_and(|name| name == "opener-timeout")
10    {
11        std::thread::sleep(Duration::from_secs(60));
12        return ExitCode::SUCCESS;
13    }
14    // argv[0] may be only a basename when launched through PATH. The test
15    // supplies its owned configuration path explicitly instead of deriving it.
16    if let Some(exit) = env::var_os("MORROW_OPENER_FIXTURE_EXIT_FILE")
17        && let Ok(code) = fs::read_to_string(exit)
18    {
19        return ExitCode::from(code.parse::<u8>().unwrap());
20    }
21    if let (Some(calls), Some(seen)) = (env::var_os("CALLS"), env::var_os("SEEN")) {
22        fs::copy(&arguments[1], seen).unwrap();
23        let mut output = Vec::new();
24        for argument in &arguments[1..] {
25            output.extend_from_slice(argument.as_bytes());
26            output.push(0);
27        }
28        fs::write(calls, output).unwrap();
29    } else {
30        fs::write(executable.with_extension("result"), arguments[1].as_bytes()).unwrap();
31    }
32    ExitCode::SUCCESS
33}
