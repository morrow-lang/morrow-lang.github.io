11/// Full-width Morrow integer represented by a canonical JSON decimal string.
12#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord)]
13pub struct Decimal(pub i64);
14impl Serialize for Decimal {
15    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
16        serializer.serialize_str(&self.0.to_string())
17    }
18}
19impl<'de> Deserialize<'de> for Decimal {
20    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
21        let text = String::deserialize(deserializer)?;
22        let value = text.parse::<i64>().map_err(serde::de::Error::custom)?;
23        if text != value.to_string() {
24            return Err(serde::de::Error::custom("noncanonical integer"));
25        }
26        Ok(Self(value))
27    }
28}

64/// Decode a bounded, closed typed schema. Serde rejects duplicate struct fields.
65pub fn decode<T: DeserializeOwned>(bytes: &[u8]) -> Result<T, Error> {
66    if bytes.len() > MAX_FRAME_BYTES {
67        return Err(Error::FrameTooLarge);
68    }
69    serde_json::from_slice(bytes).map_err(|_| Error::Malformed)
70}

72/// Serialize through a capped writer so rejected output never allocates without bound.
73pub fn encode<T: Serialize>(value: &T) -> Result<Vec<u8>, Error> {
74    struct Capped(Vec<u8>);
75    impl std::io::Write for Capped {
76        fn write(&mut self, bytes: &[u8]) -> std::io::Result<usize> {
77            if bytes.len() > MAX_FRAME_BYTES - self.0.len() {
78                return Err(std::io::Error::other("frame limit"));
79            }
80            self.0.extend_from_slice(bytes);
81            Ok(bytes.len())
82        }
83        fn flush(&mut self) -> std::io::Result<()> {
84            Ok(())
85        }
86    }
87    let mut output = Capped(Vec::new());
88    serde_json::to_writer(&mut output, value).map_err(|_| Error::FrameTooLarge)?;
89    Ok(output.0)
90}

92/// A supported mutation. IDs are room-scoped and never reused within an incarnation.
93#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
94#[serde(tag = "kind", rename_all = "snake_case", deny_unknown_fields)]
95pub enum Mutation {
96    Add { label: String },
97    SetDone { id: Decimal, done: bool },
98    Remove { id: Decimal },
99}
100impl Mutation {
101    pub(crate) fn validate(&self, label_limit: usize) -> Result<(), Error> {
102        match self {
103            Self::Add { label }
104                if label.trim().is_empty()
105                    || label.len() > label_limit
106                    || label.chars().any(char::is_control) =>
107            {
108                Err(Error::InvalidLabel)
109            }
110            Self::SetDone { id, .. } | Self::Remove { id } if id.0 <= 0 => Err(Error::Malformed),
111            _ => Ok(()),
112        }
113    }
114}
