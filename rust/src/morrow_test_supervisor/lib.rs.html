16/// Captured native test result, bounded independently for stdout and stderr.
17#[derive(Debug)]
18pub struct Captured {
19    pub status: ExitStatus,
20    pub stdout: Vec<u8>,
21    pub stderr: Vec<u8>,
22}

24/// Execute with a positive timeout up to sixty seconds and 256 KiB per stream.
25/// The deadline initiates retained-child cleanup; kernel reaping can extend it.
26#[cfg(test)]
27fn run(executable: &Path, timeout: Duration) -> Result<Captured, String> {
28    run_command(Command::new(executable), timeout)
29}

31/// Round a checked positive duration upward without allowing a zero deadline.
32fn timeout_ms(timeout: Duration) -> Result<u64, String> {
33    if timeout.is_zero() || timeout > Duration::from_secs(60) {
34        return Err("test timeout must be greater than zero and at most 60 seconds".into());
35    }
36    Ok(timeout.as_nanos().div_ceil(1_000_000) as u64)
37}

39/// Keep native spool ownership separate from Rust; only remove the empty parent.
40struct Directory(PathBuf);
41impl Directory {
42    /// Allocate an exclusive private directory; no existing path is reused.
43    fn new() -> Result<Self, String> {
44        use std::os::unix::fs::DirBuilderExt;
45        let epoch = std::time::SystemTime::now()
46            .duration_since(std::time::UNIX_EPOCH)
47            .map_err(|error| error.to_string())?
48            .as_nanos();
49        for attempt in 0..100 {
50            let path = std::env::temp_dir().join(format!(
51                ".morrow-test-{}-{epoch}-{attempt}",
52                std::process::id()
53            ));
54            match std::fs::DirBuilder::new().mode(0o700).create(&path) {
55                Ok(()) => return Ok(Self(path)),
56                Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => continue,
57                Err(error) => return Err(format!("cannot create test capture directory: {error}")),
58            }
59        }
60        Err("cannot allocate test capture directory".into())
61    }
62}
63impl Drop for Directory {
64    /// Never traverse remaining or replaced spool names; the helper owns those.
65    fn drop(&mut self) {
66        let _ = std::fs::remove_dir(&self.0);
67    }
68}

76/// Pass literal executable/arguments and capture only the bounded protocol pipe.
77/// A taken stdin guard remains alive through wait; wait must not signal parent death.
78#[cfg(test)]
79fn run_command(command: Command, timeout: Duration) -> Result<Captured, String> {
80    timeout_ms(timeout)?;
81    capture(command, Command::new(helper()?), timeout)
82}

84/// Capture the command's literal executable and arguments through a trusted supervisor.
85/// Configure working directory and environment on `supervisor`; child stdin is
86/// intentionally closed. Each stream is capped at 256 KiB and timeout at 60 seconds.
87pub fn capture(
88    command: Command,
89    mut supervisor: Command,
90    timeout: Duration,
91) -> Result<Captured, String> {
92    let milliseconds = timeout_ms(timeout)?;
93    let directory = Directory::new()?;
94    let executable = std::fs::canonicalize(command.get_program())
95        .map_err(|error| format!("cannot execute doc test: {error}"))?;
96    let mut child = supervisor
97        .arg(milliseconds.to_string())
98        .arg(&directory.0)
99        .arg("--")
100        .arg(executable)
101        .args(command.get_args())
102        .stdin(Stdio::piped())
103        .stdout(Stdio::piped())
104        .stderr(Stdio::null())
105        .spawn()
106        .map_err(|error| format!("cannot execute test supervisor: {error}"))?;
107    let liveness = child.stdin.take();
108    let mut output = Vec::with_capacity(FRAME_MAX + 1);
109    let read = child
110        .stdout
111        .take()
112        .ok_or("missing test supervisor stdout")?
113        .take(FRAME_MAX as u64 + 1)
114        .read_to_end(&mut output);
115    // On read failure/overflow, closing liveness asks the helper to finish cleanup.
116    if read.is_err() || output.len() > FRAME_MAX {
117        drop(liveness);
118    } else {
119        let status = child.wait().map_err(|error| error.to_string())?;
120        drop(liveness);
121        read.map_err(|error| format!("test supervisor protocol: {error}"))?;
122        if !status.success() {
123            return Err("test supervisor failed to publish a complete result".into());
124        }
125        return decode(&output);
126    }
127    let _ = child.wait();
128    Err("test supervisor protocol exceeds capture limit or cannot be read".into())
129}

136    /// Resolve Cargo-built Rust helpers without nested builds or global environment mutation.
137    fn artifact(variable: &str, name: &str) -> Result<PathBuf, String> {
138        if let Some(path) = std::env::var_os(variable) {
139            return Ok(PathBuf::from(path));
140        }
141        let current = std::env::current_exe().map_err(|error| error.to_string())?;
142        for parent in current.ancestors().skip(1) {
143            if ["debug", "release"]
144                .iter()
145                .any(|profile| parent.file_name().is_some_and(|name| name == *profile))
146            {
147                let candidate = parent.join(name);
148                if candidate.is_file() {
149                    return Ok(candidate);
150                }
151            }
152        }
153        Err(format!(
154            "missing Rust {name}; run cargo build -p morrow-test-supervisor --bins or set {variable}"
155        ))
156    }

168    /// Execute a fixed test body through the existing interpreter, avoiding write-to-exec races.
169    fn script(body: &str, timeout: Duration) -> Result<Captured, String> {
170        let mut command = Command::new("/bin/sh");
171        command.args(["-c", body]);
172        run_command(command, timeout)
173    }
