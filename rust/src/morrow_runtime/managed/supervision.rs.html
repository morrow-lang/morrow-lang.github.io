1//! Typed actor initializers and bounded restart ownership.
2use super::*;

4const MAX_RESTARTS: i64 = 32;
5// Historical language-visible quota; synchronization storage is physical only.
6const SUPERVISOR_BYTES: usize = 48;
7type Current = std::sync::Mutex<std::sync::Weak<control::Allocation<Actor>>>;
8
9#[repr(C)]
10pub(super) struct Supervisor {
11    heap: usize,
12    initializer: *mut c_void,
13    initializer_cost: usize,
14    mailbox: *const Type,
15    // Immutable pointer to opaque synchronization storage: GC never scans the
16    // mutex/Weak internals. Weak ownership breaks Actor -> Supervisor -> Actor.
17    current: control::Owned<Current>,
18    remaining: usize,
19}
20impl Default for Supervisor {
21    fn default() -> Self {
22        Self {
23            heap: 0,
24            initializer: null_mut(),
25            initializer_cost: 0,
26            mailbox: null(),
27            current: control::Owned::new(Current::default()),
28            remaining: 0,
29        }
30    }
31}

33/// Supervisor is retained by the caller. Its cell pointer never changes after
34/// publication; all weak ownership operations are protected by this mutex.
35unsafe fn resolve_current(supervisor: *mut Supervisor) -> Option<control::Owned<Actor>> {
36    let cell = unsafe { &*(*supervisor).current.as_ptr() };
37    let current = cell.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
38    control::Owned::upgrade(&current)
39}
40
41/// Only the owning scheduler publishes actors, already fully initialized. Neither
42/// replacing nor dropping a weak reference touches actor payloads or sender TLS.
43unsafe fn publish_current(supervisor: *mut Supervisor, actor: *mut Actor) {
44    let next = if actor.is_null() {
45        std::sync::Weak::new()
46    } else {
47        unsafe { control::Owned::retain(actor) }.downgrade()
48    };
49    let cell = unsafe { &*(*supervisor).current.as_ptr() };
50    *cell.lock().unwrap_or_else(|poisoned| poisoned.into_inner()) = next;
51}

53/// Queue a typed child with a bounded restart policy.
54/// # Safety
55/// Context, initializer and mailbox satisfy `morrow_managed_spawn`'s contract.
56#[unsafe(no_mangle)]
57pub unsafe extern "C" fn morrow_managed_supervise(
58    exec: *mut Exec,
59    initializer: *mut c_void,
60    mailbox: *const Type,
61    max_restarts: i64,
62) -> *mut c_void {
63    unsafe {
64        if exec.is_null() || (*exec).session.is_null() || *(*exec).fault != 0 {
65            return null_mut();
66        }
67        let s = (*exec).session;
68        let f = function(s, initializer);
69        if f.is_null()
70            || (*f).step.is_none()
71            || !cost::descriptor(mailbox, false, &mut 0)
72            || (!(*f).mailbox.is_null() && (*f).mailbox != mailbox)
73        {
74            fail(exec, 11);
75            return null_mut();
76        }
77        let cost = cost::frame(s, initializer);
78        if !(0..=MAX_RESTARTS).contains(&max_restarts)
79            || (*s).stopped
80            || cost.is_none()
81            || !charge(s, cost.unwrap_or(0) + SUPERVISOR_BYTES)
82        {
83            fail(exec, 9);
84            return null_mut();
85        }
86        let owner = control::Owned::new(Supervisor::default());
87        let supervisor = owner.as_ptr();
88        (*supervisor).heap = memory::create_control_heap(
89            supervisor.cast(),
90            std::mem::size_of::<Supervisor>() / 8,
91            control::Owned::retain(supervisor).token(),
92        );
93        (*supervisor).initializer_cost = cost.unwrap();
94        (*supervisor).mailbox = mailbox;
95        (*supervisor).remaining = max_restarts as usize;
96        {
97            let _initializer = memory::enter_heap((*supervisor).heap);
98            let copied = copy::frame(s, initializer);
99            (*supervisor).initializer = copied.value as *mut c_void;
100        }
101        let pid =
102            lifecycle::spawn(exec, (*supervisor).initializer, mailbox, supervisor).cast::<Pid>();
103        if pid.is_null() {
104            retire(s, supervisor);
105            return null_mut();
106        }
107        publish_current(supervisor, (*pid).actor);
108        pid.cast()
109    }
110}
111
112unsafe fn retire(s: *mut Session, supervisor: *mut Supervisor) {
113    unsafe {
114        let _owner = control::Owned::retain(supervisor);
115        if (*supervisor).heap == 0 {
116            return;
117        }
118        publish_current(supervisor, null_mut());
119        (*supervisor).initializer = null_mut();
120        release(s, (*supervisor).initializer_cost + SUPERVISOR_BYTES);
121        (*supervisor).initializer_cost = 0;
122        memory::retire_heap((*supervisor).heap);
123        (*supervisor).heap = 0;
124    }
125}

127/// Normal completion and invocation cancellation retire the retained initializer.
128pub(super) unsafe fn completed(a: *mut Actor) {
129    unsafe {
130        let supervisor = (*a).identity.supervisor;
131        if !supervisor.is_null()
132            && resolve_current(supervisor).is_some_and(|current| current.as_ptr() == a)
133            && ((*a).fault == 0 || (*(*a).exec.session).stopped)
134        {
135            retire((*a).exec.session, supervisor);
136        }
137    }
138}

140/// Handle an ordinary typed fault after every generated stack frame has returned.
141pub(super) unsafe fn recover(a: *mut Actor) -> bool {
142    unsafe {
143        let _actor = control::Owned::retain(a);
144        let supervisor = (*a).identity.supervisor;
145        if supervisor.is_null() {
146            return false;
147        }
148        let s = (*a).exec.session;
149        scheduler::finish(a);
150        #[cfg(test)]
151        parallel::after_supervised_retirement();
152        if (*supervisor).remaining == 0 || (*s).stopped {
153            retire(s, supervisor);
154            return true;
155        }
156        (*supervisor).remaining -= 1;
157        // A restart admission failure belongs to this lineage, not the caller
158        // that originally started it or unrelated actors in the invocation.
159        let mut fault = 0;
160        let mut exec = Exec {
161            session: s,
162            actor: null_mut(),
163            fault: &mut fault,
164        };
165        let pid = lifecycle::spawn(
166            &mut exec,
167            (*supervisor).initializer,
168            (*supervisor).mailbox,
169            supervisor,
170        )
171        .cast::<Pid>();
172        if pid.is_null() {
173            retire(s, supervisor);
174        } else {
175            publish_current(supervisor, (*pid).actor);
176        }
177        true
178    }
179}

181/// Resolve a supervision lineage to its current, freshly allocated actor identity.
182/// Old PIDs remain invalid for send; lookup never redirects an existing PID.
183/// # Safety
184/// Exec and original PID are live native values on their invocation thread.
185#[unsafe(no_mangle)]
186pub unsafe extern "C" fn morrow_managed_supervised_current(
187    exec: *mut Exec,
188    original: *mut c_void,
189) -> i64 {
190    unsafe {
191        if exec.is_null() || (*exec).session.is_null() || original.is_null() {
192            return abi::result_err(3);
193        }
194        let s = (*exec).session;
195        let pid = original.cast::<Pid>();
196        if !valid_pid(s, pid) || (*s).stopped {
197            return abi::result_err(3);
198        }
199        let supervisor = (*(*pid).actor).identity.supervisor;
200        if supervisor.is_null() {
201            return abi::result_err(3);
202        }
203        let Some(owner) = resolve_current(supervisor) else {
204            return abi::result_err(3);
205        };
206        let a = owner.as_ptr();
207        if !(*a).identity.alive.load(Ordering::Acquire) {
208            return abi::result_err(3);
209        }
210        // This temporary wrapper belongs to the caller heap. Any retained
211        // continuation/message charges its own copied PID graph; lookup itself
212        // must not consume an irreversible lifetime quota on every request.
213        let current = new_pid(a);
214        abi::result_ok(current as i64)
215    }
216}

223    #[test]
224    fn synchronized_lineage_upgrades_without_retaining_an_actor_supervisor_cycle() {
225        unsafe {
226            let supervisor = control::Owned::new(Supervisor::default());
227            let supervisor_weak = supervisor.downgrade();
228            let pointer = supervisor.as_ptr();
229            let actor = control::Owned::new(Actor {
230                _supervisor: Some(control::Owned::retain(pointer)),
231                ..Actor::default()
232            });
233            let actor_weak = actor.downgrade();
234            publish_current(pointer, actor.as_ptr());
235            let current = resolve_current(pointer).expect("published actor");
236            assert_eq!(current.as_ptr(), actor.as_ptr());
237            drop(supervisor);
238            drop(actor);
239            assert!(actor_weak.upgrade().is_some());
240            drop(current);
241            assert!(actor_weak.upgrade().is_none());
242            assert!(supervisor_weak.upgrade().is_none());
243        }
244    }
