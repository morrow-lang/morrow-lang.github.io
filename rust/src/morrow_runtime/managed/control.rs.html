1//! Reference ownership for scheduler control records outside collected payloads.
2//!
3//! References may be released by a moved heap. Foreign readers access only
4//! immutable identity or explicitly synchronized fields; mutable scheduler state
5//! remains owner-only. Atomic ownership does not make mutation thread safe.
6//! Destructors only release owned storage and never enter a domain.
7use std::cell::UnsafeCell;
8use std::ffi::c_void;
9use std::marker::PhantomData;
10use std::mem::ManuallyDrop;
11use std::ptr::NonNull;
12use std::sync::Arc;
13
14#[repr(C)]
15pub(super) struct Allocation<T> {
16    // Keeping value first preserves the control address in ABI records and heap
17    // metadata while the accounting guard follows ordinary Arc reclamation.
18    value: UnsafeCell<T>,
19    _accounting: crate::memory::ControlAllocation,
20}
21
22#[repr(transparent)]
23pub(super) struct Owned<T> {
24    // Preserve the into_raw pointer and keep Rust internals out of ABI records.
25    pointer: NonNull<c_void>,
26    _type: PhantomData<Arc<Allocation<T>>>,
27}
28
29impl<T> Owned<T> {
30    pub(super) fn new(value: T) -> Self {
31        Self::new_retained(value, 0, 1)
32    }
33
34    #[allow(clippy::arc_with_non_send_sync)] // Erased heap tokens release atomic ownership only.
35    pub(super) fn new_retained(value: T, retained_bytes: usize, objects: usize) -> Self {
36        let bytes = std::mem::size_of::<T>()
37            .checked_add(retained_bytes)
38            .expect("bounded control allocation size");
39        let pointer = Arc::into_raw(Arc::new(Allocation {
40            value: UnsafeCell::new(value),
41            _accounting: crate::memory::account_control(bytes, objects),
42        }));
43        Self {
44            pointer: NonNull::new(pointer.cast_mut().cast()).unwrap(),
45            _type: PhantomData,
46        }
47    }
48
49    pub(super) fn as_ptr(&self) -> *mut T {
50        self.pointer.as_ptr().cast()
51    }
52
53    /// The pointer names a live Owned allocation; no shared Rust reference to its
54    /// mutable contents exists. The repr(C) allocation starts with a transparent
55    /// UnsafeCell, so the record and Arc allocation have the same address.
56    pub(super) unsafe fn retain(pointer: *mut T) -> Self {
57        unsafe {
58            let pointer = pointer.cast::<Allocation<T>>();
59            Arc::increment_strong_count(pointer);
60            Self {
61                pointer: NonNull::new(pointer.cast()).unwrap(),
62                _type: PhantomData,
63            }
64        }
65    }
66
67    /// Transfer ownership to inert heap metadata. T must only own inert storage:
68    /// dropping it on a heap's owner thread must not invoke callbacks or a domain.
69    pub(super) unsafe fn token(self) -> crate::memory::Control {
70        unsafe fn release<T>(pointer: *const ()) {
71            // SAFETY: token consumes exactly one Arc reference from into_raw.
72            unsafe { drop(Arc::from_raw(pointer.cast::<Allocation<T>>())) };
73        }
74        let owner = ManuallyDrop::new(self);
75        unsafe { crate::memory::Control::new(owner.pointer.as_ptr().cast(), release::<T>) }
76    }
77
78    pub(super) fn downgrade(&self) -> std::sync::Weak<Allocation<T>> {
79        // SAFETY: borrow the into_raw-produced reference without consuming it.
80        let owner = ManuallyDrop::new(unsafe {
81            Arc::from_raw(self.pointer.as_ptr().cast::<Allocation<T>>())
82        });
83        Arc::downgrade(&owner)
84    }
85
86    /// Retain a published control without racing the final strong release.
87    /// This grants allocation lifetime only, never access to owner-only fields.
88    /// Dropping the resulting reference uses the same inert, TLS-independent
89    /// destruction contract as retained control tokens.
90    pub(super) fn upgrade(weak: &std::sync::Weak<Allocation<T>>) -> Option<Self> {
91        weak.upgrade().map(|owner| Self {
92            pointer: NonNull::new(Arc::into_raw(owner).cast_mut().cast()).unwrap(),
93            _type: PhantomData,
94        })
95    }
96}
97
98impl<T> Drop for Owned<T> {
99    fn drop(&mut self) {
100        // SAFETY: every Owned has one unreleased reference from into_raw or
101        // increment_strong_count; no Rust reference to its contents is exposed.
102        unsafe { drop(Arc::from_raw(self.pointer.as_ptr().cast::<Allocation<T>>())) };
103    }
104}
105
106/// Retain a record for the lifetime of a collected ABI wrapper.
107/// Both pointers must be live, and the wrapper must belong to the active heap.
108pub(super) unsafe fn attach<T>(wrapper: *const u8, record: *mut T) {
109    unsafe { crate::memory::retain_control(wrapper, Owned::retain(record).token()) };
110}

112#[cfg(test)]
113pub(super) unsafe fn observe<T>(pointer: *mut T) -> std::sync::Weak<Allocation<T>> {
114    unsafe { Owned::retain(pointer).downgrade() }
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120
121    #[test]
122    fn weak_upgrade_retains_live_control_and_cannot_revive_finalized_storage() {
123        struct Counted(std::sync::Arc<std::sync::atomic::AtomicUsize>);
124        impl Drop for Counted {
125            fn drop(&mut self) {
126                self.0.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
127            }
128        }
129        let drops = std::sync::Arc::new(std::sync::atomic::AtomicUsize::new(0));
130        let owner = Owned::new(Counted(drops.clone()));
131        let pointer = owner.as_ptr();
132        let weak = owner.downgrade();
133        let retained = Owned::upgrade(&weak).expect("live weak reference");
134        assert_eq!(retained.as_ptr(), pointer);
135        drop(owner);
136        assert_eq!(drops.load(std::sync::atomic::Ordering::SeqCst), 0);
137        drop(retained);
138        assert_eq!(drops.load(std::sync::atomic::Ordering::SeqCst), 1);
139        assert!(Owned::upgrade(&weak).is_none());
140        drop(weak);
141        assert_eq!(drops.load(std::sync::atomic::Ordering::SeqCst), 1);
142    }
143}
