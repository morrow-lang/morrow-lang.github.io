1//! Rolling restart-intensity window, matching supervisor.erl `add_restart`.
2use std::collections::VecDeque;
3
4/// Whole-second restart timestamps kept while `now <= then + period`.
5///
6/// Storage is bounded by `intensity + 1` entries: once the count exceeds the
7/// intensity the supervisor retires, so older entries never matter.
8#[derive(Clone, Debug)]
9pub struct Window {
10    intensity: usize,
11    period: u64,
12    stamps: VecDeque<u64>,
13}
14
15impl Window {
16    pub fn new(intensity: u32, period_seconds: u32) -> Self {
17        let intensity = intensity as usize;
18        Self {
19            intensity,
20            period: u64::from(period_seconds),
21            stamps: VecDeque::with_capacity(intensity.saturating_add(1)),
22        }
23    }
24
25    /// Record one restart attempt at `now` and report whether the intensity is
26    /// exceeded. The boundary is inclusive: a restart exactly `period` seconds
27    /// after another still counts against it.
28    pub fn charge(&mut self, now: u64) -> bool {
29        self.stamps.push_front(now);
30        let period = self.period;
31        self.stamps
32            .retain(|then| now <= then.saturating_add(period));
33        self.stamps.truncate(self.intensity.saturating_add(1));
34        self.stamps.len() > self.intensity
35    }
36
37    /// Restart attempts currently inside the window.
38    pub fn count(&self) -> usize {
39        self.stamps.len()
40    }
41}
