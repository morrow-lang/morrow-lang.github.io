1//! Owner-local supervisor state machine.
2//!
3//! The engine receives inputs from a driver (child acknowledgements and exits,
4//! management requests, retry ticks, injected whole-second clock readings and
5//! parent exits) and returns an ordered, bounded list of actions the driver
6//! performs. It never schedules, allocates program heaps, reads a clock or
7//! touches process identities beyond the opaque ids the driver reports.
8//!
9//! Restart and termination operations are serialized: while one operation
10//! waits for a child acknowledgement or exit, later inputs are queued in
11//! arrival order and handled after the operation completes. Exits of running
12//! children observed meanwhile are recorded immediately so a later termination
13//! step skips the dead child, and their policy handling is deferred.
14use super::spec::{
15    AutoShutdown, ChildInfo, ChildKind, ChildSpec, ChildState, Error, ExitReason, Flags,
16    MAX_CHILDREN, ProcessId, Restart, Shutdown, Strategy, Template, validate, validate_child,
17};
18use super::window::Window;
19use std::collections::VecDeque;

21/// Most management requests that may wait behind an in-flight operation.
22/// Further requests receive `ResourceLimit` immediately.
23pub const MAX_QUEUED_REQUESTS: usize = 64;
24
25/// Driver-chosen correlation token echoed in `Action::Reply`.
26#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
27pub struct RequestId(pub u64);
28
29/// Management commands. Each yields exactly one `Action::Reply`.
30#[derive(Clone, Debug, PartialEq, Eq)]
31pub enum Request<T> {
32    /// Append a dynamic child and start it.
33    StartChild(ChildSpec<T>),
34    /// Terminate a child; Permanent/Transient specs are retained as Stopped,
35    /// Temporary specs are removed.
36    TerminateChild(String),
37    /// Start a stopped child with a fresh generation; never charges intensity.
38    RestartChild(String),
39    /// Remove a stopped child specification.
40    DeleteChild(String),
41    /// Report every child in declaration order.
42    WhichChildren,
43    /// Resolve a name to its running process id or an explicit state error.
44    Current(String),
45    /// Terminate all children in reverse order and retire with `Normal`.
46    Stop,
47}
48
49/// Successful request payloads.
50#[derive(Clone, Debug, PartialEq, Eq)]
51pub enum Response {
52    Unit,
53    Children(Vec<ChildInfo>),
54    Current(ProcessId),
55}
56
57/// Request outcome carried by `Action::Reply`.
58pub type Reply = Result<Response, Error>;
59
60/// Everything a driver can tell the engine.
61#[derive(Clone, Debug, PartialEq, Eq)]
62pub enum Input<T> {
63    /// The child generation acknowledged its startup with this identity.
64    Started {
65        name: String,
66        generation: u64,
67        id: ProcessId,
68    },
69    /// The child generation declined to start.
70    Ignored { name: String, generation: u64 },
71    /// The child generation failed to start.
72    StartFailed {
73        name: String,
74        generation: u64,
75        reason: ExitReason,
76    },
77    /// A running child generation exited. Acknowledgement must precede this.
78    Exited {
79        name: String,
80        generation: u64,
81        reason: ExitReason,
82    },
83    /// A graceful deadline armed for this generation elapsed.
84    DeadlineElapsed { name: String, generation: u64 },
85    /// The driver delivers a retry requested by `Action::ScheduleRetry`.
86    Retry { name: String },
87    /// Monotonic whole-second reading used for restart-intensity charging.
88    Clock { seconds: u64 },
89    /// The parent exited; a Normal parent exit also shuts the supervisor down.
90    ParentExit { reason: ExitReason },
91    /// A management request.
92    Request { id: RequestId, request: Request<T> },
93}

95/// Everything the driver must perform, in order.
96#[derive(Clone, Debug, PartialEq, Eq)]
97pub enum Action<T> {
98    /// Start a fresh child generation from its template and acknowledge it.
99    Start {
100        name: String,
101        kind: ChildKind,
102        generation: u64,
103        template: T,
104    },
105    /// Send a shutdown signal to the running generation.
106    SendShutdown {
107        name: String,
108        generation: u64,
109        id: ProcessId,
110    },
111    /// Report `Input::DeadlineElapsed` after this many milliseconds unless the
112    /// generation exits first.
113    ArmDeadline {
114        name: String,
115        generation: u64,
116        milliseconds: u32,
117    },
118    /// Kill the running generation without a shutdown signal.
119    Kill {
120        name: String,
121        generation: u64,
122        id: ProcessId,
123    },
124    /// Deliver `Input::Retry` for this child after yielding to other inputs.
125    ScheduleRetry { name: String },
126    /// Answer a management request.
127    Reply { request: RequestId, reply: Reply },
128    /// The startup transaction finished. `Err` means every started child was
129    /// terminated again; a `Retire` follows.
130    StartupComplete(Result<(), Error>),
131    /// The supervisor is finished; no further actions follow.
132    Retire { reason: ExitReason },
133}

135/// Internal child occupancy. `Dead` is a natural exit whose policy handling
136/// waits for the in-flight operation to complete.
137#[derive(Clone, Debug)]
138enum Slot {
139    Running {
140        generation: u64,
141        id: ProcessId,
142    },
143    Dead {
144        generation: u64,
145        id: ProcessId,
146        reason: ExitReason,
147    },
148    Stopped,
149    Restarting,
150}
151
152impl Slot {
153    const fn is_alive(&self) -> bool {
154        matches!(self, Self::Running { .. } | Self::Dead { .. })
155    }
156
157    const fn state(&self) -> ChildState {
158        match self {
159            Self::Running { id, .. } | Self::Dead { id, .. } => ChildState::Running(*id),
160            Self::Stopped => ChildState::Stopped,
161            Self::Restarting => ChildState::Restarting,
162        }
163    }
164}

235/// Deterministic supervisor engine for one supervisor instance.
236pub struct Engine<T> {
237    flags: Flags,
238    children: Vec<Child<T>>,
239    next_key: u64,
240    next_generation: u64,
241    clock: u64,
242    window: Window,
243    operation: Option<Operation>,
244    queue: VecDeque<Input<T>>,
245    queued_requests: usize,
246    phase: Phase,
247}
248
249impl<T: Template> Engine<T> {
250    /// Validate the tree and begin the startup transaction.
251    ///
252    /// Returns the engine together with its first actions: the first child's
253    /// `Start`, or `StartupComplete(Ok(()))` for an empty child list. Nothing
254    /// is emitted when validation fails.
255    pub fn start(
256        flags: Flags,
257        children: Vec<ChildSpec<T>>,
258    ) -> Result<(Self, Vec<Action<T>>), Error> {
259        validate(&flags, &children)?;
260        let window = Window::new(flags.intensity, flags.period_seconds);
261        let children: Vec<Child<T>> = children
262            .into_iter()
263            .enumerate()
264            .map(|(index, spec)| Child {
265                key: index as u64,
266                spec,
267                slot: Slot::Stopped,
268            })
269            .collect();
270        let steps = children
271            .iter()
272            .map(|child| Step::Start(child.key))
273            .collect();
274        let mut engine = Self {
275            flags,
276            next_key: children.len() as u64,
277            children,
278            next_generation: 1,
279            clock: 0,
280            window,
281            operation: Some(Operation {
282                kind: Kind::Startup,
283                steps,
284                wait: None,
285            }),
286            queue: VecDeque::new(),
287            queued_requests: 0,
288            phase: Phase::Starting,
289        };
290        let mut out = Vec::new();
291        engine.run(&mut out);
292        Ok((engine, out))
293    }

295    /// Feed one input and return the actions it produces.
296    ///
297    /// Work per call is bounded by the queued inputs and twice the child count;
298    /// retries always go back through the driver.
299    pub fn handle(&mut self, input: Input<T>) -> Vec<Action<T>> {
300        let mut out = Vec::new();
301        match input {
302            Input::Clock { seconds } => self.clock = self.clock.max(seconds),
303            Input::DeadlineElapsed { name, generation } => {
304                self.deadline(&name, generation, &mut out);
305            }
306            Input::Started {
307                name,
308                generation,
309                id,
310            } => self.acknowledge(&name, generation, Ack::Started(id), &mut out),
311            Input::Ignored { name, generation } => {
312                self.acknowledge(&name, generation, Ack::Ignored, &mut out);
313            }
314            Input::StartFailed {
315                name,
316                generation,
317                reason,
318            } => self.acknowledge(&name, generation, Ack::Failed(reason), &mut out),
319            Input::Exited {
320                name,
321                generation,
322                reason,
323            } => self.exited(name, generation, reason),
324            Input::Request { id, request } => {
325                if self.phase == Phase::Retired {
326                    out.push(reply(id, Err(Error::SupervisorStopped)));
327                } else if self.queued_requests >= MAX_QUEUED_REQUESTS {
328                    out.push(reply(id, Err(Error::ResourceLimit)));
329                } else {
330                    self.queued_requests = self.queued_requests.saturating_add(1);
331                    self.queue.push_back(Input::Request { id, request });
332                }
333            }
334            Input::Retry { .. } | Input::ParentExit { .. } => {
335                if self.phase != Phase::Retired {
336                    self.queue.push_back(input);
337                }
338            }
339        }
340        self.run(&mut out);
341        out
342    }

385    /// Advance until an operation waits on the driver or nothing is queued.
386    fn run(&mut self, out: &mut Vec<Action<T>>) {
387        // Every iteration consumes a step, finishes an operation or consumes a
388        // queued input; the budget makes that bound explicit.
389        let per_operation = self.children.len().saturating_mul(2).saturating_add(2);
390        let mut budget = self
391            .queue
392            .len()
393            .saturating_add(1)
394            .saturating_mul(per_operation)
395            .saturating_add(per_operation);
396        while budget > 0 {
397            budget = budget.saturating_sub(1);
398            if self.phase == Phase::Retired {
399                self.drain_retired(out);
400                return;
401            }
402            if let Some(operation) = &mut self.operation {
403                if operation.wait.is_some() {
404                    return;
405                }
406                match operation.steps.pop_front() {
407                    Some(Step::Terminate(key)) => self.begin_terminate(key, out),
408                    Some(Step::Start(key)) => self.begin_start(key, out),
409                    None => self.finish(out),
410                }
411                continue;
412            }
413            match self.queue.pop_front() {
414                Some(input) => self.process(input, out),
415                None => return,
416            }
417        }
418    }
419
420    /// Answer queued requests after retirement and drop other queued inputs.
421    fn drain_retired(&mut self, out: &mut Vec<Action<T>>) {
422        while let Some(input) = self.queue.pop_front() {
423            if let Input::Request { id, .. } = input {
424                out.push(reply(id, Err(Error::SupervisorStopped)));
425            }
426        }
427        self.queued_requests = 0;
428    }

430    fn process(&mut self, input: Input<T>, out: &mut Vec<Action<T>>) {
431        match input {
432            Input::Request { id, request } => {
433                self.queued_requests = self.queued_requests.saturating_sub(1);
434                self.request(id, request, out);
435            }
436            Input::Exited {
437                name, generation, ..
438            } => {
439                let Some(index) = self.index_of_name(&name) else {
440                    return;
441                };
442                let matches = self.children.get(index).is_some_and(|child| {
443                    matches!(child.slot, Slot::Dead { generation: dead, .. } if dead == generation)
444                });
445                if matches {
446                    self.child_exited(index);
447                }
448            }
449            Input::Retry { name } => {
450                let Some(index) = self.index_of_name(&name) else {
451                    return;
452                };
453                if let Some(child) = self.children.get(index)
454                    && matches!(child.slot, Slot::Restarting)
455                {
456                    let key = child.key;
457                    self.restart(key);
458                }
459            }
460            Input::ParentExit { reason } => self.begin_shutdown(reason, None),
461            Input::Started { .. }
462            | Input::Ignored { .. }
463            | Input::StartFailed { .. }
464            | Input::DeadlineElapsed { .. }
465            | Input::Clock { .. } => {}
466        }
467    }

469    fn request(&mut self, id: RequestId, request: Request<T>, out: &mut Vec<Action<T>>) {
470        match request {
471            Request::StartChild(spec) => self.start_child(id, spec, out),
472            Request::TerminateChild(name) => match self.index_of_name(&name) {
473                None => out.push(reply(id, Err(Error::Removed))),
474                Some(index) => {
475                    let key = self.children.get(index).map_or(0, |child| child.key);
476                    self.operation = Some(Operation {
477                        kind: Kind::ManualTerminate { request: id },
478                        steps: VecDeque::from([Step::Terminate(key)]),
479                        wait: None,
480                    });
481                }
482            },
483            Request::RestartChild(name) => match self.lookup(&name) {
484                Err(error) => out.push(reply(id, Err(error))),
485                Ok(key) => {
486                    self.operation = Some(Operation {
487                        kind: Kind::ManualRestart { request: id },
488                        steps: VecDeque::from([Step::Start(key)]),
489                        wait: None,
490                    });
491                }
492            },
493            Request::DeleteChild(name) => {
494                let result = self.lookup(&name).map(|key| {
495                    self.children.retain(|child| child.key != key);
496                    Response::Unit
497                });
498                out.push(reply(id, result));
499            }
500            Request::WhichChildren => {
501                out.push(reply(id, Ok(Response::Children(self.children()))));
502            }
503            Request::Current(name) => {
504                let result = match self.index_of_name(&name).and_then(|i| self.children.get(i)) {
505                    None => Err(Error::Removed),
506                    Some(child) => match child.slot {
507                        Slot::Running { id, .. } | Slot::Dead { id, .. } => {
508                            Ok(Response::Current(id))
509                        }
510                        Slot::Stopped => Err(Error::Stopped),
511                        Slot::Restarting => Err(Error::Restarting),
512                    },
513                };
514                out.push(reply(id, result));
515            }
516            Request::Stop => self.begin_shutdown(ExitReason::Normal, Some(id)),
517        }
518    }

520    /// Resolve a name to the key of a stopped child, or the error describing
521    /// why it cannot be restarted or deleted.
522    fn lookup(&self, name: &str) -> Result<u64, Error> {
523        let child = self
524            .index_of_name(name)
525            .and_then(|index| self.children.get(index))
526            .ok_or(Error::Removed)?;
527        match child.slot {
528            Slot::Running { .. } | Slot::Dead { .. } => Err(Error::AlreadyRunning),
529            Slot::Restarting => Err(Error::Restarting),
530            Slot::Stopped => Ok(child.key),
531        }
532    }

534    fn start_child(&mut self, id: RequestId, spec: ChildSpec<T>, out: &mut Vec<Action<T>>) {
535        if let Err(error) = validate_child(&self.flags, &spec, 1) {
536            out.push(reply(id, Err(error)));
537            return;
538        }
539        if let Some(existing) = self
540            .index_of_name(&spec.name)
541            .and_then(|index| self.children.get(index))
542        {
543            let error = if existing.slot.is_alive() {
544                Error::AlreadyRunning
545            } else {
546                Error::AlreadyPresent
547            };
548            out.push(reply(id, Err(error)));
549            return;
550        }
551        if self.children.len() >= MAX_CHILDREN {
552            out.push(reply(id, Err(Error::ResourceLimit)));
553            return;
554        }
555        let key = self.next_key;
556        self.next_key = self.next_key.saturating_add(1);
557        self.children.push(Child {
558            key,
559            spec,
560            slot: Slot::Stopped,
561        });
562        self.operation = Some(Operation {
563            kind: Kind::ManualStart { request: id, key },
564            steps: VecDeque::from([Step::Start(key)]),
565            wait: None,
566        });
567    }

569    /// Record a running child's exit. An awaited exit completes the current
570    /// termination step; any other running generation is marked dead and its
571    /// policy handling is queued behind the in-flight operation.
572    fn exited(&mut self, name: String, generation: u64, reason: ExitReason) {
573        let Some(index) = self.index_of_name(&name) else {
574            return;
575        };
576        let Some(child) = self.children.get_mut(index) else {
577            return;
578        };
579        let Slot::Running {
580            generation: running,
581            id,
582        } = child.slot
583        else {
584            return;
585        };
586        if running != generation {
587            return;
588        }
589        let key = child.key;
590        let awaited = matches!(
591            self.operation.as_ref().and_then(|operation| operation.wait),
592            Some(Wait::Exit { key: waited, generation: expected, .. })
593                if waited == key && expected == generation
594        );
595        if awaited {
596            if let Some(operation) = &mut self.operation {
597                operation.wait = None;
598            }
599            self.stop_child(index);
600        } else {
601            child.slot = Slot::Dead {
602                generation,
603                id,
604                reason: reason.clone(),
605            };
606            self.queue.push_back(Input::Exited {
607                name,
608                generation,
609                reason,
610            });
611        }
612    }

614    /// Escalate a graceful termination that is still waiting to a kill.
615    fn deadline(&mut self, name: &str, generation: u64, out: &mut Vec<Action<T>>) {
616        let Some(Wait::Exit {
617            key,
618            generation: expected,
619            escalated: false,
620        }) = self.operation.as_ref().and_then(|operation| operation.wait)
621        else {
622            return;
623        };
624        if expected != generation {
625            return;
626        }
627        let Some(child) = self.index_of_key(key).and_then(|i| self.children.get(i)) else {
628            return;
629        };
630        if child.spec.name != name {
631            return;
632        }
633        let Slot::Running { id, .. } = child.slot else {
634            return;
635        };
636        out.push(Action::Kill {
637            name: name.to_string(),
638            generation,
639            id,
640        });
641        if let Some(operation) = &mut self.operation {
642            operation.wait = Some(Wait::Exit {
643                key,
644                generation,
645                escalated: true,
646            });
647        }
648    }

650    /// Complete the awaited start step with the driver's outcome.
651    fn acknowledge(&mut self, name: &str, generation: u64, ack: Ack, out: &mut Vec<Action<T>>) {
652        let Some(Wait::Start {
653            key,
654            generation: expected,
655        }) = self.operation.as_ref().and_then(|operation| operation.wait)
656        else {
657            return;
658        };
659        if expected != generation {
660            return;
661        }
662        let Some(index) = self.index_of_key(key) else {
663            return;
664        };
665        if self
666            .children
667            .get(index)
668            .is_none_or(|child| child.spec.name != name)
669        {
670            return;
671        }
672        let Some(mut operation) = self.operation.take() else {
673            return;
674        };
675        operation.wait = None;
676        match ack {
677            Ack::Started(id) => {
678                if let Some(child) = self.children.get_mut(index) {
679                    child.slot = Slot::Running { generation, id };
680                }
681                self.operation = Some(operation);
682            }
683            Ack::Ignored => {
684                self.stop_child(index);
685                self.operation = Some(operation);
686            }
687            Ack::Failed(reason) => self.start_failed(operation, index, reason, out),
688        }
689    }

691    fn start_failed(
692        &mut self,
693        mut operation: Operation,
694        index: usize,
695        reason: ExitReason,
696        out: &mut Vec<Action<T>>,
697    ) {
698        let name = self
699            .children
700            .get(index)
701            .map(|child| child.spec.name.clone())
702            .unwrap_or_default();
703        match operation.kind {
704            Kind::Startup => {
705                operation.kind = Kind::Rollback { name, reason };
706                operation.steps = self.terminate_steps(|_| true);
707                self.operation = Some(operation);
708            }
709            Kind::Restart => {
710                if let Some(child) = self.children.get_mut(index) {
711                    child.slot = Slot::Restarting;
712                }
713                operation.steps.clear();
714                out.push(Action::ScheduleRetry { name });
715                self.operation = Some(operation);
716            }
717            Kind::ManualStart { request, key } => {
718                self.children.retain(|child| child.key != key);
719                out.push(reply(request, Err(Error::StartFailed(name, reason))));
720            }
721            Kind::ManualRestart { request } => {
722                out.push(reply(request, Err(Error::StartFailed(name, reason))));
723            }
724            Kind::Rollback { .. } | Kind::ManualTerminate { .. } | Kind::Shutdown { .. } => {
725                // These operations never start children; keep them running.
726                self.operation = Some(operation);
727            }
728        }
729    }

731    /// Terminate steps in reverse declaration order for children matching the
732    /// predicate on their declaration index.
733    fn terminate_steps(&self, include: impl Fn(usize) -> bool) -> VecDeque<Step> {
734        self.children
735            .iter()
736            .enumerate()
737            .rev()
738            .filter(|(index, _)| include(*index))
739            .map(|(_, child)| Step::Terminate(child.key))
740            .collect()
741    }

743    fn begin_terminate(&mut self, key: u64, out: &mut Vec<Action<T>>) {
744        let Some(index) = self.index_of_key(key) else {
745            return;
746        };
747        let Some(child) = self.children.get_mut(index) else {
748            return;
749        };
750        match child.slot {
751            Slot::Running { generation, id } => {
752                let name = child.spec.name.clone();
753                let escalated = match child.spec.policy.shutdown {
754                    Shutdown::Immediate => {
755                        out.push(Action::Kill {
756                            name,
757                            generation,
758                            id,
759                        });
760                        true
761                    }
762                    Shutdown::Graceful(milliseconds) => {
763                        out.push(Action::SendShutdown {
764                            name: name.clone(),
765                            generation,
766                            id,
767                        });
768                        out.push(Action::ArmDeadline {
769                            name,
770                            generation,
771                            milliseconds,
772                        });
773                        false
774                    }
775                    Shutdown::Infinity => {
776                        out.push(Action::SendShutdown {
777                            name,
778                            generation,
779                            id,
780                        });
781                        true
782                    }
783                };
784                if let Some(operation) = &mut self.operation {
785                    operation.wait = Some(Wait::Exit {
786                        key,
787                        generation,
788                        escalated,
789                    });
790                }
791            }
792            Slot::Dead { .. } => self.stop_child(index),
793            Slot::Stopped => {}
794            Slot::Restarting => child.slot = Slot::Stopped,
795        }
796    }
797
798    fn begin_start(&mut self, key: u64, out: &mut Vec<Action<T>>) {
799        let Some(child) = self.index_of_key(key).and_then(|i| self.children.get(i)) else {
800            return;
801        };
802        let generation = self.next_generation;
803        self.next_generation = self.next_generation.saturating_add(1);
804        out.push(Action::Start {
805            name: child.spec.name.clone(),
806            kind: child.spec.kind,
807            generation,
808            template: child.spec.template.clone(),
809        });
810        if let Some(operation) = &mut self.operation {
811            operation.wait = Some(Wait::Start { key, generation });
812        }
813    }
814
815    /// A child stopped: temporary specs are removed, others become Stopped.
816    fn stop_child(&mut self, index: usize) {
817        let Some(child) = self.children.get_mut(index) else {
818            return;
819        };
820        if child.spec.policy.restart == Restart::Temporary {
821            self.children.remove(index);
822        } else {
823            child.slot = Slot::Stopped;
824        }
825    }
826
827    /// Apply restart policy and significant-child auto shutdown to a natural
828    /// exit recorded in a `Dead` slot.
829    fn child_exited(&mut self, index: usize) {
830        let Some(child) = self.children.get_mut(index) else {
831            return;
832        };
833        let Slot::Dead { reason, .. } = &child.slot else {
834            return;
835        };
836        let restart = match child.spec.policy.restart {
837            Restart::Permanent => true,
838            Restart::Transient => reason.is_abnormal(),
839            Restart::Temporary => false,
840        };
841        if restart {
842            child.slot = Slot::Restarting;
843            let key = child.key;
844            self.restart(key);
845            return;
846        }
847        let significant = child.spec.policy.significant;
848        self.stop_child(index);
849        if !significant {
850            return;
851        }
852        let shutdown = match self.flags.auto_shutdown {
853            AutoShutdown::Never => false,
854            AutoShutdown::AnySignificant => true,
855            AutoShutdown::AllSignificant => !self
856                .children
857                .iter()
858                .any(|child| child.spec.policy.significant && child.slot.is_alive()),
859        };
860        if shutdown {
861            self.begin_shutdown(ExitReason::Shutdown, None);
862        }
863    }
864
865    /// Charge one restart attempt, then start the strategy's operation for the
866    /// child identified by `key`, or retire when intensity is exceeded.
867    fn restart(&mut self, key: u64) {
868        if self.window.charge(self.clock) {
869            self.begin_shutdown(ExitReason::Shutdown, None);
870            return;
871        }
872        let Some(index) = self.index_of_key(key) else {
873            return;
874        };
875        let mut steps = match self.flags.strategy {
876            Strategy::OneForOne => VecDeque::new(),
877            Strategy::OneForAll => self.terminate_steps(|_| true),
878            Strategy::RestForOne => self.terminate_steps(|later| later > index),
879        };
880        let first = match self.flags.strategy {
881            Strategy::OneForOne => index,
882            Strategy::OneForAll => 0,
883            Strategy::RestForOne => index,
884        };
885        let last = match self.flags.strategy {
886            Strategy::OneForOne => index.saturating_add(1),
887            Strategy::OneForAll | Strategy::RestForOne => self.children.len(),
888        };
889        steps.extend(
890            self.children
891                .iter()
892                .skip(first)
893                .take(last.saturating_sub(first))
894                .map(|child| Step::Start(child.key)),
895        );
896        self.operation = Some(Operation {
897            kind: Kind::Restart,
898            steps,
899            wait: None,
900        });
901    }
902
903    fn begin_shutdown(&mut self, reason: ExitReason, request: Option<RequestId>) {
904        self.operation = Some(Operation {
905            kind: Kind::Shutdown { reason, request },
906            steps: self.terminate_steps(|_| true),
907            wait: None,
908        });
909    }
910
911    fn finish(&mut self, out: &mut Vec<Action<T>>) {
912        let Some(operation) = self.operation.take() else {
913            return;
914        };
915        match operation.kind {
916            Kind::Startup => {
917                self.phase = Phase::Running;
918                out.push(Action::StartupComplete(Ok(())));
919            }
920            Kind::Rollback { name, reason } => {
921                out.push(Action::StartupComplete(Err(Error::StartFailed(
922                    name, reason,
923                ))));
924                self.retire(ExitReason::Shutdown, out);
925            }
926            Kind::Restart => {}
927            Kind::ManualStart { request, .. }
928            | Kind::ManualRestart { request }
929            | Kind::ManualTerminate { request } => {
930                out.push(reply(request, Ok(Response::Unit)));
931            }
932            Kind::Shutdown { reason, request } => {
933                if let Some(request) = request {
934                    out.push(reply(request, Ok(Response::Unit)));
935                }
936                self.retire(reason, out);
937            }
938        }
939    }
940
941    fn retire(&mut self, reason: ExitReason, out: &mut Vec<Action<T>>) {
942        self.phase = Phase::Retired;
943        out.push(Action::Retire { reason });
944        self.drain_retired(out);
945    }
946}

948fn reply<T>(request: RequestId, reply: Reply) -> Action<T> {
949    Action::Reply { request, reply }
