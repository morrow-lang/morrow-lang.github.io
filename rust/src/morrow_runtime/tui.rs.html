16thread_local! { static CHECKED_FAULT: std::cell::Cell<*mut i64> = const { std::cell::Cell::new(std::ptr::null_mut()) }; }
17
18/// Enter a checked terminal call without unwinding through generated frames.
19/// # Safety
20/// Fault is writable until the matching leave; both calls occur on this thread.
21#[unsafe(no_mangle)]
22pub unsafe extern "C" fn morrow_tui_fault_enter(fault: *mut i64) -> *mut i64 {
23    CHECKED_FAULT.with(|slot| slot.replace(fault))
24}
25
26/// Restore the previous checked terminal scope after the native call returns.
27#[unsafe(no_mangle)]
28pub extern "C" fn morrow_tui_fault_leave(previous: *mut i64) {
29    CHECKED_FAULT.with(|slot| slot.set(previous));
30}

32fn limit_fault(message: &str) {
33    let fault = CHECKED_FAULT.with(|slot| slot.get());
34    if fault.is_null() {
35        abi::fault(message);
36    }
37    // SAFETY: the compiler's matching enter/leave pair owns this writable cell.
38    unsafe {
39        if *fault == 0 {
40            *fault = 14;
41        }
42    }
43}

45unsafe fn text<'a>(value: *const c_char) -> &'a str {
46    if value.is_null() {
47        ""
48    } else {
49        unsafe { abi::text(value) }
50    }
51}

53fn rendered(value: &str) -> *const c_char {
54    if value.len() > crate::io::TEXT_LIMIT {
55        limit_fault("terminal rendering exceeds 16 MiB");
56        return std::ptr::null();
57    }
58    abi::string(value)
59}
60
61fn repeat(value: &str, count: usize) -> String {
62    if count > crate::io::TEXT_LIMIT / value.len().max(1) {
63        limit_fault("terminal rendering exceeds 16 MiB");
64        return String::new();
65    }
66    value.repeat(count)
67}

69fn display_width(value: &str) -> usize {
70    let mut escaped = false;
71    value
72        .chars()
73        .filter(|ch| {
74            if escaped {
75                if *ch == 'm' {
76                    escaped = false;
77                }
78                false
79            } else if *ch == '\x1b' {
80                escaped = true;
81                false
82            } else {
83                *ch != '\n' && *ch != '\r'
84            }
85        })
86        .count()
87}

94    #[test]
95    fn tree_snapshots_and_log_escaping_are_exact() {
96        unsafe {
97            let root = morrow_tree_new(c"root".as_ptr());
98            let child = morrow_tree_add(
99                morrow_tree_new(c"child".as_ptr()),
100                morrow_tree_new(c"leaf".as_ptr()),
101            );
102            let one = morrow_tree_add(root, child);
103            let two = morrow_tree_add(one, morrow_tree_new(c"last".as_ptr()));
104            assert_eq!(abi::text(morrow_tree_render(root)), "root");
105            assert_eq!(
106                abi::text(morrow_tree_render(one)),
107                "root\n└── child\n    └── leaf"
108            );
109            assert_eq!(
110                abi::text(morrow_tree_render(two)),
111                "root\n├── child\n│   └── leaf\n└── last"
112            );
113            assert_eq!(
114                abi::text(morrow_log_info(c"hé\n\t\\\x1b".as_ptr())),
115                "[INFO] hé\\n\\t\\\\\\x1b"
116            );
117        }
118    }

120    #[test]
121    fn panel_table_and_progress_preserve_native_rendering() {
122        unsafe {
123            let panel = morrow_panel_new(c"hello".as_ptr());
124            assert_eq!(
125                abi::text(morrow_panel_render(panel)),
126                "╭───────╮\n│ hello │\n╰───────╯"
127            );
128            let table = morrow_table_add_column(morrow_table_new(), c"A".as_ptr());
129            morrow_table_add_row(table, abi::strings(&["b"]));
130            assert_eq!(
131                abi::text(morrow_table_render(table)),
132                "╭───╮\n│ A │\n│───│\n│ b │\n╰───╯"
133            );
134            let progress = morrow_progress_width(morrow_progress_new(4), 4);
135            morrow_progress_set(progress, 2);
136            assert_eq!(abi::text(morrow_progress_render(progress)), "[██░░]  50%");
137            let spinner = morrow_spinner_style(morrow_spinner_new(), c"line".as_ptr());
138            morrow_spinner_tick(spinner);
139            assert_eq!(abi::text(morrow_spinner_render(spinner)), "\\");
140        }
141    }
