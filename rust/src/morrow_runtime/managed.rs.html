251/// Borrow the invocation group while the caller keeps its Session alive.
252unsafe fn shared<'a>(s: *mut Session) -> Option<&'a transport::Shared> {
253    unsafe { (*s)._shared.as_ref().map(|shared| &**shared.as_ptr()) }
254}

266    }

269/// Find reusable storage without changing immutable actor generations.
270unsafe fn vacant_slot(s: *mut Session) -> Option<usize> {
271    unsafe {
272        if shared(s).is_none() && (*s).next_id == u64::MAX {
273            return None;
274        }
275        for slot in 0..(*s).used_slots {
276            if (*(*s).identities.add(slot)).is_null() {
277                return Some(slot);
278            }
279        }
280        ((*s).used_slots < IDS).then_some((*s).used_slots)
281    }
282}

284unsafe fn publish_actor(s: *mut Session, a: *mut Actor, slot: usize) {
285    unsafe {
286        let id = (*a).identity.id;
287        debug_assert_ne!(id, 0);
288        debug_assert_eq!((*a).slot, slot);
289        (*s).next_id = (*s).next_id.max(id);
290        (*s).used_slots = (*s).used_slots.max(slot + 1);
291        *(*s).identities.add(slot) = a;
292        (*s).live += 1;
293    }
294}

310/// Validate immutable identity even when its old slot now belongs to another actor.
311/// A retained PID owns Actor -> owner Session. Local keys are that retained
312/// Session address; parallel keys are its retained Shared Arc address. Migration
313/// replaces the Session with another owner in that same Shared group, so neither
314/// key can be reused while any old PID wrapper remains alive.
315unsafe fn valid_pid(s: *mut Session, pid: *const Pid) -> bool {
316    unsafe {
317        !pid.is_null()
318            && (*pid).session as usize == (*s).session_key
319            && !(*pid).actor.is_null()
320            && (*pid).id != 0
321            && (*(*pid).actor).identity.session_key == (*s).session_key
322            && (*(*pid).actor).identity.id == (*pid).id
323            && (*(*pid).actor).identity.mailbox == (*pid).mailbox
324    }
325}

327unsafe fn live_pid(s: *mut Session, pid: *const Pid) -> bool {
328    unsafe {
329        valid_pid(s, pid)
330            && (*(*pid).actor).identity.alive.load(Ordering::Acquire)
331            && !(*(*pid).actor).identity.exiting.load(Ordering::Acquire)
332    }
333}

335// SAFETY for private helpers: callers supply live invocation-owned records and
336// validated immutable compiler descriptors; callbacks never overlap Rust references.
337unsafe fn fail(exec: *mut Exec, code: i64) {
338    unsafe {
339        if !(*exec).actor.is_null() && matches!(code, 11 | 12) {
340            (*(*exec).actor).infrastructure_fault = true;
341        }
342        if *(*exec).fault == 0 {
343            // A root operation rejected during shutdown must not replace a
344            // worker's already published failure with an admission error.
345            let published = if (*exec).actor.is_null() && !(*exec).session.is_null() {
346                shared((*exec).session).map_or(0, |group| group.fault.load(Ordering::Acquire))
347            } else {
348                0
349            };
350            *(*exec).fault = if published == 0 { code } else { published };
351        }
352    }
353}
354unsafe fn charge(s: *mut Session, cost: usize) -> bool {
355    unsafe {
356        if let Some(shared) = shared(s) {
357            let Some(reservation) = shared.budget.try_charge(cost) else {
358                return false;
359            };
360            let retained = (*s)
361                .retained
362                .checked_add(cost)
363                .expect("local retained-byte mirror overflowed");
364            reservation.commit();
365            (*s).retained = retained;
366            return true;
367        }
368        if (*s)
369            .retained
370            .checked_add(reasons::retained_bytes(s))
371            .and_then(|used| used.checked_add(cost))
372            .is_none_or(|used| used > BYTES)
373        {
374            return false;
375        }
376        (*s).retained += cost;
377        true
378    }
379}
380unsafe fn release(s: *mut Session, cost: usize) {
381    unsafe {
382        assert!(cost <= (*s).retained);
383        if let Some(shared) = shared(s) {
384            shared.budget.release_bytes(cost);
385        }
386        (*s).retained -= cost;
387    }
388}

390/// Admit one actor and return the immutable invocation-wide generation.
391unsafe fn reserve_actor(s: *mut Session, total_cost: usize) -> Option<u64> {
392    unsafe {
393        if let Some(shared) = shared(s) {
394            let reservation = shared.budget.try_reserve_actor(total_cost)?;
395            let retained = (*s)
396                .retained
397                .checked_add(total_cost)
398                .expect("local retained-byte mirror overflowed");
399            let generation = reservation.commit();
400            (*s).retained = retained;
401            return Some(generation);
402        }
403        if (*s).live >= LIVE {
404            return None;
405        }
406        let generation = (*s).next_id.checked_add(1)?;
407        charge(s, total_cost).then_some(generation)
408    }
409}
410unsafe fn function_work(
411    s: *mut Session,
412    closure: *const c_void,
413    work: &mut usize,
414) -> *const Function {
415    unsafe {
416        if closure.is_null() || !cost::work(work) {
417            return null();
418        }
419        let identity = *closure.cast::<*const c_void>();
420        if let Some((descriptor, index)) = (*s).function_cache.get(identity) {
421            // Preserve the old registry-position charge and exact failure point,
422            // even though the physical comparison loop is now unnecessary.
423            if index + 1 > WORK - *work {
424                *work = WORK + 1;
425                return null();
426            }
427            *work += index + 1;
428            return descriptor;
429        }
430        for i in 0..(*s).function_count {
431            if !cost::work(work) {
432                return null();
433            }
434            let f = *(*s).functions.add(i);
435            #[cfg(test)]
436            descriptor_cache::compared();
437            if (*f).identity == identity {
438                (*s).function_cache.insert(identity, f, i);
439                return f;
440            }
441        }
442        null()
443    }
444}
445unsafe fn function(s: *mut Session, closure: *const c_void) -> *const Function {
446    unsafe { function_work(s, closure, &mut 0) }
447}

449#[cfg(test)]
450thread_local! { static CLOCK: std::cell::Cell<Option<Option<u64>>> = const { std::cell::Cell::new(None) }; }
451unsafe fn now(_session: *mut Session) -> Option<u64> {
452    #[cfg(any(test, feature = "simulation"))]
453    unsafe {
454        let state = &mut (*_session).simulation;
455        if state.enabled {
456            if std::mem::take(&mut state.fail_next) {
457                return None;
458            }
459            return Some(state.milliseconds);
460        }
461    }
462    #[cfg(test)]
463    if let Some(value) = CLOCK.with(|clock| clock.get()) {
464        return value;
465    }
466    let mut time = libc::timespec {
467        tv_sec: 0,
468        tv_nsec: 0,
469    };
470    // SAFETY: valid writable timespec and supported monotonic clock identifier.
471    if unsafe { libc::clock_gettime(libc::CLOCK_MONOTONIC, &mut time) } != 0
472        || time.tv_sec < 0
473        || !(0..1000000000).contains(&time.tv_nsec)
474    {
475        return None;
476    }
477    (time.tv_sec as u64)
478        .checked_mul(1000)?
479        .checked_add(time.tv_nsec as u64 / 1000000)
480}
