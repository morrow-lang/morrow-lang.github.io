1//! Audited native register/stack boundary. Only the current thread is inspected.
2//! Assembly loads intentionally read machine words, including padding/uninitialized
3//! stack bytes, without constructing Rust references to those bytes.
4
5/// Read one mapped machine word without imposing Rust initialization requirements.
6/// # Safety
7/// Eight bytes at `address` must be mapped and readable for the duration of the load.
8pub unsafe fn word(address: usize) -> usize {
9    let value: usize;
10    #[cfg(target_arch = "aarch64")]
11    unsafe {
12        std::arch::asm!("ldr {value}, [{address}]", value = out(reg) value, address = in(reg) address, options(nostack, readonly));
13    }
14    #[cfg(target_arch = "x86_64")]
15    unsafe {
16        std::arch::asm!("mov {value}, [{address}]", value = out(reg) value, address = in(reg) address, options(nostack, readonly));
17    }
18    value
19}

21#[cfg(target_os = "macos")]
22fn stack_top() -> usize {
23    unsafe extern "C" {
24        fn pthread_self() -> usize;
25        fn pthread_get_stackaddr_np(thread: usize) -> *mut std::ffi::c_void;
26    }
27    // SAFETY: querying the current OS thread; no foreign thread lifetime is involved.
28    unsafe { pthread_get_stackaddr_np(pthread_self()) as usize }
29}
30
31#[cfg(target_os = "linux")]
32fn stack_top() -> usize {
33    // SAFETY: pthread initializes attributes; every success destroys them after use.
34    unsafe {
35        let mut attributes = std::mem::MaybeUninit::<libc::pthread_attr_t>::uninit();
36        if libc::pthread_getattr_np(libc::pthread_self(), attributes.as_mut_ptr()) != 0 {
37            std::process::abort();
38        }
39        let mut attributes = attributes.assume_init();
40        let mut base = std::ptr::null_mut();
41        let mut size = 0;
42        let status = libc::pthread_attr_getstack(&attributes, &mut base, &mut size);
43        libc::pthread_attr_destroy(&mut attributes);
44        if status != 0 {
45            std::process::abort();
46        }
47        (base as usize)
48            .checked_add(size)
49            .unwrap_or_else(|| std::process::abort())
50    }
51}

53/// Snapshot ABI-preserved registers and the active native stack into ordinary Rust storage.
54#[inline(never)]
55pub fn snapshot() -> Vec<usize> {
56    let top = stack_top();
57    let mut registers = [0usize; 16];
58    let stack: usize;
59    #[cfg(target_arch = "aarch64")]
60    // SAFETY: output points to sixteen writable words; assembly preserves all registers.
61    unsafe {
62        std::arch::asm!(
63        "stp x19, x20, [{out}, #0]", "stp x21, x22, [{out}, #16]",
64        "stp x23, x24, [{out}, #32]", "stp x25, x26, [{out}, #48]",
65        "stp x27, x28, [{out}, #64]", "str x29, [{out}, #80]", "mov {stack}, sp",
66        out = in(reg) registers.as_mut_ptr(), stack = out(reg) stack, options(nostack));
67    }
68    #[cfg(target_arch = "x86_64")]
69    // SAFETY: output points to sixteen writable words; assembly preserves all registers.
70    unsafe {
71        std::arch::asm!(
72        "mov [{out}], rbx", "mov [{out}+8], rbp", "mov [{out}+16], r12",
73        "mov [{out}+24], r13", "mov [{out}+32], r14", "mov [{out}+40], r15", "mov {stack}, rsp",
74        out = in(reg) registers.as_mut_ptr(), stack = out(reg) stack, options(nostack));
75    }
76    if top < stack || top - stack > 1024 * 1024 * 1024 {
77        std::process::abort();
78    }
79    let mut words = Vec::with_capacity((top - stack) / 8 + registers.len());
80    words.extend_from_slice(&registers);
81    for address in (stack..top.saturating_sub(7)).step_by(8) {
82        // SAFETY: OS-provided stack top and current SP delimit the current mapped active stack.
83        words.push(unsafe { word(address) });
84    }
85    words
86}
