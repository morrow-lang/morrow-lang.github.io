1//! Real SQLite connections with bounded live slots and never-recycled handles.
2use crate::abi;
3use rusqlite::{Connection, ffi};
4use std::collections::BTreeMap;
5use std::ffi::{CStr, OsStr, c_char};
6use std::os::unix::ffi::OsStrExt;
7use std::sync::Mutex;
8
9struct Connections {
10    next: i64,
11    live: BTreeMap<i64, Connection>,
12}
13static CONNECTIONS: Mutex<Connections> = Mutex::new(Connections {
14    next: 1,
15    live: BTreeMap::new(),
16});
17
18#[unsafe(no_mangle)]
19/// # Safety
20/// Non-null string pointers must reference readable NUL-terminated storage.
21/// Object and list pointers must use the declared runtime ABI, remain live for
22/// this call, and allow any requested mutation without aliasing or concurrent access.
23pub unsafe extern "C" fn morrow_sql_open(path: *const c_char) -> i64 {
24    if path.is_null() || unsafe { *path } == 0 {
25        return abi::result_err(3);
26    }
27    let mut connections = CONNECTIONS
28        .lock()
29        .unwrap_or_else(|error| error.into_inner());
30    if connections.live.len() == 256 || connections.next == i64::MAX {
31        return abi::result_err(4);
32    }
33    let path = OsStr::from_bytes(unsafe { CStr::from_ptr(path) }.to_bytes());
34    let Ok(connection) = Connection::open(path) else {
35        return abi::result_err(3);
36    };
37    if connection.busy_timeout(std::time::Duration::ZERO).is_err() {
38        return abi::result_err(3);
39    }
40    let id = connections.next;
41    connections.next += 1;
42    connections.live.insert(id, connection);
43    abi::result_ok(id)
44}

46#[unsafe(no_mangle)]
47/// # Safety
48/// Non-null string pointers must reference readable NUL-terminated storage.
49/// Object and list pointers must use the declared runtime ABI, remain live for
50/// this call, and allow any requested mutation without aliasing or concurrent access.
51pub unsafe extern "C" fn morrow_sql_execute(handle: i64, query: *const c_char) -> i64 {
52    if handle <= 0 || query.is_null() || unsafe { *query } == 0 {
53        return abi::result_err(3);
54    }
55    let connections = CONNECTIONS
56        .lock()
57        .unwrap_or_else(|error| error.into_inner());
58    let Some(connection) = connections.live.get(&handle) else {
59        return abi::result_err(3);
60    };
61    // sqlite3_exec permits complete statement batches and ignores returned query rows.
62    // rusqlite owns the connection; the lock excludes concurrent close/statement access.
63    let status = unsafe {
64        ffi::sqlite3_exec(
65            connection.handle(),
66            query,
67            None,
68            std::ptr::null_mut(),
69            std::ptr::null_mut(),
70        )
71    };
72    if status == ffi::SQLITE_OK {
73        abi::result_ok(unsafe { ffi::sqlite3_changes(connection.handle()) } as i64)
74    } else {
75        abi::result_err(3)
76    }
77}

79#[unsafe(no_mangle)]
80pub extern "C" fn morrow_sql_close(handle: i64) -> i64 {
81    let mut connections = CONNECTIONS
82        .lock()
83        .unwrap_or_else(|error| error.into_inner());
84    let Some(connection) = connections.live.remove(&handle) else {
85        return abi::result_err(3);
86    };
87    match connection.close() {
88        Ok(()) => abi::result_ok(0),
89        Err((connection, _)) => {
90            connections.live.insert(handle, connection);
91            abi::result_err(3)
92        }
93    }
94}

96#[cfg(test)]
97mod tests {
98    use super::*;
99    use crate::services::decode;
100
101    #[test]
102    fn sqlite_lifecycle_quota_stale_ids_and_arbitrary_statements() {
103        unsafe {
104            let first = decode(morrow_sql_open(c":memory:".as_ptr())).unwrap();
105            assert_eq!(
106                decode(morrow_sql_execute(
107                    first,
108                    c"CREATE TABLE t(x INTEGER); INSERT INTO t VALUES(1),(2); SELECT * FROM t;"
109                        .as_ptr()
110                )),
111                Ok(2)
112            );
113            assert_eq!(
114                decode(morrow_sql_execute(first, c"UPDATE t SET x=x+1".as_ptr())),
115                Ok(2)
116            );
117            assert_eq!(decode(morrow_sql_close(first)), Ok(0));
118            assert_eq!(decode(morrow_sql_close(first)), Err(3));
119            assert_eq!(
120                decode(morrow_sql_execute(first, c"SELECT 1".as_ptr())),
121                Err(3)
122            );
123            let mut handles = Vec::new();
124            for _ in 0..256 {
125                handles.push(decode(morrow_sql_open(c":memory:".as_ptr())).unwrap());
126            }
127            assert_eq!(decode(morrow_sql_open(c":memory:".as_ptr())), Err(4));
128            let old = handles.pop().unwrap();
129            assert_eq!(decode(morrow_sql_close(old)), Ok(0));
130            let new = decode(morrow_sql_open(c":memory:".as_ptr())).unwrap();
131            assert!(new > old);
132            assert_eq!(
133                decode(morrow_sql_execute(old, c"SELECT 1".as_ptr())),
134                Err(3)
135            );
136            handles.push(new);
137            for handle in handles {
138                assert_eq!(decode(morrow_sql_close(handle)), Ok(0));
139            }
140        }
141    }
