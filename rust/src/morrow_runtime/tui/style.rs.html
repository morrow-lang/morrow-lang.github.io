4unsafe fn wrap(prefix: &str, value: *const c_char) -> *const c_char {

70pub(super) fn hex(value: &str) -> Option<[u8; 3]> {
71    let value = value.strip_prefix('#').unwrap_or(value);
72    if value.len() != 6 || !value.is_ascii() {
73        return None;
74    }
75    Some([
76        u8::from_str_radix(&value[0..2], 16).ok()?,
77        u8::from_str_radix(&value[2..4], 16).ok()?,
78        u8::from_str_radix(&value[4..6], 16).ok()?,
79    ])
80}

105/// Non-null string pointers must reference readable NUL-terminated storage.

108pub unsafe extern "C" fn morrow_style_reset(value: *const c_char) -> *const c_char {
109    if value.is_null() {
110        return std::ptr::null();
111    }
112    let mut escaped = false;
113    let value: String = unsafe { text(value) }
114        .chars()
115        .filter(|ch| {
116            if escaped {
117                if *ch == 'm' {
118                    escaped = false;
119                }
120                false
121            } else if *ch == '\x1b' {
122                escaped = true;
123                false
124            } else {
125                true
126            }
127        })
128        .collect();
129    rendered(&value)
130}

139pub(super) fn named_color(value: &str) -> String {
140    if value.starts_with('#')
141        && let Some([r, g, b]) = hex(value)
142    {
143        return format!("\x1b[38;2;{r};{g};{b}m");
144    }
145    let (prefix, name) = value
146        .strip_prefix("bright_")
147        .map_or((30, value), |name| (90, name));
148    [
149        "black", "red", "green", "yellow", "blue", "magenta", "cyan", "white",
150    ]
151    .iter()
152    .position(|name2| *name2 == name)
153    .map_or_else(String::new, |offset| format!("\x1b[{}m", prefix + offset))
154}
