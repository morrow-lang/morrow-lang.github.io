17/// Run dynamic native programs and atomic rejections in private working directories.
18pub fn run(root: &Path, bin: &Path) -> Result<(), String> {
19    let cases = inventory::native()?;
20    for case in &cases {
21        native(bin, case).map_err(|e| format!("{}: {e}", case["name"]))?;
22    }
23    let invalid = inventory::invalid()?;
24    for case in &invalid {
25        let work = crate::Temporary::new(&std::env::temp_dir())?;
26        let source = prepare(&work.0, case)?;
27        rejected(bin, &work.0, &source, case).map_err(|e| format!("{}: {e}", case["name"]))?;
28    }
29    let mut directory_cases = 0;
30    for (group, count) in [
31        ("aliases/invalid", 12),
32        ("json_values/invalid", 12),
33        ("labels_native", 8),
34        ("namespaces_native", 12),
35    ] {
36        let mut paths = fs::read_dir(root.join("crates/morrow/tests").join(group))
37            .map_err(|e| e.to_string())?
38            .map(|entry| entry.map(|entry| entry.path()))
39            .collect::<Result<Vec<_>, _>>()
40            .map_err(|e| e.to_string())?;
41        paths.retain(|path| {
42            path.extension().is_some_and(|ext| ext == "mr")
43                && (group.ends_with("invalid")
44                    || path
45                        .file_name()
46                        .is_some_and(|name| name.to_string_lossy().starts_with("invalid_")))
47        });
48        paths.sort();
49        if paths.len() != count {
50            return Err(format!(
51                "{group}: expected {count} negative fixtures, found {}",
52                paths.len()
53            ));
54        }
55        for source in paths {
56            let work = crate::Temporary::new(&std::env::temp_dir())?;
57            rejected(bin, &work.0, &source, &serde_json::json!({}))
58                .map_err(|e| format!("{}: {e}", source.display()))?;
59            directory_cases += 1;
60        }
61    }
62    union_tests(bin)?;
63    println!(
64        "Compatibility passed: {} dynamic native programs, {} atomic rejections, union test continuation",
65        cases.len(),
66        invalid.len() + directory_cases
67    );
68    Ok(())
69}

71/// Materialize a trusted fixture with literal metacharacters in its source path.
72fn prepare(directory: &Path, case: &Value) -> Result<std::path::PathBuf, String> {
73    if let Some(files) = case["files"].as_object() {
74        for (name, text) in files {
75            if Path::new(name).components().count() != 1
76                || !matches!(
77                    Path::new(name).components().next(),
78                    Some(std::path::Component::Normal(_))
79                )
80            {
81                return Err("invalid fixture sibling path".into());
82            }
83            fs::write(
84                directory.join(name),
85                text.as_str().ok_or("invalid fixture file")?,
86            )
87            .map_err(|e| e.to_string())?;
88        }
89    }
90    let source = directory.join("hello 'source' $literal.mr");
91    fs::write(&source, case["source"].as_str().ok_or("missing source")?)
92        .map_err(|e| e.to_string())?;
93    Ok(source)
94}
95fn command(bin: &Path, directory: &Path, action: &str, source: &Path) -> Result<Captured, String> {
96    let mut command = Command::new(bin.join("morrow"));
97    command.arg(action).arg(source);
98    capture(command, bin, directory)
99}

101/// Separate compilation allows raw non-UTF8 argv to reach generated code unchanged.
102fn native(bin: &Path, case: &Value) -> Result<(), String> {
103    let work = crate::Temporary::new(&std::env::temp_dir())?;
104    let source = prepare(&work.0, case)?;
105    fs::write(work.0.join("invalid-utf8.txt"), [0xc0, 0xaf]).map_err(|e| e.to_string())?;
106    let executable = work.0.join("program 'output' $literal");
107    let mut build = Command::new(bin.join("morrow"));
108    build.arg("build").arg(&source).arg("-o").arg(&executable);
109    let built = capture(build, bin, &work.0)?;
110    if !built.status.success() {
111        return Err(format!(
112            "build failed: {}",
113            String::from_utf8_lossy(&built.stderr)
114        ));
115    }
116    let mut run = Command::new(executable);
117    match case["name"].as_str() {
118        Some("test_rust_boundaries/split_invalid_bytes") => {
119            run.arg(OsString::from_vec(vec![0xc0, 0xaf]));
120        }
121        Some("test_rust_stdlib/arguments") => {
122            run.args(["literal ; $ value", "🌿"]);
123        }
124        _ => {}
125    }
126    let result = capture(run, bin, &work.0)?;
127    if result.status.code().map(i64::from) != case["exit"].as_i64()
128        || result.stdout != case["stdout"].as_str().ok_or("missing stdout")?.as_bytes()
129        || result.stderr != case["stderr"].as_str().ok_or("missing stderr")?.as_bytes()
130    {
131        return Err(format!(
132            "unexpected {} stdout={:?} stderr={:?}",
133            result.status,
134            String::from_utf8_lossy(&result.stdout),
135            String::from_utf8_lossy(&result.stderr)
136        ));
137    }
138    Ok(())
139}
140const SENTINEL: &[u8] = b"existing executable\0must survive failed checking\xff";
141
142/// Rejection preserves prior executable bytes, inode, mode and modification time.
143fn rejected(bin: &Path, directory: &Path, source: &Path, case: &Value) -> Result<(), String> {
144    if let Some(action) = case["preflight"].as_str() {
145        let parsed = command(bin, directory, action, source)?;
146        if !parsed.status.success() {
147            return Err(format!(
148                "negative fixture must {action}: {}",
149                String::from_utf8_lossy(&parsed.stderr)
150            ));
151        }
152    }
153    let output = directory.join("preserved executable");
154    let absent = case["no_artifact"].as_bool() == Some(true);
155    let before = if absent {
156        None
157    } else {
158        fs::write(&output, SENTINEL).map_err(|e| e.to_string())?;
159        fs::set_permissions(&output, fs::Permissions::from_mode(0o751))
160            .map_err(|e| e.to_string())?;
161        Some(fs::metadata(&output).map_err(|e| e.to_string())?)
162    };
163    let mut build = Command::new(bin.join("morrow"));
164    build.arg("build").arg(source).arg("-o").arg(&output);
165    let actual = capture(build, bin, directory)?;
166    let diagnostic = String::from_utf8_lossy(&actual.stderr);
167    let expected = case["diagnostic"].as_str().unwrap_or("error:");
168    let forbidden = case["absent_diagnostic"].as_str().unwrap_or("");
169    if actual.status.code() != Some(1)
170        || !actual.stdout.is_empty()
171        || diagnostic.contains("panicked")
172        || !diagnostic.to_lowercase().contains(&expected.to_lowercase())
173        || (!forbidden.is_empty() && diagnostic.contains(forbidden))
174    {
175        return Err(format!(
176            "invalid program must diagnose {expected:?}: {} {diagnostic}",
177            actual.status
178        ));
179    }
180    unchanged(&output, before.as_ref())
181}
182fn unchanged(output: &Path, before: Option<&fs::Metadata>) -> Result<(), String> {
183    match before {
184        None if !output.try_exists().map_err(|e| e.to_string())? => Ok(()),
185        Some(before) => {
186            let after = fs::metadata(output).map_err(|e| e.to_string())?;
187            if fs::read(output).map_err(|e| e.to_string())? == SENTINEL
188                && (
189                    before.ino(),
190                    before.mode(),
191                    before.mtime(),
192                    before.mtime_nsec(),
193                ) == (after.ino(), after.mode(), after.mtime(), after.mtime_nsec())
194            {
195                Ok(())
196            } else {
197                Err("failed build modified existing executable".into())
198            }
199        }
200        None => Err("failed build published an executable".into()),
201    }
202}
203fn union_tests(bin: &Path) -> Result<(), String> {
204    let work = crate::Temporary::new(&std::env::temp_dir())?;
205    let path = work.0.join("unit_library.mr");
206    fs::write(
207        &path,
208        include_str!("../../tests/fixtures/union-test-continuation.mr"),
209    )
210    .map_err(|e| e.to_string())?;
211    let actual = command(bin, &work.0, "test", &path)?;
212    let diagnostic = String::from_utf8_lossy(&actual.stderr);
213    if actual.status.code() != Some(1)
214        || !String::from_utf8_lossy(&actual.stdout).contains("2/4 passed")
215        || ![
216            "System.exit cannot terminate a test",
217            "test_failed",
218            "test_exit",
219        ]
220        .iter()
221        .all(|part| diagnostic.contains(part))
222        || fs::read(work.0.join("events")).map_err(|e| e.to_string())? != b"after"
223    {
224        return Err(format!("union test continuation failed: {actual:?}"));
225    }
226    Ok(())
227}

232    #[test]
233    fn atomic_checker_detects_bytes_modes_and_created_output() {
234        let work = crate::Temporary::new(&std::env::temp_dir()).unwrap();
235        let path = work.0.join("output");
236        assert!(unchanged(&path, None).is_ok());
237        fs::write(&path, SENTINEL).unwrap();
238        let before = fs::metadata(&path).unwrap();
239        assert!(unchanged(&path, Some(&before)).is_ok());
240        assert!(unchanged(&path, None).is_err());
241        fs::set_permissions(&path, fs::Permissions::from_mode(0o500)).unwrap();
242        assert!(unchanged(&path, Some(&before)).is_err());
243        fs::set_permissions(&path, before.permissions()).unwrap();
244        fs::write(&path, b"replacement").unwrap();
245        assert!(unchanged(&path, Some(&before)).is_err());
246    }
247    #[test]
248    fn fixture_siblings_cannot_escape_private_workspace() {
249        let work = crate::Temporary::new(&std::env::temp_dir()).unwrap();
250        let bad = serde_json::json!({"source":"fn main():()","files":{"../escaped.mr":""}});
251        assert!(prepare(&work.0, &bad).is_err());
252        let good =
253            serde_json::json!({"source":"fn main():()","files":{"model.mr":"pub fn value():1"}});
254        assert!(prepare(&work.0, &good).is_ok());
255        assert_eq!(
256            fs::read_to_string(work.0.join("model.mr")).unwrap(),
