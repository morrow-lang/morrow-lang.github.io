1//! Explicit local release measurements; correctness gates do not depend on noisy timing.
2use serde_json::{Value, json};
3use sha2::{Digest, Sha256};
4use std::{fs, path::Path, process::Command, time::Instant};
5
6fn summary(samples: &[f64]) -> Result<Value, String> {
7    if samples.is_empty()
8        || samples
9            .iter()
10            .any(|value| !value.is_finite() || *value < 0.0)
11    {
12        return Err("timings must be nonempty, finite and nonnegative".into());
13    }
14    let mut sorted = samples.to_vec();
15    sorted.sort_by(f64::total_cmp);
16    let middle = sorted.len() / 2;
17    let median = if sorted.len().is_multiple_of(2) {
18        (sorted[middle - 1] + sorted[middle]) / 2.0
19    } else {
20        sorted[middle]
21    };
22    Ok(
23        json!({"samples":sorted.len(),"median_ms":median,"p95_ms":sorted[(sorted.len()*95).div_ceil(100)-1]}),
24    )
25}

27/// Measure explicitly staged components and simple native programs with independent outputs.
28pub fn run(root: &Path, bin: &Path, output: &Path) -> Result<(), String> {
29    let mut components = serde_json::Map::new();
30    for name in ["morrow", "morrow-test-supervisor", "libmorrow_runtime.a"] {
31        let bytes = fs::read(bin.join(name)).map_err(|error| error.to_string())?;
32        components.insert(
33            name.into(),
34            json!({"bytes":bytes.len(),"sha256":Sha256::digest(&bytes).iter().map(|byte| format!("{byte:02x}")).collect::<String>()}),
35        );
36    }
37    let compiler = bin.join("morrow");
38    let mut startup = Vec::new();
39    for _ in 0..30 {
40        let started = Instant::now();
41        let result = Command::new(&compiler)
42            .arg("--version")
43            .output()
44            .map_err(|error| error.to_string())?;
45        startup.push(started.elapsed().as_secs_f64() * 1000.0);
46        if !result.status.success()
47            || result.stdout != format!("morrow {}\n", env!("CARGO_PKG_VERSION")).as_bytes()
48            || !result.stderr.is_empty()
49        {
50            return Err("compiler version probe failed".into());
51        }
52    }
53    let temporary = crate::Temporary::new(&std::env::temp_dir())?;
54    let mut programs = Vec::new();
55    for (name, expected, code) in [
56        ("fib", "9227465\n", 0),
57        ("sum", "50005000\n", 0),
58        ("startup", "", 42),
59    ] {
60        let executable = temporary.0.join(name);
61        let started = Instant::now();
62        let result = Command::new(&compiler)
63            .arg("build")
64            .arg(root.join(format!("benchmarks/{name}_morrow.mr")))
65            .arg("-o")
66            .arg(&executable)
67            .output()
68            .map_err(|error| error.to_string())?;
69        let compile_ms = started.elapsed().as_secs_f64() * 1000.0;
70        if !result.status.success() {
71            return Err(format!(
72                "benchmark build: {}",
73                String::from_utf8_lossy(&result.stderr)
74            ));
75        }
76        let mut samples = Vec::new();
77        for _ in 0..10 {
78            let started = Instant::now();
79            let result = Command::new(&executable)
80                .output()
81                .map_err(|error| error.to_string())?;
82            samples.push(started.elapsed().as_secs_f64() * 1000.0);
83            if result.status.code() != Some(code)
84                || result.stdout != expected.as_bytes()
85                || !result.stderr.is_empty()
86            {
87                return Err(format!(
88                    "benchmark {name} produced incorrect output: {result:?}"
89                ));
90            }
91        }
92        programs.push(json!({"name":name,"compile_ms":compile_ms,"execution":summary(&samples)?,"executable_bytes":fs::metadata(&executable).map_err(|error|error.to_string())?.len()}));
93    }
94    let report = json!({"version":env!("CARGO_PKG_VERSION"),"os":std::env::consts::OS,"arch":std::env::consts::ARCH,
95        "measurement":"local wall clock, process startup included; host contention is not controlled", "components":components,"compiler_startup":summary(&startup)?,"programs":programs});
96    fs::write(
97        output,
98        serde_json::to_string_pretty(&report).map_err(|error| error.to_string())? + "\n",
99    )
100    .map_err(|error| error.to_string())?;
101    println!("Performance report: {}", output.display());
102    Ok(())
103}

105#[cfg(test)]
106mod tests {
107    use super::*;
108    #[test]
109    fn timing_summary_uses_sorted_nearest_rank_and_rejects_empty_or_invalid_samples() {
110        let samples: Vec<_> = (1..=20).rev().map(f64::from).collect();
111        let value = summary(&samples).unwrap();
112        assert_eq!(value["samples"], 20);
113        assert_eq!(value["median_ms"], 10.5);
114        assert_eq!(value["p95_ms"], 19.0);
115        assert!(summary(&[]).is_err());
116        assert!(summary(&[f64::NAN]).is_err());
117        assert!(summary(&[-1.0]).is_err());
118    }
119}
