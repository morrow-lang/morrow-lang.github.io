10/// Compute SHA-256 over at most the permitted compressed archive size.
11pub(super) fn hash(file: &mut File) -> Result<String, String> {
12    let mut digest = Sha256::new();
13    let mut buffer = [0; 65536];
14    let mut total = 0u64;
15    loop {
16        let count = file.read(&mut buffer).map_err(|e| e.to_string())?;
17        if count == 0 {
18            break;
19        }
20        total += count as u64;
21        if total > TOTAL_LIMIT {
22            return Err("release archive exceeds byte limit".into());
23        }
24        digest.update(&buffer[..count]);
25    }
26    Ok(digest
27        .finalize()
28        .iter()
29        .map(|byte| format!("{byte:02x}"))
30        .collect())
31}
32struct Count<'a> {
33    file: &'a mut File,
34    bytes: u64,
35}
36impl Read for Count<'_> {
37    fn read(&mut self, buffer: &mut [u8]) -> io::Result<usize> {
38        let count = self.file.read(buffer)?;
39        self.bytes += count as u64;
40        Ok(count)
41    }
42}
43/// Produce deterministic USTAR headers and one gzip stream from held staged files.
44pub(super) fn write(destination: &mut File, stem: &str, files: &mut [Input]) -> Result<(), String> {
45    let gzip = flate2::GzBuilder::new()
46        .mtime(0)
47        .write(destination, flate2::Compression::default());
48    let mut archive = tar::Builder::new(gzip);
49    for input in files {
50        let mut header = tar::Header::new_ustar();
51        header
52            .set_path(format!("{stem}/{}", input.name))
53            .map_err(|e| e.to_string())?;
54        header.set_size(input.size);
55        header.set_mode(input.mode);
56        header.set_uid(0);
57        header.set_gid(0);
58        header.set_mtime(0);
59        header.set_entry_type(tar::EntryType::Regular);
60        header.set_cksum();
61        let mut reader = Count {
62            file: &mut input.file,
63            bytes: 0,
64        };
65        archive
66            .append(&header, (&mut reader).take(input.size + 1))
67            .map_err(|e| e.to_string())?;
68        if reader.bytes != input.size {
69            return Err(format!(
70                "staging component changed while archiving: {}",
71                input.name
72            ));
73        }
74    }
75    archive
76        .into_inner()
77        .map_err(|e| e.to_string())?
78        .finish()
79        .map_err(|e| e.to_string())?;
80    Ok(())
81}
82fn open(path: &Path, limit: u64) -> Result<File, String> {
83    let directory = Dir::open(path.parent().ok_or("missing archive directory")?, false)
84        .map_err(|e| e.to_string())?;
85    let name = path
86        .file_name()
87        .and_then(|name| name.to_str())
88        .ok_or("invalid archive filename")?;
89    let file = directory.file(name).map_err(|e| e.to_string())?;
90    let size = file.metadata().map_err(|e| e.to_string())?.len();
91    if size == 0 || size > limit {
92        return Err("archive/checksum exceeds bounded regular-file limit".into());
93    }
94    Ok(file)
95}
96pub(super) fn verify(path: &Path, checksum: &Path) -> Result<(), String> {
97    let mut file = open(path, TOTAL_LIMIT)?;
98    let mut checksum_file = open(checksum, 4096)?;
99    let mut text = String::new();
100    checksum_file
101        .by_ref()
102        .take(4097)
103        .read_to_string(&mut text)
104        .map_err(|e| e.to_string())?;
105    let name = path
106        .file_name()
107        .and_then(|n| n.to_str())
108        .ok_or("invalid archive filename")?;
109    let expected = format!("{}  {name}\n", hash(&mut file)?);
110    if text != expected {
111        return Err("archive checksum mismatch or invalid checksum record".into());
112    }
113    file.rewind().map_err(|e| e.to_string())?;
114    verify_contents(file)
115}
116/// Read only raw regular members under one safe root, with exact length accounting.
117/// Reject extra gzip members/trailing payloads, including hidden post-tar entries.
118pub(super) fn verify_contents(file: File) -> Result<(), String> {
119    let gzip = flate2::bufread::GzDecoder::new(BufReader::new(file));
120    let mut archive = tar::Archive::new(gzip.take(TOTAL_LIMIT + 65537));
121    let mut seen = BTreeSet::new();
122    let mut root = None;
123    let mut total = 0u64;
124    let mut tar_bytes = 0u64;
125    let entries = archive.entries().map_err(|e| e.to_string())?.raw(true);
126    for item in entries {
127        let mut item = item.map_err(|e| e.to_string())?;
128        if seen.len() >= REQUIRED.len() + OPTIONAL.len() {
129            return Err("archive has too many members".into());
130        }
131        let path = item.path_bytes();
132        if path.len() > 256 {
133            return Err("archive member name exceeds limit".into());
134        }
135        let path = std::str::from_utf8(&path).map_err(|_| "archive name must be UTF-8")?;
136        let (prefix, name) = path.split_once('/').ok_or("archive member lacks a root")?;
137        if !prefix.starts_with("morrow-")
138            || prefix.len() > 192
139            || !prefix
140                .bytes()
141                .all(|b| b.is_ascii_alphanumeric() || b"._-+".contains(&b))
142            || !REQUIRED.contains(&name) && !OPTIONAL.contains(&name)
143            || !seen.insert(name.to_owned())
144        {
145            return Err("archive has unexpected or duplicate members".into());
146        }
147        if root.as_ref().is_some_and(|old| old != prefix) {
148            return Err("archive has multiple roots".into());
149        }
150        root = Some(prefix.to_owned());
151        let name = name.to_owned();
152        let header = item.header();
153        let size = header.size().map_err(|e| e.to_string())?;
154        if header.entry_type() != tar::EntryType::Regular || size == 0 || size > FILE_LIMIT {
155            return Err("archive component must be a bounded regular file".into());
156        }
157        if header.mode().map_err(|e| e.to_string())?
158            != if executable(&name) { 0o755 } else { 0o644 }
159        {
160            return Err(format!("archive component has invalid permissions: {name}"));
161        }
162        tar_bytes += 512 + size.div_ceil(512) * 512;
163        total = total.checked_add(size).ok_or("archive size overflow")?;
164        if total > TOTAL_LIMIT {
165            return Err("archive exceeds aggregate component limit".into());
166        }
167        if name == "morrow-package.json" {
168            let mut marker = Vec::new();
169            item.by_ref()
170                .take(65537)
171                .read_to_end(&mut marker)
172                .map_err(|e| e.to_string())?;
173            validate_marker(&marker)?;
174            if marker.len() as u64 != size {
175                return Err("truncated package marker".into());
176            }
177        } else {
178            let count = io::copy(&mut item, &mut io::sink()).map_err(|e| e.to_string())?;
179            if count != size {
180                return Err("truncated archive component".into());
181            }
182        }
183    }
184    if REQUIRED.iter().any(|name| !seen.contains(*name)) {
185        return Err("archive missing required members".into());
186    }
187    // Tar stops at its zero trailer. Consume and validate the bounded remainder
188    // so gzip CRC errors and concatenated/hidden data cannot be silently ignored.
189    let mut tail = archive.into_inner();
190    let mut buffer = [0; 4096];
191    let mut padding = 0u64;
192    if TOTAL_LIMIT + 65537 - tail.limit() != tar_bytes + 512 {
193        return Err("archive lacks complete tar end blocks".into());
194    }
195    loop {
196        let count = tail.read(&mut buffer).map_err(|e| e.to_string())?;
197        if count == 0 {
198            break;
199        }
200        padding += count as u64;
201        if buffer[..count].iter().any(|&b| b != 0) {
202            return Err("archive has trailing tar data".into());
203        }
204    }
205    if padding < 512 || !padding.is_multiple_of(512) {
206        return Err("archive lacks complete tar end blocks".into());
207    }
208    if tail.limit() == 0 {
209        return Err("archive exceeds decompression limit".into());
210    }
211    let mut compressed = tail.into_inner().into_inner();
212    if !compressed.fill_buf().map_err(|e| e.to_string())?.is_empty() {
213        return Err("archive has trailing compressed data".into());
214    }
215    Ok(())
