32pub(super) fn run(path: &Path, seed: u64, invoke: &mut Invoke<'_>) -> Result<usize, String> {
33    let mut random = seed ^ 0xD1B5_4A32_D192_ED03;
34    let mut cases = 0;
35    for (name, original) in SOURCES {
36        // The original also goes through check/emit, so source acceptance and
37        // formatting invariants cover each complete feature combination.
38        for index in 0..33 {
39            random ^= random << 13;
40            random ^= random >> 7;
41            random ^= random << 17;
42            let source = if index == 0 {
43                (*original).to_owned()
44            } else {
45                edit(original, random, index % 4)
46            };
47            fs::write(path, &source).map_err(|error| error.to_string())?;
48            if index == 0 {
49                require_success("feature check", invoke("check", path)?)?;
50                require_success("feature emit", invoke("emit", path)?)?;
51            }
52            mutation(&source, path, invoke).map_err(|error| format!("language feature={name} seed={seed:#x} index={index}: {error}\nsource={source:?}"))?;
53            cases += 1;
54        }
55    }
56    Ok(cases)
57}

59fn edit(source: &str, random: u64, mode: usize) -> String {
60    let boundaries: Vec<_> = source
61        .char_indices()
62        .map(|(index, _)| index)
63        .chain(std::iter::once(source.len()))
64        .collect();
65    let selected = random as usize % boundaries.len();
66    let start = boundaries[selected];
67    let end = boundaries.get(selected + 1).copied().unwrap_or(start);
68    let mut output = source.to_owned();
69    match mode {
70        0 => output.truncate(start),
71        1 => output.replace_range(start..end, ""),
72        2 => output.insert_str(start, "\n    "),
73        _ => output.insert_str(
74            start,
75            [")", "?", "|", "\"", ":", "🌿"][(random >> 32) as usize % 6],
76        ),
77    }
78    output
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84    #[test]
85    fn edits_preserve_utf8_and_have_independent_boundary_oracles() {
86        assert_eq!(edit("a🌿z", 1, 0), "a");
87        assert_eq!(edit("a🌿z", 1, 1), "az");
88        assert_eq!(edit("a🌿z", 1, 2), "a\n    🌿z");
89        assert_eq!(edit("a🌿z", 1, 3), "a)🌿z");
90        assert_eq!(edit("a🌿z", 3, 1), "a🌿z");
91    }
