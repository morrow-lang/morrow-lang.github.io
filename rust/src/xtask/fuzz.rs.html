15/// Run the original grammar corpus and fixed mutation acceptance through bounded helpers.
16/// The first seven iterations use retained seed files, matching the original runner.
17pub fn run(root: &Path, bin: &Path, iterations: u32, seed: u64) -> Result<(), String> {
18    let temporary = crate::Temporary::new(&std::env::temp_dir())?;
19    let path = temporary.0.join("case.mr");
20    let mut invoke = |action: &str, path: &Path| {
21        let mut command = Command::new(bin.join("morrow"));
22        command.arg(action).arg(path);
23        let output = crate::acceptance::capture(command, bin, root)?;
24        if !matches!(output.status.code(), Some(0 | 1))
25            || output.stderr.windows(8).any(|part| part == b"panicked")
26        {
27            return Err(format!(
28                "{action} failed abnormally: {} {}",
29                output.status,
30                String::from_utf8_lossy(&output.stderr)
31            ));
32        }
33        Ok(output)
34    };
35    for index in 0..iterations {
36        let source = if (index as usize) < SEEDS.len() {
37            fs::read_to_string(root.join(format!("fuzz/corpus/{}.mr", SEEDS[index as usize])))
38                .map_err(|error| error.to_string())?
39        } else {
40            generator::generate(seed, index)
41        };
42        fs::write(&path, &source).map_err(|error| error.to_string())?;
43        grammar(&path, &mut invoke).map_err(|error| {
44            format!("grammar seed={seed:#x} index={index}: {error}\nsource={source:?}")
45        })?;
46        if (index + 1).is_multiple_of(64) {
47            eprintln!("Fuzz grammar: {}/{iterations}", index + 1);
48        }
49    }
50    let fixture: serde_json::Value =
51        serde_json::from_str(include_str!("../../tests/fixtures/fuzz-mutations.json"))
52            .map_err(|error| error.to_string())?;
53    let mutation_seed = fixture["seed"]
54        .as_u64()
55        .ok_or("mutation fixture has no seed")?;
56    let sources = fixture["sources"]
57        .as_array()
58        .filter(|sources| sources.len() == 192)
59        .ok_or("mutation fixture must contain 192 sources")?;
60    for (index, source) in sources.iter().enumerate() {
61        let source = source.as_str().ok_or("mutation source must be text")?;
62        fs::write(&path, source).map_err(|error| error.to_string())?;
63        mutation(source, &path, &mut invoke).map_err(|error| {
64            format!("mutation seed={mutation_seed:#x} index={index}: {error}\nsource={source:?}")
65        })?;
66    }
67    let feature_cases = language::run(&path, seed, &mut invoke)?;
68    println!(
69        "Fuzz passed: {iterations} grammar cases seed={seed:#x}, 192 mutations seed={mutation_seed:#x}, {feature_cases} language-feature mutations"
70    );
71    Ok(())
72}
73type Invoke<'a> = dyn FnMut(&str, &Path) -> Result<morrow_test_supervisor::Captured, String> + 'a;
74/// A successful formatting pass must be stable and remain parseable.
75fn grammar(path: &Path, invoke: &mut Invoke<'_>) -> Result<(), String> {
76    for action in ["parse", "fmt"] {
77        require_success(action, invoke(action, path)?)?;
78    }
79    let canonical = fs::read(path).map_err(|error| error.to_string())?;
80    require_success("second fmt", invoke("fmt", path)?)?;
81    if fs::read(path).map_err(|error| error.to_string())? != canonical {
82        return Err("formatter is not idempotent".into());
83    }
84    require_success("second parse", invoke("parse", path)?)
85}
86fn require_success(action: &str, output: morrow_test_supervisor::Captured) -> Result<(), String> {
87    if output.status.success() {
88        Ok(())
89    } else {
90        Err(format!(
91            "{action}: {} {}",
92            output.status,
93            String::from_utf8_lossy(&output.stderr)
94        ))
95    }
96}
97/// Invalid sources may fail cleanly; accepted check/format operations preserve lowering.
98fn mutation(source: &str, path: &Path, invoke: &mut Invoke<'_>) -> Result<(), String> {
99    let checked = invoke("check", path)?;
100    let emitted = invoke("emit", path)?;
101    if checked.status.success() && !emitted.status.success() {
102        return Err("checked program cannot lower".into());
103    }
104    let formatted = invoke("fmt", path)?;
105    if !formatted.status.success() {
106        if fs::read(path).map_err(|error| error.to_string())? != source.as_bytes() {
107            return Err("failed formatter modified source".into());
108        }
109    } else {
110        let canonical = fs::read(path).map_err(|error| error.to_string())?;
111        require_success("second fmt", invoke("fmt", path)?)?;
112        if fs::read(path).map_err(|error| error.to_string())? != canonical {
113            return Err("formatter is not idempotent".into());
114        }
115        let again = invoke("emit", path)?;
116        if again.status.code() != emitted.status.code()
117            || (again.status.success() && again.stdout != emitted.stdout)
118        {
119            return Err("formatting changed lowering behavior".into());
120        }
121    }
122    Ok(())
123}

127    use std::{fs, os::unix::process::ExitStatusExt, process::ExitStatus};
128    fn output(code: i32, bytes: &[u8]) -> morrow_test_supervisor::Captured {
129        morrow_test_supervisor::Captured {
130            status: ExitStatus::from_raw(code << 8),
131            stdout: bytes.to_vec(),

135    #[test]
136    fn mutations_reject_failed_formatter_writes_and_checked_lowering_failure() {
137        let temporary = crate::Temporary::new(&std::env::temp_dir()).unwrap();
138        let path = temporary.0.join("case.mr");
139        fs::write(&path, "source").unwrap();
140        assert!(
141            mutation("source", &path, &mut |action, _| Ok(output(
142                if action == "emit" { 1 } else { 0 },
143                b""
144            )))
145            .is_err()
146        );
147        assert!(
148            mutation("source", &path, &mut |action, path| {
149                if action == "fmt" {
150                    fs::write(path, "changed").unwrap();
151                }
152                Ok(output(1, b""))
153            })
154            .is_err()
155        );
156    }
157    #[test]
158    fn mutations_reject_changed_lowering_and_non_idempotent_formatting() {
159        let temporary = crate::Temporary::new(&std::env::temp_dir()).unwrap();
160        let path = temporary.0.join("case.mr");
161        fs::write(&path, "source").unwrap();
162        let mut emissions = 0;
163        assert!(
164            mutation("source", &path, &mut |action, _| {
165                if action == "emit" {
166                    emissions += 1;
167                }
168                Ok(output(0, if emissions == 2 { b"new" } else { b"old" }))
169            })
170            .is_err()
171        );
172        let mut formats = 0;
173        assert!(
174            mutation("source", &path, &mut |action, path| {
175                if action == "fmt" {
176                    formats += 1;
177                    fs::write(path, formats.to_string()).unwrap();
178                }
179                Ok(output(0, b""))
180            })
181            .is_err()
182        );
183    }
