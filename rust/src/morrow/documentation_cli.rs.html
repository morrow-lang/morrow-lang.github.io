22const HELP: &str = "Usage: morrow doc [source.mr|directory] [--html] [--inferred] [--open] [-o output]\n       morrow doc [directory] --site <directory> [--title text] [--version text] [--extras path]... [--link label=url]... [--inferred] [--open]\nGenerate source documentation without executing code. Markdown is written to stdout by default. Directory HTML includes module navigation and browser Find guidance. --inferred checks the current module graph and adds resolved signatures. --open implies HTML, retains -o output (default: morrow-docs.html in the current directory), then best-effort launches the platform opener.\n--site publishes a multi-page HTML site (one page per module and Markdown guide, sidebar navigation, local search) into a directory that is created, including any missing parents, or when it is a previously generated site, replaced atomically. --extras adds Markdown guides from a file or the direct entries of a directory (directory listings omit README.md); the first README.md file extra becomes the landing page. --link adds sidebar links. Without a source operand, --site documents guides alone.\n";

24/// Parse the doc action, validate source, then print or atomically install the complete document.
25pub(super) fn run(arguments: Vec<OsString>) -> Result<u8, String> {
26    if arguments.len() == 2 && (arguments[1] == "--help" || arguments[1] == "-h") {
27        use std::io::Write;
28        std::io::stdout()
29            .lock()
30            .write_all(HELP.as_bytes())
31            .map_err(|error| error.to_string())?;
32        return Ok(0);
33    }
34    let options = options(arguments)?;
35    if let Some(output) = &options.site {
36        let request = site::Request {
37            source: options.explicit_source.then(|| options.source.clone()),
38            output: output.clone(),
39            title: options.title.clone(),
40            version: options.version.clone(),
41            extras: options.extras.clone(),
42            links: options.links.clone(),
43            inferred: options.inferred,
44        };
45        let published = site::run(&request)?;
46        if options.open {
47            opener::open(&published.join("index.html"));
48        }
49        return Ok(0);
50    }
51    let code = generate(&options)?;
52    if options.open
53        && let Some(output) = &options.output
54    {
55        opener::open(output);
56    }
57    Ok(code)
58}

60/// Complete generation and atomic publication precede any optional external launcher.
61fn generate(options: &Options) -> Result<u8, String> {
62    if options.inferred {
63        return inferred::run(&options.source, options.output.as_deref(), options.format);
64    }
65    if options.source.is_dir() {
66        return directory::run(&options.source, options.output.as_deref(), options.format);
67    }
68    let source = read_source(&options.source, &mut 0)?;
69    let title = options
70        .source
71        .file_name()
72        .unwrap_or_default()
73        .to_string_lossy();
74    let rendered = documentation::render(&source, &title, options.format).map_err(|error| {
75        let prefix = source.get(..error.span.start).unwrap_or("");
76        let line = prefix.bytes().filter(|byte| *byte == b'\n').count() + 1;
77        let column = prefix.rsplit('\n').next().unwrap_or("").chars().count() + 1;
78        format!(
79            "{}:{line}:{column}: error: {}",
80            options.source.display(),
81            error.message
82        )
83    })?;
84    if let Some(output) = &options.output {
85        super::emit_file(&options.source, output, &rendered)?;
86    } else {
87        use std::io::Write;
88        std::io::stdout()
89            .lock()
90            .write_all(rendered.as_bytes())
91            .map_err(|error| error.to_string())?;
92    }
93    Ok(0)
94}

96/// Accept one source, one optional output and an explicit HTML switch in any argument order.
97fn options(arguments: Vec<OsString>) -> Result<Options, String> {
98    let mut source = None;
99    let mut output = None;
100    let mut html = false;
101    let mut inferred = false;
102    let mut open = false;
103    let mut literal = false;
104    let mut site = None;
105    let mut title = None;
106    let mut version = None;
107    let mut extras = Vec::new();
108    let mut links = Vec::new();
109    let mut arguments = arguments.into_iter().skip(1);
110    while let Some(argument) = arguments.next() {
111        if !literal && argument == "--" {
112            literal = true;
113            continue;
114        }
115        if literal {
116            if source.replace(PathBuf::from(argument)).is_some() {
117                return Err("doc accepts one source file or directory".into());
118            }
119            continue;
120        }
121        if argument == "--site" {
122            if site.is_some() {
123                return Err("--site specified more than once".into());
124            }
125            site = Some(PathBuf::from(
126                arguments.next().ok_or("--site requires a directory")?,
127            ));
128        } else if argument == "--title" {
129            if title.is_some() {
130                return Err("--title specified more than once".into());
131            }
132            title = Some(text_value(arguments.next(), "--title", 4096)?);
133        } else if argument == "--version" {
134            if version.is_some() {
135                return Err("--version specified more than once".into());
136            }
137            version = Some(text_value(arguments.next(), "--version", 256)?);
138        } else if argument == "--extras" {
139            if extras.len() == 256 {
140                return Err("--extras accepts at most 256 paths".into());
141            }
142            extras.push(PathBuf::from(
143                arguments
144                    .next()
145                    .ok_or("--extras requires a Markdown file or directory")?,
146            ));
147        } else if argument == "--link" {
148            if links.len() == 64 {
149                return Err("--link accepts at most 64 links".into());
150            }
151            let value = text_value(arguments.next(), "--link", 4096)?;
152            let (label, url) = value
153                .split_once('=')
154                .filter(|(label, url)| !label.is_empty() && !url.is_empty())
155                .ok_or("--link requires label=url")?;
156            if !morrow_compiler::documentation::markdown::safe_destination(url) {
157                return Err(format!(
158                    "--link {label}: only safe http, https, mailto or relative destinations are accepted"
159                ));
160            }
161            links.push((label.to_string(), url.to_string()));
162        } else if argument == "--inferred" {
163            if inferred {
164                return Err("--inferred specified more than once".into());
165            }
166            inferred = true;
167        } else if argument == "--open" {
168            if open {
169                return Err("--open specified more than once".into());
170            }
171            open = true;
172        } else if argument == "--html" {
173            if html {
174                return Err("--html specified more than once".into());
175            }
176            html = true;
177        } else if argument == "-o" || argument == "--output" {
178            if output.is_some() {
179                return Err("output specified more than once".into());
180            }
181            output = Some(PathBuf::from(arguments.next().ok_or("-o requires a path")?));
182        } else if argument.to_string_lossy().starts_with('-') {
183            return Err(format!(
184                "unknown doc option: {}",
185                argument.to_string_lossy()
186            ));
187        } else if source.replace(PathBuf::from(argument)).is_some() {
188            return Err("doc accepts one source file or directory".into());
189        }
190    }
191    if site.is_some() {
192        if output.is_some() {
193            return Err("-o cannot accompany --site; the site directory is the output".into());
194        }
195        if html {
196            return Err("--html cannot accompany --site; sites are always HTML".into());
197        }
198    } else if title.is_some() || version.is_some() || !extras.is_empty() || !links.is_empty() {
199        return Err("--title, --version, --extras and --link require --site".into());
200    }
201    if open && output.is_none() && site.is_none() {
202        output = Some(PathBuf::from("morrow-docs.html"));
203    }
204    let explicit_source = source.is_some() || extras.is_empty();
205    Ok(Options {
206        source: source.unwrap_or_else(|| PathBuf::from(".")),
207        explicit_source,
208        output,
209        format: if html || open {
210            Output::Html
211        } else {
212            Output::Markdown
213        },
214        inferred,
215        open,
216        site,
217        title,
218        version,
219        extras,
220        links,
221    })
222}

224/// Accept one UTF-8 option value within a byte bound.
225fn text_value(value: Option<OsString>, option: &str, maximum: usize) -> Result<String, String> {
226    let value = value.ok_or_else(|| format!("{option} requires a value"))?;
227    let text = value
228        .into_string()
229        .map_err(|_| format!("{option} must be UTF-8"))?;
230    if text.is_empty() || text.len() > maximum {
231        return Err(format!("{option} must be 1–{maximum} bytes"));
232    }
233    Ok(text)
234}

236/// Share bounded directory discovery with the explicit executable documentation-test command.
237pub(super) fn sources(path: &std::path::Path) -> Result<Vec<PathBuf>, String> {
238    if path.is_dir() {
239        directory::discover(path)
240    } else {
241        Ok(vec![path.to_path_buf()])
242    }
243}

245/// Check bounded raw byte lengths before decoding, including a partial UTF-8 sentinel byte.
246fn read_source(path: &std::path::Path, bytes: &mut usize) -> Result<String, String> {
247    let mut source = Vec::new();
248    fs::File::open(path)
249        .and_then(|file| file.take(1024 * 1024 + 1).read_to_end(&mut source))
250        .map_err(|error| format!("{}: {error}", path.display()))?;
251    if source.len() > 1024 * 1024 {
252        return Err(format!(
253            "{}: documentation source exceeds 1 MiB",
254            path.display()
255        ));
256    }
257    *bytes = bytes
258        .checked_add(source.len())
259        .ok_or("documentation source size overflow")?;
260    if *bytes > 8 * 1024 * 1024 {
261        return Err("project documentation source exceeds 8 MiB".into());
262    }
263    String::from_utf8(source).map_err(|error| format!("{}: invalid UTF-8: {error}", path.display()))
264}
