1//! Best-effort platform launch after complete artifact publication, without shell expansion.
2use std::{
3    path::Path,
4    process::{Child, Command, Stdio},
5    time::{Duration, Instant},
6};
7
8/// Retain the artifact on launcher failure; diagnostics do not change generation's success status.
9pub(super) fn open(path: &Path) {
10    let result = path
11        .canonicalize()
12        .map_err(|error| error.to_string())
13        .and_then(|path| {
14            eprintln!("Generated docs: {}", path.display());
15            let command = platform().ok_or("no platform documentation opener available")?;
16            launch(command, &path, Duration::from_secs(10))
17        });
18    if let Err(error) = result {
19        eprintln!(
20            "note: could not open generated docs ({}): {error}",
21            path.display()
22        );
23    }
24}

26/// Platform names are fixed executables resolved by PATH, never shell command strings.
27fn platform() -> Option<&'static str> {
28    if cfg!(target_os = "macos") {
29        Some("open")
30    } else if cfg!(target_os = "linux") {
31        Some("xdg-open")
32    } else {
33        None
34    }
35}
36
37/// A literal absolute OS path is one argument; terminal input and captured-output growth are excluded.
38fn launch(command: &str, path: &Path, timeout: Duration) -> Result<(), String> {
39    let mut child = Command::new(command)
40        .arg(path)
41        .stdin(Stdio::null())
42        .stdout(Stdio::null())
43        .stderr(Stdio::null())
44        .spawn()
45        .map_err(|error| format!("{command}: {error}"))?;
46    let started = Instant::now();
47    loop {
48        match child.try_wait() {
49            Ok(Some(status)) if status.success() => return Ok(()),
50            Ok(Some(status)) => return Err(format!("{command} failed: {status}")),
51            Err(error) => {
52                stop(&mut child);
53                return Err(format!("{command}: {error}"));
54            }
55            Ok(None) => {}
56        }
57        if started.elapsed() >= timeout {
58            stop(&mut child);
59            return Err(format!("{command} timed out; generated artifact retained"));
60        }
61        std::thread::sleep(Duration::from_millis(10));
62    }
63}
64
65/// Reap only the owned launcher; an independently launched browser is outside this ownership.
66fn stop(child: &mut Child) {
67    let _ = child.kill();
68    // A kernel-uninterruptible process can delay this reap beyond the polling deadline.
69    let _ = child.wait();
70}

75    use std::{

82    static NEXT: AtomicUsize = AtomicUsize::new(0);
83    struct Fixture(PathBuf);
84    impl Fixture {
85        fn new(helper: &str) -> Self {
86            let path = std::env::temp_dir().join(format!(
87                "morrow-opener-{}-{}",
88                std::process::id(),
89                NEXT.fetch_add(1, Ordering::Relaxed)
90            ));
91            fs::create_dir(&path).unwrap();
92            let link = path.join(if helper == "timeout" {
93                "opener-timeout"
94            } else {
95                "opener"
96            });
97            let current = std::env::current_exe().unwrap();
98            let profile = current
99                .ancestors()
100                .find(|path| {
101                    path.file_name()
102                        .is_some_and(|name| name == "debug" || name == "release")
103                })
104                .unwrap();
105            let source = profile.join("morrow-opener-fixture");
106            assert!(
107                source.is_file(),
108                "build Rust supervisor fixtures before tests"
109            );
110            symlink(source, &link).unwrap();
111            Self(path)
112        }
113        fn command(&self) -> String {
114            self.0
115                .join(if self.0.join("opener-timeout").exists() {
116                    "opener-timeout"
117                } else {
118                    "opener"
119                })
120                .into_os_string()
121                .into_string()
122                .unwrap()
123        }
124    }
125    impl Drop for Fixture {
126        fn drop(&mut self) {
127            let _ = fs::remove_dir_all(&self.0);
128        }
129    }
130    #[test]
131    fn timeout_reaps_only_the_direct_launcher() {

134        assert!(error.contains("timed out"));

137    #[test]

139        let fixture = Fixture::new("capture");

141        launch(&fixture.command(), &path, Duration::from_secs(1)).unwrap();
142        assert_eq!(
