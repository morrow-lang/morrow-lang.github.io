1//! Checked documentation loads current module snapshots and protects every source identity.
2use morrow_compiler::{
3    check::editor::{self, FunctionInfo},
4    documentation::{self, InferredDocument, Output},
5    modules,
6};
7use std::{
8    collections::HashMap,
9    io::Write,
10    path::{Path, PathBuf},
11};

13/// One documented module with its checked source-local signatures.
14pub(super) struct Input {
15    pub(super) path: PathBuf,
16    pub(super) name: String,
17    pub(super) source: String,
18    pub(super) schemes: Vec<FunctionInfo>,
19}
20struct GraphBudget {
21    instances: usize,
22    bytes: usize,
23}
24
25/// Read and check every documented module once; snapshots retain each loaded source identity.
26pub(super) fn checked_inputs(
27    path: &Path,
28) -> Result<(Vec<Input>, HashMap<PathBuf, String>), String> {
29    let mut inputs = inputs(path)?;
30    let mut snapshots: HashMap<_, _> = inputs
31        .iter()
32        .map(|input| (input.path.clone(), input.source.clone()))
33        .collect();
34    let mut budget = GraphBudget {
35        instances: 0,
36        bytes: snapshots.values().map(String::len).sum(),
37    };
38    for input in &mut inputs {
39        check(input, &mut snapshots, &mut budget)?;
40    }
41    Ok((inputs, snapshots))
42}
43
44/// Render all checked modules before printing or atomically replacing any destination.
45pub(super) fn run(path: &Path, output: Option<&Path>, format: Output) -> Result<u8, String> {
46    let (inputs, snapshots) = checked_inputs(path)?;
47    let title = path.canonicalize().map_err(|e| e.to_string())?;
48    let title = title.file_name().unwrap_or_default().to_string_lossy();
49    let text = if path.is_dir() {
50        let documents: Vec<_> = inputs
51            .iter()
52            .map(|i| InferredDocument {
53                path: &i.name,
54                source: &i.source,
55                schemes: &i.schemes,
56            })
57            .collect();
58        documentation::render_inferred_project(&documents, &title, format)
59    } else {
60        documentation::render_with_schemes(
61            &inputs[0].source,
62            &title,
63            format,
64            Some(&inputs[0].schemes),
65        )
66    }
67    .map_err(|error| error.message)?;
68    if let Some(output) = output {
69        for source in snapshots.keys() {
70            super::super::output_destination(source, output)?;
71        }
72        super::super::emit_file(&inputs[0].path, output, &text)?;
73    } else {
74        std::io::stdout()
75            .lock()
76            .write_all(text.as_bytes())
77            .map_err(|error| error.to_string())?;
78    }
79    Ok(0)
80}

82/// Capture all documented files before checking imports, retaining deterministic display paths.
83fn inputs(root: &Path) -> Result<Vec<Input>, String> {
84    let files = super::sources(root)?;
85    let mut inputs = Vec::new();
86    let mut bytes = 0;
87    for path in files {
88        let source = super::read_source(&path, &mut bytes)?;
89        let name = if root.is_dir() {
90            path.strip_prefix(root)
91                .unwrap_or(&path)
92                .components()
93                .map(|c| c.as_os_str().to_string_lossy())
94                .collect::<Vec<_>>()
95                .join("/")
96        } else {
97            path.file_name()
98                .unwrap_or_default()
99                .to_string_lossy()
100                .into_owned()
101        };
102        let path = modules::source_identity(&path).map_err(|e| e.message)?;
103        inputs.push(Input {
104            path,
105            name,
106            source,
107            schemes: Vec::new(),
108        });
109    }
110    Ok(inputs)
111}

113/// Check each graph once, cache imported snapshots and relocate only the documented source's facts.
114fn check(
115    input: &mut Input,
116    snapshots: &mut HashMap<PathBuf, String>,
117    budget: &mut GraphBudget,
118) -> Result<(), String> {
119    let loaded = modules::load_documentation_sources(&input.path, snapshots)
120        .map_err(|e| format!("{}: {}", input.path.display(), e.message))?;
121    let source = loaded
122        .sources()
123        .find(|s| s.path == input.path)
124        .ok_or("missing documented source identity")?;
125    let start = source.start;
126    let end = start + source.text.len();
127    for source in loaded.sources() {
128        budget.instances += 1;
129        budget.bytes += source.text.len();
130        if budget.instances > 1024 || budget.bytes > 16 * 1024 * 1024 {
131            return Err("inferred documentation graph budget exceeded".into());
132        }
133        if !snapshots.contains_key(source.path) {
134            budget.bytes += source.text.len();
135            if snapshots.len() == 1024 || budget.bytes > 16 * 1024 * 1024 {
136                return Err("inferred documentation snapshot budget exceeded".into());
137            }
138            snapshots.insert(source.path.to_path_buf(), source.text.into());
139        }
140    }
141    let schemes = editor::function_schemes(&loaded.program).map_err(|error| {
142        let source = loaded
143            .sources()
144            .find(|s| s.start <= error.span.start && error.span.end <= s.start + s.text.len());
145        let (path, line) = source
146            .map(|s| {
147                (
148                    s.path,
149                    1 + s.text[..error.span.start - s.start]
150                        .bytes()
151                        .filter(|b| *b == b'\n')
152                        .count(),
153                )
154            })
155            .unwrap_or((&input.path, 1));
156        format!("{}:{line}: error: {}", path.display(), error.message)
157    })?;
158    input.schemes = schemes
159        .into_iter()
160        .filter(|info| start <= info.origin.start && info.origin.end <= end)
161        .map(|mut info| {
162            info.origin.start -= start;
163            info.origin.end -= start;
164            info
165        })
166        .collect();
167    Ok(())
168}
