1//! Read-only process memory observations; no Morrow heap access is required.
2
3#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
4pub struct MemoryUsage {
5    /// OS resident set estimate for the entire process, not just Morrow heaps.
6    pub resident_bytes: Option<u64>,
7    /// OS high-water resident set size for this process.
8    pub peak_resident_bytes: Option<u64>,
9}
10
11/// Sample independently observed current and peak RSS in bytes.
12///
13/// macOS uses `PROC_PIDTASKINFO`; Linux reads at most 257 bytes from statm.
14/// Linux statm accounting is approximate. These are neither allocation totals
15/// nor per-worker metrics, and the two readings are not an atomic snapshot.
16/// Missing OS support, denied access, or invalid data yields `None` per reading.
17pub fn process_memory() -> MemoryUsage {
18    MemoryUsage {
19        resident_bytes: resident(),
20        peak_resident_bytes: peak(),
21    }
22}
23
24#[cfg(any(target_os = "linux", test))]
25fn statm_resident(reader: impl std::io::Read, page_bytes: u64) -> Option<u64> {
26    use std::io::Read;
27    let mut bytes = Vec::new();
28    reader.take(257).read_to_end(&mut bytes).ok()?;
29    if bytes.len() > 256 || page_bytes == 0 {
30        return None;
31    }
32    let text = std::str::from_utf8(&bytes).ok()?;
33    let mut fields = text.split_ascii_whitespace();
34    let mut resident_pages = 0;
35    for index in 0..7 {
36        let field = fields.next()?;
37        if !field.bytes().all(|byte| byte.is_ascii_digit()) {
38            return None;
39        }
40        let value: u64 = field.parse().ok()?;
41        if index == 1 {
42            resident_pages = value;
43        }
44    }
45    if fields.next().is_some() {
46        return None;
47    }
48    resident_pages.checked_mul(page_bytes)
49}

51#[cfg(any(target_os = "linux", target_os = "macos", test))]
52fn peak_bytes(value: i128, unit_bytes: u64) -> Option<u64> {
53    if unit_bytes == 0 {
54        return None;
55    }
56    u64::try_from(value).ok()?.checked_mul(unit_bytes)
57}

59#[cfg(target_os = "linux")]
60fn resident() -> Option<u64> {
61    use std::os::unix::fs::OpenOptionsExt;
62    // SAFETY: sysconf takes a constant selector and no pointers or shared state.
63    let page_bytes = u64::try_from(unsafe { libc::sysconf(libc::_SC_PAGESIZE) }).ok()?;
64    let file = std::fs::OpenOptions::new()
65        .read(true)
66        .custom_flags(libc::O_NOFOLLOW | libc::O_NONBLOCK)
67        .open("/proc/self/statm")
68        .ok()?;
69    if !file.metadata().ok()?.is_file() {
70        return None;
71    }
72    statm_resident(file, page_bytes)
73}

75#[cfg(target_os = "macos")]
76fn resident() -> Option<u64> {
77    let mut info = std::mem::MaybeUninit::<libc::proc_taskinfo>::uninit();
78    let size = i32::try_from(std::mem::size_of::<libc::proc_taskinfo>()).ok()?;
79    // SAFETY: libc supplies the target ABI layout. The exclusive, aligned output
80    // buffer lives through the call; it is read only after an exact-size success.
81    let written = unsafe {
82        libc::proc_pidinfo(
83            libc::getpid(),
84            libc::PROC_PIDTASKINFO,
85            0,
86            info.as_mut_ptr().cast(),
87            size,
88        )
89    };
90    if written != size {
91        return None;
92    }
93    // SAFETY: successful PROC_PIDTASKINFO fills the complete struct above.
94    Some(unsafe { info.assume_init() }.pti_resident_size)
95}
96
97#[cfg(any(target_os = "linux", target_os = "macos"))]
98fn peak() -> Option<u64> {
99    let mut usage = std::mem::MaybeUninit::<libc::rusage>::uninit();
100    // SAFETY: getrusage writes the target-defined struct to an aligned, exclusive
101    // buffer. RUSAGE_SELF queries this process without retaining the pointer.
102    if unsafe { libc::getrusage(libc::RUSAGE_SELF, usage.as_mut_ptr()) } != 0 {
103        return None;
104    }
105    // SAFETY: getrusage succeeded and initialized the output struct.
106    let value = unsafe { usage.assume_init() }.ru_maxrss;
107    // Darwin's current getrusage(2) manual specifies bytes; Linux specifies KiB.
108    let unit_bytes = if cfg!(target_os = "linux") { 1024 } else { 1 };
109    peak_bytes(i128::from(value), unit_bytes)
110}

112#[cfg(not(any(target_os = "linux", target_os = "macos")))]
113fn resident() -> Option<u64> {
114    None
115}
116#[cfg(not(any(target_os = "linux", target_os = "macos")))]
117fn peak() -> Option<u64> {
118    None

121#[cfg(test)]
122mod tests {
123    use super::*;
124
125    #[test]
126    fn statm_uses_resident_pages_and_checked_units() {
127        assert_eq!(
128            statm_resident(&b"1000 23 7 2 0 10 0\n"[..], 4096),
129            Some(94_208)
130        );
131        assert_eq!(
132            statm_resident(&b"1000 23 7 2 0 10 0\n"[..], 16384),
133            Some(376_832)
134        );
135        for input in [
136            "",
137            "1 2",
138            "1 -1 0 0 0 0 0",
139            "1 +1 0 0 0 0 0",
140            "1 2 0 0 0 0 0 8",
141            "1 18446744073709551615 0 0 0 0 0",
142        ] {
143            assert_eq!(statm_resident(input.as_bytes(), 4096), None, "{input}");
144        }
145        assert_eq!(statm_resident(&b"1 1 0 0 0 0 0"[..], 0), None);
146        assert_eq!(statm_resident(std::io::repeat(b'1'), 4096), None);
147        assert_eq!(statm_resident(&b"\xff 1 0 0 0 0 0"[..], 4096), None);
148    }
149
150    #[test]
151    fn peak_units_reject_negative_and_overflow() {
152        assert_eq!(peak_bytes(65_536, 1), Some(65_536));
153        assert_eq!(peak_bytes(65_536, 1024), Some(67_108_864));
154        assert_eq!(peak_bytes(-1, 1024), None);
155        assert_eq!(peak_bytes(i64::MAX.into(), 1024), None);
156        assert_eq!(peak_bytes(1, 0), None);
157    }
158
159    #[test]
160    #[cfg(any(target_os = "macos", target_os = "linux"))]
161    fn touched_memory_is_observed_in_an_isolated_process() {
162        const CHILD: &str = "MORROW_MEMORY_OBSERVATION_CHILD";
163        if std::env::var_os(CHILD).is_none() {
164            let output = std::process::Command::new(std::env::current_exe().unwrap())
165                .args([
166                    "--exact",
167                    "system::tests::touched_memory_is_observed_in_an_isolated_process",
168                    "--nocapture",
169                ])
170                .env(CHILD, "1")
171                .output()
172                .unwrap();
173            assert!(
174                output.status.success(),
175                "{}\n{}",
176                String::from_utf8_lossy(&output.stdout),
177                String::from_utf8_lossy(&output.stderr)
178            );
179            return;
180        }
181        let before = process_memory();
182        let mut memory = vec![0_u8; 32 * 1024 * 1024];
183        for (index, byte) in memory.iter_mut().enumerate() {
184            *byte = (index.wrapping_mul(37).wrapping_add(index / 4096)) as u8;
185        }
186        std::hint::black_box(&memory);
187        let after = process_memory();
188        eprintln!("memory before={before:?}, after={after:?}");
189        assert!(after.resident_bytes.unwrap() >= before.resident_bytes.unwrap() + 8 * 1024 * 1024);
190        assert!(after.peak_resident_bytes.unwrap() >= before.peak_resident_bytes.unwrap());
191        assert!(after.peak_resident_bytes.unwrap() >= after.resident_bytes.unwrap() / 2);
192        std::hint::black_box(&memory);
193    }
