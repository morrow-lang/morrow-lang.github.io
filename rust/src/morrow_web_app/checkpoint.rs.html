19const MAX_BYTES: usize = 8 * 1024 * 1024;
20type Identity = (u64, u64);
21fn identity(file: &File) -> io::Result<Identity> {
22    let metadata = file.metadata()?;
23    Ok((metadata.dev(), metadata.ino()))
24}

26#[derive(Clone, Serialize, Deserialize)]
27#[serde(deny_unknown_fields)]
28pub(super) struct State {
29    pub tasks: Vec<Task>,
30    pub next_id: i64,
31}
32impl State {
33    pub(super) fn change(&self) -> DomainChange {
34        DomainChange {
35            tasks: self.tasks.clone(),
36            next_id: self.next_id,
37            status: Status::Applied,
38        }
39    }
40    pub(super) fn native_json(&self) -> Result<String, Error> {
41        let tasks: Vec<_> = self
42            .tasks
43            .iter()
44            .map(|task| serde_json::json!({"id":task.id.0,"label":task.label,"done":task.done}))
45            .collect();
46        serde_json::to_string(&serde_json::json!({"tasks":tasks,"next_id":self.next_id}))
47            .map_err(|_| Error::Malformed)
48    }
49    fn validate(&self) -> io::Result<()> {
50        let mut ids = BTreeSet::new();
51        if self.next_id < 1
52            || self.tasks.len() > 100
53            || self.tasks.iter().any(|task| {
54                task.id.0 < 1
55                    || task.id.0 >= self.next_id
56                    || !ids.insert(task.id)
57                    || task.label.trim().is_empty()
58                    || task.label.len() > 256
59                    || task.label.chars().any(char::is_control)
60            })
61        {
62            return Err(io::Error::other("invalid room checkpoint"));
63        }
64        Ok(())
65    }
66}
67#[derive(Serialize, Deserialize)]
68#[serde(deny_unknown_fields)]
69struct Snapshot {
70    version: u8,
71    #[serde(deserialize_with = "unique_rooms")]
72    rooms: BTreeMap<String, State>,
73}
74fn unique_rooms<'de, D: Deserializer<'de>>(
75    deserializer: D,
76) -> Result<BTreeMap<String, State>, D::Error> {
77    struct Rooms;
78    impl<'de> Visitor<'de> for Rooms {
79        type Value = BTreeMap<String, State>;
80        fn expecting(&self, formatter: &mut std::fmt::Formatter) -> std::fmt::Result {
81            formatter.write_str("at most 128 unique room checkpoints")
82        }
83        fn visit_map<A: MapAccess<'de>>(self, mut map: A) -> Result<Self::Value, A::Error> {
84            let mut rooms = BTreeMap::new();
85            while let Some(name) = map.next_key::<String>()? {
86                if rooms.len() == 128 || rooms.contains_key(&name) {
87                    return Err(serde::de::Error::custom("duplicate room or room limit"));
88                }
89                rooms.insert(name, map.next_value()?);
90            }
91            Ok(rooms)
92        }
93    }
94    deserializer.deserialize_map(Rooms)
95}

97struct WriterLock {
98    file: File,
99}
100impl WriterLock {
101    fn acquire(file: File) -> io::Result<Self> {
102        file.try_lock().map_err(io::Error::other)?;
103        Ok(Self { file })
104    }
105}
106impl Drop for WriterLock {
107    fn drop(&mut self) {
108        // A concurrent fork retains this open file description until exec, even
109        // with CLOEXEC. Closing our descriptor alone would leave its flock held
110        // by that child. Release ownership explicitly, including load failures.
111        // Only a successfully acquired lock constructs this guard.
112        let _ = self.file.unlock();
113    }
114}

116pub(super) struct Store {
117    path: PathBuf,
118    directory: File,
119    lock: WriterLock,
120    // Retain the inode itself, so unlink/recreate cannot exploit inode reuse.
121    checkpoint: Option<File>,
122    rooms: BTreeMap<String, State>,
123}

125fn open_at(directory: &File, name: &CStr, flags: i32) -> io::Result<File> {
126    // SAFETY: retained directory FD and NUL-terminated basename remain live. New
127    // descriptor ownership transfers exactly once to File; final symlinks and
128    // blocking special files are never followed while inspecting replacements.
129    let descriptor = unsafe {
130        libc::openat(
131            directory.as_raw_fd(),
132            name.as_ptr(),
133            flags | libc::O_CLOEXEC | libc::O_NOFOLLOW | libc::O_NONBLOCK,
134            0o600,
135        )
136    };
137    if descriptor < 0 {
138        return Err(io::Error::last_os_error());
139    }
140    // SAFETY: openat returned a new owned descriptor on the successful path.
141    let file = unsafe { File::from_raw_fd(descriptor) };
142    if !file.metadata()?.is_file() {
143        return Err(io::Error::other("checkpoint entry is not a regular file"));
144    }
145    Ok(file)
146}
147fn entry_identity(directory: &File, name: &CStr) -> io::Result<Option<Identity>> {
148    match open_at(directory, name, libc::O_RDONLY) {
149        Ok(file) => identity(&file).map(Some),
150        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(None),
151        Err(error) => Err(error),
152    }
153}

155struct Temporary<'a> {
156    directory: &'a File,
157    name: CString,
158    file: File,
159    published: bool,
160}
161impl Temporary<'_> {
162    fn owns_entry(&self) -> io::Result<bool> {
163        Ok(entry_identity(self.directory, &self.name)? == Some(identity(&self.file)?))
164    }
165    fn publish(&mut self) -> io::Result<()> {
166        if !self.owns_entry()? {
167            return Err(io::Error::other("checkpoint temporary was replaced"));
168        }
169        // SAFETY: both names are basenames in the pinned, single-writer directory;
170        // the temporary was exclusively created and its identity was checked.
171        if unsafe {
172            libc::renameat(
173                self.directory.as_raw_fd(),
174                self.name.as_ptr(),
175                self.directory.as_raw_fd(),
176                c"rooms.json".as_ptr(),
177            )
178        } != 0
179        {
180            return Err(io::Error::last_os_error());
181        }
182        self.published = true;
183        Ok(())
184    }
185}
186impl Drop for Temporary<'_> {
187    fn drop(&mut self) {
188        if !self.published && self.owns_entry().unwrap_or(false) {
189            // SAFETY: unlink only the still-owned basename inside the retained
190            // private writer directory, without traversing a replacement parent.
191            unsafe {
192                libc::unlinkat(self.directory.as_raw_fd(), self.name.as_ptr(), 0);
193            }
194        }
195    }
196}

198impl Store {
199    pub(super) fn open(path: &Path) -> io::Result<Self> {
200        Self::open_scoped(path, None)
201    }
202    pub(super) fn open_scoped(path: &Path, placement: Option<&str>) -> io::Result<Self> {
203        if placement.is_some_and(|value| {
204            value.is_empty()
205                || value.len() > 256
206                || !value
207                    .bytes()
208                    .all(|byte| byte.is_ascii_alphanumeric() || b"/._-".contains(&byte))
209        }) {
210            return Err(io::Error::new(
211                io::ErrorKind::InvalidInput,
212                "invalid checkpoint placement identity",
213            ));
214        }
215        let mut builder = fs::DirBuilder::new();
216        builder.recursive(true);
217        {
218            use std::os::unix::fs::DirBuilderExt;
219            builder.mode(0o700);
220        }
221        builder.create(path)?;
222        let directory = OpenOptions::new()
223            .read(true)
224            .custom_flags(libc::O_DIRECTORY | libc::O_NOFOLLOW | libc::O_CLOEXEC)
225            .open(path)?;
226        let metadata = directory.metadata()?;
227        if metadata.mode() & 0o022 != 0 {
228            return Err(io::Error::other(
229                "checkpoint directory must not be writable by group or others",
230            ));
231        }
232        let path = path.canonicalize()?;
233        let lock = WriterLock::acquire(open_at(
234            &directory,
235            c"owner.lock",
236            libc::O_RDWR | libc::O_CREAT,
237        )?)?;
238        Self::load(path, directory, lock, placement)
239    }
240    fn load(
241        path: PathBuf,
242        directory: File,
243        lock: WriterLock,
244        placement: Option<&str>,
245    ) -> io::Result<Self> {
246        let (rooms, checkpoint) = match open_at(&directory, c"rooms.json", libc::O_RDONLY) {
247            Err(error) if error.kind() == io::ErrorKind::NotFound => (BTreeMap::new(), None),
248            Err(error) => return Err(error),
249            Ok(mut file) => {
250                let mut bytes = Vec::new();
251                Read::by_ref(&mut file)
252                    .take(MAX_BYTES as u64 + 1)
253                    .read_to_end(&mut bytes)?;
254                if bytes.len() > MAX_BYTES {
255                    return Err(io::Error::other("checkpoint exceeds 8 MiB"));
256                }
257                let snapshot: Snapshot =
258                    serde_json::from_slice(&bytes).map_err(io::Error::other)?;
259                if snapshot.version != 1 {
260                    return Err(io::Error::other("unsupported checkpoint version"));
261                }
262                for (room, state) in &snapshot.rooms {
263                    if room.is_empty() || room.len() > 128 || room.chars().any(char::is_control) {
264                        return Err(io::Error::other("invalid checkpoint room identity"));
265                    }
266                    state.validate()?;
267                }
268                (snapshot.rooms, Some(file))
269            }
270        };
271        let mut store = Self {
272            path,
273            directory,
274            lock,
275            checkpoint,
276            rooms,
277        };
278        store.check_ownership()?;
279        store.bind_placement(placement)?;
280        Ok(store)
281    }
282    fn bind_placement(&mut self, placement: Option<&str>) -> io::Result<()> {
283        let mut stored = String::new();
284        Read::by_ref(&mut self.lock.file)
285            .take(513)
286            .read_to_string(&mut stored)?;
287        if stored.len() > 512 {
288            return Err(io::Error::other(
289                "checkpoint placement metadata exceeds limit",
290            ));
291        }
292        // Keep the legacy on-disk header so acknowledged checkpoints survive the rename.
293        let requested =
294            placement.map(|identity| format!("fern-cluster-placement-v1\n{identity}\n"));
295        match requested {
296            None if stored.is_empty() => Ok(()),
297            Some(requested) if stored == requested => Ok(()),
298            Some(requested) if stored.is_empty() && self.rooms.is_empty() => {
299                // The existing retained file carries the exclusive writer lock.
300                // Pin placement before any room can be created or acknowledged.
301                self.check_ownership()?;
302                self.lock.file.write_all(requested.as_bytes())?;
303                self.lock.file.sync_all()?;
304                self.directory.sync_all()?;
305                self.check_ownership()
306            }
307            _ => Err(io::Error::other(
308                "checkpoint placement differs; explicit offline migration is required",
309            )),
310        }
311    }
312    fn check_ownership(&self) -> io::Result<()> {
313        let metadata = fs::symlink_metadata(&self.path)?;
314        if !metadata.is_dir()
315            || (metadata.dev(), metadata.ino()) != identity(&self.directory)?
316            || entry_identity(&self.directory, c"owner.lock")? != Some(identity(&self.lock.file)?)
317            || entry_identity(&self.directory, c"rooms.json")?
318                != self.checkpoint.as_ref().map(identity).transpose()?
319        {
320            return Err(io::Error::other(
321                "checkpoint directory, lock or state was replaced",
322            ));
323        }
324        Ok(())
325    }
326    pub(super) fn get(&self, room: &str) -> Option<&State> {
327        self.rooms.get(room)
328    }
329    pub(super) fn commit(&mut self, room: &str, change: &DomainChange) -> io::Result<()> {
330        self.commit_with_sync(room, change, File::sync_all)
331    }
332    fn commit_with_sync(
333        &mut self,
334        room: &str,
335        change: &DomainChange,
336        sync_directory: impl FnOnce(&File) -> io::Result<()>,
337    ) -> io::Result<()> {
338        self.check_ownership()?;
339        let state = State {
340            tasks: change.tasks.clone(),
341            next_id: change.next_id,
342        };
343        state.validate()?;
344        let mut rooms = self.rooms.clone();
345        rooms.insert(room.into(), state);
346        if rooms.len() > 128 {
347            return Err(io::Error::other("checkpoint room limit"));
348        }
349        let bytes =
350            serde_json::to_vec(&Snapshot { version: 1, rooms }).map_err(io::Error::other)?;
351        if bytes.len() > MAX_BYTES {
352            return Err(io::Error::other("checkpoint exceeds 8 MiB"));
353        }
354        static NEXT: AtomicU64 = AtomicU64::new(0);
355        let name = CString::new(format!(
356            ".rooms-{}-{}.tmp",
357            std::process::id(),
358            NEXT.fetch_add(1, Ordering::Relaxed)
359        ))
360        .map_err(io::Error::other)?;
361        let file = open_at(
362            &self.directory,
363            &name,
364            libc::O_WRONLY | libc::O_CREAT | libc::O_EXCL,
365        )?;
366        let mut temporary = Temporary {
367            directory: &self.directory,
368            name,
369            file,
370            published: false,
371        };
372        temporary.file.write_all(&bytes)?;
373        temporary.file.sync_all()?;
374        let checkpoint = temporary.file.try_clone()?;
375        self.check_ownership()?;
376        temporary.publish()?;
377        // A failed directory fsync is uncertain: the new file may be durable.
378        // Remember its identity but retain the last acknowledged in-memory state;
379        // gateway recovery resets incarnation instead of reporting false success.
380        self.checkpoint = Some(checkpoint);
381        sync_directory(&self.directory)?;
382        self.rooms.insert(
383            room.into(),
384            State {
385                tasks: change.tasks.clone(),
386                next_id: change.next_id,
387            },
388        );
389        Ok(())
390    }

417    #[test]
418    fn releasing_the_writer_unlocks_descriptors_inherited_by_a_child() {
419        let directory = Directory::new();
420        let placement = Some("cluster-a/node-a/manifest-one");
421        let store = Store::open_scoped(&directory.0, placement).unwrap();
422        // dup and fork retain the same open file description. Keeping this clone
423        // open models a concurrently spawned child before its CLOEXEC cleanup.
424        let inherited = store.lock.file.try_clone().unwrap();
425        assert!(Store::open_scoped(&directory.0, placement).is_err());
426        drop(store);
427        let replacement = Store::open_scoped(&directory.0, placement)
428            .expect("a child descriptor must not extend the previous writer's ownership");
429        assert!(Store::open_scoped(&directory.0, placement).is_err());
430        drop(inherited);
431        assert!(Store::open_scoped(&directory.0, placement).is_err());
432        drop(replacement);
433        Store::open_scoped(&directory.0, placement).unwrap();
434    }

437    fn failed_initialization_unlocks_inherited_descriptors_before_and_after_store_creation() {
438        for invalid_checkpoint in [true, false] {
439            let directory = Directory::new();
440            let placement = Some("cluster-a/node-a/manifest-one");
441            drop(Store::open_scoped(&directory.0, placement).unwrap());

448            }

453            .unwrap();
454            let inherited = lock.file.try_clone().unwrap();
455            let error = Store::load(

556    #[test]
557    fn directory_sync_failure_reports_uncertainty_without_losing_acknowledged_state() {
558        let directory = Directory::new();
559        let mut store = Store::open(&directory.0).unwrap();
560        let first = DomainChange {
561            tasks: vec![Task {
562                id: Decimal(1),
563                label: "acknowledged".into(),
564                done: false,
565            }],
566            next_id: 2,
567            status: Status::Applied,
568        };
569        store.commit("garden", &first).unwrap();
570        let mut tasks = first.tasks.clone();
571        tasks.push(Task {
572            id: Decimal(2),
573            label: "uncertain".into(),
574            done: false,
575        });
576        let next = DomainChange {
577            tasks,
578            next_id: 3,
579            status: Status::Applied,
580        };
581        assert!(
582            store
583                .commit_with_sync("garden", &next, |_| Err(io::Error::other(
584                    "injected directory sync failure"
585                )))
586                .is_err()
587        );
588        assert_eq!(store.get("garden").unwrap().tasks, first.tasks);
589        store.check_ownership().unwrap();
590        drop(store);
591        let reopened = Store::open(&directory.0).unwrap();
592        // rename completed before the injected failure. Recovery may observe this
593        // unacknowledged update; a fresh incarnation prevents automatic replay.
594        assert_eq!(reopened.get("garden").unwrap().tasks, next.tasks);
595        assert_eq!(reopened.get("garden").unwrap().tasks[0], first.tasks[0]);
596    }
