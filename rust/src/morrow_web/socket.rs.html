11/// Select once at the authenticated upgrade; payload bytes never select a parser.
12#[derive(Clone, Copy)]
13pub(crate) enum Format {
14    Json,
15    Protobuf,
16}
17impl Format {
18    pub fn negotiate(protocols: &[&str]) -> Option<Self> {
19        if protocols.contains(&morrow_web_protocol::binary::SUBPROTOCOL) {
20            Some(Self::Protobuf)
21        } else if protocols.contains(&"morrow.live.v1") {
22            Some(Self::Json)
23        } else {
24            None
25        }
26    }
27    pub fn name(self) -> &'static str {
28        match self {
29            Self::Json => "morrow.live.v1",
30            Self::Protobuf => morrow_web_protocol::binary::SUBPROTOCOL,
31        }
32    }
33    fn decode(self, message: Message) -> Result<ClientMessage, Error> {
34        match (self, message) {
35            (Self::Json, Message::Text(text)) => decode(text.as_bytes()),
36            (Self::Protobuf, Message::Binary(bytes)) => {
37                morrow_web_protocol::binary::decode_client(&bytes)
38            }
39            _ => Err(Error::Malformed),
40        }
41    }
42    fn encode(self, message: &ServerMessage) -> Result<Message, Error> {
43        match self {
44            Self::Json => Ok(Message::Text(
45                String::from_utf8(encode(message)?)
46                    .map_err(|_| Error::Malformed)?
47                    .into(),
48            )),
49            Self::Protobuf => Ok(Message::Binary(
50                morrow_web_protocol::binary::encode_server(message)?.into(),
51            )),
52        }
53    }
54}

56enum Connection {
57    Local { route: owner::Route, id: String },
58    Remote(crate::peer::Remote),
59}
60impl Connection {
61    fn send(&self, app: &App, principal: &str, message: ClientMessage) -> Result<(), ()> {
62        match self {
63            Self::Local { route, id } => app.requests.try_send(Request::Command {
64                route: *route,
65                principal: principal.into(),
66                connection: id.clone(),
67                message,
68            }),
69            Self::Remote(remote) => remote.send(message),
70        }
71    }
72    async fn disconnect(self, app: &App) -> Result<(), ()> {
73        match self {
74            Self::Local { route, id } => app.requests.disconnect(route, id).await,
75            Self::Remote(remote) => {
76                remote.disconnect().await;
77                Ok(())
78            }
79        }
80    }
81    fn retire(self, app: &App) {
82        if let Self::Local { route, id } = self {
83            let _ = app.requests.try_send(Request::Disconnect {
84                route,
85                connection: id,
86                reply: None,
87            });
88        }
89    }
90}

92async fn join(
93    app: &App,
94    auth: &Authentication,
95    room: String,
96    resume: Option<String>,
97    outcomes: mpsc::Sender<ServerMessage>,
98    snapshots: watch::Sender<Option<morrow_web_protocol::Snapshot>>,
99) -> Result<(Connection, morrow_web_protocol::Connected), Error> {
100    if let Some(cluster) = &app.cluster
101        && let Some(target) = cluster.remote_owner(&room)?
102    {
103        let (remote, connected) = cluster
104            .join(&target, auth.clone(), room, outcomes, snapshots)
105            .await?;
106        return Ok((Connection::Remote(remote), connected));
107    }
108    let route = app.requests.route(&room);
109    let (reply, response) = oneshot::channel();
110    app.requests
111        .try_send(Request::Join {
112            capability: auth.capability(),
113            room,
114            resume,
115            outcomes,
116            snapshots,
117            reply,
118        })
119        .map_err(|_| Error::ConnectionLimit)?;
120    let connected = response.await.map_err(|_| Error::Offline)??;
121    Ok((
122        Connection::Local {
123            route,
124            id: connected.connection.clone(),
125        },
126        connected,
127    ))
128}

130async fn write(socket: &mut WebSocket, message: Message, limit: Duration) -> bool {
131    matches!(
132        tokio::time::timeout(limit, socket.send(message)).await,
133        Ok(Ok(()))
134    )
135}
136async fn publish(
137    socket: &mut WebSocket,
138    format: Format,
139    message: ServerMessage,
140    limit: Duration,
141) -> bool {
142    let Ok(message) = format.encode(&message) else {
143        return false;
144    };
145    write(socket, message, limit).await
146}

148/// Coalescing may skip the exact rollover snapshot. Compare resource identity,
149/// rather than revision, so every browser observes the reset before later state.
150fn publication(incarnation: &mut Option<String>, message: ServerMessage) -> ServerMessage {
151    match message {
152        ServerMessage::Connected(connected) => {
153            *incarnation = Some(connected.snapshot.incarnation.clone());
154            ServerMessage::Connected(connected)
155        }
156        ServerMessage::Reset(snapshot) => {
157            *incarnation = Some(snapshot.incarnation.clone());
158            ServerMessage::Reset(snapshot)
159        }
160        ServerMessage::Snapshot(snapshot) => {
161            let changed = incarnation
162                .as_ref()
163                .is_some_and(|old| old != &snapshot.incarnation);
164            *incarnation = Some(snapshot.incarnation.clone());
165            if changed {
166                ServerMessage::Reset(snapshot)
167            } else {
168                ServerMessage::Snapshot(snapshot)
169            }
170        }
171        other => other,
172    }
173}
174pub(crate) async fn run(
175    mut socket: WebSocket,
176    app: App,
177    mut auth: Authentication,
178    _permit: OwnedSemaphorePermit,
179    format: Format,
180) {
181    let mut connection: Option<Connection> = None;
182    let mut incarnation = None;
183    let (_, mut outcomes) = mpsc::channel(owner::OUTCOMES);
184    let (_, mut snapshots) = watch::channel(None);
185    let mut heartbeat = tokio::time::interval(Duration::from_secs(20));
186    let join_deadline = tokio::time::Instant::now() + app.config.handshake_timeout;
187    let join_timeout = tokio::time::sleep_until(join_deadline);
188    tokio::pin!(join_timeout);
189    let mut active = Instant::now();
190    let limit = app.config.write_timeout;
191    loop {
192        if *auth.revoked.borrow() {
193            break;
194        }
195        tokio::select! {
196            biased;
197            _ = auth.revoked.changed() => break,
198            // Control frames and failed joins cannot renew admission indefinitely.
199            _ = &mut join_timeout, if connection.is_none() => break,
200            message = outcomes.recv(), if connection.is_some() => {
201                let Some(message) = message else { break; };
202                if !publish(&mut socket, format, publication(&mut incarnation, message), limit).await { break; }
203            }
204            changed = snapshots.changed(), if connection.is_some() => {
205                if changed.is_err() { break; }
206                let snapshot = snapshots.borrow_and_update().clone();
207                if let Some(snapshot) = snapshot
208                    && !publish(&mut socket, format, publication(&mut incarnation, ServerMessage::Snapshot(snapshot)), limit).await { break; }
209            }
210            incoming = socket.recv() => {
211                let Some(Ok(incoming)) = incoming else { break; };
212                active = Instant::now();
213                match incoming {
214                    payload @ (Message::Text(_) | Message::Binary(_)) => match format.decode(payload) {
215                        Ok(ClientMessage::Join { room, resume_namespace }) => {
216                            let response_deadline = tokio::time::Instant::now() + Duration::from_secs(2);
217                            let response_deadline = if connection.is_none() {
218                                response_deadline.min(join_deadline)
219                            } else { response_deadline };
220                            if let Some(old_connection) = connection.take()
221                                && !matches!(tokio::time::timeout_at(response_deadline, old_connection.disconnect(&app)).await, Ok(Ok(()))) { break; }
222                            let (outcome_tx, outcome_rx) = mpsc::channel(owner::OUTCOMES);
223                            let (snapshot_tx, snapshot_rx) = watch::channel(None);
224                            let joining = join(&app, &auth, room, resume_namespace, outcome_tx, snapshot_tx);
225                            match tokio::time::timeout_at(response_deadline, joining).await {
226                                Ok(Ok((joined, connected))) => {
227                                    connection = Some(joined);
228                                    outcomes = outcome_rx;
229                                    snapshots = snapshot_rx;
230                                    if !publish(&mut socket, format, publication(&mut incarnation, ServerMessage::Connected(connected)), limit).await { break; }
231                                }
232                                Ok(Err(error)) => { if !publish(&mut socket, format, ServerMessage::Error(error), limit).await { break; } }
233                                _ => break,
234                            }
235                        }
236                        Ok(message @ ClientMessage::Command(_)) => {
237                            let Some(connection) = &connection else {
238                                publish(&mut socket, format, ServerMessage::Error(Error::Unauthorized), limit).await;
239                                break;
240                            };
241                            if connection.send(&app, &auth.token, message).is_err() { break; }
242                        }
243                        Err(error) => { publish(&mut socket, format, ServerMessage::Error(error), limit).await; break; }
244                    },
245                    Message::Ping(bytes) => { if !write(&mut socket, Message::Pong(bytes), limit).await { break; } }
246                    Message::Pong(_) => (),
247                    Message::Close(_) => break,
248                }
249            }
250            _ = heartbeat.tick() => {
251                if active.elapsed() > Duration::from_secs(60) { break; }
252                if !write(&mut socket, Message::Ping(Vec::new().into()), limit).await { break; }
253            }
254        }
255    }
256    if let Some(connection) = connection {
257        connection.retire(&app);
258    }
259    let _ = write(&mut socket, Message::Close(None), limit).await;
260}

290    #[test]
291    fn coalesced_incarnation_change_resets_browser_and_never_replays_pending_work() {
292        let mut incarnation = None;
293        let first = connected("before");
294        publication(&mut incarnation, ServerMessage::Connected(first.clone()));
295        let mut browser = Client::new(first).unwrap();
296        browser
297            .submit(Mutation::Add {
298                label: "uncertain".into(),
299            })
300            .unwrap();
301        let latest = snapshot("after", 3);
302        let ServerMessage::Reset(reset) =
303            publication(&mut incarnation, ServerMessage::Snapshot(latest.clone()))
304        else {
305            panic!("coalesced owner rollover must be published as Reset");
306        };
307        assert_eq!(reset, latest);
308        browser.accept_snapshot(reset, true).unwrap();
309        assert!(browser.uncertain());
310        assert!(browser.pending().is_none());
311        assert_eq!(browser.snapshot(), &latest);
312        assert_eq!(
313            publication(
314                &mut incarnation,
315                ServerMessage::Snapshot(snapshot("after", 4))
316            ),
317            ServerMessage::Snapshot(snapshot("after", 4))
318        );
319    }

321    #[test]
322    fn explicit_reset_and_new_join_replace_the_publication_incarnation() {
323        let mut incarnation = None;
324        publication(
325            &mut incarnation,
326            ServerMessage::Connected(connected("first")),
327        );
328        let reset = ServerMessage::Reset(snapshot("second", 0));
329        assert_eq!(publication(&mut incarnation, reset.clone()), reset);
330        assert_eq!(incarnation.as_deref(), Some("second"));
331        let update = ServerMessage::Snapshot(snapshot("second", 1));
332        assert_eq!(publication(&mut incarnation, update.clone()), update);
333        let join = ServerMessage::Connected(connected("third"));
334        assert_eq!(publication(&mut incarnation, join.clone()), join);
335        assert_eq!(incarnation.as_deref(), Some("third"));
336        let update = ServerMessage::Snapshot(snapshot("third", 1));
337        assert_eq!(publication(&mut incarnation, update.clone()), update);
338    }
