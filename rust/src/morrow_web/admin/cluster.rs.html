1//! A local observation of configured ownership and currently authenticated peer streams.
2use super::{escape, peer};
3use std::fmt::Write;
4
5pub(super) fn render(body: &mut String, cluster: Option<&peer::Observation>) {
6    let Some(cluster) = cluster else {
7        body.push_str("<section class=\"panel cluster-panel\" aria-label=\"Server connections\"><div class=section-heading><div><p class=eyebrow>SERVER CONNECTIONS</p><h2>Standalone</h2></div><span class=badge>One server</span></div><p>Rooms run on this server. No peer connections are configured.</p></section>");
8        return;
9    };
10    let _ = write!(
11        body,
12        "<section class=\"panel cluster-panel\" aria-label=\"Server connections\"><div class=section-heading><div><p class=eyebrow>SERVER CONNECTIONS</p><h2>Node {}</h2></div><span class=badge>Fixed room ownership</span></div><dl class=cluster-grid>",
13        escape(&cluster.node)
14    );
15    for (name, value) in [
16        ("Configured nodes", cluster.configured_nodes.to_string()),
17        ("Connected peers", cluster.connected_nodes.to_string()),
18        (
19            "Inbound streams",
20            format!(
21                "{} / {}",
22                cluster.inbound_streams, cluster.stream_limit_per_direction
23            ),
24        ),
25        (
26            "Outbound streams",
27            format!(
28                "{} / {}",
29                cluster.outbound_streams, cluster.stream_limit_per_direction
30            ),
31        ),
32        ("Forwarded commands", cluster.forwarded_commands.to_string()),
33        (
34            "Rejected connections",
35            cluster.rejected_connections.to_string(),
36        ),
37    ] {
38        let _ = write!(
39            body,
40            "<div><dt>{name}</dt><dd>{}</dd></div>",
41            escape(&value)
42        );
43    }
44    body.push_str("</dl><p class=cluster-note>Configured nodes includes this server. Connected peers counts distinct authenticated remote nodes with an open stream; idle nodes may have no stream. Stream counts include connection setup. Forwarded commands counts outgoing transport writes, not confirmed changes.</p><p class=cluster-note>Unavailable owners keep their rooms unavailable. This connection snapshot does not establish cluster health, replication or automatic failover.</p></section>");
45}
