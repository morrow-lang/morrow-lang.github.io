21/// Explicit preview authentication and admission settings.
22#[derive(Clone)]
23pub struct Config {
24    /// Optional authenticated fixed-membership routing. Omit for a standalone node.
25    pub cluster: Option<morrow_cluster::NodeSettings>,
26    /// Fixed number of pinned Morrow room workers. Room state never migrates live.
27    pub workers: usize,
28    /// Optional single-writer checkpoint directory. Omit for ephemeral rooms.
29    pub data_dir: Option<std::path::PathBuf>,
30    pub origin: String,
31    /// Optional shared login secret. Empty disables POST `/session`; GET still issues a cookie.
32    pub access_key: String,
33    /// Serve `/admin` and the footer link. Defaults on; `MORROW_WEB_ADMIN=0` hides it.
34    pub admin: bool,
35    pub max_sessions: usize,
36    pub session_ttl: std::time::Duration,
37    pub write_timeout: std::time::Duration,
38    pub max_tcp_connections: usize,
39    pub handshake_timeout: std::time::Duration,
40    pub limits: morrow_web_protocol::Limits,
41}
42impl Config {
43    pub fn new(origin: String, access_key: String) -> Self {
44        Self {
45            cluster: None,
46            workers: std::thread::available_parallelism().map_or(1, |count| count.get().min(4)),
47            data_dir: None,
48            origin,
49            access_key,
50            admin: true,
51            max_sessions: 256,
52            session_ttl: std::time::Duration::from_secs(3600),
53            write_timeout: std::time::Duration::from_secs(2),
54            max_tcp_connections: 512,
55            handshake_timeout: std::time::Duration::from_secs(10),
56            limits: Default::default(),
57        }
58    }
59}

75/// Build a router inside a Tokio runtime. Serve it through [`BoundedListener`]
76/// (as [`serve`] does) to enforce admission before HTTP parsing. GET `/session`
77/// issues a same-origin cookie; POST `/session` remains available when an access
78/// key is configured. Applications need their own resource policy.
79pub fn router(config: Config, assets: Assets) -> Result<Router, std::io::Error> {
80    let origin: Uri = config.origin.parse().map_err(std::io::Error::other)?;
81    if !matches!(origin.scheme_str(), Some("http" | "https"))
82        || origin.authority().is_none()
83        || origin
84            .path_and_query()
85            .is_some_and(|path| path.as_str() != "/")
86        || config.origin.ends_with('/')
87        || config.origin.contains('@')
88        || !(config.access_key.is_empty() || (16..=256).contains(&config.access_key.len()))
89        || !(1..=1024).contains(&config.max_sessions)
90        || config.session_ttl.is_zero()
91        || config.session_ttl > Duration::from_secs(86_400)
92        || config.write_timeout.is_zero()
93        || config.write_timeout > Duration::from_secs(30)
94        || !(1..=2048).contains(&config.max_tcp_connections)
95        || config.handshake_timeout.is_zero()
96        || config.handshake_timeout > Duration::from_secs(30)
97        || !(1..=32).contains(&config.workers)
98    {
99        return Err(std::io::Error::new(
100            std::io::ErrorKind::InvalidInput,
101            "invalid origin, key or transport limits",
102        ));
103    }
104    let sockets = Arc::new(Semaphore::new(config.limits.max_connections));
105    let requests = owner::start(config.clone())?;
106    let (cluster, cluster_stop) = match config.cluster.clone() {
107        Some(settings) => {
108            let (cluster, stop) = peer::Cluster::start(settings, requests.clone())?;
109            (Some(cluster), Some(stop))
110        }
111        None => (None, None),
112    };
113    let admin = config.admin;
114    let app = App {
115        config: Arc::new(config),
116        requests,
117        sockets,
118        http: Arc::new(Semaphore::new(512)),
119        assets,
120        system: Arc::new(admin::System::new()),
121        cluster,
122        _cluster_stop: cluster_stop,
123    };
124    let mut router = Router::new()
125        .route("/session", get(session).post(login))
126        .route("/logout", post(logout))
127        .route("/ws", get(upgrade))
128        .route("/health", get(|| async { "morrow-web preview\n" }));
129    if admin {
130        router = router
131            .route("/admin", get(admin::page))
132            .route("/admin/", get(admin::page))
133            .route("/admin/status", get(admin::status))
134            .route("/admin/style.css", get(admin::stylesheet));
135    }
136    Ok(router
137        .fallback(asset)
138        .layer(DefaultBodyLimit::max(1024))
139        .layer(middleware::from_fn_with_state(app.clone(), admission))
140        .with_state(app))
141}

143/// Serve embedded assets and the preview protocol with bounded TCP admission.
144pub async fn serve(
145    listener: tokio::net::TcpListener,
146    config: Config,
147    assets: Assets,
148) -> std::io::Result<()> {
149    let listener = BoundedListener::new(
150        listener,
151        config.max_tcp_connections,
152        config.handshake_timeout,
153    )?;
154    axum::serve(listener, router(config, assets)?).await
155}

157async fn admission(State(app): State<App>, request: Request, next: Next) -> Response {
158    let Ok(_permit) = app.http.clone().try_acquire_owned() else {
159        return StatusCode::SERVICE_UNAVAILABLE.into_response();
160    };
161    match tokio::time::timeout(Duration::from_secs(5), next.run(request)).await {
162        Ok(mut response) => {
163            if response.status() != StatusCode::SWITCHING_PROTOCOLS {
164                response
165                    .headers_mut()
166                    .insert(header::CONNECTION, HeaderValue::from_static("close"));
167            }
168            response.headers_mut().insert(
169                header::X_CONTENT_TYPE_OPTIONS,
170                HeaderValue::from_static("nosniff"),
171            );
172            response.headers_mut().insert(
173                header::REFERRER_POLICY,
174                HeaderValue::from_static("no-referrer"),
175            );
176            response
177        }
178        Err(_) => StatusCode::REQUEST_TIMEOUT.into_response(),
179    }
180}
181fn origin(headers: &HeaderMap, app: &App) -> Result<(), StatusCode> {
182    if headers.get(header::ORIGIN).and_then(|v| v.to_str().ok()) != Some(&app.config.origin) {
183        Err(StatusCode::FORBIDDEN)
184    } else {
185        Ok(())
186    }
187}
188fn session_cookie(headers: &HeaderMap) -> Result<Option<String>, StatusCode> {
189    let Some(values) = headers.get(header::COOKIE).and_then(|v| v.to_str().ok()) else {
190        return Ok(None);
191    };
192    let mut found = values
193        .split(';')
194        .filter_map(|v| v.trim().strip_prefix("morrow_session="));
195    let Some(token) = found.next() else {
196        return Ok(None);
197    };
198    if found.next().is_some() || token.len() != 64 || !token.bytes().all(|b| b.is_ascii_hexdigit())
199    {
200        return Err(StatusCode::UNAUTHORIZED);
201    }
202    Ok(Some(token.into()))
203}
204fn cookie(headers: &HeaderMap) -> Result<String, StatusCode> {
205    session_cookie(headers)?.ok_or(StatusCode::UNAUTHORIZED)
206}
207async fn response<T>(
208    app: &App,
209    request: owner::Request,
210    rx: oneshot::Receiver<Result<T, StatusCode>>,
211) -> Result<T, StatusCode> {
212    app.requests
213        .try_send(request)
214        .map_err(|_| StatusCode::SERVICE_UNAVAILABLE)?;
215    tokio::time::timeout(Duration::from_secs(2), rx)
216        .await
217        .map_err(|_| StatusCode::SERVICE_UNAVAILABLE)?
218        .map_err(|_| StatusCode::SERVICE_UNAVAILABLE)?
219}

229fn session_response(authentication: owner::Authentication) -> Response {
230    let mut response = Json(SessionResponse {
231        csrf: authentication.csrf,
232    })
233    .into_response();
234    response
235        .headers_mut()
236        .insert(header::CACHE_CONTROL, HeaderValue::from_static("no-store"));
237    response
238}
239fn issued_session(app: &App, auth: owner::Authentication) -> Result<Response, StatusCode> {
240    let cookie = format!(
241        "morrow_session={}; HttpOnly; SameSite=Strict; Path=/; Max-Age={}{}",
242        auth.token,
243        app.config.session_ttl.as_secs().max(1),
244        if app.config.origin.starts_with("https:") {
245            "; Secure"
246        } else {
247            ""
248        }
249    );
250    let mut response = session_response(auth);
251    response.headers_mut().insert(
252        header::SET_COOKIE,
253        cookie
254            .parse()
255            .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?,
256    );
257    Ok(response)
258}
259async fn open_session(app: &App) -> Result<owner::Authentication, StatusCode> {
260    let (reply, rx) = oneshot::channel();
261    response(app, owner::Request::Open { reply }, rx).await
262}
263async fn login(
264    State(app): State<App>,
265    headers: HeaderMap,
266    Json(credentials): Json<Credentials>,
267) -> Result<Response, StatusCode> {
268    origin(&headers, &app)?;
269    let (reply, rx) = oneshot::channel();
270    issued_session(
271        &app,
272        response(
273            &app,
274            owner::Request::Login {
275                key: credentials.access_key,
276                reply,
277            },
278            rx,
279        )
280        .await?,
281    )
282}
283async fn session(State(app): State<App>, headers: HeaderMap) -> Result<Response, StatusCode> {
284    // Same-origin fetch may omit Origin for GET. No CORS response is provided;
285    // cross-origin callers cannot read the nonce. Explicit foreign origins fail.
286    if headers.contains_key(header::ORIGIN) {
287        origin(&headers, &app)?;
288    }
289    match session_cookie(&headers)? {
290        Some(token) => {
291            let (reply, rx) = oneshot::channel();
292            match response(
293                &app,
294                owner::Request::Authenticate {
295                    token,
296                    csrf: None,
297                    reply,
298                },
299                rx,
300            )
301            .await
302            {
303                Ok(auth) => Ok(session_response(auth)),
304                Err(StatusCode::UNAUTHORIZED) => issued_session(&app, open_session(&app).await?),
305                Err(status) => Err(status),
306            }
307        }
308        None => issued_session(&app, open_session(&app).await?),
309    }
310}
311async fn logout(State(app): State<App>, headers: HeaderMap) -> Result<Response, StatusCode> {
312    origin(&headers, &app)?;
313    let csrf = headers
314        .get("x-morrow-csrf")
315        .and_then(|v| v.to_str().ok())
316        .ok_or(StatusCode::FORBIDDEN)?
317        .to_owned();
318    let (reply, rx) = oneshot::channel();
319    response(
320        &app,
321        owner::Request::Logout {
322            token: cookie(&headers)?,
323            csrf,
324            reply,
325        },
326        rx,
327    )
328    .await?;
329    let mut response = StatusCode::NO_CONTENT.into_response();
330    response.headers_mut().insert(
331        header::SET_COOKIE,
332        HeaderValue::from_static("morrow_session=; HttpOnly; SameSite=Strict; Path=/; Max-Age=0"),
333    );
334    Ok(response)
335}
336async fn upgrade(
337    State(app): State<App>,
338    headers: HeaderMap,
339    ws: WebSocketUpgrade,
340) -> Result<Response, StatusCode> {
341    origin(&headers, &app)?;
342    let protocols: Vec<_> = ws
343        .requested_protocols()
344        .filter_map(|v| v.to_str().ok())
345        .collect();
346    let format = socket::Format::negotiate(&protocols).ok_or(StatusCode::BAD_REQUEST)?;
347    let mut nonces = protocols
348        .iter()
349        .filter_map(|p| p.strip_prefix("morrow.csrf."));
350    let csrf = nonces.next().ok_or(StatusCode::FORBIDDEN)?.to_owned();
351    if nonces.next().is_some() || csrf.len() != 64 {
352        return Err(StatusCode::FORBIDDEN);
353    }
354    let permit = app
355        .sockets
356        .clone()
357        .try_acquire_owned()
358        .map_err(|_| StatusCode::TOO_MANY_REQUESTS)?;
359    let (reply, rx) = oneshot::channel();
360    let auth = response(
361        &app,
362        owner::Request::Authenticate {
363            token: cookie(&headers)?,
364            csrf: Some(csrf),
365            reply,
366        },
367        rx,
368    )
369    .await?;
370    Ok(ws
371        .protocols([format.name()])
372        .max_message_size(morrow_web_protocol::MAX_FRAME_BYTES)
373        .max_frame_size(morrow_web_protocol::MAX_FRAME_BYTES)
374        .read_buffer_size(8192)
375        .write_buffer_size(0)
376        .max_write_buffer_size(2 * morrow_web_protocol::MAX_FRAME_BYTES)
377        .on_upgrade(move |socket| socket::run(socket, app, auth, permit, format)))
378}
379async fn asset(State(app): State<App>, uri: Uri) -> Response {
380    let name = if uri.path() == "/" {
381        "index.html"
382    } else {
383        uri.path().trim_start_matches('/')
384    };
385    let content_type = match name {
386        "index.html" => "text/html; charset=utf-8",
387        "bootstrap.js" | "morrow_browser.js" | "worker.js" => "text/javascript; charset=utf-8",
388        "morrow_browser_bg.wasm" | "morrow_app.wasm" => "application/wasm",
389        "style.css" => "text/css; charset=utf-8",
390        _ => return StatusCode::NOT_FOUND.into_response(),
391    };
392    let Some((_, bytes)) = app.assets.iter().find(|(path, _)| *path == name) else {
393        return if name == "index.html" {
394            (
395                StatusCode::SERVICE_UNAVAILABLE,
396                "Browser assets are not embedded. Run cargo xtask web-build.\n",
397            )
398                .into_response()
399        } else {
400            StatusCode::NOT_FOUND.into_response()
401        };
402    };
403    if name == "index.html" && !app.config.admin {
404        let body = match std::str::from_utf8(bytes) {
405            Ok(html) => html
406                .replace(" <a href=\"/admin\">System dashboard</a>", "")
407                .into_bytes(),
408            Err(_) => bytes.to_vec(),
409        };
410        return asset_headers(
411            (
412                [
413                    (header::CONTENT_TYPE, content_type),
414                    (header::CACHE_CONTROL, "no-cache"),
415                ],
416                body,
417            )
418                .into_response(),
419            false,
420        );
421    }
422    asset_headers(
423        (
424            [
425                (header::CONTENT_TYPE, content_type),
426                (header::CACHE_CONTROL, "no-cache"),
427            ],
428            *bytes,
429        )
430            .into_response(),
431        name == "worker.js",
432    )
433}
434fn asset_headers(mut response: Response, worker: bool) -> Response {
435    response.headers_mut().insert(header::CONTENT_SECURITY_POLICY, HeaderValue::from_static("default-src 'self'; script-src 'self' 'wasm-unsafe-eval'; style-src 'self'; connect-src 'self'; img-src 'self' data:; object-src 'none'; base-uri 'none'; frame-ancestors 'none'"));
436    if worker {
437        response
438            .headers_mut()
439            .insert("service-worker-allowed", HeaderValue::from_static("/"));
440    }
441    response
