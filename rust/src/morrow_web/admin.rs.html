6pub(super) struct System {
7    started: Instant,
8    parallelism: usize,
9    executable_bytes: Option<u64>,
10}
11impl System {
12    pub(super) fn new() -> Self {
13        Self {
14            started: Instant::now(),
15            parallelism: std::thread::available_parallelism().map_or(1, usize::from),
16            executable_bytes: std::env::current_exe()
17                .ok()
18                .and_then(|path| std::fs::metadata(path).ok())
19                .filter(|metadata| metadata.is_file())
20                .map(|metadata| metadata.len()),
21        }
22    }
23}

31#[derive(Serialize)]
32struct Status {
33    schema_version: u8,
34    version: &'static str,
35    operating_system: &'static str,
36    architecture: &'static str,
37    process_id: u32,
38    available_parallelism: usize,
39    uptime_seconds: u64,
40    executable_bytes: Option<u64>,
41    memory: Memory,
42    durability: &'static str,
43    embedded_asset_bytes: usize,
44    websocket_connections: usize,
45    websocket_limit: usize,
46    tcp_limit: usize,
47    room_limit: usize,
48    namespace_limit: usize,
49    session_limit: usize,
50    runtime: owner::PoolSnapshot,
51    cluster: Option<peer::Observation>,
52}
53impl Status {
54    fn read(app: &App) -> Self {
55        let memory = morrow_web_app::system::process_memory();
56        Self {
57            schema_version: 1,
58            version: env!("CARGO_PKG_VERSION"),
59            operating_system: std::env::consts::OS,
60            architecture: std::env::consts::ARCH,
61            process_id: std::process::id(),
62            available_parallelism: app.system.parallelism,
63            uptime_seconds: app.system.started.elapsed().as_secs(),
64            executable_bytes: app.system.executable_bytes,
65            memory: Memory {
66                resident_bytes: memory.resident_bytes,
67                peak_resident_bytes: memory.peak_resident_bytes,
68            },
69            durability: if app.config.data_dir.is_some() {
70                "checkpointed"
71            } else {
72                "ephemeral"
73            },
74            embedded_asset_bytes: app.assets.iter().map(|(_, bytes)| bytes.len()).sum(),
75            websocket_connections: app
76                .config
77                .limits
78                .max_connections
79                .saturating_sub(app.sockets.available_permits()),
80            websocket_limit: app.config.limits.max_connections,
81            tcp_limit: app.config.max_tcp_connections,
82            room_limit: app.config.limits.max_rooms,
83            namespace_limit: app.config.limits.max_namespaces,
84            session_limit: app.config.max_sessions,
85            runtime: app.requests.snapshot(),
86            cluster: app.cluster.as_ref().map(peer::Cluster::observe),
87        }
88    }
89}

91async fn authorize(app: &App, headers: &HeaderMap) -> Result<(), StatusCode> {
92    if headers.contains_key(header::ORIGIN) {
93        origin(headers, app)?;
94    }
95    let (reply, rx) = oneshot::channel();
96    response(
97        app,
98        owner::Request::Authenticate {
99            token: cookie(headers)?,
100            csrf: None,
101            reply,
102        },
103        rx,
104    )
105    .await?;
106    Ok(())
107}

109fn private(mut response: Response) -> Response {
110    response
111        .headers_mut()
112        .insert(header::CACHE_CONTROL, HeaderValue::from_static("no-store"));
113    response.headers_mut().insert(header::CONTENT_SECURITY_POLICY, HeaderValue::from_static(
114        "default-src 'none'; script-src 'none'; style-src 'self'; img-src 'self'; base-uri 'none'; frame-ancestors 'none'; form-action 'self'"));
115    response
116}

118pub(super) async fn status(State(app): State<App>, headers: HeaderMap) -> Response {
119    private(match authorize(&app, &headers).await {
120        Ok(()) => Json(Status::read(&app)).into_response(),
121        Err(code) => code.into_response(),
122    })
123}

125pub(super) async fn page(State(app): State<App>, headers: HeaderMap) -> Response {
126    private(match authorize(&app, &headers).await {
127        Ok(()) => axum::response::Html(render(&Status::read(&app))).into_response(),
128        Err(code) => {
129            let content = match code {
130                StatusCode::UNAUTHORIZED => {
131                    "<h1>Open the application to see your system.</h1><p>The dashboard uses the session the application opens automatically.</p><a class=button href=/>Open the application</a>"
132                }
133                StatusCode::FORBIDDEN => {
134                    "<h1>This request is not allowed.</h1><p>Open the dashboard from the application’s own address.</p><a class=button href=/>Open the application</a>"
135                }
136                _ => {
137                    "<h1>Snapshot temporarily unavailable.</h1><p>The server could not authorize this request within its current capacity.</p><a class=button href=/admin>Try again</a>"
138                }
139            };
140            (
141                code,
142                axum::response::Html(shell(&format!(
143                    "<main><p class=eyebrow>SERVER ACCESS</p>{content}</main>"
144                ))),
145            )
146                .into_response()
147        }
148    })
149}

167fn escape(value: &str) -> String {
168    value
169        .replace('&', "&amp;")
170        .replace('<', "&lt;")
171        .replace('>', "&gt;")
172        .replace('"', "&quot;")
173        .replace('\'', "&#39;")
174}

176fn bytes(value: usize) -> String {
177    format!("{:.2} MiB", value as f64 / 1_048_576.0)
178}

180fn optional_bytes(value: Option<u64>) -> String {
181    value.map_or_else(
182        || "Unavailable".into(),
183        |bytes| format!("{:.2} MiB", bytes as f64 / 1_048_576.0),
184    )
185}

187fn render(status: &Status) -> String {
188    let mut body = String::from(
189        "<main><section class=intro><div><p class=eyebrow>ROOTED. RUNNING. OBSERVABLE.</p><h1>A little clarity<br>for your system.</h1><p>A snapshot of the server behind your shared garden.</p></div><nav aria-label=\"Dashboard actions\"><a class=button href=/admin>Refresh snapshot</a><a href=/admin/status>JSON status ↗</a></nav></section>",
190    );
191    let workers = &status.runtime.workers;
192    let rooms: usize = workers.iter().map(|worker| worker.rooms).sum();
193    let _ = write!(
194        body,
195        "<section class=metrics aria-label=\"System summary\"><article><p class=eyebrow>UPTIME</p><strong>{}h {:02}m {:02}s</strong><p>Since this server started</p></article><article><p class=eyebrow>ACTOR WORKERS</p><strong>{}</strong><p>Independently pinned threads</p></article><article><p class=eyebrow>OPEN WEBSOCKETS</p><strong>{}<small> / {}</small></strong><p>Includes sockets awaiting a room</p></article><article><p class=eyebrow>RESIDENT MEMORY</p><strong>{}</strong><p>Server process · OS snapshot</p></article></section>",
196        status.uptime_seconds / 3600,
197        status.uptime_seconds / 60 % 60,
198        status.uptime_seconds % 60,
199        workers.len(),
200        status.websocket_connections,
201        status.websocket_limit,
202        optional_bytes(status.memory.resident_bytes)
203    );
204    cluster::render(&mut body, status.cluster.as_ref());
205    body.push_str("<section class=panel><div class=section-heading><div><p class=eyebrow>ACTOR RUNTIME</p><h2>Workers, at a glance.</h2></div><span class=badge>Native Morrow</span></div><div class=table-scroll tabindex=0 role=region aria-label=\"Worker observations\"><table><caption>Counts from each worker’s last completed observation; busy work may change them.</caption><thead><tr><th scope=col>Worker</th><th scope=col>State</th><th scope=col>Rooms</th><th scope=col>Namespaces</th><th scope=col>Connections</th><th scope=col>Subscriptions</th></tr></thead><tbody>");
206    for worker in workers {
207        let state = match worker.state {
208            owner::WorkerState::Idle => "Idle",
209            owner::WorkerState::Busy => "Busy",
210            owner::WorkerState::Stopped => "Stopped",
211        };
212        let _ = write!(
213            body,
214            "<tr><th scope=row>Worker {:02}</th><td><span class=badge>{state}</span></td><td>{}</td><td>{}</td><td>{}</td><td>{}</td></tr>",
215            worker.index + 1,
216            worker.rooms,
217            worker.namespaces,
218            worker.connections,
219            worker.subscriptions
220        );
221    }
222    body.push_str("</tbody></table></div></section><div class=details><section class=panel><p class=eyebrow>HOST &amp; BUILD</p><h2>One running binary.</h2><dl>");
223    let executable = optional_bytes(status.executable_bytes);
224    for (name, value) in [
225        ("Morrow version", status.version.to_owned()),
226        (
227            "Platform",
228            format!("{} · {}", status.operating_system, status.architecture),
229        ),
230        ("Process ID", status.process_id.to_string()),
231        (
232            "Available CPU parallelism",
233            status.available_parallelism.to_string(),
234        ),
235        ("Executable size", executable),
236        (
237            "Resident memory (RSS)",
238            optional_bytes(status.memory.resident_bytes),
239        ),
240        (
241            "Peak resident memory",
242            optional_bytes(status.memory.peak_resident_bytes),
243        ),
244        (
245            "Embedded browser assets",
246            bytes(status.embedded_asset_bytes),
247        ),
248    ] {
249        let _ = write!(
250            body,
251            "<div><dt>{name}</dt><dd>{}</dd></div>",
252            escape(&value)
253        );
254    }
255    body.push_str("</dl></section><section class=panel><p class=eyebrow>CAPACITY &amp; RECOVERY</p><h2>Explicit boundaries.</h2><dl>");
256    for (name, value) in [
257        ("Room storage", status.durability.to_owned()),
258        ("Loaded rooms", format!("{rooms} / {}", status.room_limit)),
259        (
260            "Retained sessions",
261            format!(
262                "{} / {}",
263                status.runtime.authentication.retained_sessions, status.session_limit
264            ),
265        ),
266        (
267            "Authentication owner",
268            if status.runtime.authentication.stopped {
269                "Stopped"
270            } else {
271                "Running"
272            }
273            .into(),
274        ),
275        (
276            "Ingress in use",
277            format!(
278                "{} / {}",
279                status.runtime.ingress_in_use, status.runtime.ingress_limit
280            ),
281        ),
282        ("TCP admission limit", status.tcp_limit.to_string()),
283        ("Namespace limit", status.namespace_limit.to_string()),
284    ] {
285        let _ = write!(
286            body,
287            "<div><dt>{name}</dt><dd>{}</dd></div>",
288            escape(&value)
289        );
290    }
291    body.push_str("</dl></section></div><footer><p>Read-only snapshot · Refresh to observe changes.</p><p>The dashboard uses the same session the application opens automatically. Counts are independent observations, not a globally atomic snapshot. No room contents or credentials are displayed.</p></footer></main>");
